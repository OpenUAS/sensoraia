    33/**
    34 * @brief Validated geographic placement metadata shared by heatmap consumers.
    35 * @details The renderer uses a local equirectangular frame about the center; overlay
    36 * code must transform to its map projection rather than stretch the JPEG by size.
    37 */
    38struct ear_heatmap_georef {
    39 unsigned int width_px; /**< Encoded heatmap width; must match the JPEG consumer reads. */
    40 unsigned int height_px; /**< Encoded heatmap height. */
    41 double m_per_px; /**< Ground spacing represented by each heatmap pixel. */
    42 double lat_center; /**< Latitude origin for the local projection, degrees. */
    43 double lon_center; /**< Longitude origin for the local projection, degrees. */
    44 double lat_north; /**< Latitude at the upper image edge, degrees. */
    45 double lon_west; /**< Longitude at the left image edge, degrees. */
    46 bool has_spot; /**< True only when the sidecar contains a complete spot pair. */
    47 double spot_lat; /**< Estimated spot latitude, degrees, when `has_spot`. */
    48 double spot_lon; /**< Estimated spot longitude, degrees, when `has_spot`. */
    49};

    51/**
    52 * @brief Read the georeference sidecar emitted by the heatmap renderer.
    53 * @param filename Path to the `.geo` sidecar.
    54 * @param georef Receives validated dimensions, projection and optional spot.
    55 * @return 0 on success; -1 for invalid arguments, malformed data or I/O error.
    56 * @details Required dimensions, scale and edge/origin coordinates are validated
    57 * before publication to the caller; partial parses never replace its struct.
    58 * Duplicate fields, oversized lines and incomplete optional spot pairs fail.
    59 */
    60int ear_heatmap_read_georef(const char *filename, struct ear_heatmap_georef *georef);

    62/**
    63 * @brief Write a decorated north-up JPEG of relative measured acoustic intensity.
    64 * @param filename Output path; an adjacent .geo sidecar is attempted after JPEG success.
    65 * @param samples Borrowed, immutable geotagged windows; not retained after the call.
    66 * @param count Window count; at least three usable finite, unclipped windows are required.
    67 * @param spot Optional borrowed estimate; only valid estimates receive a marker.
    68 * @return 0 for a successfully written JPEG, -1 for insufficient data or file/codec failure.
    69 * @details Track dots, an estimated-source marker and a scale bar aid interpretation.
    70 * Calls serialize on bounded shared scratch storage. Grid cells are roughly AGL/4;
    71 * bilinear upsampling adds visual smoothness, not measurement resolution.
    72 * @warning Sidecar write failure does not change the return status. Neither the
    73 * smoothed field nor the marker demonstrates localization accuracy or alarm presence.
    74 */
    75int ear_heatmap_write(const char *filename, const struct ear_sample *samples, size_t count,
    76 const struct ear_loudest_spot *spot);
    77
    78/**
    79 * @brief Write the same field without track dots, source marker or scale bar.
    80 * @param filename JPEG path; a best-effort .geo sidecar supplies overlay placement.
    81 * @param samples Borrowed immutable acoustic windows.
    82 * @param count Number of supplied windows, with the same usable-data gate as ear_heatmap_write().
    83 * @param spot Optional estimate retained in sidecar metadata, not drawn into pixels.
    84 * @return 0 for JPEG success, -1 for insufficient usable data or file/codec failure.
    85 * @details Use the same interpolation and shared workspace as the decorated
    86 * renderer, keeping annotation pixels out of the field used for overlays.
    87 * @see ear_heatmap_write for locking, resolution and sidecar-status caveats.
    88 */
    89int ear_heatmap_write_field(const char *filename, const struct ear_sample *samples, size_t count,
    90 const struct ear_loudest_spot *spot);
