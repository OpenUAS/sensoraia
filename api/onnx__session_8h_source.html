    37/** @brief Supported numeric types; values match the ONNX Runtime C ABI. */
    38enum onnx_element_type {
    39 ONNX_FLOAT32 = 1, ONNX_UINT8 = 2, ONNX_INT8 = 3,
    40 ONNX_INT32 = 6, ONNX_INT64 = 7, ONNX_FLOAT64 = 11
    41};

    43/** @brief Named concrete tensor view; its aligned backing storage must outlive the session. */
    44struct onnx_tensor {
    45 const char *name; /**< Graph name, copied by session open rather than matched by declaration order. */
    46 enum onnx_element_type type; /**< Exact numeric storage representation. */
    47 size_t rank; /**< Concrete rank, including zero for a scalar. */
    48 int64_t dimensions[ONNX_RANK_LIMIT]; /**< Positive concrete extents; unused axes are ignored. */
    49 void *data; /**< Aligned caller storage, never freed by the session. */
    50 size_t bytes; /**< Exact byte count implied by type and concrete dimensions. */
    51};

    53/** @brief Explicit execution limits so each model cannot create an unbounded CPU pool. */
    54struct onnx_session_options {
    55 unsigned threads; /**< Intra-op thread count in 1..32; inter-op execution remains sequential. */
    56 const char *library_directory; /**< Optional runtime search directory, consumed during open. */
    57};

    59#define ONNX_SESSION_OPTIONS_DEFAULT {1U, NULL}

    61/** @brief Noncopyable caller-owned session; zero-initialize and serialize open/run/close ownership.
    62 * @details Fields are implementation state, not configuration knobs. Caller tensors must
    63 * remain alive until close has completed; a failed run does not validate their output contents. */
    64struct onnx_session {
    65 void *handle, *memory, *run_options;
    66 void *inputs[ONNX_TENSOR_LIMIT], *outputs[ONNX_TENSOR_LIMIT];
    67 char input_names[ONNX_TENSOR_LIMIT][ONNX_NAME_CAPACITY];
    68 char output_names[ONNX_TENSOR_LIMIT][ONNX_NAME_CAPACITY];
    69 struct onnx_tensor output_specs[ONNX_TENSOR_LIMIT];
    70 size_t input_count, output_count;
    71 pthread_mutex_t mutex;
    72 pthread_cond_t condition;
    73 pthread_t watchdog;
    74 bool synchronized, watched, stopping, running, timed_out, cancelled;
    75 uint64_t deadline_us;
    76};

    78/** @brief Compute bounded concrete storage, rejecting unsupported types, dynamic extents and overflow.
    79 * @param type Supported numeric element type.
    80 * @param dimensions Concrete extents; NULL is allowed only for rank zero.
    81 * @param rank Number of extents, at most ONNX_RANK_LIMIT.
    82 * @param bytes Output byte count, meaningful only on success.
    83 * @return True for a representable tensor within ONNX_BUFFER_LIMIT. */
    84bool onnx_tensor_bytes(enum onnx_element_type type, const int64_t *dimensions,
    85 size_t rank, size_t *bytes);
    86/** @brief Validate graph contracts and bind reusable caller buffers; failure releases acquired resources.
    87 * @param session Zero-initialized owner with no open session; do not copy it after opening.
    88 * @param model Readable ONNX graph path.
    89 * @param inputs Named input views whose backing buffers outlive the session.
    90 * @param input_count Number of input views, in 1..ONNX_TENSOR_LIMIT.
    91 * @param outputs Named writable output views with the same lifetime rule.
    92 * @param output_count Number of output views, in 1..ONNX_TENSOR_LIMIT.
    93 * @param options Optional execution/search policy; NULL uses one CPU thread.
    94 * @param error Writable failure diagnostic, cleared on success.
    95 * @param capacity Available bytes in error.
    96 * @return True only after every name, type, shape and backing byte count is accepted.
    97 * @note Dynamic graph axes bind to the declared concrete shape; outputs are checked again after each run. */
    98bool onnx_session_open(struct onnx_session *session, const char *model,
    99 const struct onnx_tensor *inputs, size_t input_count,
    100 const struct onnx_tensor *outputs, size_t output_count,
    101 const struct onnx_session_options *options,
    102 char *error, size_t capacity);
    103/** @brief Run once; deadline is absolute CLOCK_MONOTONIC microseconds, or zero for none.
    104 * @param session Open owner; one run at a time is admitted.
    105 * @param deadline_us Absolute completion budget in the host monotonic clock domain.
    106 * @param error Writable failure diagnostic, cleared on success.
    107 * @param capacity Available bytes in error.
    108 * @return True for complete execution with valid output shapes and an unexpired budget.
    109 * @warning Ignore output buffers on false. Cancellation is cooperative, not a hard deadline. */
    110bool onnx_session_run(struct onnx_session *session, uint64_t deadline_us,
    111 char *error, size_t capacity);
    112/** @brief Request cooperative cancellation from another thread; join the owner before destroying storage.
    113 * @param session Live owner; NULL/unopened owners are ignored. */
    114void onnx_session_cancel(struct onnx_session *session);
    115/** @brief Cancel/wait for an active run and release resources; repeated closes are harmless.
    116 * @param session Exclusive lifecycle owner; NULL is ignored.
    117 * @pre No new calls may begin while close destroys synchronization objects. */
    118void onnx_session_close(struct onnx_session *session);
    119/** @brief Read the clock used by inference deadlines.
    120 * @return CLOCK_MONOTONIC microseconds, or zero when the clock cannot be read. */
    121uint64_t onnx_monotonic_us(void);
