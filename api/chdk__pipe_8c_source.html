    141static struct {
    142 int descriptor;
    143 size_t size;
    144 uint8_t bytes[CHDK_CONTROL_REPLY_SIZE];
    145 bool invalid;

    148/** @brief Collect one chunk from the nonblocking control reply pipe.
    149 * @details Retain partial bytes across polls; flag overflow and hard I/O
    150 * errors so a truncated or overlong reply cannot count as confirmation. */
    151static void drain_control_reply(void)
    152{
    153 if (camera_reply.descriptor < 0) return;
    154 uint8_t bytes[32];
    155 ssize_t received = read(camera_reply.descriptor, bytes, sizeof(bytes));
    156 if (received < 0 && errno != EAGAIN && errno != EWOULDBLOCK && errno != EINTR) camera_reply.invalid = true;
    157 for (ssize_t index = 0; index < received; ++index) {
    158 if (camera_reply.size < sizeof(camera_reply.bytes)) camera_reply.bytes[camera_reply.size++] = bytes[index];
    159 else camera_reply.invalid = true;
    160 }
    161}

    172/** @brief Read the clock used for controller supervision.
    173 * @return Monotonic milliseconds, or zero on clock failure.
    174 * @details Deadlines must not move when the wall clock is corrected. */
    175static uint64_t chdk_monotonic_ms(void)
    176{
    177 struct timespec now;
    178 if (clock_gettime(CLOCK_MONOTONIC, &now) != 0) return 0;
    179 return (uint64_t)now.tv_sec * 1000 + (uint64_t)now.tv_nsec / 1000000;
    180}

    182/** @brief Parse a decimal build-profile limit.
    183 * @param text Null-terminated numeric setting.
    184 * @param minimum Inclusive lower bound.
    185 * @param maximum Inclusive upper bound.
    186 * @return Parsed value, or zero on syntax or range error.
    187 * @details Limit callers use positive ranges so zero is an unambiguous
    188 * failure sentinel. Reject signs and suffixes rather than accepting a prefix. */
    189static unsigned long chdk_number(const char *text, unsigned long minimum, unsigned long maximum)
    190{
    191 char *end;
    192 errno = 0;
    193 unsigned long value = strtoul(text, &end, 10);
    194 return errno || *text < '0' || *text > '9' || *end || value < minimum || value > maximum ? 0 : value;
    195}

    197/** @brief Poll the owned controller for exit while collecting its reply.
    198 * @param[out] status Wait status; valid when the result is one.
    199 * @param timeout_ms Polling budget; zero checks without sleeping.
    200 * @param cancelable True to honor the service cancellation flag.
    201 * @return One if reaped, zero on timeout or cancellation, or -1 on wait/setup error.
    202 * @details Drain during waits and again on exit; the last reply can arrive
    203 * between the initial read and the successful waitpid. */
    204static int wait_camera(int *status, unsigned timeout_ms, bool cancelable)
    205{
    206 const uint64_t started = chdk_monotonic_ms();
    207 if (!started) return -1;
    208 for (;;) {
    209 drain_control_reply();
    210 pid_t result = waitpid(camera_pid, status, WNOHANG);
    211 if (result == camera_pid) { drain_control_reply(); camera_pid = -1; return 1; }
    212 if (result < 0 && errno == ECHILD) { camera_pid = -1; return -1; }
    213 if (result < 0 && errno != EINTR) return -1;
    214 if (cancelable && !capture_running(camera_running)) return 0;
    215 uint64_t now = chdk_monotonic_ms();
    216 if (!now || now - started >= timeout_ms) return 0;
    217 struct timespec pause = {.tv_sec = 0, .tv_nsec = 10000000};
    218 nanosleep(&pause, NULL);
    219 }
    220}

    222/** @brief Stop the owned controller group within bounded grace periods.
    223 * @details Allow graceful PTP cleanup before forced termination. Preserve an
    224 * unreaped PID so future commands cannot acquire the camera concurrently. */
    225static void stop_camera(void)
    226{
    227 if (camera_pid <= 0) return;
    228 int status = 0;
    229 if (wait_camera(&status, 0, false) == 1 || camera_pid <= 0) return;
    230 kill(-camera_pid, SIGTERM);
    231 /* The controller closes its PTP session on SIGTERM; SIGKILL would leave camera state behind. */
    232 if (wait_camera(&status, 3000, false) == 1 || camera_pid <= 0) return;
    233 kill(-camera_pid, SIGKILL);
    234 partial_cleanup_pending = true;
    235 if (wait_camera(&status, 1500, false) != 1 && camera_pid > 0)
    236 fprintf(stderr, "CHDK: killed controller has not exited; retaining process ownership\n");
    237}

    245/** @brief Run one CHDKbridge capture or lens operation.
    246 * @param destination Private capture path; unused for a control operation.
    247 * @param command Zero for capture, otherwise the validated control command.
    248 * @param value Control operand, unused for capture and lens retraction.
    249 * @return Zero for successful child execution, or -1 on setup, cancellation or child failure.
    250 * @details Validate build-profile limits and acquire exclusive child
    251 * ownership before spawning. Lens operations use a distinct reply pipe;
    252 * child exit alone does not establish that the camera applied the command. */
    253static int run_camera(const char *destination, uint8_t command, uint8_t value)
    254{
    255 if ((!destination && !command) || !capture_running(camera_running)) return -1;
    256 if (camera_pid > 0) {
    257 int status = 0;
    258 wait_camera(&status, 0, false);
    259 if (camera_pid > 0) return -1;
    260 }
    261 unsigned long timeout = chdk_number(USHER_CHDK_TIMEOUT, 1, 3600);
    262 if (!timeout || !chdk_number(USHER_CHDK_USB_TIMEOUT, 50, 10000) ||
    263 !chdk_number(USHER_CHDK_MAX_DOWNLOAD_MIB, 1, 256) ||
    264 (USHER_CHDK_DIRECT[0] != '0' && USHER_CHDK_DIRECT[0] != '1') || USHER_CHDK_DIRECT[1] ||
    265 (USHER_CHDK_KEEP_AWAKE[0] && ((USHER_CHDK_KEEP_AWAKE[0] != '0' && USHER_CHDK_KEEP_AWAKE[0] != '1') ||
    266 USHER_CHDK_KEEP_AWAKE[1])) ||
    267 (USHER_CHDK_PORT[0] && (USHER_CHDK_BUS[0] || USHER_CHDK_DEVICE[0])) ||
    268 ((USHER_CHDK_BUS[0] == '\0') != (USHER_CHDK_DEVICE[0] == '\0'))) {
    269 fprintf(stderr, "CHDK: invalid timeout, size limit or conflicting USB selection in build profile\n");
    270 return -1;
    271 }
    272 char *arguments[64];
    273 char timeout_text[12], value_text[4];
    274 if (command && timeout > 60) timeout = 60;
    275 if (snprintf(timeout_text, sizeof(timeout_text), "%lu", timeout) < 1
    276 || snprintf(value_text, sizeof(value_text), "%u", value) < 1) return -1;
    277 size_t count = 0;
    278 arguments[count++] = (char *)USHER_CHDK_COMMAND;
    279 arguments[count++] = "--timeout";
    280 arguments[count++] = timeout_text;
    281 arguments[count++] = "--usb-timeout";
    282 arguments[count++] = (char *)USHER_CHDK_USB_TIMEOUT;
    283 arguments[count++] = "--max-download-mib";
    284 arguments[count++] = (char *)USHER_CHDK_MAX_DOWNLOAD_MIB;
    285 if (USHER_CHDK_PORT[0]) {
    286 arguments[count++] = "--port";
    287 arguments[count++] = (char *)USHER_CHDK_PORT;
    288 }
    289 if (USHER_CHDK_BUS[0]) {
    290 arguments[count++] = "--bus";
    291 arguments[count++] = (char *)USHER_CHDK_BUS;
    292 arguments[count++] = "--device";
    293 arguments[count++] = (char *)USHER_CHDK_DEVICE;
    294 }
    295 static const char *const options[][2] = {
    296 {"--iso-mode", USHER_CHDK_ISO_MODE}, {"--flash", USHER_CHDK_FLASH},
    297 {"--resolution", USHER_CHDK_RESOLUTION}, {"--quality", USHER_CHDK_QUALITY},
    298 {"--date-stamp", USHER_CHDK_DATE_STAMP}, {"--af-assist", USHER_CHDK_AF_ASSIST},
    299 {"--raw", USHER_CHDK_RAW},
    300 {"--focus-mode", USHER_CHDK_FOCUS_MODE}, {"--focus-distance", USHER_CHDK_FOCUS_DISTANCE},
    301 {"--focus-timeout", USHER_CHDK_FOCUS_TIMEOUT}, {"--focus-tolerance", USHER_CHDK_FOCUS_TOLERANCE},
    302 {"--shutter", USHER_CHDK_SHUTTER}, {"--tv96", USHER_CHDK_TV96},
    303 {"--iso-real", USHER_CHDK_ISO_REAL}, {"--iso-market", USHER_CHDK_ISO_MARKET}, {"--sv96", USHER_CHDK_SV96},
    304 {"--aperture", USHER_CHDK_APERTURE}, {"--av96", USHER_CHDK_AV96}, {"--nd", USHER_CHDK_ND},
    305 {"--exposure-timeout", USHER_CHDK_EXPOSURE_TIMEOUT}, {"--exposure-tolerance", USHER_CHDK_EXPOSURE_TOLERANCE},
    306 {"--keep-awake", USHER_CHDK_KEEP_AWAKE}
    307 };
    308 for (size_t index = 0; index < sizeof(options) / sizeof(options[0]); ++index) {
    309 if (command) break;
    310 if (!options[index][1][0]) continue;
    311 if (index == 0 && (USHER_CHDK_ISO_REAL[0] || USHER_CHDK_ISO_MARKET[0] || USHER_CHDK_SV96[0])) continue;
    312 if (count + 2 >= sizeof(arguments) / sizeof(arguments[0])) return -1;
    313 arguments[count++] = (char *)options[index][0];
    314 arguments[count++] = (char *)options[index][1];
    315 }
    316 if (command) {
    317 arguments[count++] = "--control-result-fd";
    318 arguments[count++] = "3";
    319 arguments[count++] = command == CAM_CMD_ZOOM_ABSOLUTE ? "zoom" : "--retract-lens";
    320 if (command == CAM_CMD_ZOOM_ABSOLUTE) arguments[count++] = value_text;
    321 } else {
    322 if (USHER_CHDK_DIRECT[0] == '1') arguments[count++] = "--direct-download";
    323 arguments[count++] = (char *)destination;
    324 }
    325 arguments[count] = NULL;
    326 posix_spawnattr_t attributes;
    327 int error = posix_spawnattr_init(&attributes);
    328 if (error) return -1;
    329 sigset_t mask, defaults;
    330 sigemptyset(&mask);
    331 sigemptyset(&defaults);
    332 sigaddset(&defaults, SIGINT);
    333 sigaddset(&defaults, SIGTERM);
    334 error = posix_spawnattr_setsigmask(&attributes, &mask);
    335 if (!error) error = posix_spawnattr_setsigdefault(&attributes, &defaults);
    336 if (!error) error = posix_spawnattr_setpgroup(&attributes, 0);
    337 if (!error) error = posix_spawnattr_setflags(&attributes,
    338 POSIX_SPAWN_SETSIGMASK | POSIX_SPAWN_SETSIGDEF | POSIX_SPAWN_SETPGROUP);
    339 posix_spawn_file_actions_t actions;
    340 bool actions_ready = false;
    341 int descriptors[2] = {-1, -1};
    342 if (!error && command) {
    343 if (pipe(descriptors) != 0) error = errno;
    344 if (!error && (fcntl(descriptors[0], F_SETFD, FD_CLOEXEC) < 0
    345 || fcntl(descriptors[1], F_SETFD, FD_CLOEXEC) < 0
    346 || fcntl(descriptors[0], F_SETFL, O_NONBLOCK) < 0)) error = errno;
    347 if (!error) { error = posix_spawn_file_actions_init(&actions); actions_ready = error == 0; }
    348 if (!error) error = posix_spawn_file_actions_adddup2(&actions, descriptors[1], 3);
    349 if (!error && descriptors[0] != 3) error = posix_spawn_file_actions_addclose(&actions, descriptors[0]);
    350 if (!error && descriptors[1] != 3) error = posix_spawn_file_actions_addclose(&actions, descriptors[1]);
    351 }
    352 if (!error) error = posix_spawn(&camera_pid, USHER_CHDK_COMMAND, actions_ready ? &actions : NULL,
    353 &attributes, arguments, environ);
    354 if (actions_ready) posix_spawn_file_actions_destroy(&actions);
    355 posix_spawnattr_destroy(&attributes);
    356 if (descriptors[1] >= 0) close(descriptors[1]);
    357 if (error) {
    358 if (descriptors[0] >= 0) close(descriptors[0]);
    359 camera_pid = -1;
    360 fprintf(stderr, "CHDK: cannot start %s: %s\n", USHER_CHDK_COMMAND, strerror(error));
    361 return -1;
    362 }
    363 camera_reply.descriptor = descriptors[0];
    364 camera_reply.size = 0;
    365 camera_reply.invalid = false;
    366 errno = 0;
    367 if (setpriority(PRIO_PROCESS, (id_t)camera_pid, 10) != 0 && errno != ESRCH)
    368 fprintf(stderr, "CHDK: cannot lower controller scheduling priority: %s\n", strerror(errno));
    369 int status = 0;
    370 int result = wait_camera(&status, (unsigned)timeout * 1000 + 2000, true);
    371 if (camera_reply.descriptor >= 0) close(camera_reply.descriptor);
    372 camera_reply.descriptor = -1;
    373 if (result != 1) {
    374 fprintf(stderr, "CHDK: operation cancelled or deadline exceeded; no automatic capture retry\n");
    375 stop_camera();
    376 return -1;
    377 }
    378 if (!WIFEXITED(status) || WEXITSTATUS(status) != 0) {
    379 if (WIFSIGNALED(status)) fprintf(stderr, "CHDK: controller stopped by signal %d; no capture replay\n", WTERMSIG(status));
    380 else fprintf(stderr, "CHDK: controller exited with code %d; camera state may have changed\n", WIFEXITED(status) ? WEXITSTATUS(status) : -1);
    381 return -1;
    382 }
    383 return 0;
    384}

    386int chdk_pipe_init(void)
    387{
    388 chdk_pipe_deinit();
    389 if (camera_pid > 0 || !capture_running(camera_running)) return -1;
    390 if (access(USHER_CHDK_COMMAND, X_OK) != 0) {
    391 fprintf(stderr, "CHDK: controller unavailable at %s: %s\n", USHER_CHDK_COMMAND, strerror(errno));
    392 return -1;
    393 }
    394 camera_ready = true;
    395 return 0;

    398/** @brief True for the controller's private mkstemp name, e.g. c000123.jpg.part.Ab12Cd. */
    399static bool partial_output_name(const char *name)
    400{
    401 static const char middle[] = ".jpg.part.";
    402 if (name[0] != 'c') return false;
    403 for (size_t index = 1; index <= 6; ++index)
    404 if (name[index] < '0' || name[index] > '9') return false;
    405 for (size_t index = 0; index < sizeof(middle) - 1; ++index)
    406 if (name[7 + index] != middle[index]) return false;
    407 for (size_t index = 17; index < 23; ++index)
    408 if (!name[index] || name[index] == '/') return false;
    409 return name[23] == '\0';
    410}

    412/** @brief Delete partial outputs left by a killed or powered-off controller; none is running. */
    413static void remove_partial_outputs(const char *destination)
    414{
    415 char directory[4096];
    416 size_t separator = 0, length = 0;
    417 bool found = false;
    418 for (; destination[length]; ++length) {
    419 if (length + 1 >= sizeof(directory)) return;
    420 directory[length] = destination[length];
    421 if (destination[length] == '/') { separator = length; found = true; }
    422 }
    423 if (found) directory[separator ? separator : 1] = '\0';
    424 else { directory[0] = '.'; directory[1] = '\0'; }
    425 DIR *listing = opendir(directory);
    426 if (!listing) return;
    427 unsigned removed = 0;
    428 struct dirent *entry;
    429 while ((entry = readdir(listing))) {
    430 struct stat status;
    431 if (!partial_output_name(entry->d_name) ||
    432 fstatat(dirfd(listing), entry->d_name, &status, AT_SYMLINK_NOFOLLOW) != 0 ||
    433 !S_ISREG(status.st_mode) || unlinkat(dirfd(listing), entry->d_name, 0) != 0) continue;
    434 ++removed;
    435 }
    436 closedir(listing);
    437 if (removed) fprintf(stderr, "CHDK: removed %u partial download(s) left by an interrupted controller\n", removed);
    438}

    440int chdk_pipe_shoot(const char *filename)
    441{
    442 if (!camera_ready || !filename || !filename[0] || filename[0] == '-') return -1;
    443 if (partial_cleanup_pending && camera_pid <= 0) {
    444 remove_partial_outputs(filename);
    445 partial_cleanup_pending = false;
    446 }
    447 struct stat status;
    448 if (lstat(filename, &status) == 0 || errno != ENOENT) return -1;
    449 if (run_camera(filename, 0, 0) != 0) return -1;
    450 int descriptor = open(filename, O_RDONLY | O_NOFOLLOW | O_NONBLOCK | O_CLOEXEC);
    451 if (descriptor < 0) {

    469/** @brief Apply one supported lens command and qualify device evidence.
    470 * @param command Absolute zoom or lens retraction.
    471 * @param value Requested zoom value; must be zero for retraction.
    472 * @param[out] observed Returned device value; meaningful with applied status.
    473 * @param[out] evidence Returned evidence kind; meaningful with applied status.
    474 * @return A CAM_CONTROL status; UNKNOWN means execution could not be confirmed.
    475 * @details Check both the reply shape and echoed command. Do not retry
    476 * an ambiguous operation or infer success from the controller's exit alone. */
    477uint8_t chdk_pipe_control(uint8_t command, uint8_t value, uint8_t *observed, uint8_t *evidence)
    478{
    479 if (command != CAM_CMD_ZOOM_ABSOLUTE && command != CAM_CMD_LENS_RETRACT) return CAM_CONTROL_UNSUPPORTED;
    480 if (command == CAM_CMD_LENS_RETRACT && value) return CAM_CONTROL_INVALID_VALUE;
    481 if (!observed || !evidence || !camera_ready || !capture_running(camera_running)) return CAM_CONTROL_FAILED;
    482 if (camera_pid > 0) {
    483 int status = 0;
    484 wait_camera(&status, 0, false);
    485 if (camera_pid > 0) return CAM_CONTROL_BUSY;
    486 }
    487 *observed = 0;
    488 *evidence = CAM_CONTROL_EVIDENCE_NONE;
    489 if (run_camera(NULL, command, value) != 0 || camera_reply.invalid
    490 || camera_reply.size != CHDK_CONTROL_REPLY_SIZE || camera_reply.bytes[0] != CHDK_CONTROL_REPLY_TAG
    491 || camera_reply.bytes[1] != command
    492 || camera_reply.bytes[3] != (command == CAM_CMD_LENS_RETRACT ? CAM_CONTROL_EVIDENCE_PLAYBACK : CAM_CONTROL_EVIDENCE_VALUE)
    493 || (command == CAM_CMD_LENS_RETRACT && camera_reply.bytes[2])) return CAM_CONTROL_UNKNOWN;
    494 *observed = camera_reply.bytes[2];
    495 *evidence = camera_reply.bytes[3];
    496 return CAM_CONTROL_APPLIED;
    497}
