    122/** @brief Caller-owned scratch storage; never reserved by an imagery-only build.
    123 * @details Use static or otherwise long-lived storage, not a small thread stack.
    124 * No initialization is needed. Separate workspaces permit independent detectors.
    125 * Size is roughly 4.1 bytes per maximum native pixel plus 256 KiB of histogram.
    126 */
    127struct lwir_detection_workspace {
    128 uint8_t seen[(LWIR_MAX_WIDTH * LWIR_MAX_HEIGHT + 7) / 8]; ///< One visited bit per pixel.
    129 uint32_t queue[LWIR_MAX_WIDTH * LWIR_MAX_HEIGHT]; ///< BFS queue; afterwards lists the component's pixels.
    130 uint32_t histogram[UINT16_MAX + 1U]; ///< One bin per raw code, used for exact medians.
    131};

    133/** @brief Rank native thermal components without touching hardware, pose or files.
    134 * @param temperature Immutable little-endian Kelvin*64 plane.
    135 * @param bytes Exact accessible plane length, width * height * 2.
    136 * @param width Native width, at most LWIR_MAX_WIDTH.
    137 * @param height Native height, at most LWIR_MAX_HEIGHT.
    138 * @param workspace Exclusive scratch storage for this call, separate from input/result.
    139 * @param result Bounded candidates; ignore on failure.
    140 * @return True for a valid plane, including a plane with no candidates or no
    141 * valid codes at all (then background_celsius and threshold_celsius are NaN).
    142 * False for NULL pointers, zero or oversized dimensions, or a length mismatch.
    143 * @details Threshold = max(60 C absolute, frame median + max(20 C, 6 robust
    144 * sigma)), where robust sigma is 1.4826 x the median absolute deviation. Pixels
    145 * at or above it form 8-connected components; each must exceed its local
    146 * background by at least 15 C. See lwir_detection.c for the full algorithm.
    147 * Endpoint codes 0 and 65535 are invalid samples: they are excluded from the
    148 * frame median, noise estimate, components and local backgrounds.
    149 * @note Temperatures are apparent sensor values without emissivity or
    150 * atmospheric correction; the 60/100 C limits apply to those values.
    151 */
    152bool lwir_detect_hotspots(const uint8_t *temperature, size_t bytes, unsigned width, unsigned height,
    153 struct lwir_detection_workspace *workspace, struct lwir_hotspot_set *result);
