    18/**
    19 * @file image_exif.c
    20 * @brief JPEG EXIF updates for USHER flight and thermal-capture provenance.
    21 * @details Existing JPEG markers are copied while prior EXIF is replaced by a standards
    22 * compliant APP1 block. Default in-place updates avoid temporary files and leave
    23 * equal-position image bytes untouched. IMAGE_EXIF_ATOMIC_REPLACE selects the older
    24 * sibling-file/fsync/rename method when original-file retention is more important.
    25 * In-place writes are not atomic; interrupted mutation can damage the JPEG. Neither
    26 * method promises full power-loss durability, since the parent directory is not synced.
    27 */

    58struct exif_workspace {
    59 atomic_bool in_use;
    60 char temporary_name[PATH_MAX];
    61 unsigned char scratch[EXIF_SCRATCH_CAPACITY];
    62 unsigned char payload[EXIF_MAX_PAYLOAD];
    63 struct exif_document document;

    66static struct exif_workspace exif_workspaces[EXIF_WORKSPACE_COUNT];
    67static _Thread_local struct exif_workspace *active_workspace;

    69/** @brief Claim a bounded EXIF editing slot for the current thread.
    70 * @return Owned slot, or NULL with EBUSY for reentry or EAGAIN for exhaustion.
    71 * @details Keep large scratch buffers off worker stacks. Acquire ordering
    72 * pairs with release so a new owner does not race the previous user. */
    73static struct exif_workspace *workspace_acquire(void)
    74{
    75 if (active_workspace != NULL) {
    76 errno = EBUSY;
    77 return NULL;
    78 }
    79 for (size_t index = 0; index < EXIF_WORKSPACE_COUNT; ++index) {
    80 bool expected = false;
    81 if (atomic_compare_exchange_strong_explicit(&exif_workspaces[index].in_use, &expected, true,
    82 memory_order_acquire, memory_order_relaxed)) {
    83 struct exif_workspace *workspace = &exif_workspaces[index];
    84 workspace->temporary_name[0] = '\0';
    85 exif_document_init(&workspace->document);
    86 active_workspace = workspace;
    87 return workspace;
    88 }
    89 }
    90 errno = EAGAIN;
    91 return NULL;
    92}

    94/** @brief Return an editing slot to the shared pool.
    95 * @param workspace Thread-owned slot, or NULL.
    96 * @details Clear the thread-local reference before making the slot
    97 * available. All borrowed document and scratch pointers expire here. */
    98static void workspace_release(struct exif_workspace *workspace)
    99{
    100 if (workspace == NULL) {
    101 return;
    102 }
    103 active_workspace = NULL;
    104 atomic_store_explicit(&workspace->in_use, false, memory_order_release);
    105}

    107/** @brief Import an existing JPEG EXIF block into the active editing slot.
    108 * @param filename JPEG to read.
    109 * @return Borrowed editable document, or NULL if the slot or metadata is unavailable.
    110 * @details Parse the bounded payload before importing its entries so a
    111 * metadata-only update can retain the image's existing tags. */
    112static struct exif_document *load_exif(const char *filename)
    113{
    114 if (!active_workspace) return NULL;
    115 size_t size;
    116 struct exif_view view;
    117 if (exif_read_jpeg(filename, active_workspace->payload, sizeof(active_workspace->payload), &size) != 0
    118 || exif_parse(&view, active_workspace->payload, size) != 0
    119 || image_exif_import(&active_workspace->document, &view) != 0) return NULL;
    120 return &active_workspace->document;
    121}

    123/** @brief Encode one signed GPS coordinate as EXIF hemisphere and DMS tags.
    124 * @param[in,out] exif Editable EXIF document.
    125 * @param reference_tag GPS tag for the hemisphere letter.
    126 * @param coordinate_tag GPS tag for the three unsigned rationals.
    127 * @param coordinate Signed coordinate in units of 10^-7 degrees.
    128 * @param limit Inclusive absolute bound in the same units.
    129 * @param positive Hemisphere for nonnegative values.
    130 * @param negative Hemisphere for negative values.
    131 * @return Zero on success, or -1 on invalid range or insufficient tag storage.
    132 * @details Split the magnitude with integer arithmetic so coordinate
    133 * conversion does not introduce floating-point rollover between DMS fields.
    134 * A failure after setting the hemisphere can leave the document modified. */
    135static int set_coordinate(struct exif_document *exif, uint16_t reference_tag, uint16_t coordinate_tag,
    136 int32_t coordinate, int32_t limit, char positive, char negative)
    137{
    138 if (coordinate < -limit || coordinate > limit) {
    139 return -1;
    140 }
    141
    142 char reference[2] = {coordinate < 0 ? negative : positive, '\0'};
    143 if (exif_document_ascii(exif, EXIF_GPS, reference_tag, reference) != 0) {
    144 return -1;
    145 }
    146
    147 unsigned char *entry = exif_document_reserve(exif, EXIF_GPS, coordinate_tag, EXIF_RATIONAL, 3);
    148 if (entry == NULL) {
    149 return -1;
    150 }
    151
    152 uint64_t absolute = coordinate < 0 ? (uint64_t)(-(int64_t)coordinate) : (uint64_t)coordinate;
    153 uint32_t degrees = (uint32_t)(absolute / 10000000U);
    154 uint64_t remainder = absolute % 10000000U;
    155 uint32_t minutes = (uint32_t)((remainder * 60U) / 10000000U);
    156 remainder = remainder * 60U - (uint64_t)minutes * 10000000U;
    157 uint32_t seconds_millionths = (uint32_t)((remainder * 60U * 1000000U) / 10000000U);
    158
    159 exif_encode_long(entry, degrees);
    160 exif_encode_long(entry + 4, 1);
    161 exif_encode_long(entry + 8, minutes);
    162 exif_encode_long(entry + 12, 1);
    163 exif_encode_long(entry + 16, seconds_millionths);
    164 exif_encode_long(entry + 20, 1000000U);
    165 return 0;
    166}

    168/** @brief Convert a flight-controller heading to the EXIF angular range.
    169 * @param angle_bfp Angle in flight-controller fixed-point radians.
    170 * @return Degrees in [0, 360).
    171 * @details Wrap negative headings before encoding unsigned EXIF rationals. */
    172static double normalized_degrees(int32_t angle_bfp)
    173{
    174 double degrees = angle_bfp / ANGLE_BFP_SCALE * RAD_TO_DEG;
    175 degrees = fmod(degrees, 360.0);
    176 return degrees < 0.0 ? degrees + 360.0 : degrees;
    177}

    179/** @brief Form a nonnegative numerator for an EXIF rational.
    180 * @param value Quantity to encode.
    181 * @param scale Positive finite denominator scale.
    182 * @return Rounded numerator, saturated at UINT32_MAX; zero for nonpositive or nonfinite input.
    183 * @details Saturate before converting so an out-of-range value cannot wrap. */
    184static uint32_t scaled_unsigned(double value, double scale)
    185{
    186 if (!isfinite(value) || value <= 0.0) {
    187 return 0;
    188 }
    189 double scaled = value * scale;
    190 return scaled >= UINT32_MAX ? UINT32_MAX : (uint32_t)llround(scaled);
    191}

    193/** @brief Add GPS fields and the flight-pose description to an editable document.
    194 * @param[in,out] exif Document owned by the active workspace.
    195 * @param original Pose supplied by the caller; never modified.
    196 * @param delay Estimated capture delay in seconds.
    197 * @param compensate Nonzero to attempt ground-motion compensation.
    198 * @param timing Optional capture delivery evidence.
    199 * @return Zero on success, or -1 if a tag or the bounded description cannot be written.
    200 * @details Apply optional compensation to a local copy. Keep standard EXIF
    201 * fields for generic viewers and record fixed-point and physical units in
    202 * the description for consumers that need the full flight pose. */
    203static int add_shot_tags(struct exif_document *exif, const union dc_shot_union *original, double delay, int compensate,
    204 const struct capture_timing *timing)
    205{
    206 union dc_shot_union adjusted = *original;
    207 int32_t latitude = original->data.lat;
    208 int32_t longitude = original->data.lon;
    209 const int applied = compensate && compensate_ground_position(&latitude, &longitude,
    210 original->data.vground / SPEED_BFP_SCALE, original->data.course / ANGLE_BFP_SCALE, delay);
    211 adjusted.data.lat = latitude;
    212 adjusted.data.lon = longitude;
    213 const union dc_shot_union *shot = &adjusted;
    214 unsigned char *version = exif_document_reserve(exif, EXIF_GPS, EXIF_GPS_VERSION, EXIF_BYTE, 4);
    215 if (version == NULL) {
    216 return -1;
    217 }
    218 version[0] = 2;
    219 version[1] = 3;
    220
    221 if (set_coordinate(exif, EXIF_GPS_LATITUDE_REF, EXIF_GPS_LATITUDE,
    222 shot->data.lat, 900000000, 'N', 'S') != 0
    223 || set_coordinate(exif, EXIF_GPS_LONGITUDE_REF, EXIF_GPS_LONGITUDE,
    224 shot->data.lon, 1800000000, 'E', 'W') != 0) {
    225 return -1;
    226 }
    227
    228 unsigned char *altitude_reference = exif_document_reserve(exif, EXIF_GPS, EXIF_GPS_ALTITUDE_REF,
    229 EXIF_BYTE, 1);
    230 if (altitude_reference == NULL) {
    231 return -1;
    232 }
    233 altitude_reference[0] = shot->data.alt < 0 ? 1 : 0;
    234 uint64_t altitude_mm = shot->data.alt < 0 ? (uint64_t)(-(int64_t)shot->data.alt) : (uint64_t)shot->data.alt;
    235 if (altitude_mm > UINT32_MAX
    236 || exif_document_rational(exif, EXIF_GPS, EXIF_GPS_ALTITUDE, (uint32_t)altitude_mm, 1000) != 0) {
    237 return -1;
    238 }
    239
    240 double speed_m_s = shot->data.vground / SPEED_BFP_SCALE;
    241 double course_deg = normalized_degrees(shot->data.course);
    242 double yaw_deg = normalized_degrees(shot->data.psi);
    243 if (exif_document_ascii(exif, EXIF_GPS, EXIF_GPS_SPEED_REF, "K") != 0
    244 || exif_document_rational(exif, EXIF_GPS, EXIF_GPS_SPEED,
    245 scaled_unsigned(fabs(speed_m_s) * 3.6, 1000.0), 1000) != 0
    246 || exif_document_ascii(exif, EXIF_GPS, EXIF_GPS_TRACK_REF, "T") != 0
    247 || exif_document_rational(exif, EXIF_GPS, EXIF_GPS_TRACK,
    248 scaled_unsigned(course_deg, 10000.0), 10000) != 0
    249 || exif_document_ascii(exif, EXIF_GPS, EXIF_GPS_DIRECTION_REF, "T") != 0
    250 || exif_document_rational(exif, EXIF_GPS, EXIF_GPS_DIRECTION,
    251 scaled_unsigned(yaw_deg, 10000.0), 10000) != 0) {
    252 return -1;
    253 }
    254
    255 char *comment = (char *)active_workspace->scratch;
    256 int comment_length = snprintf(comment, SHOT_COMMENT_CAPACITY,
    257 "Paparazzi USHER MORA; nr=%" PRId32 "; lat_e7deg=%" PRId32
    258 "; lon_e7deg=%" PRId32 "; alt_mm=%" PRId32
    259 "; phi_bfp=%" PRId32 "; theta_bfp=%" PRId32
    260 "; psi_bfp=%" PRId32 "; vground_bfp=%" PRId32
    261 "; course_bfp=%" PRId32 "; groundalt_bfp=%" PRId32
    262 "; lat_deg=%.7f; lon_deg=%.7f; alt_m=%.3f"
    263 "; roll_deg=%.4f; pitch_deg=%.4f; yaw_deg=%.4f"
    264 "; speed_m_s=%.4f; course_deg=%.4f; groundalt_m=%.3f",
    265 shot->data.nr, shot->data.lat, shot->data.lon, shot->data.alt,
    266 shot->data.phi, shot->data.theta, shot->data.psi,
    267 shot->data.vground, shot->data.course, shot->data.groundalt,
    268 shot->data.lat / 1e7, shot->data.lon / 1e7, shot->data.alt / 1000.0,
    269 shot->data.phi / ANGLE_BFP_SCALE * RAD_TO_DEG,
    270 shot->data.theta / ANGLE_BFP_SCALE * RAD_TO_DEG,
    271 shot->data.psi / ANGLE_BFP_SCALE * RAD_TO_DEG,
    272 speed_m_s, course_deg, shot->data.groundalt / POSITION_BFP_SCALE);
    273 if (comment_length < 0 || (size_t)comment_length >= SHOT_COMMENT_CAPACITY) {
    274 return -1;
    275 }
    276 if (isfinite(delay) && delay >= 0) {
    277 int extra = snprintf(comment + comment_length, SHOT_COMMENT_CAPACITY - (size_t)comment_length,
    278 "; original_lat_deg=%.7f; original_lon_deg=%.7f; request_to_frame_arrival_s=%.9f"
    279 "; position_compensation=%s; timing_reference=%s"
    280 "; exposure_latency_known=false; attitude_compensated=false; altitude_compensated=false",
    281 original->data.lat / 1e7, original->data.lon / 1e7, delay,
    282 applied ? "constant_ground_velocity_estimate" : "not_applied",
    283 timing && capture_timing_valid(timing)
    284 && timing->callback_arrival ? "server_request_to_callback" : "server_request_to_uvc_return");
    285 if (extra < 0 || (size_t)extra >= SHOT_COMMENT_CAPACITY - (size_t)comment_length) return -1;
    286 comment_length += extra;
    287 }
    288 if (timing && capture_timing_valid(timing)) {
    289 char boot_id[37];
    290 usher_boot_id(boot_id);
    291 int extra = snprintf(comment + comment_length, SHOT_COMMENT_CAPACITY - (size_t)comment_length,
    292 "; mora_boot_id=%s; camera_request_monotonic_us=%" PRIu64
    293 "; frame_arrival_monotonic_us=%" PRIu64 "; capture_time_kind=%s",
    294 boot_id, timing->request_monotonic_us, timing->arrival_monotonic_us,
    295 timing->callback_arrival ? "sdk_callback_not_exposure" : "uvc_return_not_exposure");
    296 if (extra < 0 || (size_t)extra >= SHOT_COMMENT_CAPACITY - (size_t)comment_length) return -1;
    297 comment_length += extra;
    298 if (timing->callback_arrival) {
    299 extra = snprintf(comment + comment_length, SHOT_COMMENT_CAPACITY - (size_t)comment_length,
    300 "; callback_sequence=%" PRIu64 "; callback_drops=%" PRIu64,
    301 timing->callback_sequence, timing->callback_drops);
    302 if (extra < 0 || (size_t)extra >= SHOT_COMMENT_CAPACITY - (size_t)comment_length) return -1;
    303 }
    304 }
    305 if (exif_document_ascii(exif, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, comment) != 0
    306 || exif_document_ascii(exif, EXIF_PRIMARY, EXIF_SOFTWARE, "Paparazzi USHER") != 0) {
    307 return -1;
    308 }
    309 return 0;
    310}

    314/** @brief Write an EXIF UserComment composed of flight and analysis records.
    315 * @param[in,out] exif Document already carrying the flight description.
    316 * @param tag Optional section label as an empty or nonempty string, never NULL.
    317 * @param information Null-terminated analysis text, at most 8192 bytes.
    318 * @return Zero on success, or -1 on invalid input or insufficient storage.
    319 * @details Snapshot the description before reserving another entry so
    320 * rebuilding the tag storage cannot invalidate the bytes being copied. */
    321static int set_tagged_comment(struct exif_document *exif, const char *tag, const char *information)
    322{
    323 if (!exif || !tag || !information) return -1;
    324 const size_t information_length = strnlen(information, 8193);
    325 if (information_length > 8192) return -1;
    326 struct exif_value description;
    327 if (!exif_document_get(exif, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, &description)
    328 || description.format != EXIF_ASCII || description.size > COMPLETE_RECORD_CAPACITY) return -1;
    329 const size_t description_length = strnlen((const char *)description.data, description.size);
    330 unsigned char saved_description[COMPLETE_RECORD_CAPACITY];
    331 for (size_t index = 0; index < description_length; ++index) saved_description[index] = description.data[index];
    332 char separator[64];
    333 const int separator_length = tag[0] ? snprintf(separator, sizeof(separator), "\n%s\n", tag) : 0;
    334 if (separator_length < 0 || (size_t)separator_length >= sizeof(separator)) return -1;
    335 const size_t length = ASCII_COMMENT_PREFIX_SIZE + description_length + (size_t)separator_length
    336 + information_length + 1;
    337 unsigned char *entry = exif_document_reserve(exif, EXIF_CAMERA, EXIF_USER_COMMENT,
    338 EXIF_UNDEFINED, (uint32_t)length);
    339 if (!entry) return -1;
    340 const unsigned char prefix[8] = {'A', 'S', 'C', 'I', 'I', 0, 0, 0};
    341 size_t offset = 0;
    342 for (size_t index = 0; index < sizeof(prefix); ++index) entry[offset++] = prefix[index];
    343 for (size_t index = 0; index < description_length; ++index) entry[offset++] = saved_description[index];
    344 for (size_t index = 0; index < (size_t)separator_length; ++index) entry[offset++] = (unsigned char)separator[index];
    345 for (size_t index = 0; index < information_length; ++index) entry[offset++] = (unsigned char)information[index];
    346 return 0;
    347}

    349/** @brief Compatibility wrapper for analysis that becomes available after the flight EXIF write.
    350 * @details Prefer the combined writer when the result is already available, so
    351 * the original need not be streamed to a replacement file twice.
    352 */
    353static int write_tagged_comment(const char *filename, const char *tag, const char *information)
    354{
    355 if (!filename) return -1;
    356 struct exif_workspace *workspace = workspace_acquire();
    357 if (workspace == NULL) return -1;
    358 struct exif_document *exif = load_exif(filename);
    359 if (!exif) {
    360 workspace_release(workspace);
    361 return -1;
    362 }
    363 if (set_tagged_comment(exif, tag, information) != 0) {
    364 workspace_release(workspace);
    365 return -1;
    366 }
    367 int result = save_exif(filename, exif);
    368 workspace_release(workspace);
    369 return result;
    370}

    382int image_exif_write_mock_map(const char *filename, double horizontal_fov, double vertical_fov)
    383{
    384 if (filename == NULL || !isfinite(horizontal_fov) || !isfinite(vertical_fov)
    385 || horizontal_fov <= 0 || horizontal_fov >= 179 || vertical_fov <= 0 || vertical_fov >= 179) return -1;
    386 struct exif_workspace *workspace = workspace_acquire();
    387 if (workspace == NULL) return -1;
    388 struct exif_document *exif = load_exif(filename);
    389 if (exif == NULL) {
    390 workspace_release(workspace);
    391 return -1;
    392 }
    393 char record[256];
    394 int length = snprintf(record, sizeof(record),
    395 "{\"schema\":\"USHER_MOCK_MAP_V1\",\"tile_scheme\":\"XYZ\",\"zoom\":18,"
    396 "\"projection\":\"pinhole_flat_ground\",\"horizontal_fov_deg\":%.9g,"
    397 "\"vertical_fov_deg\":%.9g}", horizontal_fov, vertical_fov);
    398 if (length < 0 || (size_t)length >= sizeof(record)
    399 || exif_document_ascii(exif, EXIF_PRIMARY, EXIF_DOCUMENT_NAME, record) != 0) {
    400 workspace_release(workspace);
    401 return -1;
    402 }
    403 int result = save_exif(filename, exif);
    404 workspace_release(workspace);
    405 return result;
    406}

    425/** @brief Extend the flight description with observation provenance.
    426 * @param[in,out] exif Active document containing the existing description.
    427 * @param timing Optional timing, frame-identity and pose evidence.
    428 * @param aligned Whether the primary capture pose came from synchronized evidence.
    429 * @return Zero on success, or -1 on malformed metadata or insufficient storage.
    430 * @details Record frame identity separately from timing and pose alignment.
    431 * A bound frame may lack a valid exposure interval; navigation uncertainty
    432 * is recorded only when a pose is marked valid. */
    433static int add_observation_tags(struct exif_document *exif, const struct capture_timing *timing, bool aligned)
    434{
    435 const struct capture_observation empty = {0};
    436 const struct capture_observation *observation = timing ? &timing->observation : &empty;
    437 if (!capture_observation_valid(observation) && !observation->frame_bound) observation = &empty;
    438 struct exif_value description;
    439 if (!exif_document_get(exif, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, &description)
    440 || description.format != EXIF_ASCII || description.size > COMPLETE_RECORD_CAPACITY) return -1;
    441 char boot[37];
    442 usher_boot_id(boot);
    443 char *record = (char *)active_workspace->scratch;
    444 int length = snprintf(record, COMPLETE_RECORD_CAPACITY,
    445 "%.*s; altitude_datum=flight_controller_navigation; capture_pose_synchronized=%s"
    446 "; observation_time_kind=%s; observation_pose_aligned=%s; frame_identity_bound=%s"
    447 "; observation_earliest_monotonic_us=%" PRIu64 "; observation_latest_monotonic_us=%" PRIu64
    448 "; observation_sequence=%" PRIu64 "; sensor_timestamp_ns=%" PRIu64 "; observation_boot_id=%s",
    449 (int)description.size, description.data, aligned ? "true" : "false",
    450 capture_time_name(observation->kind), observation->pose_valid ? "true" : "false",
    451 observation->frame_bound ? "true" : "false",
    452 observation->earliest_us, observation->latest_us, observation->sequence, observation->sensor_timestamp_ns, boot);
    453 if (length < 0 || (size_t)length >= COMPLETE_RECORD_CAPACITY) return -1;
    454 if (observation->pose_valid) {
    455 const union dc_shot_union *pose = &observation->pose;
    456 if (!isfinite(observation->pose_uncertainty_m) || observation->pose_uncertainty_m < 0) return -1;
    457 int extra = snprintf(record + length, COMPLETE_RECORD_CAPACITY - (size_t)length,
    458 "; observation_lat_deg=%.7f; observation_lon_deg=%.7f; observation_alt_m=%.3f"
    459 "; observation_agl_m=%.3f; observation_roll_deg=%.4f; observation_pitch_deg=%.4f"
    460 "; observation_yaw_deg=%.4f; observation_pose_uncertainty_m=%.3f"
    461 "; navigation_position_method=%s; navigation_position_age_ms=%" PRIu32,
    462 pose->data.lat / 1e7, pose->data.lon / 1e7, pose->data.alt / 1000.0,
    463 pose->data.groundalt / POSITION_BFP_SCALE, pose->data.phi / ANGLE_BFP_SCALE * RAD_TO_DEG,
    464 pose->data.theta / ANGLE_BFP_SCALE * RAD_TO_DEG, pose->data.psi / ANGLE_BFP_SCALE * RAD_TO_DEG,
    465 observation->pose_uncertainty_m,
    466 observation->navigation_projected ? "gps_receipt_velocity_projection" : "fc_state_sample",
    467 observation->navigation_age_ms);
    468 if (extra < 0 || (size_t)extra >= COMPLETE_RECORD_CAPACITY - (size_t)length) return -1;
    469 }
    470 return exif_document_ascii(exif, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, record);
    471}

    473int image_exif_write_complete(const char *filename, const union dc_shot_union *shot,
    474 double capture_delay_s, int compensate, const struct capture_timing *timing,
    475 const union dc_shot_union *request, double uncertainty_m, const char *objects)
    476{
    477 if (filename == NULL || shot == NULL) {
    478 errno = EINVAL;
    479 return -1;
    480 }
    481
    482 struct exif_workspace *workspace = workspace_acquire();
    483 if (workspace == NULL) {
    484 return -1;
    485 }
    486 struct exif_document *exif = &workspace->document;
    487 size_t capture_size = 0;
    488 struct exif_view view;
    489 if (exif_read_jpeg(filename, workspace->payload, sizeof(workspace->payload), &capture_size) == 0
    490 && exif_parse(&view, workspace->payload, capture_size) == 0) {
    491 struct exif_value capture;
    492 for (size_t field = 0; field < sizeof(image_exif_capture_tags) / sizeof(image_exif_capture_tags[0]); ++field) {
    493 if (!exif_get(&view, EXIF_CAMERA, image_exif_capture_tags[field], &capture)) continue;
    494 unsigned char *stored = exif_document_reserve(exif, EXIF_CAMERA, image_exif_capture_tags[field],
    495 (enum exif_format)capture.format, capture.count);
    496 if (!stored) {
    497 workspace_release(workspace);
    498 return -1;
    499 }
    500 for (size_t index = 0; index < capture.size; ++index) stored[index] = capture.data[index];
    501 }
    502 }
    503 if (timing && capture_timing_valid(timing)) {
    504 capture_delay_s = (timing->arrival_monotonic_us - timing->request_monotonic_us) / 1e6;
    505 }
    506 if (add_shot_tags(exif, shot, capture_delay_s, request ? 0 : compensate, timing) != 0) {
    507 fprintf(stderr, "EXIF:\tfailed to create shot metadata for %s\n", filename);
    508 workspace_release(workspace);
    509 return -1;
    510 }
    511
    512 if (request) {
    513 /* Exposure-aligned coordinates improve the geotag, but the original FC
    514 * request remains essential when investigating latency or pose errors later. */
    515 struct exif_value description;
    516 char *record = (char *)active_workspace->scratch;
    517 if (!exif_document_get(exif, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, &description)
    518 || description.format != EXIF_ASCII || description.size > 1536
    519 || !isfinite(uncertainty_m) || uncertainty_m < 0) {
    520 workspace_release(workspace);
    521 return -1;
    522 }
    523 const int length = snprintf(record, COMPLETE_RECORD_CAPACITY,
    524 "%.*s; pose_reference=bounded_sensor_exposure_estimate; pose_uncertainty_m=%.3f"
    525 "; request_lat_e7deg=%" PRId32 "; request_lon_e7deg=%" PRId32 "; request_alt_mm=%" PRId32
    526 "; request_phi_bfp=%" PRId32 "; request_theta_bfp=%" PRId32 "; request_psi_bfp=%" PRId32
    527 "; request_groundalt_bfp=%" PRId32 "; request_vground_bfp=%" PRId32 "; request_course_bfp=%" PRId32,
    528 (int)description.size, description.data, uncertainty_m, request->data.lat, request->data.lon, request->data.alt,
    529 request->data.phi, request->data.theta, request->data.psi, request->data.groundalt, request->data.vground,
    530 request->data.course);
    531 if (length < 0 || (size_t)length >= COMPLETE_RECORD_CAPACITY
    532 || exif_document_ascii(exif, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, record) != 0) {
    533 workspace_release(workspace);
    534 return -1;
    535 }
    536 }
    537 if (add_observation_tags(exif, timing, request != NULL) != 0
    538 || set_tagged_comment(exif, objects ? "AICAM_OBJECTS_V1" : "", objects ? objects : "") != 0) {
    539 workspace_release(workspace);
    540 return -1;
    541 }
    542
    543 int result = save_exif(filename, exif);
    544 workspace_release(workspace);
    545 return result;
    546}

    548/** @brief Replace metadata while streaming the original compressed payload unchanged.
    549 * @param filename Existing image, exclusively owned by the caller during publication.
    550 * @param exif Metadata ownership is consumed on every exit.
    551 * @return Zero after replacement, -1 on failure with the original path retained.
    552 * @details The temporary sits beside the original so rename stays on one filesystem.
    553 * Flush and file sync happen before rename. This helper does not sync the parent
    554 * directory, so do not treat success as a complete power-loss durability guarantee.
    555 */
    556static int save_exif(const char *filename, struct exif_document *exif)
    557{
    558 if (active_workspace == NULL) {
    559 errno = EINVAL;
    560 return -1;
    561 }
    562 struct stat file_status, path_status;
    563 FILE *input = fopen(filename, IMAGE_EXIF_ATOMIC_REPLACE ? "rb" : "r+b");
    564 int source_result = input == NULL ? -1 : fstat(fileno(input), &file_status);
    565 if (source_result == 0) source_result = lstat(filename, &path_status);
    566 if (source_result == 0 && (!S_ISREG(file_status.st_mode) || !S_ISREG(path_status.st_mode)
    567 || file_status.st_dev != path_status.st_dev || file_status.st_ino != path_status.st_ino)) {
    568 errno = EINVAL;
    569 source_result = -1;
    570 }
    571 if (source_result != 0) {
    572 fprintf(stderr, "EXIF:\tfailed to open %s: %s\n", filename, strerror(errno));
    573 if (input != NULL) {
    574 const int saved_errno = errno;
    575 fclose(input);
    576 errno = saved_errno;
    577 }
    578 return -1;
    579 }
    580
    581 uint32_t width, height;
    582 unsigned components;
    583 int metadata_result = exif_jpeg_dimensions(input, &width, &height, &components);
    584 if (!metadata_result && components != 1 && components != 3) {
    585 errno = EINVAL;
    586 metadata_result = -1;
    587 }
    588 if (!metadata_result) {
    589 unsigned char *configuration = exif_document_reserve(exif, EXIF_CAMERA, EXIF_COMPONENTS_CONFIGURATION,
    590 EXIF_UNDEFINED, 4);
    591 if (!configuration) metadata_result = -1;
    592 else {
    593 configuration[0] = 1;
    594 if (components == 3) { configuration[1] = 2; configuration[2] = 3; }
    595 }
    596 unsigned char *dimension = exif_document_reserve(exif, EXIF_CAMERA, EXIF_PIXEL_X_DIMENSION, EXIF_LONG, 1);
    597 if (!dimension) metadata_result = -1;
    598 else exif_encode_long(dimension, width);
    599 dimension = exif_document_reserve(exif, EXIF_CAMERA, EXIF_PIXEL_Y_DIMENSION, EXIF_LONG, 1);
    600 if (!dimension) metadata_result = -1;
    601 else exif_encode_long(dimension, height);
    602 }
    603 unsigned char *exif_data = active_workspace->payload;
    604 size_t exif_size = 0;
    605 if (metadata_result != 0 || fseek(input, 0, SEEK_SET) != 0
    606 || exif_serialize(exif, exif_data, EXIF_MAX_PAYLOAD, &exif_size) != 0) {
    607 const int saved_errno = errno;
    608 fclose(input);
    609 errno = saved_errno;
    610 return -1;
    611 }
    612
    613#if IMAGE_EXIF_ATOMIC_REPLACE
    614 int name_length = snprintf(active_workspace->temporary_name, sizeof(active_workspace->temporary_name),
    615 "%s.exif.XXXXXX", filename);
    616 if (name_length < 0 || (size_t)name_length >= sizeof(active_workspace->temporary_name)) {
    617 fclose(input);
    618 errno = ENAMETOOLONG;
    619 return -1;
    620 }
    621 char *temporary_name = active_workspace->temporary_name;
    622 int temporary_fd = mkstemp(temporary_name);
    623 FILE *output = temporary_fd >= 0 ? fdopen(temporary_fd, "wb") : NULL;
    624 int result = 0;
    625 if (output == NULL || fchmod(temporary_fd, file_status.st_mode & 0777) != 0
    626 || exif_rewrite_jpeg(input, output, exif_data, exif_size) != 0
    627 || fflush(output) != 0 || fsync(temporary_fd) != 0) {
    628 result = -1;
    629 }
    630 if (fclose(input) != 0) {
    631 result = -1;
    632 }
    633 if (output != NULL && fclose(output) != 0) {
    634 result = -1;
    635 } else if (output == NULL && temporary_fd >= 0) {
    636 close(temporary_fd);
    637 }
    638
    639 if (result == 0) {
    640 if (lstat(filename, &path_status) != 0 || !S_ISREG(path_status.st_mode)
    641 || file_status.st_dev != path_status.st_dev || file_status.st_ino != path_status.st_ino
    642 || file_status.st_size != path_status.st_size
    643 || file_status.st_mtim.tv_sec != path_status.st_mtim.tv_sec
    644 || file_status.st_mtim.tv_nsec != path_status.st_mtim.tv_nsec) {
    645 errno = ESTALE;
    646 result = -1;
    647 } else if (rename(temporary_name, filename) != 0) {
    648 result = -1;
    649 }
    650 }
    651 if (result != 0) {
    652 int saved_errno = errno;
    653 unlink(temporary_name);
    654 fprintf(stderr, "EXIF:\tfailed to update %s: %s\n", filename, strerror(saved_errno));
    655 errno = saved_errno;
    656 }
    657
    658 return result;
    659#else
    660 int result = 0;
    661 if (lstat(filename, &path_status) != 0 || !S_ISREG(path_status.st_mode) ||
    662 file_status.st_dev != path_status.st_dev || file_status.st_ino != path_status.st_ino ||
    663 file_status.st_size != path_status.st_size ||
    664 file_status.st_mtim.tv_sec != path_status.st_mtim.tv_sec ||
    665 file_status.st_mtim.tv_nsec != path_status.st_mtim.tv_nsec) {
    666 errno = ESTALE;
    667 result = -1;
    668 } else if (exif_update_jpeg(input, exif_data, exif_size) != 0 || fsync(fileno(input)) != 0) {
    669 result = -1;
    670 } else if (lstat(filename, &path_status) != 0 || !S_ISREG(path_status.st_mode) ||
    671 file_status.st_dev != path_status.st_dev || file_status.st_ino != path_status.st_ino) {
    672 errno = ESTALE;
    673 result = -1;
    674 }
    675 int saved_errno = errno;
    676 if (fclose(input) != 0 && result == 0) {
    677 saved_errno = errno;
    678 result = -1;
    679 }
    680 if (result != 0) {
    681 fprintf(stderr, "EXIF:\tfailed to update %s in place: %s\n", filename, strerror(saved_errno));
    682 errno = saved_errno;
    683 }
    684 return result;
    685#endif
    686}
