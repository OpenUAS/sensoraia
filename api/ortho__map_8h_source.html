    19 * @file ortho_map.h
    20 * @brief Describe the validated photo set and render projected map windows.
    21 * @details Rendering retains source provenance and a per-pixel confidence class
    22 * alongside colour so publication can distinguish observed colour from
    23 * completion. The API keeps rendering separate from JPEG/XYZ publication,
    24 * allowing both output formats to share the same map engine.
    25 */

    43/**
    44 * @brief Compute a north-up, non-overlapping image grid for the requested bounds.
    45 * @param[in] config Validated geographic bounds, pixel size, and tile dimensions.

    48OrthoLayout orthoLayout(const MapConfig& config);
    49/**
    50 * @brief One fixed-session source image and its camera/pose evidence.
    51 * @details Coordinates are WGS-84 degrees; elevations are AMSL metres; camera
    52 * intrinsics are pixels; timestamps are 100 ns monotonic counts. The optional
    53 * capture cell lets Great use the planned flight grid without inventing pose data.
    54 */
    55struct OrthoSource {
    56 std::filesystem::path path;
    57 std::optional<CameraPlane> camera;
    58 double latitude = 0, longitude = 0, altitude = 0;
    59 double focal = 0, cx = 0, cy = 0;
    60 double focalY = 0, poseUncertainty = 0;
    61 double pitch = 0, bank = 0, heading = 0, fov = 0;
    62 long long poseTime = 0, frameTime = 0;
    63 int width = 0, height = 0;
    64 std::optional<UtcDateTime> captureUtc;
    65 double utcAbsoluteSeconds = 0;
    66 long long utcRead100ns = 0;
    67 double referencePlane = std::numeric_limits<double>::quiet_NaN();
    68 bool distorted = false;
    69 LensEffects lens;
    70 std::optional<std::pair<int, int>> captureCell;
    71};
    72/**
    73 * @brief Rendered BGRA map plus coverage, source list, and method provenance.
    74 * @details Confidence is independent of colour: zero means no supported colour,
    75 * and Great's reconstruction classes must not be reported as observed camera
    76 * geometry. `validFraction` is image colour coverage, not map accuracy.
    77 */
    78struct OrthoImage {
    79 MapTile map{};
    80 int width = 0, height = 0;
    81 std::vector<unsigned char> bgra, confidence;
    82 std::vector<OrthoSource> sources;
    83 double validFraction = 0;
    84 double geometryFraction = 0, corroboratedFraction = 0;
    85 std::string mode = "simple";
    86 double referencePlane = std::numeric_limits<double>::quiet_NaN();
    87 int seamScale = 0, blendBands = 0;
    88 std::string seamMethod;
    89 bool exposureCompensated = false;
    90 double reconstructedFraction = 0, reconstructionBlendFraction = 0;
    91};
    92/**
    93 * @brief Load a fixed-session source set and render its projected tiles/windows.
    94 * @details The implementation owns bounded decoded-photo and rendered-block
    95 * caches. Returned images own their pixel/provenance vectors; cancellation is
    96 * cooperative and throws before an incomplete render can be treated as complete.
    97 */
    98class Orthorectifier {
    99 struct Impl;

    154 * @param[in] progress Optional render fraction callback in `[0, 1]`.

    163 * @brief Rectify JPEG sources into working JPEGs and write a new CSV.

    165 * @param[in] directory New directory for normalized JPEGs and CSV.
    166 * @param[in] stopped Cooperative cancellation predicate.
    167 * @param[in] workers Upper bound on independent source-preparation workers.
    168 * @param[in] memoryMiB Working-memory budget used to reduce the worker count.
    169 * @return Path to the normalized source CSV.

    172 static std::filesystem::path normalizeSources(const std::filesystem::path& csv,

    174 unsigned workers = 1, unsigned memoryMiB = 128);
    175};
