    33{
    34 unsigned i;

    41

    66 /* between lb+al and ub-au transformation is the identity */
    67 t->al[i] = fmin((ub[i] - lb[i]) / 2., (1. + fabs(lb[i])) / 20.);

    78void cmaes_boundary_transformation(cmaes_boundary_transformation_t *t,
    79 double const *x, double *y, unsigned long len)
    80{
    81 double lb, ub, al, au;
    82 unsigned long i;
    83 cmaes_boundary_transformation_shift_into_feasible_preimage(t, x, y, len);
    84 for(i = 0; i < len; ++i) {
    85 lb = t->lower_bounds[_index(t, i)];
    86 ub = t->upper_bounds[_index(t, i)];
    87 al = t->al[_index(t, i)];
    88 au = t->au[_index(t, i)];
    89 if (y[i] < lb + al)
    90 y[i] = lb + (y[i] - (lb - al)) * (y[i] - (lb - al)) / 4. / al;
    91 else if (y[i] > ub - au)
    92 y[i] = ub - (y[i] - (ub + au)) * (y[i] - (ub + au)) / 4. / au;
    93 }
    94}
    95void cmaes_boundary_transformation_shift_into_feasible_preimage(
    96 cmaes_boundary_transformation_t *t, double const *x, double *y, unsigned long len)
    97{
    98 double lb, ub, al, au, r, xlow, xup;
    99 unsigned long i;
    100
    101 for(i = 0; i < len; ++i) {
    102 lb = t->lower_bounds[_index(t, i)];
    103 ub = t->upper_bounds[_index(t, i)];
    104 al = t->al[_index(t, i)];
    105 au = t->au[_index(t, i)];
    106 xlow = lb - 2 * al - (ub - lb) / 2.0;
    107 xup = ub + 2 * au + (ub - lb) / 2.0;
    108 r = 2 * (ub - lb + al + au); /* == xup - xlow == period of the transformation */
    109
    110 y[i] = x[i];
    111
    112 if (y[i] < xlow) { /* shift up */
    113 y[i] += r * (1 + (int)((xlow - y[i]) / r));
    114 }
    115 if (y[i] > xup) { /* shift down */
    116 y[i] -= r * (1 + (int)((y[i] - xup) / r));
    117 /* printf(" \n%f\n", fmod(y[i] - ub - au, r)); */
    118 }
    119 if (y[i] < lb - al) /* mirror */
    120 y[i] += 2 * (lb - al - y[i]);
    121 if (y[i] > ub + au)
    122 y[i] -= 2 * (y[i] - ub - au);
    123
    124 if ((y[i] < lb - al - 1e-15) || (y[i] > ub + au + 1e-15)) {
    125 printf("BUG in cmaes_boundary_transformation_shift_into_feasible_preimage: lb=%f, ub=%f, al=%f au=%f, y=%f\n",
    126 lb, ub, al, au, y[i]);
    127 _FatalError("BUG");
    128 }
    129 }

    131void cmaes_boundary_transformation_inverse(cmaes_boundary_transformation_t *t,
    132 double const *x, double *y, unsigned long len)
    133{
    134 double lb, ub, al, au;
    135 unsigned long i;
    136
    137 for (i = 0; i < len; ++i) {
    138 lb = t->lower_bounds[_index(t, i)];
    139 ub = t->upper_bounds[_index(t, i)];
    140 al = t->al[_index(t, i)];
    141 au = t->au[_index(t, i)];
    142 y[i] = x[i];
    143 if (y[i] < lb + al)
    144 y[i] = (lb - al) + 2 * sqrt(al * (y[i] - lb));
    145 else if (y[i] > ub - au)
    146 y[i] = (ub + au) - 2 * sqrt(au * (ub - y[i]));
    147 }
    148 if (11 < 3 || do_assertions) {
    149 double *z = (double *) calloc(len, sizeof(double));
    150 for (i = 0; i < len; ++i)
    151 z[i] = y[i];
    152 cmaes_boundary_transformation(t, z, y, len);
    153 for (i = 0; i < len; ++i)
    154 if (fabs(y[i] - x[i]) > 1e-14)
    155 printf(" difference for index %ld should be zero, is %f ", i, y[i] - x[i]);
    156 for (i = 0; i < len; ++i)
    157 y[i] = z[i];
    158 free(z);
    159 }
