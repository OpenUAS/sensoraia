    18/**
    19 * @file usher_protocol.h
    20 * @brief Shared FC/MORA byte-stream UART framing, messages and payload layouts.
    21 * @details This is the shared FC/MORA contract: commands and replies cross only the
    22 * UART datastream. Both peers compile these message IDs, payload sizes and layouts;
    23 * camera implementation, storage and capture scheduling are not part of the wire API.
    24 * Multi-byte fields use the shared native layout; both peers must be upgraded
    25 * together. The unions are serialization storage, not portable network byte
    26 * order. There is no protocol-wide version field; the separate MORA survey
    27 * extension declares its own version but is not negotiated by the FC.
    28 *
    29 * |STX|length|message ID|payload (length-5 bytes)|Checksum A|Checksum B|
    30 *
    31 * Length counts the entire frame. The checksum covers length, message ID and payload:
    32 * @code
    33 * usher_ck_a = usher_ck_b = length
    34 * for each byte b in message ID followed by payload
    35 * usher_ck_a += b;
    36 * usher_ck_b += usher_ck_a;
    37 * @endcode
    38 */

    245/** @brief Check the structural consistency of pose-sample GPS flags.
    246 * @param flags Received wire flags including the encoded age field.
    247 * @return True if all bits are known and the GPS presence and age dependencies hold.
    248 * @details Held GPS requires a present fix and valid age. Reject stray age
    249 * bits rather than treating them as qualified timing evidence. */
    250static inline bool usher_pose_flags_valid(uint32_t flags)
    251{
    252 const uint32_t known = USHER_POSE_SAMPLE_GPS_PRESENT | USHER_POSE_SAMPLE_GPS_HELD
    253 | USHER_POSE_SAMPLE_GPS_AGE_VALID | USHER_POSE_SAMPLE_GPS_AGE_MASK;
    254 return !(flags & ~known)
    255 && (!(flags & (USHER_POSE_SAMPLE_GPS_HELD | USHER_POSE_SAMPLE_GPS_AGE_VALID))
    256 || (flags & USHER_POSE_SAMPLE_GPS_PRESENT))
    257 && (!(flags & (USHER_POSE_SAMPLE_GPS_HELD | USHER_POSE_SAMPLE_GPS_AGE_MASK))
    258 || (flags & USHER_POSE_SAMPLE_GPS_AGE_VALID));
    259}

    378#define STX 0x99

    380#define UsherSizeOf(_payload) (_payload+5)

    382#define UsherPutUint8( _byte) { \
    383 usher_ck_a += _byte; \
    384 usher_ck_b += usher_ck_a; \
    385 CameraLinkTransmit(_byte); \
    386 }
    387
    388#define UsherHeader(msg_id, payload_len) { \
    389 CameraLinkTransmit(STX); \
    390 uint8_t msg_len = UsherSizeOf( payload_len); \
    391 CameraLinkTransmit(msg_len); \
    392 usher_ck_a = msg_len; usher_ck_b = msg_len; \
    393 UsherPutUint8(msg_id); \
    394 }
    395
    396#define UsherTrailer() { \
    397 CameraLinkTransmit(usher_ck_a); \
    398 CameraLinkTransmit(usher_ck_b); \
    399 }

    402 uint8_t _x = *(_byte); \
    403 UsherPutUint8( _x); \

    409/**
    410 * @brief Persistent state for incremental USHER UART frame parsing.
    411 * @details One instance must remain associated with one ordered byte stream. A completed
    412 * message remains in @c payload until its consumer clears @c msg_received; feeding later
    413 * bytes before that acknowledgement is counted as an error to avoid silently overwriting it.
    414 */
    415struct usher_transport {
    416 /** Storage for the validated payload of the most recently completed message. */
    417 uint8_t payload[256];
    418 /** Count of framing or checksum faults observed while consuming this stream. */
    419 uint8_t error;
    420 /** Message identifier from the current or most recently completed frame. */
    421 uint8_t msg_id;
    422 /** True when @c payload contains one complete validated message awaiting dispatch. */
    423 bool msg_received;
    424 /** Number of payload bytes expected in the current frame. */
    425 uint8_t payload_len;
    426 /** Internal parser state: the next expected position in the frame. */
    427 uint8_t status;
    428 /** Internal offset into @c payload while a frame is being assembled. */
    429 uint8_t payload_idx;
    430 /** Running Fletcher-style checksum accumulators for the in-progress frame. */
    431 uint8_t ck_a, ck_b;
    432};
