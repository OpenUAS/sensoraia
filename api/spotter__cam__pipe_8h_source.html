    21/**
    22 * @file spotter_cam_pipe.h
    23 * @brief Native USHER adapter for on-sensor detection or photo-only capture.
    24 * @details Uses the neutral native-worker supervisor,
    25 * then interprets only the matching detector sidecar. Capture success and usable
    26 * analysis are separate outcomes; the last result is retrieved separately for EXIF.
    27 * @pre Configuration, capture, binding and result retrieval are serialized by
    28 * the owning backend. This adapter has one parser workspace and one last result.
    29 */
    30

    76/** @brief Request one JPEG and, when enabled, parse its matching detection sidecar.
    77 * @param filename Output path buffer, cleared on failure.
    78 * @param filename_size Output path capacity.
    79 * @param image_number USHER shot sequence.
    80 * @return 0 only for a qualified saved image, otherwise -1.
    81 * @details Missing or malformed analysis leaves the photograph usable and makes
    82 * the summary report analysis_failed. Clearing the old result prevents a previous
    83 * hit being attached to the next exposure. */
    84int spotter_cam_pipe_shoot(char *filename, size_t filename_size, int image_number);

    87/** @brief Format the most recent shot's detection result for EXIF writing.
    88 * @param buffer Caller-owned destination for a short human-readable summary.
    89 * @param buffer_size Capacity of @p buffer.
    90 * @details Reflects the last spotter_cam_pipe_shoot() call only; safe to call even
    91 * when that call failed (reports status=analysis_failed) or no server ever ran. */
    92void spotter_cam_pipe_last_detection_summary(char *buffer, size_t buffer_size);
    93/** @brief Reject a detection result whose identity differs from the validated saved frame.
    94 * @param sequence Camera sequence obtained from that image's validated frame evidence.
    95 * @param sensor_timestamp_ns Matching sensor timestamp, not the FC shot time.
    96 * @return True for matching detector identity; false clears the retained analysis.
    97 * @details INI parsing alone does not bind detections to pixels. Compare both
    98 * fields after shared frame/file validation and before EXIF replacement changes the inode. */
    99bool spotter_cam_pipe_bind_frame(uint64_t sequence, uint64_t sensor_timestamp_ns);
    100/** @brief Copy @p filename into a "detections" subdirectory of the photo directory

    102 * @param filename Path to the just-finished image (ideally already EXIF-tagged by the
    103 * caller, so the copy carries the same metadata as the original).
    104 * @details No-op when the last shot found nothing, never completed, or @p filename is
    105 * NULL/empty; safe to call unconditionally, mirroring spotter_cam_pipe_last_detection_summary(). */
    106void spotter_cam_pipe_save_detection_copy(const char *filename);
    107/** @brief Select the detector assignment before either capture or saved-picture analysis starts.
