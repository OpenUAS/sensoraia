    55/** @brief Read cumulative CPU time across the mapping process's threads.
    56 * @return CPU seconds, independent of elapsed wall time.
    57 * @throws std::runtime_error If the process CPU clock cannot be read. */
    58static double processCpuSeconds() {
    59 struct timespec value{};
    60 if (clock_gettime(CLOCK_PROCESS_CPUTIME_ID, &value) != 0)
    61 throw std::runtime_error("Cannot measure mapping CPU time");
    62 return value.tv_sec + value.tv_nsec / 1e9;
    63}
    64/** @brief Check cancellation at a cooperative mapping boundary.
    65 * @return True when a stop signal has been recorded.
    66 * @details Wait out operator pauses and throttle against cumulative CPU time.
    67 * Poll in short sleeps so cancellation remains responsive while paused. */
    68static bool isStopping() {
    69 while (!stopping && !activePauseFile.empty() && fs::exists(activePauseFile))
    70 std::this_thread::sleep_for(std::chrono::milliseconds(250));
    71 if (liveCpuPercent) {
    72 const auto target = (processCpuSeconds() - liveCpuStarted) * 100 / liveCpuPercent;
    73 while (!stopping && std::chrono::duration<double>(Clock::now() - liveStarted).count() < target) {
    74 if (resources) resources->checkpoint();
    75 std::this_thread::sleep_for(std::chrono::milliseconds(10));
    76 }
    77 }
    78 // Check headroom at the same cooperative boundaries used for cancellation.
    79 if (resources && !stopping) resources->checkpoint();
    80 return stopping != 0;
    81}

    101/** @brief Create a readable, millisecond-resolution session label.
    102 * @return Local wall-clock label, not a uniqueness guarantee. */
    103static std::string sessionName() {
    104 auto now = std::chrono::system_clock::now();
    105 auto tt = std::chrono::system_clock::to_time_t(now);
    106 std::tm tm{};
    107 localtime_r(&tt, &tm);
    108 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()).count() % 1000;
    109 std::ostringstream text;
    110 text << std::setfill('0') << std::setw(4) << (tm.tm_year + 1900) << std::setw(2) << (tm.tm_mon + 1)
    111 << std::setw(2) << tm.tm_mday << '_' << std::setw(2) << tm.tm_hour
    112 << std::setw(2) << tm.tm_min << std::setw(2) << tm.tm_sec << '_'
    113 << std::setw(3) << ms << '_' << getpid();
    114 return text.str();
    115}

    123static void testOpenCvParallelBackend() {
    124 const int previousThreads = cv::getNumThreads();
    125 struct RestoreThreads {
    126 int count;
    127 ~RestoreThreads() { cv::setNumThreads(count); }
    128 } restore{previousThreads};
    129
    130 constexpr int totalItems = 513;
    131 for (const int team : {4, 3}) {
    132 cv::setNumThreads(team);
    133 require(cv::getNumThreads() == team, "OpenCV parallel backend did not apply thread count");
    134 std::vector<std::atomic<int>> hits(totalItems);
    135 for (auto& hit : hits) hit.store(0, std::memory_order_relaxed);
    136 std::atomic<bool> validThreadIds{true};
    137
    138 cv::parallel_for_(cv::Range(0, totalItems), [&](const cv::Range& range) {
    139 const int tid = cv::getThreadNum();
    140 if (tid < 0 || tid >= team) validThreadIds.store(false, std::memory_order_relaxed);
    141 int nestedCount = 0;
    142 cv::parallel_for_(cv::Range(0, 4), [&](const cv::Range& inner) {
    143 nestedCount += inner.size();
    144 });
    145 if (nestedCount != 4) validThreadIds.store(false, std::memory_order_relaxed);
    146 for (int index = range.start; index < range.end; ++index)
    147 hits[static_cast<std::size_t>(index)].fetch_add(1, std::memory_order_relaxed);
    148 });
    149
    150 require(validThreadIds.load(std::memory_order_relaxed), "OpenCV parallel backend reported invalid thread index");
    151 for (const auto& hit : hits)
    152 require(hit.load(std::memory_order_relaxed) == 1, "OpenCV parallel backend missed or duplicated stripe work");
    153 if (team == 4) {
    154 cv::setNumThreads(1);
    155 require(cv::getNumThreads() == 1, "OpenCV parallel backend did not shrink to single thread");
    156 }
    157 }
    158
    159 bool caught = false;
    160 try {
    161 cv::parallel_for_(cv::Range(0, 64), [&](const cv::Range& range) {
    162 if (range.start <= 17 && 17 < range.end)
    163 throw std::runtime_error("synthetic parallel_for failure");
    164 });
    165 } catch (const std::exception& error) {
    166 caught = std::string(error.what()).find("synthetic parallel_for failure") != std::string::npos;
    167 }
    168 require(caught, "OpenCV parallel backend lost body exception");
    169
    170 cv::setNumThreads(1);
    171 require(cv::getNumThreads() == 1, "OpenCV parallel backend did not shrink to single thread");
    172 std::atomic<bool> serialOnCaller{true};
    173 cv::parallel_for_(cv::Range(0, 32), [&](const cv::Range&) {
    174 if (cv::getThreadNum() != 0) serialOnCaller.store(false, std::memory_order_relaxed);
    175 });
    176 require(serialOnCaller.load(std::memory_order_relaxed), "Single-thread OpenCV region used worker thread");

    220 " pangea --self-test\n"
    221#endif
    222 " pangea --version\n";
    223 return 0;
    224 }
    225 if (argc == 2 && std::strcmp(argv[1], "--self-test") == 0) {
    226#if SENSORAIA_BUILD_TESTING

    228 testSurveyTiming();
    229 pangea_survey::MapProgress::selfTest();
    230 pangea_survey::Orthorectifier::selfTest();
    231 testXyzGrid();
    232 testMapExif();
    233 testSplitPipeline();
    234 return 0;
    235#else
    236 throw std::runtime_error("Self-tests require the private development checkout and BUILD_TESTING=ON");
    237#endif
    238 }
    239 const std::string lockPath = "/tmp/pangea-survey-" + std::to_string(getuid()) + ".lock";
    240 if (argc == 2 && std::strcmp(argv[1], "--stop") == 0) {
    241 std::ifstream lockFile(lockPath);
    242 pid_t pid = 0;
    243 if (lockFile >> pid && pid > 1) {
    244 const auto process = fs::path("/proc") / std::to_string(pid);
    245 std::ifstream arguments(process / "cmdline", std::ios::binary);
    246 std::string program, command;
    247 std::getline(arguments, program, '\0'); std::getline(arguments, command, '\0');
    248 std::error_code error;
    249 const auto executable = fs::canonical(process / "exe", error);
    250 if (!error && executable == executablePath() &&
    251 (command == "--run" || command == "--diagnostic" || command == "--update-great" ||
    252 command == "--follow-great" || command == "--finalize-great") &&
    253 kill(pid, SIGTERM) == 0) {
    254 std::cout << "Stopped Pangea survey PID " << pid << ".\n";
    255 return 0;
    256 }
    257 }
    258 std::cerr << "No Pangea survey is running.\n";
    259 return 1;
    260 }
    261 if (argc == 3 && (std::strcmp(argv[1], "--update-great") == 0 ||
    262 std::strcmp(argv[1], "--follow-great") == 0 ||
    263 std::strcmp(argv[1], "--finalize-great") == 0)) {
    264 const auto ini = fs::path(argv[2]);
    265 const auto config = pangea_survey::loadMapConfig(ini);
    266 pangea_survey::GreatSessionLease lease;
    267 resources = std::make_unique<pangea_survey::MapResources>(config);
    268 signal(SIGINT, signalHandler);
    269 signal(SIGTERM, signalHandler);
    270 activePauseFile = config.livePauseFile;
    271 liveCpuPercent = config.liveCpuPercent;
    272 liveCpuStarted = processCpuSeconds();
    273 liveStarted = Clock::now();
    274 const bool follow = std::strcmp(argv[1], "--follow-great") == 0;
    275 const bool finalize = std::strcmp(argv[1], "--finalize-great") == 0;
    276 auto updated = Clock::now() - std::chrono::seconds(config.liveUpdateSeconds);
    277 std::string previous;
    278 std::string prepared;
    279 bool pressureReported = false;
    280 do {
    281 try {
    282 if (isStopping()) break;
    283 const bool final = finalize || (!config.liveFinalFile.empty() && fs::exists(config.liveFinalFile));
    284 auto snapshot = pangea_survey::greatInputText(config);
    285 const auto end = snapshot.rfind('\n');
    286 if (end != std::string::npos) snapshot.resize(end + 1);
    287 const auto added = std::count(snapshot.begin(), snapshot.end(), '\n') -
    288 std::count(previous.begin(), previous.end(), '\n');

    291 if (!follow || final || (snapshot != previous &&

    293 if (pangea_survey::updateGreatSession(config, ini, final)) {
