    18/** @file
    19 * @brief Serial saved-image analysis adapter, not part of the camera driver.
    20 *
    21 * @details Built only when `LWIR_ENABLE_ANALYSIS=1`. Works on an already saved
    22 * LWIR JPEG: extracts the embedded exact Kelvin*64 temperature plane, detects
    23 * hotspots, reads the observation pose from EXIF ImageDescription, optionally
    24 * projects each hotspot to the ground, and writes a text record into EXIF
    25 * UserComment. The entropy-coded display image is not decoded or recompressed.
    26 *
    27 * Memory: large buffers are file-scope statics (no heap, no large stack frames
    28 * apart from the 8 KiB record and EXIF payload copy in read_pose()), which is
    29 * why the adapter is strictly serial.
    30 */

    61/**
    62 * @brief Decode one unsigned 16-bit little-endian value.
    63 * @details Byte-wise assembly is independent of host endianness and alignment,
    64 * so the plane can be read straight from an arbitrary byte buffer.
    65 * @param bytes At least two readable bytes.
    66 * @return The decoded value.
    67 */
    68static uint16_t read_u16_le(const uint8_t *bytes)
    69{
    70 return (uint16_t)bytes[0] | (uint16_t)((uint16_t)bytes[1] << 8);
    71}

    73/**
    74 * @brief Raw Kelvin*64 code of one pixel in the static temperature plane.
    75 * @param pixel Row-major native index (y * width + x); caller bounds-checks it.
    76 * @return Raw code; 0 and 65535 are sensor endpoint codes, not real temperatures.
    77 * @note Celsius is code / 64 - 273.15.
    78 */
    79static uint16_t temperature_at(size_t pixel)
    80{
    81 return read_u16_le(temperature_plane + pixel * 2);
    82}

    84/**
    85 * @brief Intermediate pose values parsed from the ImageDescription key=value list.
    86 * @details Angles are still in degrees here; read_pose() converts them.
    87 * `agl` holds height above ground from `altitude_agl_m` or `groundalt_m`.
    88 * `seen` is a bitmask indexed like the key table in read_pose(), used both for
    89 * duplicate-key rejection and required-key checks. `uncertainty` stays -1 when
    90 * absent, which later disqualifies `synchronized`.
    91 */
    92struct pose_fields {
    93 double latitude, longitude, altitude, agl, roll, pitch, yaw, uncertainty;
    94 unsigned seen;
    95 bool synchronized, exposure_interval;
    96};

    98/**
    99 * @brief Parse a finite decimal number from a non-NUL-terminated slice.
    100 * @details Trims spaces/tabs, copies the slice into a bounded local buffer so
    101 * strtod() has a terminator (EXIF text is not guaranteed to be NUL-terminated at
    102 * the slice end), and requires strtod() to consume every remaining character.
    103 * @param value Start of the value slice (may be unterminated).
    104 * @param length Slice length in bytes.
    105 * @param[out] number Written only on success.
    106 * @return True for a complete finite number; false for empty, overlong (>= 96
    107 * bytes after trimming), partial, out-of-range (ERANGE), NaN or infinite input.
    108 * @note Locale-dependent strtod(); presumably the process uses the "C" locale.
    109 */
    110static bool parse_description_number(const char *value, size_t length, double *number)
    111{
    112 char text[96];
    113 while (length && (*value == ' ' || *value == '\t')) { ++value; --length; }
    114 while (length && (value[length - 1] == ' ' || value[length - 1] == '\t')) --length;
    115 if (!length || length >= sizeof(text)) return false;
    116 for (size_t index = 0; index < length; ++index) text[index] = value[index];
    117 text[length] = '\0';
    118 errno = 0;
    119 char *end = NULL;
    120 const double parsed = strtod(text, &end);
    121 if (errno || end == text || *end || !isfinite(parsed)) return false;
    122 *number = parsed;
    123 return true;
    124}

    126/**
    127 * @brief Read the observation pose recorded in a saved image's EXIF ImageDescription.
    128 * @details The description is a `key=value; key=value` list written at capture
    129 * time. Segments end at ';' or NUL; segments without '=' and unknown keys are
    130 * ignored, but a duplicated known key or an unparsable number rejects the whole
    131 * pose rather than guessing which value is right.
    132 *
    133 * Required: `lat_deg`, `lon_deg`, `alt_m` (WGS84 ellipsoidal height, as sent by
    134 * the flight controller), `roll_deg`, `pitch_deg`, `yaw_deg`, and either
    135 * `altitude_agl_m` or `groundalt_m`. Both carry height above ground: USHER's
    136 * `groundalt_m` is the flight controller's `groundalt` field, which Paparazzi
    137 * fills with altitude above ground, not the ground's own altitude. When both are
    138 * present `altitude_agl_m` wins.
    139 *
    140 * Timing qualification: `synchronized` is true only when
    141 * `capture_pose_synchronized=true`, `observation_time_kind` names an
    142 * exposure-bounding interval, and a non-negative `pose_uncertainty_m` is present.
    143 * Any other time kind (for example a plain arrival time) leaves it false.
    144 * @param filename Saved JPEG to read; opened independently of the publisher.
    145 * @param[out] pose Written only on success; angles converted to radians.
    146 * @return True when a complete pose was parsed; false when EXIF or the
    147 * description is absent/malformed or required keys are missing. Callers treat
    148 * false as "missing pose", never as a zero pose.
    149 * @note Uses a stack copy of the EXIF payload (EXIF_MAX_PAYLOAD bytes).
    150 */
    151static bool read_pose(const char *filename, struct lwir_camera_pose *pose)
    152{
    153 unsigned char payload[EXIF_MAX_PAYLOAD];
    154 size_t payload_size = 0;
    155 struct exif_view view;
    156 struct exif_value description;
    157 if (exif_read_jpeg(filename, payload, sizeof(payload), &payload_size) != 0
    158 || exif_parse(&view, payload, payload_size) != 0
    159 || !exif_get(&view, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, &description)
    160 || description.format != EXIF_ASCII || description.size > 8192) return false;
    161 struct pose_fields fields = {.uncertainty = -1};
    162 size_t offset = 0;
    163 while (offset < description.size) {
    164 const size_t start = offset;
    165 while (offset < description.size && description.data[offset] != ';' && description.data[offset] != 0) ++offset;
    166 const size_t end = offset;
    167 if (offset < description.size) {
    168 if (description.data[offset] == ';') ++offset;
    169 else offset = description.size;
    170 }
    171 size_t separator = start;
    172 while (separator < end && description.data[separator] != '=') ++separator;
    173 if (separator == end) continue;
    174 size_t key_start = start, key_end = separator;
    175 while (key_start < key_end && (description.data[key_start] == ' ' || description.data[key_start] == '\t')) ++key_start;
    176 while (key_end > key_start && (description.data[key_end - 1] == ' ' || description.data[key_end - 1] == '\t')) --key_end;
    177 const char *key = (const char *)description.data + key_start;
    178 const size_t key_length = key_end - key_start;
    179 const char *value = (const char *)description.data + separator + 1;
    180 size_t value_length = end - separator - 1;
    181 while (value_length && (*value == ' ' || *value == '\t')) { ++value; --value_length; }
    182 while (value_length && (value[value_length - 1] == ' ' || value[value_length - 1] == '\t')) --value_length;
    183 static const char *const names[] = {"lat_deg", "lon_deg", "alt_m", "roll_deg", "pitch_deg", "yaw_deg",
    184 "altitude_agl_m", "groundalt_m", "pose_uncertainty_m", "capture_pose_synchronized", "observation_time_kind"};
    185 unsigned field = 0;
    186 while (field < sizeof(names) / sizeof(names[0])
    187 && (strlen(names[field]) != key_length || memcmp(key, names[field], key_length))) ++field;
    188 if (field == sizeof(names) / sizeof(names[0])) continue;
    189 const unsigned field_bit = 1U << field;
    190 if (fields.seen & field_bit) return false;
    191 fields.seen |= field_bit;
    192 if (field == 9) {
    193 while (value_length && (*value == ' ' || *value == '\t')) { ++value; --value_length; }
    194 fields.synchronized = value_length == 4 && !memcmp(value, "true", 4);
    195 if (!fields.synchronized && !(value_length == 5 && !memcmp(value, "false", 5))) return false;
    196 continue;
    197 }
    198 if (field == 10) {
    199 /* Command-to-download presumably qualifies because it brackets the
    200 * exposure, unlike an arrival time that only follows it. */
    201 fields.exposure_interval = (value_length == 24 && !memcmp(value, "sensor_exposure_interval", 24))
    202 || (value_length == 28 && !memcmp(value, "calibrated_exposure_interval", 28))
    203 || (value_length == sizeof("command_to_download_interval") - 1 &&
    204 !memcmp(value, "command_to_download_interval", sizeof("command_to_download_interval") - 1));
    205 continue;
    206 }
    207 double number;
    208 if (!parse_description_number(value, value_length, &number)) return false;
    209 switch (field) {
    210 case 0: fields.latitude = number; break;
    211 case 1: fields.longitude = number; break;
    212 case 2: fields.altitude = number; break;
    213 case 3: fields.roll = number; break;
    214 case 4: fields.pitch = number; break;
    215 case 5: fields.yaw = number; break;
    216 case 6: fields.agl = number; break;
    217 /* groundalt_m is a fallback for the same quantity; never overwrite an explicit AGL. */
    218 case 7: if (!(fields.seen & (1U << 6))) fields.agl = number; break;
    219 case 8: fields.uncertainty = number; break;
    220 default: return false;
    221 }
    222 }
    223 const unsigned required = (1U << 0) | (1U << 1) | (1U << 2) | (1U << 3) | (1U << 4) | (1U << 5);
    224 if ((fields.seen & required) != required || !(fields.seen & ((1U << 6) | (1U << 7)))) return false;
    225 *pose = (struct lwir_camera_pose){
    226 .latitude_deg = fields.latitude, .longitude_deg = fields.longitude,
    227 .altitude_ellipsoid_m = fields.altitude, .agl_m = fields.agl,
    228 .roll_rad = fields.roll * (3.14159265358979323846 / 180),
    229 .pitch_rad = fields.pitch * (3.14159265358979323846 / 180),
    230 .yaw_rad = fields.yaw * (3.14159265358979323846 / 180),
    231 .uncertainty_m = fields.uncertainty,
    232 .synchronized = fields.synchronized && fields.exposure_interval && fields.uncertainty >= 0};
    233 return true;
    234}

    236/**
    237 * @brief Append printf-formatted text to a fixed record buffer.
    238 * @details Fails instead of truncating, so a published record is never silently
    239 * cut in the middle of a key/value pair.
    240 * @param record Destination buffer, NUL-terminated after each successful append.
    241 * @param capacity Total size of @p record in bytes.
    242 * @param[in,out] used Bytes already used (excluding the terminator); advanced on success.
    243 * @param format printf format string.
    244 * @param ... Format arguments.
    245 * @return True on success; false on encoding error or insufficient space.
    246 */
    247static bool append_record(char *record, size_t capacity, size_t *used, const char *format, ...)
    248{
    249 if (*used >= capacity) return false;
    250 va_list arguments;
    251 va_start(arguments, format);
    252 const int count = vsnprintf(record + *used, capacity - *used, format, arguments);
    253 va_end(arguments);
    254 if (count < 0 || (size_t)count >= capacity - *used) return false;
    255 *used += (size_t)count;
    256 return true;
    257}

    259/**
    260 * @brief Store the analysis record in the in-memory EXIF document's UserComment.
    261 * @details Layout: the 8-byte EXIF character-code prefix `ASCII\0\0\0`, the
    262 * original ImageDescription text (so pose evidence is visible in one field), a
    263 * `\nLWIR_HOTSPOTS_V1\n` version separator, the record, and a terminating NUL.
    264 * The description is copied out first because exif_document_reserve() can move
    265 * existing values that `description.data` borrows.
    266 * @param information NUL-terminated record, at most 8192 characters.
    267 * @return True when the tag was reserved and filled; false for an oversized or
    268 * non-ASCII-format description, an oversized record, or exhausted document
    269 * capacity (errno ENOSPC/EINVAL from exif_document_reserve()).
    270 * @note exif_document_reserve() replaces an existing UserComment, so re-running
    271 * analysis overwrites rather than stacks records.
    272 */
    273static bool set_hotspot_comment(const char *information)
    274{
    275 struct exif_value description = {0};
    276 size_t description_length = 0;
    277 if (exif_document_get(&geolocation_exif_document, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, &description)) {
    278 if (description.format != EXIF_ASCII || description.size > sizeof(geolocation_description)) return false;
    279 description_length = exif_text_length(&description);
    280 if (description_length > sizeof(geolocation_description)) return false;
    281 for (size_t index = 0; index < description_length; ++index)
    282 geolocation_description[index] = description.data[index];
    283 }
    284 const size_t information_length = strnlen(information, 8193);
    285 if (information_length > 8192) return false;
    286 static const unsigned char prefix[8] = {'A', 'S', 'C', 'I', 'I', 0, 0, 0};
    287 const size_t tag_length = sizeof("\nLWIR_HOTSPOTS_V1\n") - 1;
    288 const size_t total = sizeof(prefix) + description_length + tag_length + information_length + 1;
    289 if (total > UINT32_MAX) return false;
    290 unsigned char *comment = exif_document_reserve(&geolocation_exif_document, EXIF_CAMERA,
    291 EXIF_USER_COMMENT, EXIF_UNDEFINED, (uint32_t)total);
    292 if (!comment) return false;
    293 size_t offset = 0;
    294 for (size_t index = 0; index < sizeof(prefix); ++index) comment[offset++] = prefix[index];
    295 for (size_t index = 0; index < description_length; ++index) comment[offset++] = geolocation_description[index];
    296 static const char separator[] = "\nLWIR_HOTSPOTS_V1\n";
    297 for (size_t index = 0; index < tag_length; ++index) comment[offset++] = (unsigned char)separator[index];
    298 for (size_t index = 0; index < information_length; ++index) comment[offset++] = (unsigned char)information[index];
    299 comment[offset] = 0;
    300 return true;
    301}

    303/**
    304 * @brief Rewrite a JPEG's EXIF with the analysis record, keeping image data intact.
    305 * @details Opens with O_NOFOLLOW and requires a regular file, so a symlink cannot
    306 * redirect the write. Existing EXIF is imported (ENODATA means none, which is
    307 * acceptable); other read/parse failures abort rather than discard metadata.
    308 *
    309 * With IMAGE_EXIF_ATOMIC_REPLACE the JPEG is streamed to a sibling mkstemp()
    310 * file (same directory, so rename() stays atomic on one filesystem) with the
    311 * original permission bits, fsync()ed, and renamed over the original only if the
    312 * path still names the same inode (dev/ino), so a concurrently replaced file is
    313 * not clobbered. Otherwise the EXIF segment is updated in place.
    314 * @param filename Target JPEG path.
    315 * @param information NUL-terminated analysis record.
    316 * @return 0 on success, -1 on any open, stat, EXIF, write, sync or rename
    317 * failure (errno may reflect the failing call; not guaranteed).
    318 * @note The containing directory is not fsync()ed after rename(), so the rename
    319 * itself may not survive a sudden power loss.
    320 * @warning In-place mode cannot roll back a partially written update.
    321 */
    322static int publish_hotspot_comment(const char *filename, const char *information)
    323{
    324 if (!filename || !information) return -1;
    325 const int input_fd = open(filename, O_RDWR | O_CLOEXEC | O_NOFOLLOW);
    326 if (input_fd < 0) return -1;
    327 struct stat original;
    328 if (fstat(input_fd, &original) != 0 || !S_ISREG(original.st_mode)) { close(input_fd); return -1; }
    329 FILE *input = fdopen(input_fd, "r+b");
    330 if (!input) { close(input_fd); return -1; }
    331 size_t payload_size = 0;
    332 struct exif_view view;
    333 exif_document_init(&geolocation_exif_document);
    334 bool success = fseeko(input, 0, SEEK_SET) == 0;
    335 if (success) {
    336 const int read_result = exif_read_stream(input, geolocation_exif_payload,
    337 sizeof(geolocation_exif_payload), &payload_size);
    338 if (read_result == 0) {
    339 success = exif_parse(&view, geolocation_exif_payload, payload_size) == 0
    340 && image_exif_import(&geolocation_exif_document, &view) == 0;
    341 } else if (errno != ENODATA) {
    342 success = false;
    343 }
    344 }
    345 success = success && set_hotspot_comment(information)
    346 && exif_serialize(&geolocation_exif_document, geolocation_exif_payload,
    347 sizeof(geolocation_exif_payload), &payload_size) == 0;
    348#if IMAGE_EXIF_ATOMIC_REPLACE
    349 char temporary[PATH_MAX];
    350 int temporary_length = snprintf(temporary, sizeof(temporary), "%s.lwir.XXXXXX", filename);
    351 int output_fd = -1;
    352 bool temporary_created = false;
    353 FILE *output = NULL;
    354 if (success && (temporary_length < 0 || (size_t)temporary_length >= sizeof(temporary))) success = false;
    355 if (success) output_fd = mkstemp(temporary);
    356 if (success && output_fd < 0) success = false;
    357 if (output_fd >= 0) temporary_created = true;
    358 if (success && fchmod(output_fd, original.st_mode & 0777) != 0) success = false;
    359 if (success) output = fdopen(output_fd, "wb");
    360 if (success && !output) success = false;
    361 /* The FILE now owns the descriptor; avoid a double close below. */
    362 if (output) output_fd = -1;
    363 if (success) success = fseeko(input, 0, SEEK_SET) == 0
    364 && exif_rewrite_jpeg(input, output, geolocation_exif_payload, payload_size) == 0
    365 && fflush(output) == 0 && fsync(fileno(output)) == 0;
    366 if (output && fclose(output) != 0) success = false;
    367 if (output_fd >= 0) close(output_fd);
    368 struct stat current;
    369 if (success) success = lstat(filename, &current) == 0
    370 && current.st_dev == original.st_dev && current.st_ino == original.st_ino;
    371 if (success) success = rename(temporary, filename) == 0;
    372 if (!success && temporary_created) unlink(temporary);
    373#else
    374 success = success && fseeko(input, 0, SEEK_SET) == 0
    375 && exif_update_jpeg(input, geolocation_exif_payload, payload_size) == 0
    376 && fsync(fileno(input)) == 0;
    377#endif
    378 if (fclose(input) != 0) success = false;
    379 return success ? 0 : -1;
    380}

    382/**
    383 * @brief Detect, optionally geolocate and record hotspots for one saved JPEG.
    384 * @details See lwir_analysis.h for the contract. Order of evaluation:
    385 * -# Read the embedded temperature plane: invalid -> `status=invalid_native_frame`;
    386 * absent -> `status=no_temperature_plane` (both published, return value from publishing).
    387 * -# Read pose and calibration (each may be missing without aborting).
    388 * -# Detect hotspots, then derive a single status with precedence
    389 * missing pose > missing calibration > invalid calibration > unverified
    390 * calibration, finally overridden by `no_hotspots`.
    391 * -# For each hotspot, report the rounded pixel, its nearest-pixel temperature
    392 * and the subpixel projection input; project only when status is `estimated`.
    393 * @return 0 when a record was published, -1 otherwise.
    394 */
    395int lwir_geolocate_image(const char *filename, const char *calibration_path)
    396{
    397 if (!filename || !*filename) return -1;
    398 unsigned width = 0, height = 0;
    399 bool synthetic = false;
    400 const int layer = lwir_read_temperature_layer(filename, temperature_plane, sizeof(temperature_plane),
    401 &width, &height, &synthetic);
    402 char record[8192];
    403 size_t used = 0;
    404 if (layer < 0)
    405 return publish_hotspot_comment(filename, "status=invalid_native_frame; count=0; coordinates_omitted=true");
    406 /* Header states the model limits next to every result: flat ground, zero GPS
    407 * antenna offset, apparent (uncorrected) temperatures, unknown accuracy. */
    408 if (!append_record(record, sizeof(record), &used,
    409 "coordinate_datum=WGS84; ground_model=local_horizontal_plane; gps_camera_offset_m=0; "
