    290double *
    291cmaes_init_final(cmaes_t *t /* "this" */)
    292/*
    293 * */
    294{
    295 int i, j, N;
    296 double dtest, trace;
    297
    298 if (t->version == NULL) {
    299 ERRORMESSAGE("cmaes_init_final called (probably) without calling cmaes_init_para first",
    300 ", which will likely lead to unexpected results",0,0);
    301 printf("Error: cmaes_init_final called (probably) without calling cmaes_init_para first\n");
    302 }
    303 if (strcmp(c_cmaes_version, t->version) != 0) {
    304 ERRORMESSAGE("cmaes_init_final called twice, which will lead to a memory leak",
    305 "; use cmaes_exit() first",0,0);
    306 printf("Error: cmaes_init_final called twice, which will lead to a memory leak; use cmaes_exit first\n");
    307 }
    308 /* assign_string(&t->signalsFilename, "cmaes_signals.par"); */
    309
    310 if (!t->sp.flgsupplemented) {
    311 cmaes_readpara_SupplementDefaults(&t->sp);
    312 if (!isNoneStr(t->sp.filename)) /* TODO: should this be done in readpara_SupplementDefaults? */
    313 cmaes_readpara_WriteToFile(&t->sp, "actparcmaes.par");
    314 }
    315
    316 t->sp.seed = cmaes_random_init( &t->rand, (long unsigned int) t->sp.seed);
    317
    318 N = t->sp.N; /* for convenience */
    319
    320 /* initialization */
    321 for (i = 0, trace = 0.; i < N; ++i)
    322 trace += t->sp.rgInitialStds[i]*t->sp.rgInitialStds[i];
    323 t->sigma = sqrt(trace/N); /* t->sp.mueff/(0.2*t->sp.mueff+sqrt(N)) * sqrt(trace/N); */
    324
    325 t->chiN = sqrt((double) N) * (1. - 1./(4.*N) + 1./(21.*N*N));
    326 t->flgEigensysIsUptodate = 1;
    327 t->flgCheckEigen = 0;
    328 t->genOfEigensysUpdate = 0;
    329 cmaes_timings_init(&t->eigenTimings);
    330 t->flgIniphase = 0; /* do not use iniphase, hsig does the job now */
    331 t->flgresumedone = 0;
    332 t->flgStop = 0;
    333
    334 for (dtest = 1.; dtest && dtest < 1.1 * dtest; dtest *= 2.)
    335 if (dtest == dtest + 1.)
    336 break;
    337 t->dMaxSignifKond = dtest / 1000.; /* not sure whether this is really save, 100 does not work well enough */
    338
    339 t->gen = 0;
    340 t->countevals = 0;
    341 t->state = 0;
    342 t->dLastMinEWgroesserNull = 1.0;
    343 t->printtime = t->writetime = t->firstwritetime = t->firstprinttime = 0;
    344
    345 t->rgpc = new_double(N);
    346 t->rgps = new_double(N);
    347 t->rgdTmp = new_double(N+1);
    348 t->rgBDz = new_double(N);
    349 t->rgxmean = new_double(N+2); t->rgxmean[0] = N; ++t->rgxmean;
    350 t->rgxold = new_double(N+2); t->rgxold[0] = N; ++t->rgxold;
    351 t->rgxbestever = new_double(N+3); t->rgxbestever[0] = N; ++t->rgxbestever;
    352 t->rgout = new_double(N+2); t->rgout[0] = N; ++t->rgout;
    353 t->rgD = new_double(N);
    354 t->C = (double**)new_void(N, sizeof(double*));
    355 t->B = (double**)new_void(N, sizeof(double*));
    356 t->publicFitness = new_double(t->sp.lambda);
    357 t->rgFuncValue = new_double(t->sp.lambda+1);
    358 t->rgFuncValue[0]=t->sp.lambda; ++t->rgFuncValue;
    359 t->arFuncValueHist = new_double(10+(int)ceil(3.*10.*N/t->sp.lambda)+1);
    360 t->arFuncValueHist[0] = (double)(10+(int)ceil(3.*10.*N/t->sp.lambda));
    361 t->arFuncValueHist++;
    362
    363 for (i = 0; i < N; ++i) {
    364 t->C[i] = new_double(i+1);
    365 t->B[i] = new_double(N);
    366 }
    367 t->index = (int *) new_void(t->sp.lambda, sizeof(int));
    368 for (i = 0; i < t->sp.lambda; ++i)
    369 t->index[i] = i; /* should not be necessary */
    370 t->rgrgx = (double **)new_void(t->sp.lambda, sizeof(double*));
    371 for (i = 0; i < t->sp.lambda; ++i) {
    372 t->rgrgx[i] = new_double(N+2);
    373 t->rgrgx[i][0] = N;
    374 t->rgrgx[i]++;
    375 }
    376
    377 /* Initialize newed space */
    378
    379 for (i = 0; i < N; ++i)
    380 for (j = 0; j < i; ++j)
    381 t->C[i][j] = t->B[i][j] = t->B[j][i] = 0.;
    382
    383 for (i = 0; i < N; ++i)
    384 {
    385 t->B[i][i] = 1.;
    386 t->C[i][i] = t->rgD[i] = t->sp.rgInitialStds[i] * sqrt(N / trace);
    387 t->C[i][i] *= t->C[i][i];
    388 t->rgpc[i] = t->rgps[i] = 0.;
    389 }
    390
    391 t->minEW = rgdouMin(t->rgD, N); t->minEW = t->minEW * t->minEW;
    392 t->maxEW = rgdouMax(t->rgD, N); t->maxEW = t->maxEW * t->maxEW;
    393
    394 t->maxdiagC=t->C[0][0]; for(i=1;i<N;++i) if(t->maxdiagC<t->C[i][i]) t->maxdiagC=t->C[i][i];
    395 t->mindiagC=t->C[0][0]; for(i=1;i<N;++i) if(t->mindiagC>t->C[i][i]) t->mindiagC=t->C[i][i];
    396
    397 /* set xmean */
    398 for (i = 0; i < N; ++i)
    399 t->rgxmean[i] = t->rgxold[i] = t->sp.xstart[i];
    400 /* use in case xstart as typicalX */
    401 if (t->sp.typicalXcase)
    402 for (i = 0; i < N; ++i)
    403 t->rgxmean[i] += t->sigma * t->rgD[i] * cmaes_random_Gauss(&t->rand);
    404
    405 if (strcmp(t->sp.resumefile, "_no_") != 0)
    406 cmaes_resume_distribution(t, t->sp.resumefile);
    407
    408 return (t->publicFitness);
    409
    410} /* cmaes_init_final() */

    437void
    438cmaes_resume_distribution(cmaes_t *t, char *filename)
    439{
    440 int i, j, res, n;
    441 double d;
    442 FILE *fp = fopen( filename, "r");
    443 if(fp == NULL) {
    444 ERRORMESSAGE("cmaes_resume_distribution(): could not open '",
    445 filename, "'",0);
    446 return;
    447 }
    448 /* count number of "resume" entries */
    449 i = 0; res = 0;
    450 while (1) {
    451 if ((res = fscanf(fp, " resume %lg", &d)) == EOF)
    452 break;
    453 else if (res==0)
    454 fscanf(fp, " %*s");
    455 else if(res > 0)
    456 i += 1;
    457 }
    458
    459 /* go to last "resume" entry */
    460 n = i; i = 0; res = 0; rewind(fp);
    461 while (i<n) {
    462 if ((res = fscanf(fp, " resume %lg", &d)) == EOF)
    463 FATAL("cmaes_resume_distribution(): Unexpected error, bug",0,0,0);
    464 else if (res==0)
    465 fscanf(fp, " %*s");
    466 else if(res > 0)
    467 ++i;
    468 }
    469 if (d != t->sp.N)
    470 FATAL("cmaes_resume_distribution(): Dimension numbers do not match",0,0,0);
    471
    472 /* find next "xmean" entry */
    473 while (1) {
    474 if ((res = fscanf(fp, " xmean %lg", &d)) == EOF)
    475 FATAL("cmaes_resume_distribution(): 'xmean' not found",0,0,0);
    476 else if (res==0)
    477 fscanf(fp, " %*s");
    478 else if(res > 0)
    479 break;
    480 }
    481
    482 /* read xmean */
    483 t->rgxmean[0] = d; res = 1;
    484 for(i = 1; i < t->sp.N; ++i)
    485 res += fscanf(fp, " %lg", &t->rgxmean[i]);
    486 if (res != t->sp.N)
    487 FATAL("cmaes_resume_distribution(): xmean: dimensions differ",0,0,0);
    488
    489 /* find next "path for sigma" entry */
    490 while (1) {
    491 if ((res = fscanf(fp, " path for sigma %lg", &d)) == EOF)
    492 FATAL("cmaes_resume_distribution(): 'path for sigma' not found",0,0,0);
    493 else if (res==0)
    494 fscanf(fp, " %*s");
    495 else if(res > 0)
    496 break;
    497 }
    498
    499 /* read ps */
    500 t->rgps[0] = d; res = 1;
    501 for(i = 1; i < t->sp.N; ++i)
    502 res += fscanf(fp, " %lg", &t->rgps[i]);
    503 if (res != t->sp.N)
    504 FATAL("cmaes_resume_distribution(): ps: dimensions differ",0,0,0);
    505
    506 /* find next "path for C" entry */
    507 while (1) {
    508 if ((res = fscanf(fp, " path for C %lg", &d)) == EOF)
    509 FATAL("cmaes_resume_distribution(): 'path for C' not found",0,0,0);
    510 else if (res==0)
    511 fscanf(fp, " %*s");
    512 else if(res > 0)
    513 break;
    514 }
    515 /* read pc */
    516 t->rgpc[0] = d; res = 1;
    517 for(i = 1; i < t->sp.N; ++i)
    518 res += fscanf(fp, " %lg", &t->rgpc[i]);
    519 if (res != t->sp.N)
    520 FATAL("cmaes_resume_distribution(): pc: dimensions differ",0,0,0);
    521
    522 /* find next "sigma" entry */
    523 while (1) {
    524 if ((res = fscanf(fp, " sigma %lg", &d)) == EOF)
    525 FATAL("cmaes_resume_distribution(): 'sigma' not found",0,0,0);
    526 else if (res==0)
    527 fscanf(fp, " %*s");
    528 else if(res > 0)
    529 break;
    530 }
    531 t->sigma = d;
    532
    533 /* find next entry "covariance matrix" */
    534 while (1) {
    535 if ((res = fscanf(fp, " covariance matrix %lg", &d)) == EOF)
    536 FATAL("cmaes_resume_distribution(): 'covariance matrix' not found",0,0,0);
    537 else if (res==0)
    538 fscanf(fp, " %*s");
    539 else if(res > 0)
    540 break;
    541 }
    542 /* read C */
    543 t->C[0][0] = d; res = 1;
    544 for (i = 1; i < t->sp.N; ++i)
    545 for (j = 0; j <= i; ++j)
    546 res += fscanf(fp, " %lg", &t->C[i][j]);
    547 if (res != (t->sp.N*t->sp.N+t->sp.N)/2)
    548 FATAL("cmaes_resume_distribution(): C: dimensions differ",0,0,0);
    549
    550 fclose(fp);
    551
    552 t->flgIniphase = 0;
    553 t->flgEigensysIsUptodate = 0;
    554 t->flgresumedone = 1;
    555 cmaes_UpdateEigensystem(t, 1);
    556
    557} /* cmaes_resume_distribution() */

    623/* --------------------------------------------------------- */
    624double * const *
    625cmaes_SamplePopulation(cmaes_t *t)
    626{
    627 int iNk, i, j, N=t->sp.N;
    628 int flgdiag = ((t->sp.diagonalCov == 1) || (t->sp.diagonalCov >= t->gen));
    629 double sum;
    630 double const *xmean = t->rgxmean;
    631
    632 /* cmaes_SetMean(t, xmean); * xmean could be changed at this point */
    633
    634 /* calculate eigensystem */
    635 if (!t->flgEigensysIsUptodate) {
    636 if (!flgdiag)
    637 cmaes_UpdateEigensystem(t, 0);
    638 else {
    639 for (i = 0; i < N; ++i)
    640 t->rgD[i] = sqrt(t->C[i][i]);
    641 t->minEW = douSquare(rgdouMin(t->rgD, N));
    642 t->maxEW = douSquare(rgdouMax(t->rgD, N));
    643 t->flgEigensysIsUptodate = 1;
    644 cmaes_timings_start(&t->eigenTimings);
    645 }
    646 }
    647
    648 /* treat minimal standard deviations and numeric problems */
    649 TestMinStdDevs(t);
    650
    651 for (iNk = 0; iNk < t->sp.lambda; ++iNk)
    652 { /* generate scaled cmaes_random vector (D * z) */
    653 for (i = 0; i < N; ++i)
    654 if (flgdiag)
    655 t->rgrgx[iNk][i] = xmean[i] + t->sigma * t->rgD[i] * cmaes_random_Gauss(&t->rand);
    656 else
    657 t->rgdTmp[i] = t->rgD[i] * cmaes_random_Gauss(&t->rand);
    658 if (!flgdiag)
    659 /* add mutation (sigma * B * (D*z)) */
    660 for (i = 0; i < N; ++i) {
    661 for (j = 0, sum = 0.; j < N; ++j)
    662 sum += t->B[i][j] * t->rgdTmp[j];
    663 t->rgrgx[iNk][i] = xmean[i] + t->sigma * sum;
    664 }
    665 }
    666 if(t->state == 3 || t->gen == 0)
    667 ++t->gen;
    668 t->state = 1;
    669
    670 return(t->rgrgx);
    671} /* SamplePopulation() */

    807/* --------------------------------------------------------- */
    808/* --------------------------------------------------------- */
    809double *
    810cmaes_UpdateDistribution( cmaes_t *t, const double *rgFunVal)
    811{
    812 int i, j, iNk, hsig, N=t->sp.N;
    813 int flgdiag = ((t->sp.diagonalCov == 1) || (t->sp.diagonalCov >= t->gen));
    814 double sum;
    815 double psxps;
    816
    817 if(t->state == 3)
    818 FATAL("cmaes_UpdateDistribution(): You need to call \n",
    819 "SamplePopulation() before update can take place.",0,0);
    820 if(rgFunVal == NULL)
    821 FATAL("cmaes_UpdateDistribution(): ",
    822 "Fitness function value array input is missing.",0,0);
    823
    824 if(t->state == 1) /* function values are delivered here */
    825 t->countevals += t->sp.lambda;
    826 else
    827 ERRORMESSAGE("cmaes_UpdateDistribution(): unexpected state",0,0,0);
    828
    829 /* assign function values */
    830 for (i=0; i < t->sp.lambda; ++i)
    831 t->rgrgx[i][N] = t->rgFuncValue[i] = rgFunVal[i];
    832
    833
    834 /* Generate index */
    835 Sorted_index(rgFunVal, t->index, t->sp.lambda);
    836
    837 /* Test if function values are identical, escape flat fitness */
    838 if (t->rgFuncValue[t->index[0]] ==
    839 t->rgFuncValue[t->index[(int)t->sp.lambda/2]]) {
    840 t->sigma *= exp(0.2+t->sp.cs/t->sp.damps);
    841 ERRORMESSAGE("Warning: sigma increased due to equal function values\n",
    842 " Reconsider the formulation of the objective function",0,0);
    843 }
    844
    845 /* update function value history */
    846 for(i = (int)*(t->arFuncValueHist-1)-1; i > 0; --i) /* for(i = t->arFuncValueHist[-1]-1; i > 0; --i) */
    847 t->arFuncValueHist[i] = t->arFuncValueHist[i-1];
    848 t->arFuncValueHist[0] = rgFunVal[t->index[0]];
    849
    850 /* update xbestever */
    851 if (t->rgxbestever[N] > t->rgrgx[t->index[0]][N] || t->gen == 1)
    852 for (i = 0; i <= N; ++i) {
    853 t->rgxbestever[i] = t->rgrgx[t->index[0]][i];
    854 t->rgxbestever[N+1] = t->countevals;
    855 }
    856
    857 /* calculate xmean and rgBDz~N(0,C) */
    858 for (i = 0; i < N; ++i) {
    859 t->rgxold[i] = t->rgxmean[i];
    860 t->rgxmean[i] = 0.;
    861 for (iNk = 0; iNk < t->sp.mu; ++iNk)
    862 t->rgxmean[i] += t->sp.weights[iNk] * t->rgrgx[t->index[iNk]][i];
    863 t->rgBDz[i] = sqrt(t->sp.mueff)*(t->rgxmean[i] - t->rgxold[i])/t->sigma;
    864 }
    865
    866 /* calculate z := D^(-1) * B^(-1) * rgBDz into rgdTmp */
    867 for (i = 0; i < N; ++i) {
    868 if (!flgdiag)
    869 for (j = 0, sum = 0.; j < N; ++j)
    870 sum += t->B[j][i] * t->rgBDz[j];
    871 else
    872 sum = t->rgBDz[i];
    873 t->rgdTmp[i] = sum / t->rgD[i];
    874 }
    875
    876 /* TODO?: check length of t->rgdTmp and set an upper limit, e.g. 6 stds */
    877 /* in case of manipulation of arx,
    878 this can prevent an increase of sigma by several orders of magnitude
    879 within one step; a five-fold increase in one step can still happen.
    880 */
    881 /*
    882 for (j = 0, sum = 0.; j < N; ++j)
    883 sum += t->rgdTmp[j] * t->rgdTmp[j];
    884 if (sqrt(sum) > chiN + 6. * sqrt(0.5)) {
    885 rgdTmp length should be set to upper bound and hsig should become zero
    886 }
    887 */
    888
    889 /* cumulation for sigma (ps) using B*z */
    890 for (i = 0; i < N; ++i) {
    891 if (!flgdiag)
    892 for (j = 0, sum = 0.; j < N; ++j)
    893 sum += t->B[i][j] * t->rgdTmp[j];
    894 else
    895 sum = t->rgdTmp[i];
    896 t->rgps[i] = (1. - t->sp.cs) * t->rgps[i] +
    897 sqrt(t->sp.cs * (2. - t->sp.cs)) * sum;
    898 }
    899
    900 /* calculate norm(ps)^2 */
    901 for (i = 0, psxps = 0.; i < N; ++i)
    902 psxps += t->rgps[i] * t->rgps[i];
    903
    904 /* cumulation for covariance matrix (pc) using B*D*z~N(0,C) */
    905 hsig = sqrt(psxps) / sqrt(1. - pow(1.-t->sp.cs, 2*t->gen)) / t->chiN
    906 < 1.4 + 2./(N+1);
    907 for (i = 0; i < N; ++i) {
    908 t->rgpc[i] = (1. - t->sp.ccumcov) * t->rgpc[i] +
    909 hsig * sqrt(t->sp.ccumcov * (2. - t->sp.ccumcov)) * t->rgBDz[i];
    910 }
    911
    912 /* stop initial phase */
    913 if (t->flgIniphase &&
    914 t->gen > douMin(1/t->sp.cs, 1+N/t->sp.mucov))
    915 {
    916 if (psxps / t->sp.damps / (1.-pow((1. - t->sp.cs), t->gen))
    917 < N * 1.05)
    918 t->flgIniphase = 0;
    919 }
    920
    921#if 0
    922 /* remove momentum in ps, if ps is large and fitness is getting worse */
    923 /* This is obsolete due to hsig and harmful in a dynamic environment */
    924 if(psxps/N > 1.5 + 10.*sqrt(2./N)
    925 && t->arFuncValueHist[0] > t->arFuncValueHist[1]
    926 && t->arFuncValueHist[0] > t->arFuncValueHist[2]) {
    927 double tfac = sqrt((1 + douMax(0, log(psxps/N))) * N / psxps);
    928 for (i=0; i<N; ++i)
    929 t->rgps[i] *= tfac;
    930 psxps *= tfac*tfac;
    931 }
    932#endif
    933
    934 /* update of C */
    935
    936 Adapt_C2(t, hsig);
    937
    938 /* Adapt_C(t); not used anymore */
    939
    940#if 0
    941 if (t->sp.ccov != 0. && t->flgIniphase == 0) {
    942 int k;
    943
    944 t->flgEigensysIsUptodate = 0;
    945
    946 /* update covariance matrix */
    947 for (i = 0; i < N; ++i)
    948 for (j = 0; j <=i; ++j) {
    949 t->C[i][j] = (1 - t->sp.ccov) * t->C[i][j]
    950 + t->sp.ccov * (1./t->sp.mucov)
    951 * (t->rgpc[i] * t->rgpc[j]
    952 + (1-hsig)*t->sp.ccumcov*(2.-t->sp.ccumcov) * t->C[i][j]);
    953 for (k = 0; k < t->sp.mu; ++k) /* additional rank mu update */
    954 t->C[i][j] += t->sp.ccov * (1-1./t->sp.mucov) * t->sp.weights[k]
    955 * (t->rgrgx[t->index[k]][i] - t->rgxold[i])
    956 * (t->rgrgx[t->index[k]][j] - t->rgxold[j])
    957 / t->sigma / t->sigma;
    958 }
    959 }
    960#endif
    961
    962
    963 /* update of sigma */
    964 t->sigma *= exp(((sqrt(psxps)/t->chiN)-1.)*t->sp.cs/t->sp.damps);
    965
    966 t->state = 3;
    967
    968 return (t->rgxmean);
    969
    970} /* cmaes_UpdateDistribution() */

    973/* --------------------------------------------------------- */
    974/* --------------------------------------------------------- */
    975static void
    976Adapt_C2(cmaes_t *t, int hsig)
    977{
    978 int i, j, k, N=t->sp.N;
    979 int flgdiag = ((t->sp.diagonalCov == 1) || (t->sp.diagonalCov >= t->gen));
    980
    981 if (t->sp.ccov != 0. && t->flgIniphase == 0) {
    982
    983 /* definitions for speeding up inner-most loop */
    984 double ccov1 = douMin(t->sp.ccov * (1./t->sp.mucov) * (flgdiag ? (N+1.5) / 3. : 1.), 1.);
    985 double ccovmu = douMin(t->sp.ccov * (1-1./t->sp.mucov)* (flgdiag ? (N+1.5) / 3. : 1.), 1.-ccov1);
    986 double sigmasquare = t->sigma * t->sigma;
    987
    988 t->flgEigensysIsUptodate = 0;
    989
    990 /* update covariance matrix */
    991 for (i = 0; i < N; ++i)
    992 for (j = flgdiag ? i : 0; j <= i; ++j) {
    993 t->C[i][j] = (1 - ccov1 - ccovmu) * t->C[i][j]
    994 + ccov1
    995 * (t->rgpc[i] * t->rgpc[j]
    996 + (1-hsig)*t->sp.ccumcov*(2.-t->sp.ccumcov) * t->C[i][j]);
    997 for (k = 0; k < t->sp.mu; ++k) { /* additional rank mu update */
    998 t->C[i][j] += ccovmu * t->sp.weights[k]
    999 * (t->rgrgx[t->index[k]][i] - t->rgxold[i])
    1000 * (t->rgrgx[t->index[k]][j] - t->rgxold[j])
    1001 / sigmasquare;
    1002 }
    1003 }
    1004 /* update maximal and minimal diagonal value */
    1005 t->maxdiagC = t->mindiagC = t->C[0][0];
    1006 for (i = 1; i < N; ++i) {
    1007 if (t->maxdiagC < t->C[i][i])
    1008 t->maxdiagC = t->C[i][i];
    1009 else if (t->mindiagC > t->C[i][i])
    1010 t->mindiagC = t->C[i][i];
    1011 }
    1012 } /* if ccov... */
    1013}

    1020 /* increases sigma */

    1543/* tests stopping criteria
    1544 * returns a string of satisfied stopping criterion for each line
    1545 * otherwise NULL
    1546*/
    1547const char *
    1548cmaes_TestForTermination( cmaes_t *t)
    1549{
    1550 double range, fac;
    1551 int iAchse, iKoo;
    1552 int flgdiag = ((t->sp.diagonalCov == 1) || (t->sp.diagonalCov >= t->gen));
    1553 static char sTestOutString[3024];
    1554 char * cp = sTestOutString;
    1555 int i, cTemp, N=t->sp.N;
    1556 cp[0] = '\0';
    1557
    1558 /* function value reached */
    1559 if ((t->gen > 1 || t->state > 1) && t->sp.stStopFitness.flg &&
    1560 t->rgFuncValue[t->index[0]] <= t->sp.stStopFitness.val)
    1561 cp += sprintf(cp, "Fitness: function value %7.2e <= stopFitness (%7.2e)\n",
    1562 t->rgFuncValue[t->index[0]], t->sp.stStopFitness.val);
    1563
    1564 /* TolFun */
    1565 range = douMax(rgdouMax(t->arFuncValueHist, (int)douMin(t->gen,*(t->arFuncValueHist-1))),
    1566 rgdouMax(t->rgFuncValue, t->sp.lambda)) -
    1567 douMin(rgdouMin(t->arFuncValueHist, (int)douMin(t->gen, *(t->arFuncValueHist-1))),
    1568 rgdouMin(t->rgFuncValue, t->sp.lambda));
    1569
    1570 if (t->gen > 0 && range <= t->sp.stopTolFun) {
    1571 cp += sprintf(cp,
    1572 "TolFun: function value differences %7.2e < stopTolFun=%7.2e\n",
    1573 range, t->sp.stopTolFun);
    1574 }
    1575
    1576 /* TolFunHist */
    1577 if (t->gen > *(t->arFuncValueHist-1)) {
    1578 range = rgdouMax(t->arFuncValueHist, (int)*(t->arFuncValueHist-1))
    1579 - rgdouMin(t->arFuncValueHist, (int)*(t->arFuncValueHist-1));
    1580 if (range <= t->sp.stopTolFunHist)
    1581 cp += sprintf(cp,
    1582 "TolFunHist: history of function value changes %7.2e stopTolFunHist=%7.2e",
    1583 range, t->sp.stopTolFunHist);
    1584 }
    1585
    1586 /* TolX */
    1587 for(i=0, cTemp=0; i<N; ++i) {
    1588 cTemp += (t->sigma * sqrt(t->C[i][i]) < t->sp.stopTolX) ? 1 : 0;
    1589 cTemp += (t->sigma * t->rgpc[i] < t->sp.stopTolX) ? 1 : 0;
    1590 }
    1591 if (cTemp == 2*N) {
    1592 cp += sprintf(cp,
    1593 "TolX: object variable changes below %7.2e \n",
    1594 t->sp.stopTolX);
    1595 }
    1596
    1597 /* TolUpX */
    1598 for(i=0; i<N; ++i) {
    1599 if (t->sigma * sqrt(t->C[i][i]) > t->sp.stopTolUpXFactor * t->sp.rgInitialStds[i])
    1600 break;
    1601 }
    1602 if (i < N) {
    1603 cp += sprintf(cp,
    1604 "TolUpX: standard deviation increased by more than %7.2e, larger initial standard deviation recommended \n",
    1605 t->sp.stopTolUpXFactor);
    1606 }
    1607
    1608 /* Condition of C greater than dMaxSignifKond */
    1609 if (t->maxEW >= t->minEW * t->dMaxSignifKond) {
    1610 cp += sprintf(cp,
    1611 "ConditionNumber: maximal condition number %7.2e reached. maxEW=%7.2e,minEW=%7.2e,maxdiagC=%7.2e,mindiagC=%7.2e\n",
    1612 t->dMaxSignifKond, t->maxEW, t->minEW, t->maxdiagC, t->mindiagC);
    1613 } /* if */
    1614
    1615 /* Principal axis i has no effect on xmean, ie.
    1616 x == x + 0.1 * sigma * rgD[i] * B[i] */
    1617 if (!flgdiag) {
    1618 for (iAchse = 0; iAchse < N; ++iAchse)
    1619 {
    1620 fac = 0.1 * t->sigma * t->rgD[iAchse];
    1621 for (iKoo = 0; iKoo < N; ++iKoo){
    1622 if (t->rgxmean[iKoo] != t->rgxmean[iKoo] + fac * t->B[iKoo][iAchse])
    1623 break;
    1624 }
    1625 if (iKoo == N)
    1626 {
    1627 /* t->sigma *= exp(0.2+t->sp.cs/t->sp.damps); */
    1628 cp += sprintf(cp,
    1629 "NoEffectAxis: standard deviation 0.1*%7.2e in principal axis %d without effect\n",
    1630 fac/0.1, iAchse);
    1631 break;
    1632 } /* if (iKoo == N) */
    1633 } /* for iAchse */
    1634 } /* if flgdiag */
    1635 /* Component of xmean is not changed anymore */
    1636 for (iKoo = 0; iKoo < N; ++iKoo)
    1637 {
    1638 if (t->rgxmean[iKoo] == t->rgxmean[iKoo] +
    1639 0.2*t->sigma*sqrt(t->C[iKoo][iKoo]))
    1640 {
    1641 /* t->C[iKoo][iKoo] *= (1 + t->sp.ccov); */
    1642 /* flg = 1; */
    1643 cp += sprintf(cp,
    1644 "NoEffectCoordinate: standard deviation 0.2*%7.2e in coordinate %d without effect\n",
    1645 t->sigma*sqrt(t->C[iKoo][iKoo]), iKoo);
    1646 break;
    1647 }
    1648
    1649 } /* for iKoo */
    1650 /* if (flg) t->sigma *= exp(0.05+t->sp.cs/t->sp.damps); */
    1651
    1652 if(t->countevals >= t->sp.stopMaxFunEvals)
    1653 cp += sprintf(cp, "MaxFunEvals: conducted function evaluations %.0f >= %g\n",
    1654 t->countevals, t->sp.stopMaxFunEvals);
    1655 if(t->gen >= t->sp.stopMaxIter)
    1656 cp += sprintf(cp, "MaxIter: number of iterations %.0f >= %g\n",
    1657 t->gen, t->sp.stopMaxIter);
    1658 if(t->flgStop)
    1659 cp += sprintf(cp, "Manual: stop signal read\n");
    1660
    1661#if 0
    1662 else if (0) {
    1663 for(i=0, cTemp=0; i<N; ++i) {
    1664 cTemp += (sigma * sqrt(C[i][i]) < stopdx) ? 1 : 0;
    1665 cTemp += (sigma * rgpc[i] < stopdx) ? 1 : 0;
    1666 }
    1667 if (cTemp == 2*N)
    1668 flgStop = 1;
    1669 }
    1670#endif
    1671
    1672 if (cp - sTestOutString>320)
    1673 ERRORMESSAGE("Bug in cmaes_t:Test(): sTestOutString too short",0,0,0);
    1674
    1675 if (cp != sTestOutString) {
    1676 return sTestOutString;
    1677 }
    1678
    1679 return(NULL);

    1855static int
    1856Check_Eigen( int N, double **C, double *diag, double **Q)
    1857/*
    1858 exhaustive test of the output of the eigendecomposition
    1859 needs O(n^3) operations
    1860
    1861 writes to error file
    1862 returns number of detected inaccuracies
    1863*/
    1864{
    1865 /* compute Q diag Q^T and Q Q^T to check */
    1866 int i, j, k, res = 0;
    1867 double cc, dd;
    1868 static char s[324];
    1869
    1870 for (i=0; i < N; ++i)
    1871 for (j=0; j < N; ++j) {
    1872 for (cc=0.,dd=0., k=0; k < N; ++k) {
    1873 cc += diag[k] * Q[i][k] * Q[j][k];
    1874 dd += Q[i][k] * Q[j][k];
    1875 }
    1876 /* check here, is the normalization the right one? */
    1877 if (fabs(cc - C[i>j?i:j][i>j?j:i])/sqrt(C[i][i]*C[j][j]) > 1e-10
    1878 && fabs(cc - C[i>j?i:j][i>j?j:i]) > 3e-14) {
    1879 sprintf(s, "%d %d: %.17e %.17e, %e",
    1880 i, j, cc, C[i>j?i:j][i>j?j:i], cc-C[i>j?i:j][i>j?j:i]);
    1881 ERRORMESSAGE("cmaes_t:Eigen(): imprecise result detected ",
    1882 s, 0, 0);
    1883 ++res;
    1884 }
    1885 if (fabs(dd - (i==j)) > 1e-10) {
    1886 sprintf(s, "%d %d %.17e ", i, j, dd);
    1887 ERRORMESSAGE("cmaes_t:Eigen(): imprecise result detected (Q not orthog.)",
    1888 s, 0, 0);
    1889 ++res;
    1890 }
    1891 }
    1892 return res;

    2001static void
    2002QLalgo2 (int n, double *d, double *e, double **V) {
    2003 /*
    2004 -> n : Dimension.
    2005 -> d : Diagonale of tridiagonal matrix.
    2006 -> e[1..n-1] : off-diagonal, output from Householder
    2007 -> V : matrix output von Householder
    2008 <- d : eigenvalues
    2009 <- e : garbage?
    2010 <- V : basis of eigenvectors, according to d
    2011
    2012 Symmetric tridiagonal QL algorithm, iterative
    2013 Computes the eigensystem from a tridiagonal matrix in roughtly 3N^3 operations
    2014
    2015 code adapted from Java JAMA package, function tql2.
    2016 */
    2017
    2018 int i, k, l, m;
    2019 double f = 0.0;
    2020 double tst1 = 0.0;
    2021 double eps = 2.22e-16; /* Math.pow(2.0,-52.0); == 2.22e-16 */
    2022
    2023 /* shift input e */
    2024 for (i = 1; i < n; i++) {

    2146static void
    2147Householder2(int n, double **V, double *d, double *e) {
    2148 /*
    2149 Householder transformation of a symmetric matrix V into tridiagonal form.
    2150 -> n : dimension
    2151 -> V : symmetric nxn-matrix
    2152 <- V : orthogonal transformation matrix:
    2153 tridiag matrix == V * V_in * V^t
    2154 <- d : diagonal
    2155 <- e[0..n-1] : off diagonal (elements 1..n-1)
    2156
    2157 code slightly adapted from the Java JAMA package, function private tred2()
    2158
    2159 */
    2160
    2161 int i,j,k;
    2162
    2163 for (j = 0; j < n; j++) {
    2164 d[j] = V[n-1][j];
    2165 }
    2166
    2167 /* Householder reduction to tridiagonal form */
    2168
    2169 for (i = n-1; i > 0; i--) {
    2170
    2171 /* Scale to avoid under/overflow */
    2172
    2173 double scale = 0.0;
    2174 double h = 0.0;
    2175 for (k = 0; k < i; k++) {
    2176 scale = scale + fabs(d[k]);
    2177 }
    2178 if (scale == 0.0) {
    2179 e[i] = d[i-1];
    2180 for (j = 0; j < i; j++) {
    2181 d[j] = V[i-1][j];
    2182 V[i][j] = 0.0;
    2183 V[j][i] = 0.0;
    2184 }
    2185 } else {
    2186
    2187 /* Generate Householder vector */
    2188
    2189 double f, g, hh;
    2190
    2191 for (k = 0; k < i; k++) {
    2192 d[k] /= scale;
    2193 h += d[k] * d[k];
    2194 }
    2195 f = d[i-1];
    2196 g = sqrt(h);
    2197 if (f > 0) {
    2198 g = -g;
    2199 }
    2200 e[i] = scale * g;
    2201 h = h - f * g;
    2202 d[i-1] = f - g;
    2203 for (j = 0; j < i; j++) {
    2204 e[j] = 0.0;
    2205 }
    2206
    2207 /* Apply similarity transformation to remaining columns */
    2208
    2209 for (j = 0; j < i; j++) {
    2210 f = d[j];
    2211 V[j][i] = f;
    2212 g = e[j] + V[j][j] * f;
    2213 for (k = j+1; k <= i-1; k++) {
    2214 g += V[k][j] * d[k];
    2215 e[k] += V[k][j] * f;
    2216 }
    2217 e[j] = g;
    2218 }
    2219 f = 0.0;
    2220 for (j = 0; j < i; j++) {
    2221 e[j] /= h;
    2222 f += e[j] * d[j];
    2223 }
    2224 hh = f / (h + h);
    2225 for (j = 0; j < i; j++) {
    2226 e[j] -= hh * d[j];
    2227 }
    2228 for (j = 0; j < i; j++) {
    2229 f = d[j];
    2230 g = e[j];
    2231 for (k = j; k <= i-1; k++) {
    2232 V[k][j] -= (f * e[k] + g * d[k]);
    2233 }
    2234 d[j] = V[i-1][j];
    2235 V[i][j] = 0.0;
    2236 }
    2237 }
    2238 d[i] = h;
    2239 }
    2240
    2241 /* Accumulate transformations */
    2242
    2243 for (i = 0; i < n-1; i++) {
    2244 double h;
    2245 V[n-1][i] = V[i][i];
    2246 V[i][i] = 1.0;
    2247 h = d[i+1];
    2248 if (h != 0.0) {
    2249 for (k = 0; k <= i; k++) {
    2250 d[k] = V[k][i+1] / h;
    2251 }
    2252 for (j = 0; j <= i; j++) {
    2253 double g = 0.0;
    2254 for (k = 0; k <= i; k++) {
    2255 g += V[k][i+1] * V[k][j];
    2256 }
    2257 for (k = 0; k <= i; k++) {
    2258 V[k][j] -= g * d[k];
    2259 }
    2260 }
    2261 }
    2262 for (k = 0; k <= i; k++) {
    2263 V[k][i+1] = 0.0;
    2264 }
    2265 }
    2266 for (j = 0; j < n; j++) {
    2267 d[j] = V[n-1][j];
    2268 V[n-1][j] = 0.0;
    2269 }
    2270 V[n-1][n-1] = 1.0;
    2271 e[0] = 0.0;
    2272
    2273} /* Housholder() */
