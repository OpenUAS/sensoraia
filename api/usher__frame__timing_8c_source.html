    46/** @brief Convert a validated POSIX clock sample to nanoseconds without overflow.
    47 * @param[in] value Clock sample returned by clock_gettime().
    48 * @param[out] result Nanoseconds from that clock's origin.
    49 * @return True only for normalized, nonnegative values representable in uint64_t.
    50 * @note A malformed sample must invalidate alignment rather than wrap the age.
    51 */
    52static bool timespec_to_nanoseconds(const struct timespec *value, uint64_t *result)
    53{
    54 if (!value || !result || value->tv_sec < 0 || value->tv_nsec < 0
    55 || value->tv_nsec >= 1000000000L) {
    56 return false;
    57 }
    58 const uint64_t seconds = (uint64_t)value->tv_sec;
    59 const uint64_t nanoseconds = (uint64_t)value->tv_nsec;
    60 if (seconds > (UINT64_MAX - nanoseconds) / UINT64_C(1000000000)) {
    61 return false;
    62 }
    63 *result = seconds * UINT64_C(1000000000) + nanoseconds;
    64 return true;
    65}

    67struct usher_frame_timing usher_frame_timing_estimate(uint64_t sequence, uint64_t sensor_ns,
    68 int64_t exposure_us, int64_t duration_us)
    69{
    70 struct usher_frame_timing result = {
    71 .sequence = sequence,
    72 .sensor_ns = sensor_ns,
    73 .monotonic_us = 0,
    74 .time_error_us = unqualified_time_error_us,
    75 .exposure_us = exposure_us,
    76 .duration_us = duration_us
    77 };
    78 struct timespec mono_before, boot, mono_after;
    79 uint64_t before_ns, boot_ns, after_ns;
    80
    81 /* Reject implausible input before any unsigned conversion or clock work. */
    82 if (!result.sensor_ns || exposure_us < 0 || exposure_us > maximum_exposure_us
    83 || duration_us <= 0 || duration_us > maximum_frame_duration_us
    84 || clock_gettime(CLOCK_MONOTONIC, &mono_before) != 0
    85 || clock_gettime(CLOCK_BOOTTIME, &boot) != 0
    86 || clock_gettime(CLOCK_MONOTONIC, &mono_after) != 0
    87 || !timespec_to_nanoseconds(&mono_before, &before_ns)
    88 || !timespec_to_nanoseconds(&boot, &boot_ns)
    89 || !timespec_to_nanoseconds(&mono_after, &after_ns)) {
    90 return result;
    91 }
    92
    93 /* Slow or impossible samples cannot safely translate one clock into the other. */
    94 if (after_ns < before_ns || after_ns - before_ns > maximum_clock_sample_ns
    95 || boot_ns < result.sensor_ns || boot_ns - result.sensor_ns > maximum_sensor_age_ns
    96 || before_ns < boot_ns - result.sensor_ns) {
    97 return result;
    98 }
    99
    100 /* Frame age maps the BOOTTIME stamp into MONOTONIC; center exposure/readout. */
    101 const uint64_t age_ns = boot_ns - result.sensor_ns;
    102 const uint64_t monotonic_midpoint_ns = before_ns + (after_ns - before_ns) / 2 - age_ns;
    103 const uint64_t interval_us = (uint64_t)exposure_us + (uint64_t)duration_us;
    104 const uint64_t midpoint_us = monotonic_midpoint_ns / 1000;
    105 const uint64_t exposure_offset_us = interval_us / 2;
    106 if (UINT64_MAX - midpoint_us < exposure_offset_us) {
    107 return result;
    108 }
    109 result.monotonic_us = midpoint_us + exposure_offset_us;
    110 result.time_error_us = (after_ns - before_ns + 1999) / 2000 + (interval_us + 1) / 2 + 2;
    111 return result;
    112}

    114bool usher_frame_document(const struct usher_frame_timing *timing, unsigned width, unsigned height,
    115 struct exif_document *document)
    116{
    117 if (!timing || !document) {
    118 return false;
    119 }
    120 struct sensor_capture record = {0};
    121 usher_boot_id(record.boot_id);
    122 record.width = width;
    123 record.height = height;
    124 record.process_id = (uint32_t)getpid();
    125 record.sequence = timing->sequence;
    126 record.sensor_timestamp_ns = timing->sensor_ns;
    127 record.monotonic_us = timing->monotonic_us;
    128 record.time_error_us = timing->time_error_us;
    129 record.exposure_us = timing->exposure_us;
    130 record.frame_duration_us = timing->duration_us;
    131
    132 exif_document_init(document);
    133 unsigned char *data = exif_document_reserve(document, EXIF_CAMERA, SENSOR_CAPTURE_TAG,
    134 EXIF_UNDEFINED, SENSOR_CAPTURE_BYTES);
    135 return data && sensor_capture_encode(&record, data, SENSOR_CAPTURE_BYTES);
    136}
