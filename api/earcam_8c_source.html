    199/**
    200 * @brief Subtract unread PCM duration from a monotonic ALSA timestamp.
    201 * @return End of consumed audio in microseconds, or zero if the calculation is unusable.
    202 * @details Arrival time alone includes buffered audio delay. This correction lets
    203 * USHER associate a window with flight poses closer to its actual capture time.
    204 */
    205static uint64_t audio_read_end_us(uint64_t hardware_us, uint64_t unread_frames, unsigned int rate)
    206{
    207 if (!rate || unread_frames > UINT64_MAX / MICROSECONDS_PER_SECOND) return 0;
    208 const uint64_t delay_us = unread_frames * MICROSECONDS_PER_SECOND / rate;
    209 return hardware_us > delay_us ? hardware_us - delay_us : 0;
    210}

    219/**
    220 * @brief Normalize --name value and --name=value into bounded getopt arguments.
    221 * @return Argument count, or -1 for unknown options, missing values or capacity exhaustion.
    222 * @details Pointers borrow argv/static alias strings; no copied argument payload or heap is needed.
    223 */
    224static int normalize_long_options(int argc, char *const argv[], char *normalized[],
    225 size_t capacity, const struct option_alias aliases[],
    226 size_t alias_count)
    227{
    228 if (argc < 1 || argc > EARCAM_MAX_ARGUMENTS || capacity == 0) {
    229 return -1;
    230 }
    231 size_t output_count = 0;
    232 normalized[output_count++] = argv[0];
    233 for (int index = 1; index < argc; ++index) {
    234 const char *argument = argv[index];
    235 if (strcmp(argument, "--") == 0) {
    236 if (output_count >= capacity) {
    237 return -1;
    238 }
    239 normalized[output_count++] = argv[index];
    240 for (++index; index < argc; ++index) {
    241 if (output_count >= capacity) {
    242 return -1;
    243 }
    244 normalized[output_count++] = argv[index];
    245 }
    246 break;
    247 }
    248 if (strncmp(argument, "--", 2) != 0 || argument[2] == '\0') {
    249 if (output_count >= capacity) {
    250 return -1;
    251 }
    252 normalized[output_count++] = argv[index];
    253 continue;
    254 }
    255 const char *name = argument + 2;
    256 size_t name_length = strcspn(name, "=");
    257 const char *inline_value = name[name_length] == '=' ? name + name_length + 1 : NULL;
    258 const struct option_alias *alias = NULL;
    259 for (size_t alias_index = 0; alias_index < alias_count; ++alias_index) {
    260 if (strlen(aliases[alias_index].name) == name_length
    261 && strncmp(name, aliases[alias_index].name, name_length) == 0) {
    262 alias = &aliases[alias_index];
    263 break;
    264 }
    265 }
    266 if (alias == NULL || (!alias->requires_argument && inline_value != NULL)
    267 || output_count >= capacity) {
    268 return -1;
    269 }
    270 normalized[output_count++] = (char *)alias->option;
    271 if (alias->requires_argument) {
    272 if (inline_value == NULL) {
    273 if (++index >= argc) {
    274 return -1;
    275 }
    276 inline_value = argv[index];
    277 }
    278 if (output_count >= capacity) {
    279 return -1;
    280 }
    281 normalized[output_count++] = (char *)inline_value;
    282 }
    283 }
    284 if (output_count >= capacity) {
    285 return -1;
    286 }
    287 normalized[output_count] = NULL;
    288 return (int)output_count;
    289}

    300/** @brief Return monotonic milliseconds for correlating samples without wall-clock jumps.
    301 * @return Monotonic timestamp in milliseconds. */
    302static uint64_t monotonic_ms(void)
    303{
    304 struct timespec now;
    305 if (clock_gettime(CLOCK_MONOTONIC, &now) != 0 || now.tv_sec < 0 || now.tv_nsec < 0 ||
    306 now.tv_nsec >= NANOSECONDS_PER_SECOND) return 0;
    307 return (uint64_t)now.tv_sec * MILLISECONDS_PER_SECOND
    308 + (uint64_t)now.tv_nsec / NANOSECONDS_PER_MILLISECOND;
    309}

    360// Find the first USB audio card with a capture PCM. Writes "hw:CARD=<id>,DEV=<n>"
    361// into name; returns 0 on success, -1 when no USB capture device is present.
    362/** @brief Discover the first USB ALSA card offering a capture PCM.
    363 * @param name Destination ALSA device name.
    364 * @param name_size Capacity of @p name.
    365 * @return 0 when a usable device is found, otherwise -1.
    366 * @details USB-Audio matching prevents an arbitrary onboard sound device from silently
    367 * becoming the payload sensor when `--device auto` is used. */
    368static int find_usb_capture_device(char *name, size_t name_size)
    369{
    370 if (snd_ctl_card_info_sizeof() > sizeof(card_info_storage.bytes)
    371 || snd_pcm_info_sizeof() > sizeof(pcm_info_storage.bytes)) {
    372 errno = EOVERFLOW;
    373 return -1;
    374 }
    375 snd_ctl_card_info_t *info = (snd_ctl_card_info_t *)card_info_storage.bytes;
    376 snd_pcm_info_t *pcm_info = (snd_pcm_info_t *)pcm_info_storage.bytes;
    377 int found = -1;
    378 int card = -1;
    379 while (found != 0 && snd_card_next(&card) == 0 && card >= 0) {
    380 char control_name[CONTROL_DEVICE_NAME_CAPACITY];
    381 if (snprintf(control_name, sizeof(control_name), "hw:%d", card) <= 0) {
    382 continue;
    383 }
    384 snd_ctl_t *control = NULL;
    385 if (snd_ctl_open(&control, control_name, 0) < 0) {
    386 continue;
    387 }
    388 if (snd_ctl_card_info(control, info) < 0) {
    389 snd_ctl_close(control);
    390 continue;
    391 }
    392 const char *driver = snd_ctl_card_info_get_driver(info);
    393 const char *card_id = snd_ctl_card_info_get_id(info);
    394 if (driver == NULL || card_id == NULL || strcmp(driver, "USB-Audio") != 0) {
    395 snd_ctl_close(control);
    396 continue;
    397 }
    398 int device = -1;
    399 while (snd_ctl_pcm_next_device(control, &device) == 0 && device >= 0) {
    400 snd_pcm_info_set_device(pcm_info, (unsigned int)device);
    401 snd_pcm_info_set_subdevice(pcm_info, 0);
    402 snd_pcm_info_set_stream(pcm_info, SND_PCM_STREAM_CAPTURE);
    403 if (snd_ctl_pcm_info(control, pcm_info) < 0) {
    404 continue;
    405 }
    406 int length = snprintf(name, name_size, "hw:CARD=%s,DEV=%d", card_id, device);
    407 found = (length > 0 && (size_t)length < name_size) ? 0 : -1;
    408 break;
    409 }
    410 snd_ctl_close(control);
    411 }
    412 return found;
    413}

    433/** @brief Parse and cross-validate EARcam command-line configuration.
    434 * @return 0 for a runnable configuration, 1 after help/version output, or -1 on error.
    435 * @details Nyquist and band-order checks happen here rather than in the DSP loop so a
    436 * misconfigured filter cannot produce misleading field measurements. */
    437static int parse_options(int argc, char **argv, struct options *options)
    438{
    439 static const struct option_alias long_options[] = {
    440 {"device", "-d", true},
    441 {"rate", "-r", true},
    442 {"interval", "-i", true},
    443 {"low-cut", "-l", true},
    444 {"high-cut", "-u", true},
    445 {"min-level", "-n", true},
    446 {"calibration", "-c", true},
    447 {"self-test", "-S", false},
    448 {"server", "-M", false},
    449 {"settings", "-Z", true},
    450 {"analyze-audio", "-X", true},
    451 {"check-model", "-Q", false},
    452 {"version", "-V", false},
    453 {"help", "-h", false}
    454 };
    455 int option;
    456 char *normalized_argv[EARCAM_MAX_PARSED_ARGUMENTS];
    457 int normalized_argc = normalize_long_options(argc, argv, normalized_argv,
    458 EARCAM_MAX_PARSED_ARGUMENTS, long_options,
    459 sizeof(long_options) / sizeof(long_options[0]));
    460 if (normalized_argc < 0) {
    461 fprintf(stderr, "Invalid or excessive command-line arguments.\n");
    462 return -1;
    463 }
    464 optind = 1;
    465 while ((option = getopt(normalized_argc, normalized_argv,
    466 "d:r:i:l:u:n:c:hSMVZ:X:Q")) != -1) {
    467 switch (option) {
    468 case 'Z': options->settings = optarg; break;
    469 case 'X': options->audio_file = optarg; break;
    470 case 'Q': options->check_model = true; break;
    471 case 'd':
    472 options->device = optarg;
    473 break;
    474 case 'r':
    475 if (parse_unsigned(optarg, MIN_RATE_HZ, MAX_RATE, &options->rate) != 0) {
    476 fprintf(stderr, "Invalid sample rate: %s\n", optarg);
    477 return -1;
    478 }
    479 break;
    480 case 'i':
    481 if (parse_unsigned(optarg, MIN_INTERVAL_MS, MAX_INTERVAL_MS, &options->interval_ms) != 0) {
    482 fprintf(stderr, "Invalid display interval: %s\n", optarg);
    483 return -1;
    484 }
    485 break;
    486 case 'l':
    487 if (parse_double_value(optarg, 0.0, MAX_CUTOFF_HZ, &options->low_cut_hz) != 0) {

    551/** @brief Configure a second-order Butterworth high- or low-pass section.
    552 * @details Two identical sections are cascaded to give useful out-of-band rejection
    553 * before tone analysis without the latency and allocation cost of an FFT filter bank. */
    554static void configure_biquad(struct biquad *filter, double cutoff_hz,
    555 unsigned int sample_rate, bool high_pass)
    556{
    557 const double q = BUTTERWORTH_Q;
    558 double omega;
    559 double cosine;
    560 double alpha;
    561 double a0;
    562
    563 memset(filter, 0, sizeof(*filter));
    564 if (cutoff_hz <= 0.0) {
    565 return;
    566 }
    567
    568 /* Factors of two and unity below belong to the biquad equations, not tuning:
    569 * omega is radians/sample; a0 normalizes both numerator and feedback terms. */
    570 omega = 2.0 * PI * cutoff_hz / (double)sample_rate;
    571 cosine = cos(omega);
    572 alpha = sin(omega) / (2.0 * q);
    573 a0 = 1.0 + alpha;
    574
    575 if (high_pass) {
    576 filter->b0 = (1.0 + cosine) / (2.0 * a0);
    577 filter->b1 = -(1.0 + cosine) / a0;
    578 filter->b2 = filter->b0;
    579 } else {
    580 filter->b0 = (1.0 - cosine) / (2.0 * a0);
    581 filter->b1 = (1.0 - cosine) / a0;
    582 filter->b2 = filter->b0;
    583 }
    584 filter->a1 = (-2.0 * cosine) / a0;
    585 filter->a2 = (1.0 - alpha) / a0;
    586 filter->enabled = true;
    587}

    589/** @brief Filter one PCM sample while retaining the section delay state.
    590 * @return Filtered sample, or @p input when the section is disabled. */
    591static double apply_biquad(struct biquad *filter, double input)
    592{
    593 double output;
    594
    595 if (!filter->enabled) {
    596 return input;
    597 }
    598 output = filter->b0 * input + filter->z1;
    599 filter->z1 = filter->b1 * input - filter->a1 * output + filter->z2;
    600 filter->z2 = filter->b2 * input - filter->a2 * output;
    601 return output;
    602}

    614/** @brief Initialize evenly spaced Goertzel bins across the selected search band.
    615 * @details The small fixed bank is cheaper and more targeted than a broad FFT because
    616 * mission alarms occupy a known band and only the strongest frequency is needed. */
    617static void configure_tone_bins(struct tone_bin *bins,
    618 const struct options *options)
    619{
    620 double frequency_step = (options->high_cut_hz - options->low_cut_hz) /
    621 (double)(TONE_BIN_COUNT - 1U);
    622 unsigned int index;
    623
    624 for (index = 0; index < TONE_BIN_COUNT; ++index) {
    625 double frequency = options->low_cut_hz + frequency_step * (double)index;
    626 bins[index].coefficient =
    627 2.0 * cos(2.0 * PI * frequency / (double)options->rate);
    628 bins[index].first = 0.0;
    629 bins[index].second = 0.0;
    630 bins[index].frequency_hz = frequency;
    631 }
    632}

    634/** @brief Feed one windowed, band-passed sample into every Goertzel recurrence.
    635 * @param bins Persistent Goertzel states for the search frequencies.
    636 * @param sample Current band-passed PCM value normalized to [-1, 1].
    637 * @param window Precomputed Hann weight for this sample's interval position.
    638 * @details Windowing reduces spectral leakage; precomputation keeps trigonometry out of
    639 * the realtime sample loop. */
    640static void update_tone_bins(struct tone_bin *bins, double sample, double window)
    641{
    642 unsigned int index;
    643
    644 for (index = 0; index < TONE_BIN_COUNT; ++index) {
    645 double value = sample * window + bins[index].coefficient * bins[index].first -
    646 bins[index].second;
    647 bins[index].second = bins[index].first;
    648 bins[index].first = value;
    649 }
    650}

    652/** @brief Extract and reset the strongest Goertzel-bin power estimate.
    653 * @param bins Goertzel states to inspect and reset for the next interval.
    654 * @param sample_count Number of samples represented by the states.
    655 * @param frequency_hz Receives the winning bin frequency.
    656 * @return Window-normalized tone level in dBFS. */
    657static double strongest_tone_db(struct tone_bin *bins, uint64_t sample_count,
    658 double *frequency_hz)
    659{
    660 double maximum_power = 0.0;
    661 unsigned int index;
    662
    663 for (index = 0; index < TONE_BIN_COUNT; ++index) {
    664 double power = bins[index].first * bins[index].first +
    665 bins[index].second * bins[index].second -
    666 bins[index].coefficient * bins[index].first *
    667 bins[index].second;
    668 if (power > maximum_power) {
    669 maximum_power = power;
    670 *frequency_hz = bins[index].frequency_hz;
    671 }
    672 bins[index].first = 0.0;
    673 bins[index].second = 0.0;
    674 }
    675 return amplitude_to_db(HANN_TONE_AMPLITUDE_SCALE * sqrt(maximum_power) / (double)sample_count);
    676}

    678/** @brief Update envelope and fast/slow trend estimates for operator guidance.
    679 * @return Stable direction label for the current acoustic evidence.
    680 * @details The peak-hold envelope falls gradually while two exponential averages make
    681 * direction less sensitive to individual noisy windows. */
    682static const char *update_homing_state(struct homing_state *state,
    683 double tone_db, bool alarm_present,
    684 const struct options *options,
    685 double *trend_db)
    686{
    687 double elapsed = (double)options->interval_ms / MILLISECONDS_PER_SECOND_F;
    688 double fast_alpha = 1.0 - exp(-elapsed / HOMING_FAST_TIME_CONSTANT_S);
    689 double slow_alpha = 1.0 - exp(-elapsed / HOMING_SLOW_TIME_CONSTANT_S);
    690
    691 if (!state->initialized) {
    692 state->envelope_db = tone_db;
    693 state->fast_db = tone_db;
    694 state->slow_db = tone_db;
    695 state->initialized = true;
    696 }
    697
    698 state->envelope_db -= ENVELOPE_RELEASE_DB_PER_SECOND * elapsed;
    699 if (alarm_present) {
    700 if (tone_db > state->envelope_db) {
    701 state->envelope_db = tone_db;
    702 }
    703 ++state->alarm_windows;
    704 }
    705 if (state->envelope_db < DB_FLOOR) {
    706 state->envelope_db = DB_FLOOR;
    707 }
    708
    709 state->fast_db += fast_alpha * (state->envelope_db - state->fast_db);
    710 state->slow_db += slow_alpha * (state->envelope_db - state->slow_db);
    711 *trend_db = state->fast_db - state->slow_db;
    712
    713 if (state->alarm_windows < HOMING_ACQUISITION_WINDOWS) {
    714 return "ACQUIRING ";
    715 }
    716 if (*trend_db > TREND_THRESHOLD_DB) {
    717 return "APPROACH ";
    718 }
    719 if (*trend_db < -TREND_THRESHOLD_DB) {
    720 return "RECEDING ";
    721 }
    722 return "STEADY ";
    723}

    725/** @brief Analyze a deterministic synthetic tone mixture for self-test coverage. */
    726static void analyze_test_window(struct tone_bin *bins,
    727 const struct options *options,
    728 double target_amplitude,
    729 double interference_amplitude,
    730 double *tone_db, double *contrast_db)
    731{
    732 uint64_t sample_count =
    733 ((uint64_t)options->rate * options->interval_ms) / MILLISECONDS_PER_SECOND;
    734 uint64_t sample_index;
    735 double sum_squares = 0.0;
    736 double frequency_hz = 0.0;
    737 if (sample_count == 0U || sample_count > MAX_INTERVAL_FRAMES) {
    738 *tone_db = DB_FLOOR;
    739 *contrast_db = 0.0;
    740 return;
    741 }
    742 for (sample_index = 0; sample_index < sample_count; ++sample_index) {
    743 analysis_window[sample_index] = sample_count > 1U
    744 ? 0.5 - 0.5 * cos(2.0 * PI * (double)sample_index /
    745 (double)(sample_count - 1U))
    746 : 1.0;
    747 }
    748
    749 for (sample_index = 0; sample_index < sample_count; ++sample_index) {
    750 double time = (double)sample_index / (double)options->rate;
    751 double sample = target_amplitude * sin(2.0 * PI * TEST_TARGET_HZ * time) +
    752 interference_amplitude * sin(2.0 * PI * TEST_INTERFERENCE_HZ * time);
    753 update_tone_bins(bins, sample, analysis_window[sample_index]);
    754 sum_squares += sample * sample;
    755 }
    756 *tone_db = strongest_tone_db(bins, sample_count, &frequency_hz);
    757 *contrast_db = *tone_db -
    758 amplitude_to_db(sqrt(sum_squares / (double)sample_count));
    759}

    761/** @brief Verify selectivity and trend classification without ALSA hardware.
    762 * @return 0 on expected detector behavior, otherwise -1. */
    763static int run_self_test(const struct options *options)
    764{
    765 /* A 10 ms unread tail must shift the timestamp back by exactly that duration;
    766 * underflow and missing sample rate must not manufacture a plausible time. */
    767 if (audio_read_end_us(TEST_TIMESTAMP_US, TEST_UNREAD_FRAMES, TEST_RATE_HZ) != TEST_EXPECTED_READ_END_US
    768 || audio_read_end_us(TEST_UNDERFLOW_TIMESTAMP_US, TEST_UNREAD_FRAMES, TEST_RATE_HZ) != 0
    769 || audio_read_end_us(TEST_TIMESTAMP_US, TEST_UNREAD_FRAMES, 0) != 0) return -1;
    770 struct tone_bin bins[TONE_BIN_COUNT];
    771 struct homing_state approaching = {0};
    772 struct homing_state receding = {0};
    773 double tone_db;
    774 double contrast_db;
    775 double trend_db = 0.0;
    776 const char *direction = NULL;
    777 unsigned int index;
    778
    779 configure_tone_bins(bins, options);
    780 /* A weak in-band tone amid stronger out-of-band interference checks bin
    781 * selectivity. The tolerance is software-fixture evidence, not sensor accuracy. */
    782 analyze_test_window(bins, options, TEST_TARGET_AMPLITUDE, TEST_INTERFERENCE_AMPLITUDE, &tone_db, &contrast_db);
    783 if (tone_db < TEST_TONE_MIN_DB || tone_db > TEST_TONE_MAX_DB || contrast_db > TEST_CONTRAST_MAX_DB) {
    784 fprintf(stderr, "Self-test failed: tone selectivity (%.1f, %.1f dB)\n",
    785 tone_db, contrast_db);
    786 return -1;
    787 }
    788
    789 for (index = 0; index < TEST_TREND_WINDOWS; ++index) {
    790 double level = TEST_APPROACH_START_DB + (double)index * TEST_TREND_STEP_DB;
    791 direction = update_homing_state(&approaching, level, true, options,
    792 &trend_db);
    793 }
    794 if (strcmp(direction, "APPROACH ") != 0) {
    795 fprintf(stderr, "Self-test failed: approaching trend (%.1f dB)\n",
    796 trend_db);
    797 return -1;
    798 }
    799
    800 for (index = 0; index < TEST_TREND_WINDOWS; ++index) {
    801 double level = TEST_RECEDE_START_DB - (double)index * TEST_TREND_STEP_DB;
    802 direction = update_homing_state(&receding, level, true, options,
    803 &trend_db);
    804 }
    805 if (strcmp(direction, "RECEDING ") != 0) {
    806 fprintf(stderr, "Self-test failed: receding trend (%.1f dB)\n", trend_db);
    807 return -1;
    808 }
    809
    810 printf("motionSCOUT detector self-test passed\n");
    811 return 0;
    812}

    957 "SAMPLE t=%llu level=%.2f freq=%.0f contrast=%.2f "
    958 "trend=%.2f env=%.2f alarm=%d clip=%d cal=%d"
    959 " begin_us=%llu end_us=%llu error_us=%llu sequence=%llu\n",

    986// Returns 1 when the controller asked us to stop, 0 otherwise.
    987/** @brief Consume non-blocking STOP/QUIT commands from USHER.
    988 * @return 1 when capture should stop, otherwise 0.
    989 * @details Chunked input accepts fragmented commands while avoiding one syscall per
    990 * character and never delaying audio acquisition. */
    991static int poll_server_stdin(void)
    992{
    993 struct pollfd descriptor = {.fd = STDIN_FILENO, .events = POLLIN, .revents = 0};
    994 static char command[SERVER_COMMAND_CAPACITY];
    995 static size_t command_size = 0;
    996
    997 if (poll(&descriptor, 1, 0) <= 0) {
    998 return 0;
    999 }
    1000 if ((descriptor.revents & (POLLHUP | POLLERR | POLLNVAL)) != 0
    1001 && (descriptor.revents & POLLIN) == 0) {
    1002 return 1;
    1003 }
    1004 char chunk[SERVER_READ_CHUNK_CAPACITY];
    1005 ssize_t count = read(STDIN_FILENO, chunk, sizeof(chunk));
    1006 if (count == 0) {
    1007 return 1;
    1008 }
    1009 if (count < 0) {
    1010 return (errno == EAGAIN || errno == EINTR) ? 0 : 1;
    1011 }
    1012 for (ssize_t index = 0; index < count; ++index) {
    1013 if (chunk[index] == '\n') {
    1014 command[command_size] = '\0';
    1015 command_size = 0;
    1016 if (strcmp(command, "STOP") == 0 || strcmp(command, "QUIT") == 0) {
    1017 return 1;
    1018 }
    1019 } else if (command_size + 1 < sizeof(command)) {
    1020 command[command_size++] = chunk[index];
    1021 }
    1022 }
    1023 return 0;
    1024}

    1026/** @brief Capture, filter, analyze, and report audio until stopped.
    1027 * @return 0 for orderly stop or -1 for unrecoverable ALSA or bounds failure.
    1028 * @details ALSA recovery handles transient overruns. Each reporting window produces
    1029 * band RMS, strongest tone, contrast, and homing trend. */
