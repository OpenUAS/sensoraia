    21/** @file
    22 * @brief Optional camera calibration and native pixel-to-ground geometry only.
    23 *
    24 * @details Compiled only in analysis builds (`LWIR_ENABLE_ANALYSIS=1`). Used by
    25 * lwir_analysis.c to turn hotspot centroids found in the native temperature plane
    26 * of a saved LWIR JPEG into estimated WGS84 positions. No detection, JPEG or EXIF
    27 * code lives here, so the geometry can be tested with synthetic poses.
    28 *
    29 * Frames (all right-handed):
    30 * - camera: x right, y down, z forward along the optical axis (OpenCV);
    31 * - body: x forward, y right, z down;
    32 * - local NED at the navigation reference: north, east, down.
    33 *
    34 * Model limits callers must respect:
    35 * - pinhole plus Brown-Conrady distortion; strong fisheye lenses are outside it;
    36 * - the ground is a horizontal plane `agl_m` below the navigation reference, so
    37 * terrain relief and slope translate directly into horizontal error;
    38 * - pose quality is the caller's responsibility: a pose stamped at frame arrival
    39 * rather than over the exposure interval can be displaced by aircraft motion,
    40 * so projection results must be qualified with the pose timing evidence;
    41 * - an unverified calibration yields plausible-looking but unqualified numbers.
    42 */

    51/** @brief Projection calibration in native coordinates, never display coordinates.
    52 * @details Lens coefficients follow OpenCV's pinhole/Brown-Conrady convention.
    53 * Keeping them in the stored native pixel frame prevents display rotation or
    54 * preview enlargement from silently changing the ray being projected.
    55 */
    56struct lwir_camera_geometry {
    57 double fx, fy, cx, cy;
    58 double k1, k2, p1, p2, k3;
    59 double camera_to_body[9]; ///< Row-major rotation: camera right/down/forward to body forward/right/down.
    60 double lever_arm_body_m[3]; ///< Navigation reference to camera origin, in body forward/right/down metres.
    61 double minimum_ground_cosine; ///< Minimum normalized downward ray component; rejects unstable horizon intersections.
    62 unsigned image_width, image_height;
    63 bool verified; ///< Operator verification claim; loading coefficients does not establish it.
    64};

    175/** @brief Load bounded projection calibration, requiring exact native dimensions.
    176 * @param[in] path Nonempty regular calibration file, at most 8192 bytes.
    177 * @param width Expected native image width, within the supported LWIR bounds.
    178 * @param height Expected native image height, within the supported LWIR bounds.
    179 * @param[out] geometry Result, unchanged on failure.
    180 * @return True for structurally valid calibration, not proof of physical accuracy.
    181 * @details This is a bounded profile reader, not a general YAML implementation.
    182 * Each scalar or bracketed numeric array must fit on one line. Required fields
    183 * include all lens coefficients, dimensions, verification flag and the nine-value
    184 * rotation; duplicate known fields fail. Optional lever arm defaults to zero and
    185 * minimum_ground_cosine to 0.2. Unknown keys are ignored.
    186 * @note An unverified profile is intentionally loadable for workshop checks.
    187 * Loading does not validate rotation orthonormality; projection checks it before
    188 * use. Neither operation establishes that a mount was physically measured.
    189 * @note Lines longer than 511 characters, a missing ':' on a non-comment line,
    190 * `verified` other than 0/1, nonpositive focal lengths, or a
    191 * minimum_ground_cosine outside 0.05..1 all fail. Lines starting with '#',
    192 * '%' or consisting of `---` are skipped. errno is not part of the failure contract.
    193 */
    194bool lwir_geolocation_load_calibration(const char *path, unsigned width, unsigned height,
    195 struct lwir_camera_geometry *geometry);
    196/** @brief Project a native subpixel location using distortion, mount and observation pose.
    197 * @param pixel_x Native column, independent of JPEG rotation.
    198 * @param pixel_y Native row.
    199 * @param[in] camera Intrinsics and proper camera-to-body rotation.
    200 * @param[in] pose Observation position, attitude and positive height above the ground plane.
    201 * @param[out] point Result, unchanged on failure.
    202 * @return False for invalid/nonfinite geometry, nonconvergence or near-horizon rays.
    203 * @details Inverse distortion is solved in camera coordinates before mount and
    204 * attitude rotations. A finite iteration budget rejects nonconvergent rays, and
    205 * the downward-cosine guard avoids arbitrarily long intersections near the
    206 * horizon. The intersection uses a local horizontal plane, not a terrain model.
    207 * The local north/east offset is converted to latitude/longitude on the WGS84
    208 * ellipsoid via ECEF, so no small-angle flat-earth approximation is involved.
    209 * @warning The caller must enforce verification, observation synchronization and
    210 * uncertainty policy; this geometry function does not gate on those flags.
    211 */
    212bool lwir_geolocation_project(double pixel_x, double pixel_y,
    213 const struct lwir_camera_geometry *geometry,
    214 const struct lwir_camera_pose *pose,
    215 struct lwir_ground_point *point);
