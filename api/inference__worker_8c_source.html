    37/** @brief Probe sibling INT8 then FP32 ONNX exports when automatic settings point to an RPK or unsupported INT8 graph. */
    38static bool open_fallback_onnx(struct inference_context *context,
    39 struct inference_settings *settings,
    40 const struct inference_image_processor *processor,
    41 char *error, size_t capacity)
    42{
    43 const char *suffix = strrchr(settings->model, '.');
    44 const char *slash = strrchr(settings->model, '/');
    45 if (!suffix || !slash || (settings->backend != INFERENCE_IMX500 &&
    46 strcmp(slash + 1, "vehicle_yolo11n_int8.onnx") != 0)) return false;
    47 static const char *const candidates[][2] = {
    48 {"vehicle_yolo11n_int8.onnx", "vehicle_yolo11n_int8.ini"},
    49 {"vehicle_yolo11n_fp32.onnx", "vehicle_yolo11n_fp32.ini"}
    50 };
    51 const int directory_length = (int)(slash - settings->model);
    52 for (size_t index = 0; index < sizeof(candidates) / sizeof(candidates[0]); ++index) {
    53 struct inference_settings candidate = *settings;
    54 candidate.backend = INFERENCE_ONNX;
    55 const int model_len = snprintf(candidate.model, sizeof(candidate.model), "%.*s/%s",
    56 directory_length, settings->model, candidates[index][0]);
    57 const int desc_len = snprintf(candidate.descriptor, sizeof(candidate.descriptor), "%.*s/%s",
    58 directory_length, settings->model, candidates[index][1]);
    59 if (model_len < 0 || (size_t)model_len >= sizeof(candidate.model) ||
    60 desc_len < 0 || (size_t)desc_len >= sizeof(candidate.descriptor)) continue;
    61 char probe_error[2048];
    62 if (inference_context_open(context, &candidate, processor->adapter, probe_error, sizeof(probe_error))) {
    63 if (!processor->validate || processor->validate(processor->context, context, probe_error, sizeof(probe_error))) {
    64 *settings = candidate;
    65 if (capacity) error[0] = '\0';
    66 return true;
    67 }
    68 inference_context_close(context);
    69 }
    70 if (probe_error[0]) snprintf(error, capacity, "%s", probe_error);
    71 }
    72 return false;
    73}

    75int inference_worker_cli(int argc, char **argv, const struct inference_image_processor *processor)
    76{
    77 bool server = false, check = false, selected = false, private_images = false;
    78 const char *image = NULL, *settings_path = NULL;
    79 uint32_t sequence = 1;
    80 if (!processor || !processor->write || !processor->application || !processor->module || !processor->adapter) return 2;
    81 for (int index = 1; index < argc; ++index) {
    82 if (!strcmp(argv[index], "--analyze") || !strcmp(argv[index], "--analyze-server") || !strcmp(argv[index], "--check-model")) selected = true;
    83 }
    84 if (!selected) return -1;
    85 for (int index = 1; index < argc; ++index) {
    86 if (!strcmp(argv[index], "--analyze-server")) { if (server) return 2; server = true; }
    87 else if (!strcmp(argv[index], "--check-model")) { if (check) return 2; check = true; }
    88 else if (!strcmp(argv[index], "--private-images")) { if (private_images) return 2; private_images = true; }
    89 else {
    90 const char *option = argv[index];
    91 if (++index == argc) return 2;
    92 if (!strcmp(option, "--analyze")) { if (image) return 2; image = argv[index]; }
    93 else if (!strcmp(option, "--settings")) { if (settings_path) return 2; settings_path = argv[index]; }
    94 else if (!strcmp(option, "--sequence")) {
    95 uint64_t number;
    96 if (!ini_uint(argv[index], &number) || !number || number > UINT32_MAX) return 2;
    97 sequence = (uint32_t)number;
    98 } else return 2;
    99 }
    100 }
    101 if ((unsigned)server + (unsigned)check + (unsigned)(image != NULL) != 1) return 2;
    102 char discovered[INFERENCE_PATH_CAPACITY], error[2048];
    103 if (!settings_path) {
    104 if (!inference_settings_find(processor->application, processor->module, discovered, sizeof(discovered))) {
    105 fprintf(stderr, "%s: settings not found; use --settings FILE\n", processor->application);
    106 return 1;
    107 }
    108 settings_path = discovered;
    109 }
    110 struct inference_settings settings;
    111 inference_settings_init(&settings, INFERENCE_ONNX);
    112 if (!inference_settings_read(&settings, settings_path, processor->setting, processor->context, error, sizeof(error))) {
    113 fprintf(stderr, "%s: %s\n", processor->application, error);
    114 return 1;
    115 }
    116 struct inference_context context = {0};
    117 if (!inference_context_open(&context, &settings, processor->adapter, error, sizeof(error)) &&
    118 !open_fallback_onnx(&context, &settings, processor, error, sizeof(error))) {
    119 fprintf(stderr, "%s: %s\n", processor->application, error);
    120 return 1;
    121 }
    122 if (processor->validate && !processor->validate(processor->context, &context, error, sizeof(error))) {
    123 fprintf(stderr, "%s: %s\n", processor->application, error);
    124 inference_context_close(&context);
    125 return 1;
    126 }
    127 int result = 0;
    128 context.private_image = private_images;
    129 /* Only a producer may opt into mutation; normal analysis must leave published source images immutable. */
    130 if (check) {
    131 if (!inference_context_run(&context, error, sizeof(error))) result = 1;
    132 else printf("%s model ready adapter=%s sha256=%s\n", processor->application, context.model.adapter, context.model.graph_sha256);
    133 } else if (image) {
    134 if (!analyze_image(&context, processor, image, sequence, error, sizeof(error))) result = 1;
    135 else puts("OK");
    136 } else {
    137 puts("READY");
    138 if (fflush(stdout)) result = 1;
    139 uint32_t previous = 0;
    140 char command[INFERENCE_PATH_CAPACITY + 64];
    141 while (!result && fgets(command, sizeof(command), stdin)) {
    142 const size_t length = strlen(command);
    143 if (!length || command[length - 1] != '\n' || strncmp(command, "ANALYZE ", 8)) { result = 2; break; }
    144 command[length - 1] = '\0';
    145 char *number = command + 8, *path = strchr(number, ' ');
    146 if (!path) { result = 2; break; }
    147 *path++ = '\0';
    148 uint64_t parsed;
    149 /* Reject replay/reordering before running the graph, not after an old result has already been written. */
    150 if (!ini_uint(number, &parsed) || parsed <= previous || parsed > UINT32_MAX || !path[0] || strchr(path, '\r')) { result = 2; break; }
    151 previous = (uint32_t)parsed;
    152 const bool analyzed = analyze_image(&context, processor, path, previous, error, sizeof(error));
    153 if (!analyzed) fprintf(stderr, "%s: %s\n", processor->application, error);
    154 puts(analyzed ? "OK" : "ERROR");
    155 if (fflush(stdout)) result = 1;
    156 }
    157 if (ferror(stdin)) result = 1;
    158 }
    159 if (result == 1) fprintf(stderr, "%s: %s\n", processor->application, error);
    160 inference_context_close(&context);
    161 return result;
    162}
