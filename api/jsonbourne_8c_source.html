    18/**
    19 * @file jsonbourne.c
    20 * @brief Bounded, two-pass JSON parsing and iterative DOM operations.
    21 * @details Validation and materialization deliberately share the same grammar.
    22 * A counting pass establishes exact record/text requirements before one backing
    23 * allocation is split into aligned records and byte strings. Writers and copies
    24 * likewise reuse a single walker so their sizing and output passes agree.
    25 *
    26 * @par Implementation invariants
    27 * Cursors never exceed their extents; arithmetic is checked before advancing.
    28 * A NULL pool/output base means measurement, not allocation failure. Fixed
    29 * traversal frames replace recursion, and all per-call state remains local.
    30 * The API still requires accessible, stable input and DOM pointers: structural
    31 * checks cannot make arbitrary addresses safe to dereference.
    32 * @see jsonbourne.h for public ownership, argument and return-value contracts.
    33 */

    117/**
    118 * @brief Two-region arena shared by measurement and materialization.
    119 * @details Fixed-size DOM records are kept together so variable-length text
    120 * cannot misalign the next record. NULL bases count bytes without writing them.
    121 */
    122struct jsonbourne_pool {
    123 unsigned char *val_pool; /**< Record region, or NULL while measuring. */
    124 char *str_pool; /**< Text region, or NULL while measuring. */
    125 size_t val_len; /**< Record bytes counted/reserved so far. */
    126 size_t str_len; /**< Text bytes counted/reserved, including terminators. */
    127 size_t val_cap; /**< Maximum record bytes available in this pass. */
    128 size_t str_cap; /**< Maximum text bytes available in this pass. */
    129 size_t max_memory; /**< Shared bound across both regions. */
    130 jsonbourne_read_code code; /**< Arena failure carried separately from grammar errors. */
    131};

    133/**
    134 * @brief Reserve checked space in one arena region or merely account for it.
    135 * @param[in,out] pool Arena counters, capacities and sticky error state.
    136 * @param[in] len Bytes required, including any caller-supplied terminator space.
    137 * @param[in] is_str Nonzero selects text; zero selects typed records.
    138 * @return Region address, or NULL while measuring/on failure; inspect pool->code
    139 * to distinguish a successful count from an error.
    140 * @details Subtractions are guarded before use so neither capacity checks nor
    141 * the combined-region budget can wrap around size_t.
    142 */
    143static inline void *jsonbourne_pool_take(struct jsonbourne_pool *pool, size_t len, int is_str) {
    144 size_t *used = is_str ? &pool->str_len : &pool->val_len;
    145 const size_t other = is_str ? pool->val_len : pool->str_len;
    146 const size_t cap = is_str ? pool->str_cap : pool->val_cap;
    147 void *out = NULL;
    148 if (*used > cap || len > cap - *used || other > pool->max_memory ||
    149 *used > pool->max_memory - other || len > pool->max_memory - other - *used) {
    150 pool->code = JSONBOURNE_READ_ERROR_LIMIT;
    151 return NULL;
    152 }
    153 if (is_str && pool->str_pool)
    154 out = pool->str_pool + *used;
    155 if (!is_str && pool->val_pool)
    156 out = pool->val_pool + *used;
    157 *used += len;
    158 return out;
    159}

    161/**
    162 * @brief Account for or copy a byte extent into the text region.
    163 * @param[in,out] pool Destination arena or counting state.
    164 * @param[in] src Accessible source bytes; NULL is permitted only for len zero.
    165 * @param[in] len Exact byte count, independent of embedded NULs.
    166 * @return 1 on success, 0 when reservation fails.
    167 * @note Counting skips source reads; materialization uses a bounded copy loop.
    168 */
    169static inline int jsonbourne_pool_copy(struct jsonbourne_pool *pool, const char *src, size_t len) {
    170 char *dst = (char *)jsonbourne_pool_take(pool, len, 1);
    171 size_t idx;
    172 if (pool->code)
    173 return 0;
    174 if (dst)
    175 for (idx = 0; idx < len; ++idx)
    176 dst[idx] = src[idx];
    177 return 1;
    178}

    180/**
    181 * @brief Find bytes that need no string escaping or Unicode decoding.
    182 * @param[in] src Accessible input extent; no padding or alignment is assumed.
    183 * @param[in] len Available bytes.
    184 * @param[in] quote Current string delimiter, which terminates the fast path.
    185 * @return Prefix length containing printable ASCII other than quote/backslash.
    186 * @details Vector loads require a complete 16-byte block. The scalar tail keeps
    187 * page-boundary inputs safe; high-bit bytes return to strict UTF-8 validation.
    188 */
    189static inline size_t jsonbourne_ascii_span(const unsigned char *src, size_t len,
    190 unsigned char quote) {
    191 size_t pos = 0;
    192#if defined(JSONBOURNE_HAS_SSE2)
    193 const __m128i quotes = _mm_set1_epi8((char)quote);
    194 const __m128i slashes = _mm_set1_epi8('\\');
    195 const __m128i controls = _mm_set1_epi8(0x20);
    196 while (len - pos >= 16) {
    197 const __m128i block = _mm_loadu_si128((const __m128i *)(const void *)(src + pos));
    198 /* Signed lanes also classify 0x80..0xff as below 0x20, stopping before UTF-8. */
    199 const __m128i special =
    200 _mm_or_si128(_mm_or_si128(_mm_cmpeq_epi8(block, quotes), _mm_cmpeq_epi8(block, slashes)),
    201 _mm_cmplt_epi8(block, controls));
    202 const unsigned mask = (unsigned)_mm_movemask_epi8(special);
    203 if (mask)
    204 return pos + (size_t)__builtin_ctz(mask);
    205 pos += 16;
    206 }
    207#elif defined(JSONBOURNE_HAS_NEON)
    208 while (len - pos >= 16) {
    209 const uint8x16_t block = vld1q_u8(src + pos);
    210 const uint8x16_t special =
    211 vorrq_u8(vorrq_u8(vceqq_u8(block, vdupq_n_u8(quote)), vceqq_u8(block, vdupq_n_u8('\\'))),
    212 vorrq_u8(vcltq_u8(block, vdupq_n_u8(0x20)), vcgeq_u8(block, vdupq_n_u8(0x80))));
    213 if (vmaxvq_u8(special))
    214 break;
    215 pos += 16;
    216 }
    217#endif
    218 while (pos < len && src[pos] >= 0x20 && src[pos] < 0x80 && src[pos] != quote && src[pos] != '\\')
    219 ++pos;
    220 return pos;
    221}

    223/**
    224 * @brief Recognize only the four JSON whitespace bytes, independently of locale.
    225 * @param[in] chr Byte to classify.
    226 * @return Nonzero for space, tab, CR or LF; zero otherwise.

    228static inline int jsonbourne_char_is_space(unsigned char chr) {
    229 return chr == ' ' || chr == '\t' || chr == '\r' || chr == '\n';

    232/**
    233 * @brief Skip a bounded run of JSON whitespace with the available SIMD baseline.
    234 * @param[in] src Accessible input extent without padding requirements.
    235 * @param[in] len Bytes available for inspection.
    236 * @return Whitespace prefix length, never greater than len.
    237 * @note The first non-whitespace byte is left for the grammar, including UTF-8.
    238 */
    239static inline size_t jsonbourne_space_span(const unsigned char *src, size_t len) {
    240 size_t pos = 0;
    241#if defined(JSONBOURNE_HAS_SSE2)
    242 while (len - pos >= 16) {
    243 const __m128i block = _mm_loadu_si128((const __m128i *)(const void *)(src + pos));
    244 const __m128i spaces = _mm_or_si128(_mm_or_si128(_mm_cmpeq_epi8(block, _mm_set1_epi8(' ')),
    245 _mm_cmpeq_epi8(block, _mm_set1_epi8('\t'))),
    246 _mm_or_si128(_mm_cmpeq_epi8(block, _mm_set1_epi8('\r')),
    247 _mm_cmpeq_epi8(block, _mm_set1_epi8('\n'))));
    248 const unsigned mask = (unsigned)_mm_movemask_epi8(spaces);
    249 if (mask != 0xffffU)
    250 return pos + (size_t)__builtin_ctz((~mask) & 0xffffU);
    251 pos += 16;
    252 }
    253#elif defined(JSONBOURNE_HAS_NEON)
    254 while (len - pos >= 16) {
    255 const uint8x16_t block = vld1q_u8(src + pos);
    256 const uint8x16_t spaces =
    257 vorrq_u8(vorrq_u8(vceqq_u8(block, vdupq_n_u8(' ')), vceqq_u8(block, vdupq_n_u8('\t'))),
    258 vorrq_u8(vceqq_u8(block, vdupq_n_u8('\r')), vceqq_u8(block, vdupq_n_u8('\n'))));
    259 if (vminvq_u8(spaces) != 255)
    260 break;
    261 pos += 16;
    262 }
    263#endif
    264 while (pos < len && jsonbourne_char_is_space(src[pos]))
    265 ++pos;
    266 return pos;
    267}

    269/**
    270 * @brief Decode an ASCII hexadecimal digit without locale-sensitive conversion.
    271 * @param[in] chr Byte to inspect.
    272 * @return Integer 0 through 15, or -1 if the byte is not a hex digit.
    273 */
    274static inline int jsonbourne_hex_to_digit(unsigned char chr) {
    275 if (chr >= '0' && chr <= '9')
    276 return chr - '0';
    277 if (chr >= 'a' && chr <= 'f')
    278 return chr - 'a' + 10;
    279 if (chr >= 'A' && chr <= 'F')
    280 return chr - 'A' + 10;
    281 return -1;
    282}

    291/**
    292 * @brief Classify this reader's relaxed, ASCII-only unquoted-key alphabet.
    293 * @param[in] chr Byte to inspect.
    294 * @return Nonzero for letters, digits, underscore or dollar sign.
    295 * @note This convenience alphabet is not a complete ECMAScript IdentifierName

    300 jsonbourne_char_is_digit(chr) || chr == '_' || chr == '$';

    303/**
    304 * @brief Validate exactly one UTF-8 scalar without reading beyond the extent.
    305 * @param[in] src First byte of a possible UTF-8 sequence.
    306 * @param[in] len Available bytes, possibly fewer than a complete sequence.
    307 * @return Sequence length 1 through 4, or zero for an invalid/truncated sequence.
    308 * @details Lead-specific second-byte bounds reject overlong encodings, UTF-16
    309 * surrogate code points and values above U+10FFFF, not just bad continuation bits.
    310 */
    311static inline size_t jsonbourne_utf8_len(const unsigned char *src, size_t len) {
    312 unsigned char first;
    313 size_t count;
    314 size_t idx;
    315 if (!len)
    316 return 0;
    317 first = src[0];
    318 if (first < 0x80)
    319 return 1;
    320 if (first >= 0xc2 && first <= 0xdf)
    321 count = 2;
    322 else if (first >= 0xe0 && first <= 0xef)
    323 count = 3;
    324 else if (first >= 0xf0 && first <= 0xf4)
    325 count = 4;
    326 else
    327 return 0;
    328 if (len < count)
    329 return 0;
    330 for (idx = 1; idx < count; ++idx)
    331 if ((src[idx] & 0xc0) != 0x80)
    332 return 0;
    333 if ((first == 0xe0 && src[1] < 0xa0) || (first == 0xed && src[1] >= 0xa0) ||
    334 (first == 0xf0 && src[1] < 0x90) || (first == 0xf4 && src[1] >= 0x90))
    335 return 0;
    336 return count;
    337}

    339/**
    340 * @brief Recognize a numeric prefix without converting or rounding its value.
    341 * @param[in] src Accessible token candidate.
    342 * @param[in] len Candidate extent, which may also include following input.
    343 * @param[in] flg Enabled numeric spelling extensions.
    344 * @param[out] read_len Consumed token bytes on success; unchanged on failure.
    345 * @return 1 for a valid numeric prefix, 0 for invalid/incomplete grammar.
    346 * @note The caller must check the following delimiter or require read_len == len;
    347 * otherwise a valid prefix could conceal an invalid suffix.
    348 */
    349static inline int jsonbourne_read_num_token(const unsigned char *src, size_t len,
    350 jsonbourne_read_flag flg, size_t *read_len) {
    351 size_t pos = 0;
    352 size_t int_pos;
    353 size_t int_len;
    354 size_t frac_len = 0;
    355 int decimal = 0;
    356 if (!len)
    357 return 0;
    358 if (src[pos] == '-' || src[pos] == '+') {
    359 if (src[pos] == '+' && !(flg & JSONBOURNE_READ_ALLOW_PLUS_SIGN))
    360 return 0;
    361 if (++pos == len)
    362 return 0;
    363 }
    364 if (flg & JSONBOURNE_READ_ALLOW_INF_AND_NAN) {
    365 if (len - pos >= 8 && memcmp(src + pos, "Infinity", 8) == 0) {
    366 *read_len = pos + 8;
    367 return 1;
    368 }
    369 if (len - pos >= 3 && memcmp(src + pos, "NaN", 3) == 0) {
    370 *read_len = pos + 3;
    371 return 1;
    372 }
    373 }
    374 if ((flg & JSONBOURNE_READ_ALLOW_HEX_NUM) && len - pos >= 2 && src[pos] == '0' &&
    375 (src[pos + 1] == 'x' || src[pos + 1] == 'X')) {
    376 pos += 2;
    377 int_pos = pos;
    378 while (pos < len && jsonbourne_hex_to_digit(src[pos]) >= 0)
    379 ++pos;
    380 if (pos == int_pos)
    381 return 0;
    382 *read_len = pos;
    383 return 1;
    384 }
    385 int_pos = pos;
    386 while (pos < len && jsonbourne_char_is_digit(src[pos]))
    387 ++pos;
    388 int_len = pos - int_pos;
    389 if (int_len > 1 && src[int_pos] == '0')
    390 return 0;
    391 if (pos < len && src[pos] == '.') {
    392 size_t frac_pos = ++pos;
    393 decimal = 1;
    394 while (pos < len && jsonbourne_char_is_digit(src[pos]))
    395 ++pos;
    396 frac_len = pos - frac_pos;
    397 }
    398 if (!int_len && !frac_len)
    399 return 0;
    400 if ((!int_len || (decimal && !frac_len)) && !(flg & JSONBOURNE_READ_ALLOW_DECIMAL_POINT))
    401 return 0;
    402 if (pos < len && (src[pos] == 'e' || src[pos] == 'E')) {
    403 size_t exp_pos;
    404 ++pos;
    405 if (pos < len && (src[pos] == '+' || src[pos] == '-'))
    406 ++pos;
    407 exp_pos = pos;
    408 while (pos < len && jsonbourne_char_is_digit(src[pos]))
    409 ++pos;
    410 if (pos == exp_pos)
    411 return 0;
    412 }
    413 *read_len = pos;
    414 return 1;
    415}

    432/**
    433 * @brief One open container, replacing an equivalent recursive parser call.
    434 * @details state 0 permits an initial member or empty closure; state 1 expects
    435 * an item after a separator; state 2 follows an item. Keeping 1 distinct from 0
    436 * is what lets closure distinguish a trailing comma from an empty container.
    437 */
    438struct jsonbourne_read_frame {
    439 void *ctn; /**< Object/array payload, NULL during measurement. */
    440 void *tail; /**< Last entry for constant-time ordered appends. */
    441 size_t len; /**< Completed/reserved member or element count. */
    442 unsigned char type; /**< Object or array tag. */
    443 unsigned char state; /**< Initial, after-separator or after-item state. */
    444 unsigned char is_global; /**< Brace-less object closes at end of input. */
    445};

    447/**
    448 * @brief Preserve the first grammar failure while supporting short-circuit returns.
    449 * @param[in,out] ctx Read context whose code may still indicate success.
    450 * @param[in] code Failure to record if no earlier grammar error exists.
    451 * @return Always zero, allowing direct use in a failing parser expression.
    452 */
    453static inline int jsonbourne_set_read_error(struct jsonbourne_read_ctx *ctx,
    454 jsonbourne_read_code code) {
    455 if (!ctx->code)
    456 ctx->code = code;
    457 return 0;
    458}

    460/**
    461 * @brief Advance a lazy location cursor instead of rescanning each node's prefix.
    462 * @param[in,out] ctx Materializing read state with cached line information.
    463 * @param[in] pos Requested input byte offset; must not precede ctx->loc_pos.
    464 * @param[out] line One-based line number at pos.
    465 * @param[out] row Zero-based byte column at pos.
    466 * @note Monotonic requests keep total location bookkeeping linear in input size.
    467 */
    468static inline void jsonbourne_read_locate(struct jsonbourne_read_ctx *ctx, size_t pos, size_t *line,
    469 size_t *row) {
    470 while (ctx->loc_pos < pos) {
    471 if (ctx->src[ctx->loc_pos++] == '\n') {
    472 ++ctx->line;
    473 ctx->line_pos = ctx->loc_pos;
    474 }
    475 }
    476 *line = ctx->line;
    477 *row = pos - ctx->line_pos;
    478}

    480/**
    481 * @brief Consume whitespace and enabled comments without swallowing grammar bytes.
    482 * @param[in,out] ctx Input cursor and enabled syntax options.
    483 * @return 1 at the next significant byte/end, 0 for an unterminated block comment.
    484 * @note A failing block comment restores its opening offset so diagnostics point
    485 * at the construct that needs repair rather than merely at end of input.
    486 */
    487static inline int jsonbourne_skip_trivia(struct jsonbourne_read_ctx *ctx) {
    488 for (;;) {
    489 ctx->pos += jsonbourne_space_span(ctx->src + ctx->pos, ctx->len - ctx->pos);
    490 if (!(ctx->flg & JSONBOURNE_READ_ALLOW_COMMENTS) || ctx->len - ctx->pos < 2 ||
    491 ctx->src[ctx->pos] != '/')
    492 return 1;
    493 if (ctx->src[ctx->pos + 1] == '/') {
    494 ctx->pos += 2;
    495 while (ctx->pos < ctx->len && ctx->src[ctx->pos] != '\n' && ctx->src[ctx->pos] != '\r')
    496 ++ctx->pos;
    497 } else if (ctx->src[ctx->pos + 1] == '*') {
    498 const size_t start = ctx->pos;
    499 ctx->pos += 2;
    500 while (ctx->len - ctx->pos >= 2 &&
    501 !(ctx->src[ctx->pos] == '*' && ctx->src[ctx->pos + 1] == '/'))
    502 ++ctx->pos;
    503 if (ctx->len - ctx->pos < 2) {
    504 ctx->pos = start;
    505 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_COMMENT);
    506 }
    507 ctx->pos += 2;
    508 } else
    509 return 1;
    510 }
    511}

    513/**
    514 * @brief Enforce a decoded-string budget before reserving any more text bytes.
    515 * @param[in,out] ctx Parser and arena state.
    516 * @param[in] start Text counter at the beginning of the current decoded string.
    517 * @param[in] bytes Decoded source bytes to append when materializing.
    518 * @param[in] count Number of bytes, excluding any final terminator.
    519 * @return 1 if appended/counted, 0 for a string or arena limit failure.
    520 * @note The budget follows decoded size, so escapes cannot bypass it.
    521 */
    522static inline int jsonbourne_read_str_bytes(struct jsonbourne_read_ctx *ctx, size_t start,
    523 const char *bytes, size_t count) {
    524 if (ctx->pool.str_len - start > ctx->limits.max_str_len ||
    525 count > ctx->limits.max_str_len - (ctx->pool.str_len - start))
    526 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_LIMIT);
    527 return jsonbourne_pool_copy(&ctx->pool, bytes, count);
    528}

    530/**
    531 * @brief Consume the four hexadecimal digits following a Unicode escape marker.
    532 * @param[in,out] ctx Cursor positioned at the first hex digit.
    533 * @param[out] unit Decoded UTF-16 code unit; undefined as a result on failure.
    534 * @return 1 for four valid digits, 0 with INVALID_ESCAPE otherwise.
    535 * @note A code unit is not necessarily a scalar; the string reader pairs surrogates.
    536 */
    537static inline int jsonbourne_read_hex4(struct jsonbourne_read_ctx *ctx, uint32_t *unit) {
    538 size_t idx;
    539 *unit = 0;
    540 if (ctx->len - ctx->pos < 4)
    541 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    542 for (idx = 0; idx < 4; ++idx) {
    543 const int digit = jsonbourne_hex_to_digit(ctx->src[ctx->pos]);
    544 if (digit < 0)
    545 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    546 *unit = (*unit << 4) | (unsigned)digit;
    547 ++ctx->pos;
    548 }
    549 return 1;
    550}

    552/**
    553 * @brief Encode a validated Unicode scalar into the decoded text arena.
    554 * @param[in,out] ctx Parser/arena and decoded-string budget.
    555 * @param[in] start Current string's initial text counter.
    556 * @param[in] codepoint Scalar already checked for surrogate/range validity.
    557 * @return 1 if its UTF-8 bytes fit, 0 on an arena/string limit.
    558 * @pre codepoint is at most U+10FFFF and is not a surrogate code point.
    559 */
    560static inline int jsonbourne_read_utf8_encode(struct jsonbourne_read_ctx *ctx, size_t start,
    561 uint32_t codepoint) {
    562 char bytes[4];
    563 size_t len;
    564 if (codepoint < 0x80) {
    565 bytes[0] = (char)codepoint;
    566 len = 1;
    567 } else if (codepoint < 0x800) {
    568 bytes[0] = (char)(0xc0 | (codepoint >> 6));
    569 bytes[1] = (char)(0x80 | (codepoint & 0x3f));
    570 len = 2;
    571 } else if (codepoint < 0x10000) {
    572 bytes[0] = (char)(0xe0 | (codepoint >> 12));
    573 bytes[1] = (char)(0x80 | ((codepoint >> 6) & 0x3f));
    574 bytes[2] = (char)(0x80 | (codepoint & 0x3f));
    575 len = 3;
    576 } else {
    577 bytes[0] = (char)(0xf0 | (codepoint >> 18));
    578 bytes[1] = (char)(0x80 | ((codepoint >> 12) & 0x3f));
    579 bytes[2] = (char)(0x80 | ((codepoint >> 6) & 0x3f));
    580 bytes[3] = (char)(0x80 | (codepoint & 0x3f));
    581 len = 4;
    582 }
    583 return jsonbourne_read_str_bytes(ctx, start, bytes, len);
    584}

    586/**
    587 * @brief Decode one quoted string or enabled unquoted key through the same two passes.
    588 * @param[in,out] ctx Cursor at the string/key and arena state.
    589 * @param[in] key Nonzero permits the unquoted-key path when its option is enabled.
    590 * @param[out] out String record, or NULL on a successful counting pass.
    591 * @return 1 for a complete decoded string, 0 with parser/arena failure recorded.
    592 * @details Escape decoding and raw UTF-8 validation converge on the same decoded
    593 * byte budget. A separate trailing NUL is stored without counting it in len.
    594 */
    595static inline int jsonbourne_read_str(struct jsonbourne_read_ctx *ctx, int key,
    596 jsonbourne_str **out) {
    597 const size_t pos = ctx->pos;
    598 const size_t start = ctx->pool.str_len;
    599 const int has_loc = (ctx->flg & JSONBOURNE_READ_TRACK_LOCATION) != 0;
    600 jsonbourne_str *str = (jsonbourne_str *)jsonbourne_pool_take(
    601 &ctx->pool, has_loc ? sizeof(struct jsonbourne_str_loc) : sizeof(struct jsonbourne_str), 0);
    602 unsigned char quote;
    603 int complete = 0;
    604 if (ctx->pool.code)
    605 return 0;
    606 if (ctx->pos >= ctx->len)
    607 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    608 quote = ctx->src[ctx->pos];
    609 if (quote != '"' && !(quote == '\'' && (ctx->flg & JSONBOURNE_READ_ALLOW_SINGLE_QUOTED_STR))) {
    610 size_t len;
    611 if (!key || !(ctx->flg & JSONBOURNE_READ_ALLOW_UNQUOTED_KEY))
    612 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_EXPECTED_KEY);
    613 while (ctx->pos < ctx->len && jsonbourne_char_is_id(ctx->src[ctx->pos]))
    614 ++ctx->pos;
    615 len = ctx->pos - pos;
    616 if (!len)
    617 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_EXPECTED_KEY);
    618 if (!jsonbourne_read_str_bytes(ctx, start, (const char *)ctx->src + pos, len))
    619 return 0;
    620 complete = 1;
    621 } else {
    622 ++ctx->pos;
    623 while (ctx->pos < ctx->len) {
    624 const size_t run = jsonbourne_ascii_span(ctx->src + ctx->pos, ctx->len - ctx->pos, quote);
    625 unsigned char chr;
    626 if (!jsonbourne_read_str_bytes(ctx, start, (const char *)ctx->src + ctx->pos, run))
    627 return 0;
    628 ctx->pos += run;
    629 if (ctx->pos == ctx->len)
    630 break;
    631 chr = ctx->src[ctx->pos++];
    632 if (chr == quote) {
    633 complete = 1;
    634 break;
    635 }
    636 if (chr == '\\') {
    637 char decoded;
    638 if (ctx->pos == ctx->len)
    639 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    640 chr = ctx->src[ctx->pos++];
    641 switch (chr) {
    642 case '"':
    643 case '\\':
    644 case '/':
    645 decoded = (char)chr;
    646 break;
    647 case 'b':
    648 decoded = '\b';
    649 break;
    650 case 'f':
    651 decoded = '\f';
    652 break;
    653 case 'n':
    654 decoded = '\n';
    655 break;
    656 case 'r':
    657 decoded = '\r';
    658 break;
    659 case 't':
    660 decoded = '\t';
    661 break;
    662 case '\'':
    663 if (!(ctx->flg & JSONBOURNE_READ_ALLOW_SINGLE_QUOTED_STR))
    664 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    665 decoded = '\'';
    666 break;
    667 case 'u': {
    668 uint32_t codepoint;
    669 if (!jsonbourne_read_hex4(ctx, &codepoint))
    670 return 0;
    671 /* UTF-16 halves are valid only as a pair; neither half may leak into UTF-8. */
    672 if (codepoint >= 0xd800 && codepoint <= 0xdbff) {
    673 uint32_t low;
    674 if (ctx->len - ctx->pos < 2 || ctx->src[ctx->pos] != '\\' ||
    675 ctx->src[ctx->pos + 1] != 'u')
    676 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    677 ctx->pos += 2;
    678 if (!jsonbourne_read_hex4(ctx, &low))
    679 return 0;
    680 if (low < 0xdc00 || low > 0xdfff)
    681 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    682 codepoint = 0x10000 + ((codepoint - 0xd800) << 10) + low - 0xdc00;
    683 } else if (codepoint >= 0xdc00 && codepoint <= 0xdfff)
    684 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    685 if (!jsonbourne_read_utf8_encode(ctx, start, codepoint))
    686 return 0;
    687 continue;
    688 }
    689 case '\n':
    690 case '\r':
    691 if (!(ctx->flg & JSONBOURNE_READ_ALLOW_MULTILINE_STR))
    692 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    693 if (chr == '\r' && ctx->pos < ctx->len && ctx->src[ctx->pos] == '\n')
    694 ++ctx->pos;
    695 continue;
    696 default:
    697 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_ESCAPE);
    698 }
    699 if (!jsonbourne_read_str_bytes(ctx, start, &decoded, 1))
    700 return 0;
    701 } else if (chr >= 0x80) {
    702 const size_t position = ctx->pos - 1;
    703 const size_t len = jsonbourne_utf8_len(ctx->src + position, ctx->len - position);
    704 if (!len) {
    705 ctx->pos = position;
    706 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_UTF8);
    707 }
    708 if (!jsonbourne_read_str_bytes(ctx, start, (const char *)ctx->src + position, len))
    709 return 0;
    710 ctx->pos = position + len;
    711 } else {
    712 const char decoded = (char)chr;
    713 if ((chr != '\n' && chr != '\r') || !(ctx->flg & JSONBOURNE_READ_ALLOW_MULTILINE_STR))
    714 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_STRING);
    715 if (!jsonbourne_read_str_bytes(ctx, start, &decoded, 1))
    716 return 0;
    717 }
    718 }
    719 }
    720 if (!complete)
    721 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    722 if (!jsonbourne_pool_copy(&ctx->pool, "", 1))
    723 return 0;
    724 if (str) {
    725 str->str = ctx->pool.str_pool + start;
    726 str->len = ctx->pool.str_len - start - 1;
    727 if (has_loc) {
    728 jsonbourne_str_loc *extended = (jsonbourne_str_loc *)str;
    729 extended->pos = pos;
    730 jsonbourne_read_locate(ctx, pos, &extended->line, &extended->col);
    731 }
    732 }
    733 *out = str;
    734 return 1;
    735}

    737/**
    738 * @brief Verify the token boundary separately from the token's own spelling.
    739 * @param[in] ctx Input extent and comment policy.
    740 * @param[in] pos Candidate end offset, no greater than ctx->len.
    741 * @return Nonzero at input end or a permissible token boundary.
    742 * @details Prevents a valid prefix such as true in truex, or 1 in 1abc, from
    743 * being admitted as a complete scalar value.
    744 */
    745static inline int jsonbourne_is_delimiter(const struct jsonbourne_read_ctx *ctx, size_t pos) {
    746 unsigned char chr;
    747 if (pos == ctx->len)
    748 return 1;
    749 chr = ctx->src[pos];
    750 return jsonbourne_char_is_space(chr) || chr == ',' || chr == ']' || chr == '}' ||
    751 (chr == '/' && (ctx->flg & JSONBOURNE_READ_ALLOW_COMMENTS));
    752}

    754/**
    755 * @brief Consume a scalar or push a new container without a recursive call.
    756 * @param[in,out] ctx Read cursor, budgets and arena.
    757 * @param[out] out Result record, or NULL while counting.
    758 * @param[in,out] stack Fixed storage for open containers.
    759 * @param[in,out] depth Number of active frames; checked before every push.
    760 * @param[in] is_global Nonzero requests a brace-less object at the current cursor.
    761 * @return 1 for a scalar/opened container, 0 for grammar or resource failure.
    762 * @note Containers are completed by jsonbourne_read_document(), allowing nesting
    763 * to consume bounded frame storage rather than the C call stack.
    764 */
    765static inline int jsonbourne_read_val(struct jsonbourne_read_ctx *ctx, jsonbourne_val **out,
    766 struct jsonbourne_read_frame *stack, size_t *depth,
    767 int is_global) {
    768 const int has_loc = (ctx->flg & JSONBOURNE_READ_TRACK_LOCATION) != 0;
    769 jsonbourne_val *val;
    770 unsigned char chr;
    771 if (ctx->val_count == ctx->limits.max_vals)
    772 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_LIMIT);
    773 ++ctx->val_count;
    774 val = (jsonbourne_val *)jsonbourne_pool_take(
    775 &ctx->pool, has_loc ? sizeof(struct jsonbourne_val_loc) : sizeof(struct jsonbourne_val), 0);
    776 if (ctx->pool.code)
    777 return 0;
    778 if (val) {
    779 val->ptr = NULL;
    780 if (has_loc) {
    781 jsonbourne_val_loc *extended = (jsonbourne_val_loc *)val;
    782 extended->pos = ctx->pos;
    783 jsonbourne_read_locate(ctx, ctx->pos, &extended->line, &extended->col);
    784 }
    785 }
    786 *out = val;
    787 if (ctx->pos == ctx->len && !is_global)
    788 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    789 chr = ctx->pos < ctx->len ? ctx->src[ctx->pos] : 0;
    790 if (is_global || chr == '{' || chr == '[') {
    791 struct jsonbourne_read_frame *frame;
    792 const int obj = is_global || chr == '{';
    793 void *ctn;
    794 if (*depth == ctx->limits.max_depth || *depth == JSONBOURNE_READER_DEPTH_LIMIT)
    795 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_DEPTH);
    796 ctn = jsonbourne_pool_take(
    797 &ctx->pool, obj ? sizeof(struct jsonbourne_obj) : sizeof(struct jsonbourne_arr), 0);
    798 if (ctx->pool.code)
    799 return 0;
    800 if (val) {
    801 val->type = obj ? JSONBOURNE_TYPE_OBJ : JSONBOURNE_TYPE_ARR;
    802 val->ptr = ctn;
    803 }
    804 frame = &stack[(*depth)++];
    805 frame->ctn = ctn;
    806 frame->tail = NULL;
    807 frame->len = 0;
    808 frame->type = (unsigned char)(obj ? JSONBOURNE_TYPE_OBJ : JSONBOURNE_TYPE_ARR);
    809 frame->state = 0;
    810 frame->is_global = (unsigned char)is_global;
    811 if (ctn) {
    812 if (obj)
    813 ((jsonbourne_obj *)ctn)->first = NULL;
    814 else
    815 ((jsonbourne_arr *)ctn)->first = NULL;
    816 }
    817 if (!is_global)
    818 ++ctx->pos;
    819 return 1;
    820 }
    821 if (chr == '"' || (chr == '\'' && (ctx->flg & JSONBOURNE_READ_ALLOW_SINGLE_QUOTED_STR))) {
    822 jsonbourne_str *str;
    823 if (!jsonbourne_read_str(ctx, 0, &str))
    824 return 0;
    825 if (val) {
    826 val->type = JSONBOURNE_TYPE_STR;
    827 val->ptr = str;
    828 }
    829 return 1;
    830 }
    831 if (chr == 't' || chr == 'f' || chr == 'n') {
    832 const char *literal = chr == 't' ? "true" : chr == 'f' ? "false" : "null";
    833 const size_t len = chr == 'f' ? 5 : 4;
    834 if (ctx->len - ctx->pos < len || memcmp(ctx->src + ctx->pos, literal, len) != 0 ||
    835 !jsonbourne_is_delimiter(ctx, ctx->pos + len))
    836 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_LITERAL);
    837 ctx->pos += len;
    838 if (val)
    839 val->type = chr == 't' ? JSONBOURNE_TYPE_TRUE
    840 : chr == 'f' ? JSONBOURNE_TYPE_FALSE
    841 : JSONBOURNE_TYPE_NULL;
    842 return 1;
    843 }
    844 {
    845 size_t len = 0;
    846 const size_t start = ctx->pool.str_len;
    847 jsonbourne_num *num;
    848 if (!jsonbourne_read_num_token(ctx->src + ctx->pos, ctx->len - ctx->pos, ctx->flg, &len) ||
    849 !jsonbourne_is_delimiter(ctx, ctx->pos + len))
    850 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_INVALID_NUMBER);
    851 num = (jsonbourne_num *)jsonbourne_pool_take(&ctx->pool, sizeof(struct jsonbourne_num), 0);
    852 if (ctx->pool.code ||
    853 !jsonbourne_pool_copy(&ctx->pool, (const char *)ctx->src + ctx->pos, len) ||
    854 !jsonbourne_pool_copy(&ctx->pool, "", 1))
    855 return 0;
    856 ctx->pos += len;
    857 if (val) {
    858 val->type = JSONBOURNE_TYPE_NUM;
    859 val->ptr = num;
    860 num->str = ctx->pool.str_pool + start;
    861 num->len = len;
    862 }
    863 return 1;
    864 }
    865}

    867/**
    868 * @brief Drive one complete JSON document using explicit container frames.
    869 * @param[in,out] ctx Initialized counting or materializing read context.
    870 * @param[out] root Parsed root, or NULL while counting.
    871 * @return 1 only when the root and all trailing trivia consume the full input.
    872 * @details The frame state distinguishes empty closure, post-separator closure
    873 * and a completed item. Sharing this state machine across both passes prevents
    874 * the allocator's size prediction from using a different grammar.
    875 */
    876static inline int jsonbourne_read_document(struct jsonbourne_read_ctx *ctx, jsonbourne_val **root) {
    877 struct jsonbourne_read_frame stack[JSONBOURNE_READER_DEPTH_LIMIT];
    878 size_t depth = 0;
    879 int is_global;
    880 if (!jsonbourne_skip_trivia(ctx))
    881 return 0;
    882 is_global = (ctx->flg & JSONBOURNE_READ_ALLOW_GLOBAL_OBJECT) &&
    883 (ctx->pos == ctx->len || ctx->src[ctx->pos] != '{');
    884 if (!jsonbourne_read_val(ctx, root, stack, &depth, is_global))
    885 return 0;
    886 while (depth) {
    887 struct jsonbourne_read_frame *frame = &stack[depth - 1];
    888 jsonbourne_val *child = NULL;
    889 jsonbourne_val **dst = &child;
    890 const unsigned char close = frame->type == JSONBOURNE_TYPE_OBJ ? '}' : ']';
    891 int closing;
    892 if (!jsonbourne_skip_trivia(ctx))
    893 return 0;
    894 closing = frame->is_global ? ctx->pos == ctx->len
    895 : ctx->pos < ctx->len && ctx->src[ctx->pos] == close;
    896 if (frame->state == 2 && !closing) {
    897 if (ctx->pos == ctx->len)
    898 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    899 if (ctx->src[ctx->pos] == ',')
    900 ++ctx->pos;
    901 else if (!(ctx->flg & JSONBOURNE_READ_ALLOW_NO_COMMAS))
    902 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_EXPECTED_SEPARATOR);
    903 frame->state = 1;
    904 if (!jsonbourne_skip_trivia(ctx))
    905 return 0;
    906 closing = frame->is_global ? ctx->pos == ctx->len
    907 : ctx->pos < ctx->len && ctx->src[ctx->pos] == close;
    908 }
    909 if (closing) {
    910 if (frame->state == 1 && !(ctx->flg & JSONBOURNE_READ_ALLOW_TRAILING_COMMAS))
    911 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_EXPECTED_SEPARATOR);
    912 if (frame->ctn) {
    913 if (frame->type == JSONBOURNE_TYPE_OBJ)
    914 ((jsonbourne_obj *)frame->ctn)->len = frame->len;
    915 else
    916 ((jsonbourne_arr *)frame->ctn)->len = frame->len;
    917 }
    918 if (!frame->is_global)
    919 ++ctx->pos;
    920 --depth;
    921 continue;
    922 }
    923 if (ctx->pos == ctx->len)
    924 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    925 if (frame->len == SIZE_MAX)
    926 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_LIMIT);
    927 if (frame->type == JSONBOURNE_TYPE_OBJ) {
    928 jsonbourne_obj_entry *elem = (jsonbourne_obj_entry *)jsonbourne_pool_take(
    929 &ctx->pool, sizeof(struct jsonbourne_obj_entry), 0);
    930 jsonbourne_str *name;
    931 if (ctx->pool.code || !jsonbourne_read_str(ctx, 1, &name) || !jsonbourne_skip_trivia(ctx))
    932 return 0;
    933 if (ctx->pos == ctx->len)
    934 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_END);
    935 if (ctx->src[ctx->pos] != ':' &&
    936 !(ctx->src[ctx->pos] == '=' && (ctx->flg & JSONBOURNE_READ_ALLOW_EQUALS)))
    937 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_EXPECTED_COLON);
    938 ++ctx->pos;
    939 if (elem) {
    940 elem->key = name;
    941 elem->val = NULL;
    942 elem->next = NULL;
    943 if (frame->tail)
    944 ((jsonbourne_obj_entry *)frame->tail)->next = elem;
    945 else
    946 ((jsonbourne_obj *)frame->ctn)->first = elem;
    947 dst = &elem->val;
    948 }
    949 frame->tail = elem;
    950 } else {
    951 jsonbourne_arr_entry *elem = (jsonbourne_arr_entry *)jsonbourne_pool_take(
    952 &ctx->pool, sizeof(struct jsonbourne_arr_entry), 0);
    953 if (ctx->pool.code)
    954 return 0;
    955 if (elem) {
    956 elem->val = NULL;
    957 elem->next = NULL;
    958 if (frame->tail)
    959 ((jsonbourne_arr_entry *)frame->tail)->next = elem;
    960 else
    961 ((jsonbourne_arr *)frame->ctn)->first = elem;
    962 dst = &elem->val;
    963 }
    964 frame->tail = elem;
    965 }
    966 ++frame->len;
    967 frame->state = 2;
    968 if (!jsonbourne_skip_trivia(ctx) || !jsonbourne_read_val(ctx, dst, stack, &depth, 0))
    969 return 0;
    970 }
    971 if (!jsonbourne_skip_trivia(ctx))
    972 return 0;
    973 if (ctx->pos != ctx->len)
    974 return jsonbourne_set_read_error(ctx, JSONBOURNE_READ_ERROR_UNEXPECTED_CONTENT);
    975 return 1;

    1192/** @brief Suspended container traversal with constant-space sibling-cycle detection. */
    1193struct jsonbourne_walk_frame {
    1194 const jsonbourne_val *val; /**< Container owning this frame. */
    1195 const void *next; /**< Next unvisited member/element. */
    1196 const void *slow; /**< One-step Floyd cycle cursor. */
    1197 const void *fast; /**< Two-step Floyd cycle cursor. */
    1198 size_t remaining; /**< Declared list entries not yet visited. */
    1199 size_t idx; /**< Zero-based next entry index, used for separators. */
    1200};

    1202/**
    1203 * @brief Advance either entry type without relying on their layouts being interchangeable.
    1204 * @param[in] elem Accessible entry, or NULL at list end.
    1205 * @param[in] type Object tag selects object entries; array traversal uses array entries.
    1206 * @return Next entry, or NULL.
    1207 */
    1208static inline const void *jsonbourne_next(const void *elem, size_t type) {
    1209 if (!elem)
    1210 return NULL;
    1211 return type == JSONBOURNE_TYPE_OBJ ? (const void *)((const jsonbourne_obj_entry *)elem)->next
    1212 : (const void *)((const jsonbourne_arr_entry *)elem)->next;
    1213}

    1215/**
    1216 * @brief Advance Floyd's two cursors without allocating a visited-address set.
    1217 * @param[in,out] slow Cursor advanced by one sibling link.
    1218 * @param[in,out] fast Cursor advanced by two sibling links.
    1219 * @param[in] type Entry layout: object or array.
    1220 * @return 1 if no cycle is encountered on this step, 0 if non-NULL cursors meet.
    1221 * @note All linked entries must still be accessible; this validates topology,
    1222 * not the validity of arbitrary pointer addresses.
    1223 */
    1224static inline int jsonbourne_acyclic_step(const void **slow, const void **fast, size_t type) {
    1225 *slow = jsonbourne_next(*slow, type);
    1226 *fast = jsonbourne_next(jsonbourne_next(*fast, type), type);
    1227 return !*slow || *slow != *fast;
    1228}

    1230/**
    1231 * @brief Validate/traverse an accessible DOM with a fixed explicit stack.
    1232 * @param[in] val Root value to inspect.
    1233 * @param[in] visit Event sink receiving context, event, value, optional key,
    1234 * depth and entry index; returning zero aborts traversal.
    1235 * @param[in,out] ctx Opaque event-sink state.
    1236 * @return 1 for a complete accepted traversal, 0 for malformed structure, nesting
    1237 * exhaustion or a rejecting visitor.
    1238 * @details Exact list lengths and Floyd cursors detect malformed sibling chains.
    1239 * Cycles through value/container pointers cannot grow without bound because
    1240 * every nested container also consumes one checked frame.
    1241 */
    1242static inline int jsonbourne_walk(const jsonbourne_val *val,
    1243 int (*visit)(void *, enum jsonbourne_event,
    1244 const jsonbourne_val *, const jsonbourne_str *,
    1245 size_t, size_t),
    1246 void *ctx) {
    1247 struct jsonbourne_walk_frame stack[JSONBOURNE_WRITER_DEPTH_LIMIT];
    1248 size_t depth = 0;
    1249 if (!val)
    1250 return 0;
    1251 for (;;) {
    1252 if (val) {
    1253 if (val->type > JSONBOURNE_TYPE_NULL || (val->type <= JSONBOURNE_TYPE_ARR && !val->ptr))
    1254 return 0;
    1255 if (!visit(ctx, JSONBOURNE_EVENT_VAL, val, NULL, depth, 0))
    1256 return 0;
    1257 if (val->type == JSONBOURNE_TYPE_OBJ || val->type == JSONBOURNE_TYPE_ARR) {
    1258 struct jsonbourne_walk_frame *frame;
    1259 if (depth == JSONBOURNE_WRITER_DEPTH_LIMIT)
    1260 return 0;
    1261 frame = &stack[depth++];
    1262 frame->val = val;
    1263 if (val->type == JSONBOURNE_TYPE_OBJ) {
    1264 const jsonbourne_obj *obj = (const jsonbourne_obj *)val->ptr;
    1265 frame->remaining = obj->len;
    1266 frame->next = obj->first;
    1267 } else {
    1268 const jsonbourne_arr *arr = (const jsonbourne_arr *)val->ptr;
    1269 frame->remaining = arr->len;
    1270 frame->next = arr->first;
    1271 }
    1272 frame->slow = frame->next;
    1273 frame->fast = frame->next;
    1274 frame->idx = 0;
    1275 }
    1276 }
    1277 val = NULL;
    1278 if (!depth)
    1279 return 1;
    1280 {
    1281 struct jsonbourne_walk_frame *frame = &stack[depth - 1];
    1282 if (!frame->remaining) {
    1283 if (frame->next ||
    1284 !visit(ctx, JSONBOURNE_EVENT_END, frame->val, NULL, depth - 1, frame->idx))
    1285 return 0;
    1286 --depth;
    1287 } else {
    1288 const jsonbourne_str *name = NULL;
    1289 if (!frame->next)
    1290 return 0;
    1291 if (frame->val->type == JSONBOURNE_TYPE_OBJ) {
    1292 const jsonbourne_obj_entry *elem = (const jsonbourne_obj_entry *)frame->next;
    1293 name = elem->key;
    1294 val = elem->val;
    1295 if (!name)
    1296 return 0;
    1297 } else
    1298 val = ((const jsonbourne_arr_entry *)frame->next)->val;
    1299 if (!val)
    1300 return 0;
    1301 frame->next = jsonbourne_next(frame->next, frame->val->type);
    1302 if (!jsonbourne_acyclic_step(&frame->slow, &frame->fast, frame->val->type))
    1303 return 0;
    1304 if (!visit(ctx, name ? JSONBOURNE_EVENT_KEY : JSONBOURNE_EVENT_ELEM, frame->val, name,
    1305 depth, frame->idx))
    1306 return 0;
    1307 --frame->remaining;
    1308 ++frame->idx;
    1309 }
    1310 }
    1311 }
    1312}

    1346/**
    1347 * @brief Emit a UTF-8 string with JSON quoting and exact control-byte escaping.
    1348 * @param[in,out] out Measuring/writing output sink.
    1349 * @param[in] str Accessible length-aware string view, possibly containing NULs.
    1350 * @return 1 for valid text that fits, 0 for invalid UTF-8/arguments or capacity.
    1351 * @details Printable ASCII takes the span fast path; all remaining bytes are
    1352 * escaped or validated as complete scalars, never passed through blindly.
    1353 */
    1354static inline int jsonbourne_write_str(struct jsonbourne_write_ctx *out,
    1355 const jsonbourne_str *str) {
    1356 static const char digits[] = "0123456789abcdef";
    1357 size_t pos = 0;
    1358 if (!str || (!str->str && str->len) || str->len > (size_t)PTRDIFF_MAX ||
    1359 !jsonbourne_write_bytes(out, "\"", 1))
    1360 return 0;
    1361 while (pos < str->len) {
    1362 const unsigned char *src = (const unsigned char *)str->str;
    1363 const size_t run = jsonbourne_ascii_span(src + pos, str->len - pos, '"');
    1364 unsigned char chr;
    1365 if (!jsonbourne_write_bytes(out, str->str + pos, run))
    1366 return 0;
    1367 pos += run;
    1368 if (pos == str->len)
    1369 break;
    1370 chr = src[pos++];
    1371 if (chr == '"' || chr == '\\') {
    1372 const char escaped[2] = {'\\', (char)chr};
    1373 if (!jsonbourne_write_bytes(out, escaped, 2))
    1374 return 0;
    1375 } else if (chr < 0x20) {
    1376 const char *escape = chr == '\b' ? "\\b"
    1377 : chr == '\f' ? "\\f"
    1378 : chr == '\n' ? "\\n"
    1379 : chr == '\r' ? "\\r"
    1380 : chr == '\t' ? "\\t"
    1381 : NULL;
    1382 if (escape) {
    1383 if (!jsonbourne_write_bytes(out, escape, 2))
    1384 return 0;
    1385 } else {
    1386 const char escaped[6] = {'\\', 'u', '0', '0', digits[chr >> 4], digits[chr & 15]};
    1387 if (!jsonbourne_write_bytes(out, escaped, 6))
    1388 return 0;
    1389 }
    1390 } else {
    1391 const size_t len = jsonbourne_utf8_len(src + pos - 1, str->len - pos + 1);
    1392 if (!len || !jsonbourne_write_bytes(out, str->str + pos - 1, len))
    1393 return 0;
    1394 pos += len - 1;
    1395 }
    1396 }
    1397 return jsonbourne_write_bytes(out, "\"", 1);
    1398}

    1400/**
    1401 * @brief Convert hexadecimal magnitude text exactly using bounded decimal digits.
    1402 * @param[in,out] out Measuring/writing output sink.
    1403 * @param[in] src Validated hex digits without sign or 0x prefix.
    1404 * @param[in] len Hex digit count, including any leading zeroes.
    1405 * @return 1 for emitted decimal text, 0 on significant-digit/capacity limits.
    1406 * @details Each input digit updates a little-endian decimal accumulator by
    1407 * multiply-by-16 plus carry. This avoids floating-point loss and an unbounded
    1408 * big-integer dependency; two decimal slots per hex digit are a conservative bound.
    1409 */
    1410static inline int jsonbourne_write_hex_num(struct jsonbourne_write_ctx *out, const char *src,
    1411 size_t len) {
    1412 unsigned char decimal[JSONBOURNE_MAX_HEX_DIGITS * 2];
    1413 size_t used = 1;
    1414 size_t pos;
    1415 while (len && *src == '0') {
    1416 ++src;
    1417 --len;
    1418 }
    1419 if (len > JSONBOURNE_MAX_HEX_DIGITS)
    1420 return 0;
    1421 decimal[0] = 0;
    1422 for (pos = 0; pos < len; ++pos) {
    1423 unsigned carry = (unsigned)jsonbourne_hex_to_digit((unsigned char)src[pos]);
    1424 size_t idx;
    1425 for (idx = 0; idx < used; ++idx) {
    1426 carry += 16U * decimal[idx];
    1427 decimal[idx] = (unsigned char)(carry % 10U);
    1428 carry /= 10U;
    1429 }
    1430 while (carry) {
    1431 if (used == sizeof(decimal))
    1432 return 0;
    1433 decimal[used++] = (unsigned char)(carry % 10U);
    1434 carry /= 10U;
    1435 }
    1436 }
    1437 while (used) {
    1438 const char digit = (char)('0' + decimal[--used]);
    1439 if (!jsonbourne_write_bytes(out, &digit, 1))
    1440 return 0;
    1441 }
    1442 return 1;
    1443}

    1445/**
    1446 * @brief Preserve strict numeric text or normalize enabled relaxed spellings.
    1447 * @param[in,out] out Measuring/writing output sink.
    1448 * @param[in] num Accessible token view whose full extent must form one number.
    1449 * @return 1 for valid normalized output that fits, 0 otherwise.
    1450 * @details Strict decimals retain every original digit. Relaxed tokens lose a
    1451 * leading plus, gain missing decimal zeroes or use exact hex conversion. The
    1452 * retained nonfinite policy emits zero for NaN and signed DBL_MAX text for infinity.
    1453 */
    1454static inline int jsonbourne_write_num(struct jsonbourne_write_ctx *out,
    1455 const jsonbourne_num *num) {
    1456 size_t len = 0;
    1457 size_t pos = 0;
    1458 const unsigned char *src;
    1459 if (!num || !num->str || num->len > (size_t)PTRDIFF_MAX)
    1460 return 0;
    1461 if (jsonbourne_read_num_token((const unsigned char *)num->str, num->len, 0, &len) &&
    1462 len == num->len)
    1463 return jsonbourne_write_bytes(out, num->str, num->len);
    1464 if (!jsonbourne_read_num_token(
    1465 (const unsigned char *)num->str, num->len,
    1466 JSONBOURNE_READ_ALLOW_EXT_NUMBER | JSONBOURNE_READ_ALLOW_INF_AND_NAN, &len) ||
    1467 len != num->len)
    1468 return 0;
    1469 src = (const unsigned char *)num->str;
    1470 if (src[pos] == '+' || src[pos] == '-')
    1471 ++pos;
    1472 if (src[pos] == 'N')
    1473 return jsonbourne_write_bytes(out, "0", 1);
    1474 if (src[0] == '-' && !jsonbourne_write_bytes(out, "-", 1))
    1475 return 0;
    1476 if (src[pos] == 'I')
    1477 return jsonbourne_write_bytes(out, "1.7976931348623158e308", 22);
    1478 if (len - pos >= 2 && src[pos] == '0' && (src[pos + 1] == 'x' || src[pos + 1] == 'X'))
    1479 return jsonbourne_write_hex_num(out, num->str + pos + 2, len - pos - 2);
    1480 if (src[pos] == '.' && !jsonbourne_write_bytes(out, "0", 1))
    1481 return 0;
    1482 for (; pos < len; ++pos) {
    1483 if (!jsonbourne_write_bytes(out, num->str + pos, 1))
    1484 return 0;
    1485 if (src[pos] == '.' && (pos + 1 == len || src[pos + 1] == 'e' || src[pos + 1] == 'E') &&
    1486 !jsonbourne_write_bytes(out, "0", 1))
    1487 return 0;
    1488 }
    1489 return 1;
    1490}

    1512 * @brief Translate shared walker events into JSON punctuation and scalar text.

    1518 * @param[in] idx Entry index or completed count, used to avoid trailing separators.
    1519 * @return 1 if output remains valid and fits, 0 to stop traversal.
    1520 */

    1522 const jsonbourne_val *val, const jsonbourne_str *name,
    1523 size_t depth, size_t idx) {

    1527 return 0;

    1533 return 0;
    1534 return jsonbourne_write_bytes(out, val->type == JSONBOURNE_TYPE_OBJ ? "}" : "]", 1);
    1535 }
    1536 switch (val->type) {
    1537 case JSONBOURNE_TYPE_STR:
    1538 return jsonbourne_write_str(out, (const jsonbourne_str *)val->ptr);
    1539 case JSONBOURNE_TYPE_NUM:
    1540 return jsonbourne_write_num(out, (const jsonbourne_num *)val->ptr);
    1541 case JSONBOURNE_TYPE_OBJ:
    1542 return jsonbourne_write_bytes(out, "{", 1);
    1543 case JSONBOURNE_TYPE_ARR:
    1544 return jsonbourne_write_bytes(out, "[", 1);
    1545 case JSONBOURNE_TYPE_TRUE:
    1546 return jsonbourne_write_bytes(out, "true", 4);
    1547 case JSONBOURNE_TYPE_FALSE:
    1548 return jsonbourne_write_bytes(out, "false", 5);
    1549 case JSONBOURNE_TYPE_NULL:
    1550 return jsonbourne_write_bytes(out, "null", 4);
    1551 default:
    1552 return 0;
    1553 }
    1554}

    1705/** @brief Copy visitor state; the arena layout is shared with the parser. */

    1713/**
    1714 * @brief Validate and duplicate a decoded string instead of borrowing source text.
    1715 * @param[in,out] pool Destination arena or sizing state.
    1716 * @param[in] src Accessible decoded UTF-8 view.
    1717 * @return Copied record, or NULL during successful counting/on failure; inspect pool->code.
    1718 * @note A manually assembled DOM may contain invalid UTF-8 even if parser output
    1719 * cannot, so copying revalidates rather than trusting its provenance.
    1720 */
    1721static inline jsonbourne_str *jsonbourne_copy_str(struct jsonbourne_pool *pool,
    1722 const jsonbourne_str *src) {
    1723 const size_t start = pool->str_len;
    1724 jsonbourne_str *str;
    1725 size_t pos = 0;
    1726 if (!src || (!src->str && src->len) || src->len > (size_t)PTRDIFF_MAX) {
    1727 pool->code = JSONBOURNE_READ_ERROR_INVALID_STRING;
    1728 return NULL;
    1729 }
    1730 while (pos < src->len) {
    1731 const size_t len = jsonbourne_utf8_len((const unsigned char *)src->str + pos, src->len - pos);
    1732 if (!len) {
    1733 pool->code = JSONBOURNE_READ_ERROR_INVALID_UTF8;
    1734 return NULL;
    1735 }
    1736 pos += len;
    1737 }
    1738 str = (jsonbourne_str *)jsonbourne_pool_take(pool, sizeof(struct jsonbourne_str), 0);
    1739 if (pool->code || !jsonbourne_pool_copy(pool, src->str, src->len) ||
    1740 !jsonbourne_pool_copy(pool, "", 1))
    1741 return NULL;
    1742 if (str) {
    1743 str->str = pool->str_pool + start;
    1744 str->len = src->len;
    1745 }
    1746 return str;
    1747}

    1749/**
    1750 * @brief Build copied records from the same validated events used for writing.
    1751 * @param[in,out] ctx Copy visitor state and destination arena.
    1752 * @param[in] event Walker event selecting entry setup or value duplication.
    1753 * @param[in] src Accessible source value/container.
    1754 * @param[in] name Source object key for key events, NULL otherwise.
    1755 * @param[in] depth Source depth, also indexing destination append frames.
    1756 * @param[in] idx Source entry index, unused because destination tails track appends.
    1757 * @return 1 when duplication/counting succeeds, 0 on invalid text/number or limits.
    1758 * @details Entry events choose the next destination slot before the child-value
    1759 * event fills it. Measurement never dereferences those materialization-only slots.
    1760 */
    1761static inline int jsonbourne_copy_visit(void *ctx, enum jsonbourne_event event,
    1762 const jsonbourne_val *src, const jsonbourne_str *name,
    1763 size_t depth, size_t idx) {
    1764 struct jsonbourne_copy_ctx *copy_ctx = (struct jsonbourne_copy_ctx *)ctx;
    1765 struct jsonbourne_pool *pool = &copy_ctx->pool;
    1766 (void)idx;
    1767 if (event == JSONBOURNE_EVENT_END)
    1768 return 1;
    1769 if (event == JSONBOURNE_EVENT_KEY || event == JSONBOURNE_EVENT_ELEM) {
    1770 struct jsonbourne_copy_frame *frame = &copy_ctx->frames[depth - 1];
    1771 if (name) {
    1772 jsonbourne_obj_entry *elem = (jsonbourne_obj_entry *)jsonbourne_pool_take(
    1773 pool, sizeof(struct jsonbourne_obj_entry), 0);
    1774 jsonbourne_str *str = jsonbourne_copy_str(pool, name);
    1775 if (pool->code)
    1776 return 0;
    1777 if (elem) {
    1778 elem->key = str;
    1779 elem->next = NULL;
    1780 if (frame->tail)
    1781 ((jsonbourne_obj_entry *)frame->tail)->next = elem;
    1782 else
    1783 ((jsonbourne_obj *)frame->ctn)->first = elem;
    1784 frame->tail = elem;
    1785 copy_ctx->dst = &elem->val;
    1786 }
    1787 } else {
    1788 jsonbourne_arr_entry *elem = (jsonbourne_arr_entry *)jsonbourne_pool_take(
    1789 pool, sizeof(struct jsonbourne_arr_entry), 0);
    1790 if (pool->code)
    1791 return 0;
    1792 if (elem) {
    1793 elem->next = NULL;
    1794 if (frame->tail)
    1795 ((jsonbourne_arr_entry *)frame->tail)->next = elem;
    1796 else
    1797 ((jsonbourne_arr *)frame->ctn)->first = elem;
    1798 frame->tail = elem;
    1799 copy_ctx->dst = &elem->val;
    1800 }
    1801 }
    1802 return 1;
    1803 }
    1804 {
    1805 jsonbourne_val *val =
    1806 (jsonbourne_val *)jsonbourne_pool_take(pool, sizeof(struct jsonbourne_val), 0);
    1807 void *payload = NULL;
    1808 if (pool->code)
    1809 return 0;
    1810 if (val) {
    1811 *copy_ctx->dst = val;
    1812 val->type = src->type;
    1813 }
    1814 switch (src->type) {
    1815 case JSONBOURNE_TYPE_STR:
    1816 payload = jsonbourne_copy_str(pool, (const jsonbourne_str *)src->ptr);
    1817 break;
    1818 case JSONBOURNE_TYPE_NUM: {
    1819 const jsonbourne_num *num = (const jsonbourne_num *)src->ptr;
    1820 const size_t start = pool->str_len;
    1821 size_t len = 0;
    1822 jsonbourne_num *copy;
    1823 if (!num->str || num->len > (size_t)PTRDIFF_MAX ||
    1824 !jsonbourne_read_num_token(
    1825 (const unsigned char *)num->str, num->len,
    1826 JSONBOURNE_READ_ALLOW_EXT_NUMBER | JSONBOURNE_READ_ALLOW_INF_AND_NAN, &len) ||
    1827 len != num->len)
    1828 return 0;
    1829 copy = (jsonbourne_num *)jsonbourne_pool_take(pool, sizeof(struct jsonbourne_num), 0);
    1830 if (pool->code || !jsonbourne_pool_copy(pool, num->str, num->len) ||
    1831 !jsonbourne_pool_copy(pool, "", 1))
    1832 return 0;
    1833 if (copy) {
    1834 copy->str = pool->str_pool + start;
    1835 copy->len = num->len;
    1836 }
    1837 payload = copy;
    1838 break;
    1839 }
    1840 case JSONBOURNE_TYPE_OBJ:
    1841 payload = jsonbourne_pool_take(pool, sizeof(struct jsonbourne_obj), 0);
    1842 if (payload) {
    1843 ((jsonbourne_obj *)payload)->first = NULL;
    1844 ((jsonbourne_obj *)payload)->len = ((const jsonbourne_obj *)src->ptr)->len;
    1845 }
    1846 break;
    1847 case JSONBOURNE_TYPE_ARR:
    1848 payload = jsonbourne_pool_take(pool, sizeof(struct jsonbourne_arr), 0);
    1849 if (payload) {
    1850 ((jsonbourne_arr *)payload)->first = NULL;
    1851 ((jsonbourne_arr *)payload)->len = ((const jsonbourne_arr *)src->ptr)->len;
    1852 }
    1853 break;
    1854 default:
    1855 break;
    1856 }
    1857 if (pool->code)
    1858 return 0;
    1859 if (val)
    1860 val->ptr = payload;
    1861 if ((src->type == JSONBOURNE_TYPE_OBJ || src->type == JSONBOURNE_TYPE_ARR) &&
    1862 depth < JSONBOURNE_WRITER_DEPTH_LIMIT) {
    1863 copy_ctx->frames[depth].ctn = payload;
    1864 copy_ctx->frames[depth].tail = NULL;
    1865 }
    1866 return 1;
    1867 }
    1868}

    1940/**
    1941 * @brief Share exact-byte member comparison while keeping ambiguity policy explicit.
    1942 * @param[in] val Accessible object value.
    1943 * @param[in] key Decoded key bytes; NULL only with len zero.
    1944 * @param[in] len Key byte count, including any embedded NULs.
    1945 * @param[in] unique Nonzero scans for exactly one match; zero returns the first.
    1946 * @return Borrowed matching value, or NULL for absent/ambiguous/invalid traversal.
    1947 * @note Unique lookup must inspect the whole declared list: a later escaped
    1948 * spelling can decode to the same key as an earlier member.
    1949 */
    1950static inline const jsonbourne_val *jsonbourne_obj_lookup(const jsonbourne_val *val,
    1951 const char *key, size_t len, int unique) {
    1952 const jsonbourne_obj *obj;
    1953 const jsonbourne_obj_entry *elem;
    1954 const jsonbourne_val *found = NULL;
    1955 const void *slow;
    1956 const void *fast;
    1957 size_t idx;
    1958 if (!val || val->type != JSONBOURNE_TYPE_OBJ || !val->ptr || (!key && len) ||
    1959 len > (size_t)PTRDIFF_MAX)
    1960 return NULL;
    1961 obj = (const jsonbourne_obj *)val->ptr;
    1962 elem = obj->first;
    1963 slow = elem;
    1964 fast = elem;
    1965 for (idx = 0; idx < obj->len; ++idx) {
    1966 if (!elem || !elem->key || (!elem->key->str && elem->key->len))
    1967 return NULL;
    1968 if (elem->key->len == len && (!len || memcmp(elem->key->str, key, len) == 0)) {
