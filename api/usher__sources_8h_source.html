    19 * @file usher_sources.h
    20 * @brief Convert validated USHER AI Camera evidence into GREAT source records.
    21 * @details Inputs must be real bounded JPEGs with complete EXIF, profile-matched
    22 * dimensions, and qualified pose evidence. Synthetic map imagery and nominal
    23 * lens-only geometry are rejected rather than promoted into survey sources.
    24 */

    36/**
    37 * @brief Read a photo snapshot and verify each source against profile/evidence.
    38 * @param[in] config GREAT session settings, measured camera profile and source directory.
    39 * @param[in] manifest Bounded `USHER_AICAM_V1` filename snapshot.
    40 * @return Fully validated sources with camera geometry and capture provenance.
    41 * @throws std::runtime_error If any JPEG, EXIF field, profile, frame identity,
    42 * pose qualification, or shared-coverage check fails.
    43 * @details Exposure-aligned pose is required unless configuration explicitly
    44 * admits the recorded-pose approximation policy.
    45 */
    46inline std::vector<OrthoSource> readUsherSources(const MapConfig& config, const std::filesystem::path& manifest) {
    47 usher_camera_profile profile{};
    48 if (!usher_camera_profile_read(config.cameraProfile.c_str(), &profile) || profile.camera_id != 2)
    49 throw std::runtime_error("GREAT requires an AI Camera profile with filename_prefix=a");
    50 const auto& calibration = profile.calibration;
    51 std::ifstream input(manifest);
    52 std::string name;
    53 if (!std::getline(input, name) || name != "USHER_AICAM_V1")
    54 throw std::runtime_error("AI Camera input requires a GREAT session photo snapshot");
    55 std::vector<OrthoSource> result;
    56 std::map<std::string, bool> names;
    57 while (std::getline(input, name)) {
    58 if (result.size() == 4096 || name.empty() || name.size() > 160 || name.front() != 'a' ||
    59 name.find_first_not_of("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789_.-") != std::string::npos ||
    60 std::filesystem::path(name).extension() != ".jpg" || !names.emplace(name, true).second)
    61 throw std::runtime_error("Invalid or duplicate AI Camera photo basename");
    62 OrthoSource source;
    63 source.path = (config.sourceDirectory.empty() ? config.photoDirectory : config.sourceDirectory) / name;
    64 struct stat status{};
    65 if (lstat(source.path.c_str(), &status) || !S_ISREG(status.st_mode) || status.st_size <= 0 ||
    66 status.st_size > 128 * 1024 * 1024)
    67 throw std::runtime_error("AI Camera input must be a bounded regular JPEG, not a symlink");
    68 std::array<unsigned char, EXIF_MAX_PAYLOAD> payload{};
    69 std::size_t size = 0;
    70 exif_view view{};
    71 if (exif_read_jpeg(source.path.c_str(), payload.data(), payload.size(), &size) || exif_parse(&view, payload.data(), size))
    72 throw std::runtime_error("AI Camera JPEG has no valid complete EXIF: " + name);
    73 const auto text = [&](exif_directory directory, uint16_t tag) {
    74 exif_value field{};
    75 if (!exif_get(&view, directory, tag, &field)) return std::string();
    76 return std::string(reinterpret_cast<const char*>(field.data), exif_text_length(&field));
    77 };
    78 const auto comment = text(EXIF_CAMERA, EXIF_USER_COMMENT);
    79 if (comment.find("USHER_MOCK_MAP_V1") != std::string::npos || comment.find("USHER_LENS_V1") != std::string::npos)
    80 throw std::runtime_error("Synthetic or lens-derived photos are not raw AI Camera inputs");
    81 const auto description = text(EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION);
    82 if (description.rfind("Paparazzi USHER ", 0) != 0)
    83 throw std::runtime_error("AI Camera input lacks USHER provenance");
    84 std::map<std::string, std::string> fields;
    85 std::istringstream parts(description);
    86 std::string part;
    87 std::getline(parts, part, ';');
    88 while (std::getline(parts, part, ';')) {
    89 const auto separator = part.find('=');
    90 if (separator == std::string::npos) throw std::runtime_error("Invalid USHER EXIF field");
    91 const auto trim = [](std::string value) {
    92 const auto first = value.find_first_not_of(" \t\r\n");
    93 if (first == std::string::npos) return std::string();
    94 return value.substr(first, value.find_last_not_of(" \t\r\n") - first + 1);
    95 };
    96 if (!fields.emplace(trim(part.substr(0, separator)), trim(part.substr(separator + 1))).second)
    97 throw std::runtime_error("Duplicate USHER EXIF field");
    98 }
    99 const auto required = [&](const char* key) -> const std::string& {
    100 const auto found = fields.find(key);
    101 if (found == fields.end() || found->second.empty()) throw std::runtime_error(std::string("USHER EXIF lacks ") + key);
    102 return found->second;
    103 };
    104 const auto real = [&](const char* key, double low, double high) {
    105 const double value = number(required(key), key);
    106 range(value, low, high, key);
    107 return value;
    108 };
    109 const auto integer = [&](const char* key) {
    110 const auto& value = required(key);
    111 uint64_t parsed = 0;
    112 const auto converted = std::from_chars(value.data(), value.data() + value.size(), parsed);
    113 if (converted.ec != std::errc() || converted.ptr != value.data() + value.size())
    114 throw std::runtime_error(std::string("Invalid USHER integer ") + key);
    115 return parsed;
    116 };
    117 bool exposurePose = true;
    118 for (const auto* key : {"capture_pose_synchronized", "observation_pose_aligned", "frame_identity_bound"}) {
    119 const auto& value = required(key);
    120 if (value != "true" && value != "false") throw std::runtime_error("Invalid USHER pose qualification");
    121 exposurePose = exposurePose && value == "true";
    122 }
    123 if (required("altitude_datum") != "flight_controller_navigation")
    124 throw std::runtime_error("AI Camera mapping requires FC navigation altitude");
    125 if (!exposurePose && !config.allowRecordedPose)
    126 throw std::runtime_error("AI Camera mapping requires exposure-aligned, frame-bound pose; recorded policy is explicitly approximate");
    127 if (exposurePose) {
    128 if (required("observation_time_kind") != "sensor_exposure_interval")
    129 throw std::runtime_error("AI Camera mapping requires sensor exposure timing");
    130 const auto earliest = integer("observation_earliest_monotonic_us"), latest = integer("observation_latest_monotonic_us");
    131 if (!earliest || latest < earliest || latest - earliest > 50000 || latest > uint64_t(LLONG_MAX / 10))
    132 throw std::runtime_error("AI Camera observation interval exceeds 50 ms or is invalid");
    133 sensor_capture capture{};
    134 exif_value evidence{};
    135 if (!exif_get(&view, EXIF_CAMERA, SENSOR_CAPTURE_TAG, &evidence) ||
    136 !sensor_capture_decode(&capture, evidence.data, evidence.size) ||
    137 !capture.sequence || !capture.sensor_timestamp_ns || capture.sequence != integer("observation_sequence") ||
    138 capture.sensor_timestamp_ns != integer("sensor_timestamp_ns") || capture.boot_id != required("observation_boot_id") ||
    139 capture.width != calibration.width || capture.height != calibration.height ||
    140 capture.monotonic_us < earliest || capture.monotonic_us > latest)
    141 throw std::runtime_error("AI Camera frame identity/dimensions disagree with its observation pose");
    142 source.poseTime = static_cast<long long>(earliest * 10); source.frameTime = static_cast<long long>(latest * 10);
    143 }
    144 FILE* jpeg = fopen(source.path.c_str(), "rb");
    145 uint32_t width = 0, height = 0;
    146 unsigned components = 0;
    147 const bool dimensions = jpeg && !exif_jpeg_dimensions(jpeg, &width, &height, &components);
    148 const int closed = jpeg ? fclose(jpeg) : -1;
    149 if (!dimensions || closed || width != calibration.width || height != calibration.height)
    150 throw std::runtime_error("AI Camera JPEG dimensions disagree with its profile");
    151 exif_value orientation{};
    152 uint32_t upright = 1;
    153 if (exif_get(&view, EXIF_PRIMARY, EXIF_ORIENTATION, &orientation) &&
    154 (!exif_value_unsigned(&orientation, 0, &upright) || upright != 1))
    155 throw std::runtime_error("AI Camera input must be upright sensor pixels");
    156 source.latitude = real(exposurePose ? "observation_lat_deg" : "lat_deg", -80, 84);
    157 source.longitude = real(exposurePose ? "observation_lon_deg" : "lon_deg", -180, 180);
    158 source.altitude = real(exposurePose ? "observation_alt_m" : "alt_m", -1000, 20000);
    159 source.bank = real(exposurePose ? "observation_roll_deg" : "roll_deg", -180, 180);
    160 source.pitch = real(exposurePose ? "observation_pitch_deg" : "pitch_deg", -90, 90);
    161 source.heading = real(exposurePose ? "observation_yaw_deg" : "yaw_deg", -360, 360);
    162 source.poseUncertainty = exposurePose ? real("observation_pose_uncertainty_m", 0, config.maxPoseUncertainty) :
    163 std::numeric_limits<double>::quiet_NaN();
    164 source.width = int(calibration.width); source.height = int(calibration.height);
    165 source.focal = calibration.fx; source.focalY = calibration.fy;
    166 source.cx = calibration.cx; source.cy = calibration.cy;
    167 source.referencePlane = config.mosaicReferencePlane;
    168 source.distorted = true;
    169 source.lens = {PiAiOptics::k1, PiAiOptics::k2, PiAiOptics::blueScale, PiAiOptics::redScale};
    170 coverage_pose pose{};
    171 pose.latitude_deg = source.latitude; pose.longitude_deg = source.longitude;
    172 pose.agl_m = source.altitude - source.referencePlane;
    173 pose.roll_deg = source.bank; pose.pitch_deg = source.pitch; pose.yaw_deg = source.heading;
    174 pose.uncertainty_m = exposurePose ? source.poseUncertainty : 0;
    175 coverage_footprint footprint{};
    176 if (!coverage_project(&calibration, &pose, source.latitude, source.longitude, &footprint))
    177 throw std::runtime_error("AI Camera calibration/pose fails shared coverage geometry validation");
    178 source.camera.emplace(source.focal, source.focalY, source.cx, source.cy, source.bank, source.pitch, source.heading,
    179 calibration.mount_roll_deg, calibration.mount_pitch_deg, calibration.mount_yaw_deg);
    180 for (const double column : {0.0, double(source.width - 1)}) for (const double row : {0.0, double(source.height - 1)})
    181 source.camera->ground(column, row, source.altitude - source.referencePlane);
    182 result.push_back(std::move(source));
    183 }
    184 if (!input.eof() || result.empty()) throw std::runtime_error("Empty or unreadable AI Camera snapshot");
    185 return result;
    186}
    187}
