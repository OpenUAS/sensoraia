    36/** @brief Estimated ground source, not a surveyed target or guaranteed alarm detection. */
    37struct ear_loudest_spot {

    39 double lat_deg; /**< Estimated source latitude, degrees. */
    40 double lon_deg; /**< Estimated source longitude, degrees. */
    41 double agl_m; /**< Representative microphone height from selected loud windows, metres. */
    42 double alt_m; /**< Ground altitude: peak-window navigation altitude minus its AGL. */
    43 double level_db; /**< Strongest retained level; dBFS unless the capture chain is calibrated. */

    49/**
    50 * @brief One audio window associated with microphone position and provenance.
    51 * @details Keep gain fixed across a session: relative level differences drive
    52 * localization. False pose_aligned preserves legacy arrival-time association;
    53 * it must not be reported as exposure/audio-window synchronization.
    54 */
    55struct ear_sample {
    56 uint64_t timestamp_ms; /**< Producer monotonic report timestamp, milliseconds; capture bounds are separate. */
    57 int32_t shot_nr; /**< Associated FC photo/request number; not the audio sequence. */
    58 double lat_deg; /**< Microphone latitude from associated navigation evidence, degrees. */
    59 double lon_deg; /**< Microphone longitude from associated navigation evidence, degrees. */
    60 double agl_m; /**< Height above the supplied ground reference, metres. */
    61 double alt_m; /**< Microphone navigation altitude, metres; datum supplied by caller. */
    62 double level_db; /**< Band loudness, dBFS or calibrated dB SPL; never mix conventions. */
    63 double trend_db; /**< Fast-minus-slow envelope level used to favor approaching evidence. */
    64 double contrast_db; /**< Tone prominence against the capture detector's background estimate. */
    65 double frequency_hz; /**< Dominant detector-bank frequency, hertz. */
    66 bool alarm; /**< Capture detector's tone-like alarm flag, not solver confidence. */
    67 bool clipped; /**< ADC saturation makes level comparisons unreliable. */
    68 bool pose_aligned; /**< Pose association used bounded audio-window timing evidence. */
    69 double pose_uncertainty_m; /**< Adapter-supplied spatial uncertainty, metres; not solver error. */
    70 /** Audio-window bounds in host monotonic microseconds, followed by producer sequence. */
    71 uint64_t window_earliest_us, window_latest_us, window_sequence;
    72 /** Original FC coordinates in 1e7 degrees, retained for timing/pose diagnostics. */
    73 int32_t request_lat, request_lon;
    74 bool classifier; /**< Distinguish PCM model acceptance from the native tone detector's alarm rule. */
    75 double classifier_score; /**< Model probability, never the acoustic level or localization confidence. */
    76 unsigned classifier_class; /**< Model-owned class index; compare only within a fixed assignment. */
    77};

    79/**
    80 * @brief Estimate a ground source from relative loudness along a survey track.
    81 * @param samples Caller-owned immutable windows; never modified or retained.
    82 * @param count Input count, bounded by EAR_MAX_SAMPLES.
    83 * @param result Required destination; cleared before validation when non-NULL.
    84 * @return Zero for an estimate, -1 for invalid input or insufficient retained evidence.
    85 * @details A robust centroid seeds a damped inverse-distance fit; unstable fits
    86 * fall back to that centroid. Shared bounded scratch storage is mutex-protected,
    87 * so concurrent calls serialize without allocating another survey-sized buffer.
    88 * @warning A valid result is not an absolute alarm gate. With too few alarm
    89 * windows the solver deliberately uses other finite levels, even silent ones.
    90 * Mission policy must consider level, quality, geometry and independent evidence.
    91 * The legacy near-zero-latitude placeholder gate also excludes equatorial tracks.
    92 */
    93int calculated_loudestspot(const struct ear_sample *samples, size_t count, struct ear_loudest_spot *result);
