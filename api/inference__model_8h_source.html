    42/** @brief Independent, noncopyable model owner; mapped tensor buffers and session are reused between jobs. */
    43struct inference_context {
    44 struct inference_model model;

    46 struct inference_image_transform transform;
    47 void *storage; /**< Private mapped arena; ORT's own allocations are additional. */

    50 uint64_t frame_sequence, sensor_timestamp_ns;
    51 bool capture_identity; /**< Sequence/time came from this retained capture rather than an offline request. */

    55/** @brief Load a bounded descriptor and verify its exact graph digest; hashes are not authentication.
    56 * @param model Destination contract; discard it on failure.
    57 * @param descriptor SENSORAIA_MODEL_V1 flat INI path.
    58 * @param graph Graph whose bytes must match the descriptor hash.
    59 * @param error Writable failure diagnostic.
    60 * @param capacity Available bytes in error.
    61 * @return True for a complete descriptor and matching graph identity. */

    63 const char *graph, char *error, size_t capacity);
    64/** @brief Parse a descriptor, not graph bytes; graph integrity remains a separate read-time check.
    65 * @param model Destination with name storage backing its own tensor views.
    66 * @param contents Destructively parsed input with at least length+1 writable bytes.
    67 * @param length Input bytes excluding the writable terminator.
    68 * @param error Writable failure diagnostic.
    69 * @param capacity Available bytes in error.
    70 * @return True for supported schema, bounded concrete tensors and complete preprocessing. */
    71bool inference_model_parse(struct inference_model *model, char *contents, size_t length,
    72 char *error, size_t capacity);
    73/** @brief Hash a stable nonempty regular file using bounded streaming scratch.
    74 * @param filename Immutable input path during hashing.
    75 * @param digest Destination for 64 lowercase hex characters and a NUL.
    76 * @return True for a complete stable read; digest is undefined on failure. */
    77bool inference_file_sha256(const char *filename, char digest[65]);
    78/** @brief Resolve one model-owned label without retaining a pointer into a mutable list.
    79 * @param model Parsed descriptor with comma-separated labels.
    80 * @param category Zero-based class index.
    81 * @param label Writable nontruncated class name.
    82 * @param capacity Available bytes in label.

    86/** @brief Verify a bundle, map bounded tensors and bind one warm session.
    87 * @param context Zero-initialized noncopyable owner.
    88 * @param settings Resolved ONNX configuration; values are consumed during open.
    89 * @param adapter Required application ABI, or NULL when the caller validates it separately.
    90 * @param error Writable failure diagnostic.
    91 * @param capacity Available bytes in error.
    92 * @return True after the graph can use this context's tensor buffers.
    93 * @note Mapping tensor storage does not make ORT allocations heap-free or bound whole-board RAM. */

    101 * @return True for valid completed output; ignore output on failure. */
    102bool inference_context_run(struct inference_context *context, char *error, size_t capacity);
    103/** @brief Prepare a single image input without running the model, so camera leases can be released first.

    109bool inference_context_prepare_image(struct inference_context *context,

    112/** @brief Release the session before unmapping its borrowed tensor storage.
    113 * @param context Exclusive owner; NULL and repeated closes are harmless. */
    114void inference_context_close(struct inference_context *context);
