    46/** @brief Store an owned bounded name so parser-buffer destruction cannot invalidate tensor views. */
    47static bool copy_name(char *destination, size_t capacity, const char *value)
    48{
    49 const size_t length = strlen(value);
    50 if (!length || length >= capacity) return false;
    51 for (size_t index = 0; index <= length; ++index) {
    52 if ((unsigned char)value[index] < 32 && value[index]) return false;
    53 destination[index] = value[index];
    54 }
    55 return true;
    56}

    58/** @brief Accept concrete positive extents only; export-time dynamic axes need an explicit execution shape. */
    59static bool parse_dimensions(struct onnx_tensor *tensor, const char *text)
    60{
    61 tensor->rank = 0;
    62 while (*text) {
    63 if (tensor->rank == ONNX_RANK_LIMIT || *text < '1' || *text > '9') return false;
    64 errno = 0;
    65 char *end = NULL;
    66 const unsigned long long number = strtoull(text, &end, 10);
    67 if (errno || end == text || number > INT64_MAX || (*end && *end != ',')) return false;
    68 tensor->dimensions[tensor->rank++] = (int64_t)number;
    69 text = *end ? end + 1 : end;
    70 if (*end && !*text) return false;
    71 }
    72 return tensor->rank != 0;
    73}

    75/** @brief Accept one shared channel value or exactly three, without silently broadcasting an incomplete pair. */
    76static bool parse_channels(float values[3], const char *text)
    77{
    78 for (unsigned channel = 0; channel < 3; ++channel) {
    79 char token[64];
    80 const char *end = strchr(text, ',');
    81 const size_t length = end ? (size_t)(end - text) : strlen(text);
    82 if (!length || length >= sizeof(token)) return false;
    83 for (size_t index = 0; index < length; ++index) token[index] = text[index];
    84 token[length] = '\0';
    85 double number = 0;
    86 if (!ini_real(token, &number) || !isfinite((float)number)) return false;
    87 values[channel] = (float)number;
    88 if (!end) {
    89 if (channel) return channel == 2;
    90 values[1] = values[2] = values[0];
    91 return true;
    92 }
    93 text = end + 1;
    94 }
    95 return false;
    96}

    98/** @brief Parse one indexed tensor while keeping duplicate detection separate for each tensor and direction. */
    99static bool tensor_field(struct model_parser *parser, const char *key, const char *value, bool input)
    100{
    101 const char *number = key + (input ? 6 : 7);
    102 const char *separator = strchr(number, '.');
    103 char index_text[21];
    104 if (!separator || separator == number || (size_t)(separator - number) >= sizeof(index_text)) return false;
    105 const size_t length = (size_t)(separator - number);
    106 for (size_t index = 0; index < length; ++index) index_text[index] = number[index];
    107 index_text[length] = '\0';
    108 uint64_t slot = 0;
    109 if (!ini_uint(index_text, &slot) || slot >= ONNX_TENSOR_LIMIT) return false;
    110 const char *field = separator + 1;
    111 struct onnx_tensor *tensor = input ? &parser->model->inputs[slot] : &parser->model->outputs[slot];
    112 unsigned char *seen = input ? &parser->inputs[slot] : &parser->outputs[slot];
    113 unsigned char bit;
    114 if (!strcmp(field, "name")) bit = 1;
    115 else if (!strcmp(field, "type")) bit = 2;
    116 else if (!strcmp(field, "shape")) bit = 4;
    117 else return false;
    118 if (*seen & bit) return false;
    119 *seen |= bit;
    120 if (bit == 1) {
    121 char *name = input ? parser->model->input_names[slot] : parser->model->output_names[slot];
    122 if (!copy_name(name, ONNX_NAME_CAPACITY, value)) return false;
    123 tensor->name = name;
    124 return true;
    125 }
    126 if (bit == 4) return parse_dimensions(tensor, value);
    127 static const char *const names[] = {"float32", "uint8", "int8", "int32", "int64", "float64"};
    128 static const enum onnx_element_type types[] = {ONNX_FLOAT32, ONNX_UINT8, ONNX_INT8, ONNX_INT32, ONNX_INT64, ONNX_FLOAT64};
    129 for (size_t index = 0; index < sizeof(types) / sizeof(types[0]); ++index) {
    130 if (!strcmp(value, names[index])) { tensor->type = types[index]; return true; }
    131 }
    132 return false;
    133}

    135/** @brief Recognize descriptor policy, not arbitrary network-specific configuration or custom operators. */
    136static bool model_field(void *context, const char *key, const char *value)
    137{
    138 struct model_parser *parser = context;
    139 struct inference_model *model = parser->model;
    140 if (!strncmp(key, "input.", 6)) return tensor_field(parser, key, value, true);
    141 if (!strncmp(key, "output.", 7)) return tensor_field(parser, key, value, false);
    142 static const char *const names[] = {"schema", "adapter", "model_sha256", "input_count", "output_count",
    143 "image.layout", "image.color", "image.resize", "image.scale", "image.mean", "image.padding",
    144 "image.units", "audio.sample_rate", "audio.samples", "image.interpolation", "labels"};
    145 size_t field = 0;
    146 while (field < sizeof(names) / sizeof(names[0]) && strcmp(key, names[field])) ++field;
    147 if (field == sizeof(names) / sizeof(names[0]) || (parser->seen & (UINT32_C(1) << field))) return false;
    148 parser->seen |= UINT32_C(1) << field;
    149 if (!field) return !strcmp(value, "SENSORAIA_MODEL_V1");
    150 if (field == 1) return copy_name(model->adapter, sizeof(model->adapter), value);
    151 if (field == 2) return strlen(value) == 64 && strspn(value, "0123456789abcdef") == 64 &&
    152 copy_name(model->graph_sha256, sizeof(model->graph_sha256), value);
    153 if (field == 15) return !strchr(value, ';') && copy_name(model->labels, sizeof(model->labels), value);
    154 if (field == 3 || field == 4 || field == 12 || field == 13) {
    155 uint64_t number;
    156 if (!ini_uint(value, &number) || !number || number > (field < 5 ? ONNX_TENSOR_LIMIT : 192000U)) return false;
    157 if (field == 3) model->input_count = (size_t)number;
    158 else if (field == 4) model->output_count = (size_t)number;
    159 else if (field == 12) model->audio_rate = (unsigned)number;
    160 else model->audio_samples = (unsigned)number;
    161 return true;
    162 }
    163 if (field == 5) {
    164 if (!strcmp(value, "nchw")) model->image.layout = INFERENCE_NCHW;
    165 else if (!strcmp(value, "nhwc")) model->image.layout = INFERENCE_NHWC;
    166 else return false;
    167 } else if (field == 6) {
    168 model->image.bgr = !strcmp(value, "bgr");
    169 model->image.channels = !strcmp(value, "gray") ? 1 : 3;
    170 if (strcmp(value, "rgb") && strcmp(value, "bgr") && strcmp(value, "gray")) return false;
    171 } else if (field == 7) {
    172 model->image.letterbox = !strcmp(value, "letterbox");
    173 if (strcmp(value, "letterbox") && strcmp(value, "stretch")) return false;
    174 } else if (field == 8 || field == 9) return parse_channels(field == 8 ? model->image.scale : model->image.mean, value);
    175 else if (field == 10) {
    176 double number;
    177 if (!ini_real(value, &number) || !isfinite((float)number)) return false;
    178 model->image.padding = (float)number;
    179 } else if (field == 11) {
    180 if (!strcmp(value, "pixel")) model->image.units = INFERENCE_PIXEL;
    181 else if (!strcmp(value, "kelvin64")) model->image.units = INFERENCE_RAW_KELVIN64;
    182 else if (!strcmp(value, "kelvin")) model->image.units = INFERENCE_KELVIN;
    183 else if (!strcmp(value, "celsius")) model->image.units = INFERENCE_CELSIUS;
    184 else return false;
    185 } else if (field == 14) return !strcmp(value, "nearest");
    186 return true;
    187}

    189bool inference_model_parse(struct inference_model *model, char *contents, size_t length,
    190 char *error, size_t capacity)
    191{
    192 if (!model || !contents || !error || !capacity) return false;
    193 *model = (struct inference_model){0};
    194 /* Hash the original bytes before destructive parsing so evidence identifies the exact preprocessing contract. */
    195 sha256_hex(contents, length, model->descriptor_sha256);
    196 struct model_parser parser = {.model = model};
    197 if (!ini_parse(contents, length, model_field, &parser) || (parser.seen & 31U) != 31U) goto invalid;
    198 size_t total = 0;
    199 for (size_t slot = 0; slot < ONNX_TENSOR_LIMIT; ++slot) {
    200 const bool input = slot < model->input_count, output = slot < model->output_count;
    201 if (parser.inputs[slot] != (input ? 7 : 0) || parser.outputs[slot] != (output ? 7 : 0)) goto invalid;
    202 struct onnx_tensor *tensors[] = {&model->inputs[slot], &model->outputs[slot]};
    203 for (size_t side = 0; side < 2; ++side) {
    204 if ((side == 0 && !input) || (side == 1 && !output)) continue;
    205 struct onnx_tensor *tensor = tensors[side];
    206 if (!onnx_tensor_bytes(tensor->type, tensor->dimensions, tensor->rank, &tensor->bytes) ||
    207 tensor->bytes > ONNX_BUFFER_LIMIT - total) goto invalid;
    208 total += tensor->bytes;
    209 for (size_t previous = 0; previous < slot; ++previous)
    210 if (!strcmp(tensor->name, side ? model->outputs[previous].name : model->inputs[previous].name)) goto invalid;
    211 }
    212 }
    213 if (model->image.channels) {
    214 const uint32_t required = (UINT32_C(1) << 5) | (UINT32_C(1) << 6) | (UINT32_C(1) << 7) |
    215 (UINT32_C(1) << 8) | (UINT32_C(1) << 9) | (UINT32_C(1) << 11) | (UINT32_C(1) << 14);
    216 const struct onnx_tensor *input = &model->inputs[0];
    217 if ((parser.seen & required) != required || input->rank != 4 || input->dimensions[0] != 1) goto invalid;
    218 const unsigned channel_axis = model->image.layout == INFERENCE_NCHW ? 1 : 3;
    219 const unsigned height_axis = model->image.layout == INFERENCE_NCHW ? 2 : 1;
    220 const unsigned width_axis = model->image.layout == INFERENCE_NCHW ? 3 : 2;
    221 if (input->dimensions[channel_axis] != model->image.channels || input->dimensions[width_axis] > 2048 ||
    222 input->dimensions[height_axis] > 2048) goto invalid;
    223 model->image.width = (unsigned)input->dimensions[width_axis];
    224 model->image.height = (unsigned)input->dimensions[height_axis];
    225 model->image.type = input->type;
    226 if (input->type != ONNX_FLOAT32 && input->type != ONNX_UINT8 && input->type != ONNX_INT8) goto invalid;
    227 for (unsigned channel = 0; channel < model->image.channels; ++channel)
    228 if (!model->image.scale[channel]) goto invalid;
    229 }
    230 error[0] = '\0';
    231 return true;
    232invalid:
    233 snprintf(error, capacity, "Invalid model descriptor: schema, duplicate, tensor bounds or preprocessing contract");
    234 return false;
    235}

    237bool inference_model_label(const struct inference_model *model, unsigned category,
    238 char *label, size_t capacity)
    239{
    240 if (!model || !label || !capacity) return false;
    241 const char *cursor = model->labels;
    242 for (unsigned index = 0; index < category; ++index) {
    243 cursor = strchr(cursor, ',');
    244 if (!cursor) return false;
    245 ++cursor;

    247 const char *end = strchr(cursor, ',');
    248 const size_t length = end ? (size_t)(end - cursor) : strlen(cursor);

    258bool inference_model_read(struct inference_model *model, const char *descriptor,
    259 const char *graph, char *error, size_t capacity)
    260{
    261 if (!model || !descriptor || !graph || !error || !capacity) return false;
    262 char contents[16385], digest[65];
    263 size_t length = 0;
    264 if (!ini_read_file(descriptor, contents, sizeof(contents), &length)) {
    265 snprintf(error, capacity, "Cannot read model descriptor: %s", descriptor);
    266 return false;
    267 }
    268 if (!inference_model_parse(model, contents, length, error, capacity)) return false;
    269 /* A well-formed descriptor for a different graph is still an incompatible assignment bundle. */
    270 if (!inference_file_sha256(graph, digest) || strcmp(digest, model->graph_sha256)) {
    271 snprintf(error, capacity, "Model graph checksum mismatch or unreadable graph: %s", graph);
    272 return false;
    273 }
    274 return true;
    275}

    277bool inference_context_open(struct inference_context *context,
    278 const struct inference_settings *settings, const char *adapter,
    279 char *error, size_t capacity)
    280{
    281 if (!context || !settings || !error || !capacity) return false;
    282 error[0] = '\0';
    283 if (context->storage || context->session.handle || settings->backend != INFERENCE_ONNX ||
    284 !settings->model[0] || !settings->descriptor[0]) {
    285 snprintf(error, capacity, "ONNX requires a model and model_descriptor in settings");
    286 return false;
    287 }
    288 if (!inference_model_read(&context->model, settings->descriptor, settings->model, error, capacity)) return false;
    289 if (adapter && strcmp(context->model.adapter, adapter)) {
    290 snprintf(error, capacity, "Unsupported model adapter: %s (expected %s)", context->model.adapter, adapter);
    291 return false;
    292 }
    293 size_t total = 0;
    294 /* Align every tensor independently; mixed numeric types may otherwise receive misaligned storage. */
    295 for (size_t side = 0; side < 2; ++side) {
    296 const size_t count = side ? context->model.output_count : context->model.input_count;
    297 const struct onnx_tensor *tensors = side ? context->model.outputs : context->model.inputs;
    298 for (size_t index = 0; index < count; ++index) {
    299 const size_t aligned = (tensors[index].bytes + 63U) & ~(size_t)63U;
    300 if (aligned > ONNX_BUFFER_LIMIT - total) {
    301 snprintf(error, capacity, "Model tensor workspace exceeds bounded storage");
    302 return false;
    303 }
    304 total += aligned;
    305 }
    306 }
    307 if (!total) {
    308 snprintf(error, capacity, "Model tensor workspace is empty");
    309 return false;
    310 }
    311 /* POSIX-backed private mapping avoids anonymous-mapping extensions and gives tensor storage a clear lifetime. */
    312 const int descriptor = open("/dev/zero", O_RDWR | O_CLOEXEC);
    313 if (descriptor < 0) {
    314 snprintf(error, capacity, "Cannot open model workspace backing: %s", strerror(errno));
    315 return false;
    316 }
    317 void *storage = mmap(NULL, total, PROT_READ | PROT_WRITE, MAP_PRIVATE, descriptor, 0);
    318 const int closed = close(descriptor);
    319 if (storage == MAP_FAILED || closed) {
    320 if (storage != MAP_FAILED) munmap(storage, total);
    321 snprintf(error, capacity, "Cannot map bounded model tensor workspace");
    322 return false;
    323 }
    324 context->storage = storage;
    325 context->storage_bytes = total;
    326 context->timeout_ms = settings->timeout_ms;
    327 size_t offset = 0;
    328 for (size_t side = 0; side < 2; ++side) {
    329 const size_t count = side ? context->model.output_count : context->model.input_count;
    330 struct onnx_tensor *tensors = side ? context->model.outputs : context->model.inputs;
    331 for (size_t index = 0; index < count; ++index) {
    332 tensors[index].data = (unsigned char *)storage + offset;
    333 offset += (tensors[index].bytes + 63U) & ~(size_t)63U;
    334 }
    335 }
    336 const struct onnx_session_options options = {.threads = settings->threads,
    337 .library_directory = settings->library_directory};
    338 if (!onnx_session_open(&context->session, settings->model, context->model.inputs, context->model.input_count,
    339 context->model.outputs, context->model.output_count, &options, error, capacity)) {
    340 inference_context_close(context);
    341 return false;
    342 }
    343 return true;
    344}

    347{
