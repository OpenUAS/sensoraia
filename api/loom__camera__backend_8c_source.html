    56 if (!settings || !settings[0] || capture_owner.pid > 0) return -1;

    68/** @brief Select the native capture protocol only when this owner is acquired.
    69 * @param source Reserved common-backend input; native capture uses the configured RPK.
    70 * @return Zero after the camera child reports READY, otherwise -1.
    71 * @details Stream startup may happen earlier without camera acquisition. Delaying
    72 * physical ownership until capture keeps saved-photo processing independent. */
    73static int loom_capture_init(const char *source)
    74{
    75 (void)source;
    76 capture_owner.command = USHER_LOOMCAM_COMMAND;
    77 capture_owner.model = network_model;
    78 capture_owner.settings = inference_settings_path ? inference_settings_path :
    79 network_model && !strcmp(network_model, USHER_LOOMCAM_MODEL) ? "-" : NULL;
    80 capture_owner.directory = USHER_AI_CAM_PHOTO_DIR;
    81 capture_owner.prefix = LOOM_IMAGE_PREFIX;
    82 return imx500_pipe_start(&capture_owner);
    83}
    84
    85/** @brief Request pixels through the neutral supervisor without detection parsing.
    86 * @param filename Caller-owned destination buffer.
    87 * @param capacity Writable path bytes including the terminator.
    88 * @param number FC photo number used only for the output filename.
    89 * @return Zero for a qualified saved image, otherwise -1. */
    90static int loom_capture_shoot(char *filename, size_t capacity, int number)
    91{
    92 return imx500_pipe_shoot(&capture_owner, filename, capacity, number);
    93}

    95/** @brief Stop only LOOM's capture child; the saved-photo stream drains separately. */
    96static void loom_capture_deinit(void)
    97{
    98 imx500_pipe_stop(&capture_owner);
    99}

    101/** @brief Start optional saved-photo processing without acquiring another sensor.
    102 * @return Zero for worker startup, otherwise -1; photograph capture remains a separate capability. */
    103static int loom_stream_init(void)
    104{
    105 const int result = loom_cam_init(reference_directory, network_model);
    106 if (result != 0)
    107 fprintf(stderr, "LOOM_cam: worker unavailable; photos continue without visual navigation\n");
    108 return result;
    109}
