    18#ifndef SENSORAIA_LWIR_DEVICE_H
    19#define SENSORAIA_LWIR_DEVICE_H
    20/** @file
    21 * @brief Tiny1C/TDK acquisition and Tiny1C state, without image or mission logic.
    22 *
    23 * @details
    24 * This interface is the only boundary between LWIRcam and the camera hardware.
    25 * It delivers raw camera-layout frames plus timing evidence; image conversion,
    26 * calibration, geolocation and mission policy live elsewhere.
    27 *
    28 * @par Supported cameras
    29 * - Tiny1-C (USB 0bda:5840): combined YUYV transport of 256x384 pixels, i.e.
    30 * a 256x192 image plane followed by a 256x192 temperature plane holding
    31 * Kelvin*64 little-endian values; 196608 bytes per frame at about 25 fps.
    32 * Its separate 256x192 image-only UVC mode is not combined data and is not used.
    33 * - TDK (USB 5474:5411): 640x512 image-only YUYV; no temperature plane.
    34 *
    35 * @par Backend
    36 * Capture runs in lwir_device_v4l2.c directly on the Linux uvcvideo kernel
    37 * driver with V4L2 MMAP buffers and usbfs control requests for Tiny1-C state.
    38 *
    39 * @par Ownership model
    40 * Every object is caller-owned storage; the backend performs no heap
    41 * allocation. A session owns one V4L2 descriptor (held under an exclusive
    42 * flock) and up to #LWIR_VIDEO_BUFFERS kernel buffers mapped into the process.
    43 * Mapped buffers are only borrowed between dequeue and requeue; callers never
    44 * see their addresses and receive copies instead.
    45 *
    46 * @par Threading
    47 * A session is not thread-safe: one thread must own all calls on it. The only
    48 * cross-context input is the borrowed sig_atomic_t cancellation flag, which
    49 * may be set from a signal handler or another thread and is polled at least
    50 * every 25 ms while waiting.
    51 *
    52 * @par Lifecycle invariants
    53 * - Zero-initialize, lwir_device_open(), acquire/discard/state calls, then
    54 * lwir_device_close(). Close before reopening.
    55 * - A fatal kernel or protocol error poisons the session (EIO) until it is
    56 * closed and reopened; no partial recovery is attempted.
    57 * - The eight-frame startup qualification runs once per USB power instance,
    58 * remembered across reopen and process restart, and is never repeated for
    59 * later requests, gaps, invalid frames or reconnection of the same instance.
    60 */
    61#include "lwir_frame.h"
    62#include "lwir_state.h"
    63#include <signal.h>

    65/** @brief Number of V4L2 MMAP buffers requested from uvcvideo.
    66 * @details Four keeps one buffer in user space for inspection while the
    67 * kernel still has spare buffers to fill, so a short inspection does not drop
    68 * the next frame. The driver may grant fewer; at least two are required.
    69 */

    87/** @brief Bounded camera policy; Tiny1C requires full 256x192 image and temperature planes.
    88 * @details Zero dimensions select native geometry; Tiny1C's 256x192 image-only UVC mode is not combined data.
    89 * The same structure is used for open and for each acquisition: open writes
    90 * the negotiated geometry and layout back, and acquisition requires them to
    91 * still match the session.
    92 */
    93struct lwir_device_options {
    94 /** Optional USB serial string that must match exactly; NULL accepts any serial.
    95 * Must fit in lwir_device::serial including its terminator. */
    96 const char *serial;
    97 /** Optional explicit V4L2 node (for example /dev/video2). NULL scans
    98 * /dev/video0..63 and fails with EEXIST when more than one camera matches. */
    99 const char *device_path;
    100 /** @c width / @c height: per-plane native geometry, or zero for "native"
    101 * before open; written on open success.
    102 * @c warmup_ms: extra delay after the request starts, applied only on the
    103 * first startup of a USB power instance; must be below @c timeout_ms.
    104 * @c timeout_ms: whole-call budget in milliseconds, 1..120000. */
    105 unsigned width, height, warmup_ms, timeout_ms;
    106 /** @c scene_check: also require lwir_frame_plausible(); combined layout only,
    107 * so enabling it for image-only sessions rejects every frame.
    108 * @c settle: require consecutive frames to agree per lwir_frame_settled()
    109 * before accepting, at the cost of at least one extra frame period. */
    110 bool scene_check, settle;
    111 /** Optional measurement region validated against the native geometry at open; NULL skips it. */
    112 const struct lwir_measurement *measurement;
    113 /** Borrowed cancellation flag; non-zero aborts waits with EINTR. NULL disables cancellation. */
    114 const volatile sig_atomic_t *cancelled;
    115 /** Output of open: the detected camera's frame layout. */
    116 enum lwir_frame_layout layout;
    117 /** Fail open with ENOTSUP when only an image-only camera is present. */
    118 bool requires_thermal;
    119};

    121/** @brief Caller-owned session, initialized to zero before opening.
    122 * @details Handles belong only to the driver. Keeping their storage here avoids
    123 * a Sensoraia allocation; callers use the functions below, not backend handles directly.
    124 */
    125struct lwir_device {
    126 const volatile sig_atomic_t *cancelled; /**< Borrowed from the options at open; may be NULL. */
    127 char serial[64]; /**< USB serial read from sysfs, empty when the device reports none. */
    128 enum lwir_frame_layout layout; /**< Combined Tiny1-C or image-only TDK, fixed by USB identity. */
    129 int video_fd; /**< V4L2 descriptor holding the exclusive flock; -1 once closed. */
    130 /** @c video_streaming is set just before STREAMON so close always issues
    131 * STREAMOFF, even after a partially failed start. @c video_have_sequence
    132 * says whether @c video_last_sequence holds a real kernel sequence yet. */
    133 bool video_open, video_streaming, video_have_sequence;
    134 /** Sticky poison after a fatal kernel/protocol error; every later call
    135 * fails with EIO until close and reopen. */
    136 bool video_failed;
    137 /** @c video_count: buffers granted by REQBUFS (2..#LWIR_VIDEO_BUFFERS).
    138 * @c video_width / @c video_height: per-plane geometry, not transport height. */
    139 unsigned video_count, video_width, video_height;
    140 uint32_t video_last_sequence; /**< Last raw 32-bit V4L2 sequence number seen. */
    141 /** 64-bit extension of the kernel sequence; skipped kernel numbers also
    142 * advance it, so gaps stay visible to consumers. */
    143 uint64_t video_sequence;
    144 uint64_t video_good_arrival_us; /**< Host MONOTONIC arrival of the last qualifying frame. */
    145 /** Consecutive good frames, saturating at eight. Once eight is reached it
    146 * latches for the rest of the USB power instance. */
    147 unsigned video_good_frames;
    148 /** True when this USB power instance already completed startup
    149 * qualification, either now or according to the runtime marker. */
    150 bool video_power_ready;
    151 /** "bootid:sysfs-inode:busnum:devnum"; changes whenever the camera
    152 * re-enumerates, which is the only observable power boundary on Linux. */
    153 char video_power_key[160];
    154 char video_power_port[64]; /**< Physical USB port name (for example 1-1.2) used to name the marker. */
    155 /** USB bus and address used to reach the same device through usbfs for
    156 * vendor state/control requests. */
    157 unsigned video_usb_bus, video_usb_address;
    158 size_t video_bytes; /**< Expected bytesused per frame (sizeimage) for the layout. */
    159 struct lwir_video_buffer video_buffers[LWIR_VIDEO_BUFFERS]; /**< Mappings; only the first @c video_count are used. */
    160};

    162/** @brief Host observation plus optional kernel clock evidence, never automatic exposure qualification. */
    163struct lwir_device_observation {
    164 /** @c sequence: session-extended frame sequence (see lwir_device::video_sequence).
    165 * @c arrival_us: host CLOCK_MONOTONIC time when the frame was dequeued, in
    166 * microseconds; it includes USB delivery delay after the sensor frame start. */
    167 uint64_t sequence, arrival_us;
    168 /** Kernel buffer timestamp in CLOCK_MONOTONIC microseconds (uvcvideo takes
    169 * it at start of frame); zero when the buffer clock is not MONOTONIC. */
    170 uint64_t kernel_monotonic_us;
    171 /** Raw V4L2 timestamp type and source flags, so consumers can judge what
    172 * @c kernel_monotonic_us actually measures. */
    173 unsigned kernel_timestamp_flags;
    174};

    176/** @brief Open/start a bounded camera-specific combined or image-only YUY2 stream.
    177 * @param options Requested policy; native dimensions are written on success.
    178 * @param device Empty caller-owned session, closed on failure.
    179 * @return True on success; errno describes failure. Close before reopening.
    180 * @par Errors
    181 * EINVAL invalid options; EBUSY session still owns a descriptor or mapping;
    182 * ENODEV no matching camera; EEXIST more than one matching camera during a
    183 * scan; EWOULDBLOCK another process holds the camera; ENOTSUP unsupported
    184 * format or image-only camera when thermal data is required; ETIMEDOUT or
    185 * EINTR deadline or cancellation; other values come from the kernel.
    186 */
    187bool lwir_device_open(struct lwir_device_options *options, struct lwir_device *device);
    188/** @brief Copy a qualified, fresh camera-layout frame into caller-owned storage.
    189 * @param device Open session with an optional borrowed cancellation flag.
    190 * @param options Native geometry, warmup and qualification policy.
    191 * @param destination Writable frame buffer, retained through the call.
    192 * @param capacity Accessible bytes; at least lwir_frame_bytes(width, height).
    193 * @param observation Sequence and host arrival, written only on success.
    194 * @return True on acceptance; ignore destination on failure and never auto-retry publication.
    195 * @details Native image/observation buffers must not alias each other, the session
    196 * or mapped camera buffers. Fatal kernel ownership failures require close/reopen.
    197 * Startup qualification is retained for the USB instance across reopen/restart;
    198 * no subsequent request, gap or invalid frame imposes another eight-frame wait.
    199 * Freshness: when the kernel timestamp is CLOCK_MONOTONIC, the accepted frame
    200 * started at or after this call began, so it never predates the trigger.
    201 * @par Errors
    202 * EINVAL bad arguments or aliasing; EIO poisoned session; ETIMEDOUT deadline;
    203 * EINTR cancellation; EPROTO/EOVERFLOW inconsistent kernel data or clock.
    204 */
    205bool lwir_device_acquire(struct lwir_device *device, const struct lwir_device_options *options,
    206 uint8_t *destination, size_t capacity, struct lwir_device_observation *observation);
    207/** @brief Drain one pending frame without retaining the borrowed UVC buffer.
    208 * @param device Open session.
    209 * @param received Whether a frame was present, written on success.
    210 * @return True on successful polling, even when no frame is present.
    211 * @details Non-blocking. A drained frame still counts toward startup
    212 * qualification, so idle draining can complete it before the first request.
    213 */
    214bool lwir_device_discard(struct lwir_device *device, bool *received);
    215/** @brief Read bounded optional state after acquisition, never writing device settings.
    216 * @param device Open session with its cancellation flag.
    217 * @param state Caller-owned result with per-field failure statuses.
    218 * @details The snapshot is not frame-atomic and cannot establish gain at exposure.
    219 * Image-only TDK sessions return a not-requested snapshot without USB commands.
    220 */
    221void lwir_device_read_state(struct lwir_device *device, struct lwir_device_state *state);
    222/** @brief Submit one Tiny1-C vendor control write (or FFC action) as a standalone operation.
    223 * @param device Open combined-layout session; image-only cameras fail with ENOTSUP.
    224 * @param request Validated control kind and value.
    225 * @param result Caller-owned reply, always initialized.
    226 * @return True when the camera acknowledged the request.
    227 * @details Uses vendor control transfers through usbfs while uvcvideo keeps
    228 * streaming. Normal capture never calls this.
    229 * @warning The write is submitted once and never replayed automatically. A
    230 * failure (including a close error after submission) does not prove the
    231 * camera ignored it; read the value back rather than blindly retrying.
    232 * @par Errors
    233 * EINVAL invalid request; ENOTSUP image-only camera; ENODEV identity changed;
    234 * EINTR cancelled; ETIMEDOUT camera did not answer; EIO other failures.
    235 */
    236bool lwir_device_control(struct lwir_device *device, const struct lwir_control_request *request,
    237 struct lwir_control_result *result);
    238/** @brief Read the current value of one Tiny1-C control without writing it.
    239 * @param device Open combined-layout session.
    240 * @param kind Control to read; LWIR_CONTROL_FFC is an action and is rejected (EINVAL).
    241 * @param result Caller-owned reply, always initialized.
    242 * @return True when the value was read; errors as for lwir_device_control().
    243 */
    244bool lwir_device_control_read(struct lwir_device *device, enum lwir_control_kind kind,
    245 struct lwir_control_result *result);
    246/** @brief Close an initialized session; safe repeatedly and preserves errno.
    247 * @details Native failed unmaps remain owned for a subsequent close and prevent
    248 * reopening. Linux descriptor-close errors are never retried on a reused number.
    249 */
    250void lwir_device_close(struct lwir_device *device);
