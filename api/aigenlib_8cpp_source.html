    63void validate_sensor(const Sensor& sensor, std::size_t cells) {
    64 if (cells > max_extent || sensor.rows.size() > max_extent || sensor.variances.size() != sensor.rows.size()) {
    65 throw std::invalid_argument("Invalid aigenlib sensor dimensions");

    67 for (std::size_t row = 0; row < sensor.rows.size(); ++row) {

    85/** @brief Factor the lower triangle in place. @param factor Scaled symmetric workspace. @return Whether positive definite. */
    86bool cholesky(Matrix& factor) {
    87 for (std::size_t row = 0; row < factor.rows(); ++row) {
    88 double* current = factor.row_data(row);
    89 for (std::size_t column = 0; column <= row; ++column) {
    90 const double* previous = factor.row_data(column);
    91 double value = current[column];
    92 for (std::size_t inner = 0; inner < column; ++inner) {
    93 value -= current[inner] * previous[inner];
    94 }
    95 if (column == row) {
    96 if (!(value > 0.0) || !std::isfinite(value)) return false;
    97 current[column] = std::sqrt(value);
    98 } else {
    99 current[column] = value / previous[column];
    100 if (!std::isfinite(current[column])) return false;
    101 }
    102 }
    103 }
    104 return true;
    105}

    107/** @brief Apply two triangular solves. @param factor Lower Cholesky factor. @param result Scaled RHS, replaced by solution. */
    108void triangular_solve(const Matrix& factor, Matrix& result) {
    109 for (std::size_t row = 0; row < factor.rows(); ++row) {
    110 double* destination = result.row_data(row);
    111 const double* coefficients = factor.row_data(row);
    112 for (std::size_t inner = 0; inner < row; ++inner) {
    113 const double* source = result.row_data(inner);
    114 for (std::size_t column = 0; column < result.cols(); ++column) {
    115 destination[column] -= coefficients[inner] * source[column];
    116 }
    117 }
    118 for (std::size_t column = 0; column < result.cols(); ++column) destination[column] /= coefficients[row];
    119 }
    120 for (std::size_t remaining = factor.rows(); remaining > 0; --remaining) {
    121 const std::size_t row = remaining - 1;
    122 double* destination = result.row_data(row);
    123 for (std::size_t inner = row + 1; inner < factor.rows(); ++inner) {
    124 const double coefficient = factor(inner, row);
    125 const double* source = result.row_data(inner);
    126 for (std::size_t column = 0; column < result.cols(); ++column) destination[column] -= coefficient * source[column];
    127 }
    128 const double diagonal = factor(row, row);
    129 for (std::size_t column = 0; column < result.cols(); ++column) destination[column] /= diagonal;
    130 }
    131}

    133/**
    134 * @brief Diagonalize a scaled symmetric matrix with cyclic Jacobi rotations.
    135 * @details The stable hypot rotation follows OpenCV JacobiImpl_; see NOTICE.md.
    136 * Eigenvectors are columns. At most 80 sweeps, with an explicit convergence
    137 * check, avoid returning an unfinished factorization as a valid result.
    138 * @param diagonal Workspace, replaced by a near-diagonal matrix.
    139 * @param vectors Output orthonormal eigenvector matrix.
    140 */
    141void jacobi(Matrix& diagonal, Matrix& vectors) {
    142 for (std::size_t index = 0; index < vectors.rows(); ++index) vectors(index, index) = 1.0;
    143 for (unsigned sweep = 0; sweep <= jacobi_sweep_limit; ++sweep) {
    144 double largest = 0.0;
    145 double scale = 0.0;
    146 for (std::size_t row = 0; row < diagonal.rows(); ++row) {
    147 scale = std::max(scale, std::abs(diagonal(row, row)));
    148 for (std::size_t column = row + 1; column < diagonal.cols(); ++column) {
    149 largest = std::max(largest, std::abs(diagonal(row, column)));
    150 }
    151 }
    152 const double tolerance = jacobi_epsilon_multiplier * std::numeric_limits<double>::epsilon() * std::max(scale, 1.0);
    153 if (largest <= tolerance) return;
    154 if (sweep == jacobi_sweep_limit) break;
    155 for (std::size_t first = 0; first < diagonal.rows(); ++first) {
    156 for (std::size_t second = first + 1; second < diagonal.cols(); ++second) {
    157 const double off = diagonal(first, second);
    158 if (std::abs(off) <= tolerance) continue;
    159 const double delta = 0.5 * (diagonal(second, second) - diagonal(first, first));
    160 double tangent = off / (std::abs(delta) + std::hypot(off, delta));
    161 if (delta < 0.0) tangent = -tangent;
    162 const double cosine = 1.0 / std::hypot(1.0, tangent);
    163 const double sine = tangent * cosine;
    164 diagonal(first, first) -= tangent * off;
    165 diagonal(second, second) += tangent * off;
    166 diagonal(first, second) = diagonal(second, first) = 0.0;
    167 for (std::size_t index = 0; index < diagonal.rows(); ++index) {
    168 if (index != first && index != second) {
    169 const double before_first = diagonal(index, first);
    170 const double before_second = diagonal(index, second);
    171 diagonal(index, first) = diagonal(first, index) = cosine * before_first - sine * before_second;
    172 diagonal(index, second) = diagonal(second, index) = sine * before_first + cosine * before_second;
    173 }
    174 const double before_first = vectors(index, first);
    175 const double before_second = vectors(index, second);
    176 vectors(index, first) = cosine * before_first - sine * before_second;
    177 vectors(index, second) = sine * before_first + cosine * before_second;
    178 }
    179 }
    180 }
    181 }
    182 throw std::runtime_error("aigenlib symmetric eigensolver did not converge");
    183}

    269Matrix solve_symmetric(const Matrix& system, const Matrix& right, SolveMethod* method) {
    270 if (system.rows() != system.cols() || right.rows() != system.rows() || !system.finite() || !right.finite()) {
    271 throw std::invalid_argument("Invalid aigenlib symmetric system");
    272 }
    273 double scale = 0.0;
    274 for (std::size_t row = 0; row < system.rows(); ++row) {
    275 for (std::size_t column = 0; column < system.cols(); ++column) scale = std::max(scale, std::abs(system(row, column)));
    276 }
    277 Matrix scaled(system.rows(), system.cols());
    278 for (std::size_t row = 0; row < system.rows(); ++row) {
    279 for (std::size_t column = 0; column <= row; ++column) {
    280 const double lower = scale == 0.0 ? 0.0 : system(row, column) / scale;
    281 const double upper = scale == 0.0 ? 0.0 : system(column, row) / scale;
    282 if (std::abs(lower - upper) > symmetry_epsilon_multiplier * std::numeric_limits<double>::epsilon()) {
    283 throw std::invalid_argument("aigenlib system is not symmetric");
    284 }
    285 scaled(row, column) = scaled(column, row) = lower;
    286 }
    287 }
    288 Matrix factor = scaled;
    289 Matrix result(right.rows(), right.cols());
    290 if (cholesky(factor)) {
    291 for (std::size_t row = 0; row < right.rows(); ++row) {
    292 for (std::size_t column = 0; column < right.cols(); ++column) result(row, column) = right(row, column) / scale;
    293 }
    294 triangular_solve(factor, result);
    295 if (method) *method = SolveMethod::cholesky;
    296 } else {
    297 Matrix vectors(system.rows(), system.cols());
    298 jacobi(scaled, vectors);
    299 double largest = 0.0;
    300 for (std::size_t index = 0; index < system.rows(); ++index) largest = std::max(largest, std::abs(scaled(index, index)));
    301 Vector coefficients(right.cols());
    302 for (std::size_t mode = 0; mode < system.rows(); ++mode) {
    303 const double eigenvalue = scaled(mode, mode);
    304 if (std::abs(eigenvalue) <= pseudoinverse_relative_cutoff * largest) continue;
    305 std::fill(coefficients.begin(), coefficients.end(), 0.0);
    306 for (std::size_t row = 0; row < right.rows(); ++row) {
    307 const double weight = vectors(row, mode);
    308 const double* source = right.row_data(row);
    309 for (std::size_t column = 0; column < right.cols(); ++column) coefficients[column] += weight * source[column];
    310 }
    311 for (double& coefficient : coefficients) coefficient = (coefficient / eigenvalue) / scale;
    312 for (std::size_t row = 0; row < result.rows(); ++row) {
    313 double* destination = result.row_data(row);
    314 const double weight = vectors(row, mode);
    315 for (std::size_t column = 0; column < result.cols(); ++column) destination[column] += weight * coefficients[column];
    316 }
    317 }
    318 if (method) *method = SolveMethod::pseudoinverse;
    319 }
    320 if (!result.finite()) throw std::overflow_error("Nonfinite aigenlib symmetric solution");
    321 return result;
    322}

    324Matrix matern32(const std::vector<Point>& positions, double sigma2, double lengthscale) {
    325 if (positions.size() > max_extent || !(sigma2 > 0.0) || !(lengthscale > 0.0) ||
    326 !std::isfinite(sigma2) || !std::isfinite(lengthscale) ||
    327 !std::all_of(positions.begin(), positions.end(), [](const Point& point) { return point.finite(); })) {
    328 throw std::invalid_argument("Invalid aigenlib Matern parameters");
    329 }
    330 Matrix result(positions.size(), positions.size());
    331 for (std::size_t row = 0; row < positions.size(); ++row) {
    332 for (std::size_t column = 0; column <= row; ++column) {
    333 const double distance = (positions[row] - positions[column]).norm();
    334 const double scaled = std::sqrt(3.0) * (distance / lengthscale);
    335 const double value = std::isinf(scaled) ? 0.0 : sigma2 * ((1.0 + scaled) * std::exp(-scaled));
    336 result(row, column) = result(column, row) = value;
    337 }
    338 }
    339 return result;
    340}

    342Matrix project(const Sensor& sensor, const Matrix& covariance) {
    343 validate_covariance_shape(covariance);
    344 validate_sensor(sensor, covariance.rows());
    345 Matrix result(sensor.rows.size(), covariance.cols());
    346 for (std::size_t row = 0; row < sensor.rows.size(); ++row) {
    347 double* destination = result.row_data(row);
    348 for (const int index : sensor.rows[row]) {
    349 const double* source = covariance.row_data(static_cast<std::size_t>(index));
    350 for (std::size_t column = 0; column < covariance.cols(); ++column) destination[column] += source[column];
    351 }
    352 const double count = static_cast<double>(sensor.rows[row].size());
    353 for (std::size_t column = 0; column < covariance.cols(); ++column) destination[column] /= count;
    354 }
    355 if (!result.finite()) throw std::overflow_error("Nonfinite aigenlib covariance projection");
    356 return result;
    357}

    359Matrix innovation(const Sensor& sensor, const Matrix& projected) {
    360 validate_sensor(sensor, projected.cols());
    361 if (projected.rows() != sensor.rows.size() || !projected.finite()) throw std::invalid_argument("Invalid aigenlib projection");
    362 Matrix result(sensor.rows.size(), sensor.rows.size());
    363 for (std::size_t row = 0; row < result.rows(); ++row) {
    364 for (std::size_t column = 0; column < result.cols(); ++column) {
    365 double value = 0.0;
    366 const double* source = projected.row_data(column);
    367 for (const int index : sensor.rows[row]) value += source[index];
    368 result(row, column) = value / static_cast<double>(sensor.rows[row].size());
    369 }
    370 result(row, row) += sensor.variances[row];
    371 }
    372 if (!result.finite()) throw std::overflow_error("Nonfinite aigenlib innovation covariance");
    373 return result;
    374}

    388double expected_reduction(const Matrix& covariance, const std::vector<bool>& mask, const Sensor& sensor) {
    389 validate_covariance_shape(covariance);
    390 validate_sensor(sensor, covariance.rows());
    391 if (mask.size() != covariance.rows()) throw std::invalid_argument("Invalid aigenlib importance mask");
    392 if (sensor.rows.empty()) return 0.0;
    393 const Matrix projected = project(sensor, covariance);
    394 const Matrix solved = solve_symmetric(innovation(sensor, projected), projected);
    395 double gain = 0.0;
    396 for (std::size_t column = 0; column < covariance.cols(); ++column) {
    397 if (!mask[column]) continue;
    398 const double variance = covariance(column, column);
    399 if (!std::isfinite(variance)) throw std::invalid_argument("Nonfinite aigenlib selected variance");
    400 double reduction = 0.0;
    401 for (std::size_t row = 0; row < projected.rows(); ++row) reduction += projected(row, column) * solved(row, column);
    402 if (!std::isfinite(reduction)) throw std::overflow_error("Nonfinite aigenlib variance reduction");
    403 gain += std::clamp(reduction, 0.0, std::max(0.0, variance));
    404 }
    405 if (!std::isfinite(gain)) throw std::overflow_error("Nonfinite aigenlib information gain");
    406 return gain;
    407}

    409void condition(Matrix& covariance, Vector* mean, const Sensor& sensor, const Vector* measurement) {
    410 validate_covariance_shape(covariance);
    411 if (!covariance.finite()) throw std::invalid_argument("Nonfinite aigenlib covariance");
    412 validate_sensor(sensor, covariance.rows());
    413 if ((mean == nullptr) != (measurement == nullptr) ||
    414 (mean && (mean->size() != covariance.rows() || measurement->size() != sensor.rows.size() ||
    415 !finite_vector(*mean) || !finite_vector(*measurement)))) {
    416 throw std::invalid_argument("Invalid aigenlib GP measurement or mean");
    417 }
    418 if (sensor.rows.empty()) return;
    419 const Matrix projected = project(sensor, covariance);
    420 const Matrix solved = solve_symmetric(innovation(sensor, projected), projected);
    421 Vector updated_mean;
    422 if (mean) {
    423 Vector residual = observe(sensor, *mean);
    424 for (std::size_t row = 0; row < residual.size(); ++row) residual[row] = (*measurement)[row] - residual[row];
    425 updated_mean = *mean;
    426 for (std::size_t row = 0; row < solved.rows(); ++row) {
    427 const double* source = solved.row_data(row);
    428 for (std::size_t column = 0; column < solved.cols(); ++column) updated_mean[column] += source[column] * residual[row];
    429 }
    430 if (!finite_vector(updated_mean)) throw std::overflow_error("Nonfinite aigenlib posterior mean");
    431 }
    432 constexpr std::size_t tile_size = 32;
    433 if (projected.rows() == 1) {
    434 const double* projection = projected.row_data(0);
    435 const double* solution = solved.row_data(0);

    462 const std::size_t row_count = std::min(update_rows, last_row - row_begin);

    464 const std::size_t column_count = std::min(update_columns, covariance.cols() - column_begin);
    465 for (std::size_t row = 0; row < row_count; ++row)

    467 for (std::size_t inner = 0; inner < projected.rows(); ++inner) {
    468 const double* weights = projected.row_data(inner) + row_begin;
    469 const double* source = solved.row_data(inner) + column_begin;
    470 for (std::size_t row = 0; row < row_count; ++row) {
    471 const double weight = weights[row];
    472 for (std::size_t column = 0; column < column_count; ++column)
    473 scratch[row][column] += weight * source[column];
    474 }
    475 }
    476 for (std::size_t row = 0; row < row_count; ++row) {
    477 double* destination = covariance.row_data(row_begin + row) + column_begin;
    478 for (std::size_t column = 0; column < column_count; ++column)
    479 destination[column] -= scratch[row][column];
    480 }
    481 }
    482 }
    483 };
    484 if (covariance.rows() >= 1024 && projected.rows() >= 16) {
    485 constexpr std::size_t lane_count = 3;
    486 std::array<std::thread, lane_count - 1> workers;
    487 std::array<std::exception_ptr, lane_count> errors{};
    488 const auto run_lane = [&](std::size_t lane) {
    489 try { update(covariance.rows() * lane / lane_count, covariance.rows() * (lane + 1) / lane_count); }
    490 catch (...) { errors[lane] = std::current_exception(); }

    492 std::size_t launched = 0;
    493 try {
    494 for (; launched < workers.size(); ++launched)
    495 workers[launched] = std::thread(run_lane, launched);
    496 } catch (...) {}
    497 for (std::size_t lane = launched; lane < lane_count; ++lane) run_lane(lane);
    498 for (auto& worker : workers) if (worker.joinable()) worker.join();
    499 for (const auto& error : errors) if (error) std::rethrow_exception(error);
    500 } else update(0, covariance.rows());
    501 for (std::size_t row_begin = 0; row_begin < covariance.rows(); row_begin += tile_size) {
    502 const std::size_t row_end = std::min(row_begin + tile_size, covariance.rows());
    503 for (std::size_t column_begin = 0; column_begin <= row_begin; column_begin += tile_size) {
    504 const std::size_t column_end = std::min(column_begin + tile_size, covariance.cols());
    505 for (std::size_t row = row_begin; row < row_end; ++row) {
    506 double* destination = covariance.row_data(row);
    507 for (std::size_t column = column_begin; column < std::min(column_end, row); ++column) {
    508 double* opposite = covariance.row_data(column);
    509 const double average = 0.5 * destination[column] + 0.5 * opposite[row];
    510 destination[column] = opposite[row] = average;
    511 }
    512 }
    513 }
    514 }
    515 }
    516 if (!covariance.finite()) throw std::overflow_error("Nonfinite aigenlib posterior covariance");
    517 if (mean) mean->swap(updated_mean);
