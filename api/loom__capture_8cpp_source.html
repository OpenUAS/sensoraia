    50/** @brief Establish finite-storage feature transport limits without claiming a trained visual match. */
    51bool validate_onnx_features(void *configuration, const inference_context *context, char *error, size_t capacity)
    52{
    53 const auto &options = *static_cast<const loom_return_options *>(configuration);
    54 if (!loom_return_options_valid(&options)) {
    55 std::snprintf(error, capacity, "Inconsistent LOOM return limits");
    56 return false;
    57 }
    58 if (context->model.input_count != 1 || !context->model.image.channels) {
    59 std::snprintf(error, capacity, "LOOM requires one declared image input");
    60 return false;
    61 }
    62 size_t values = 0;
    63 for (size_t index = 0; index < context->model.output_count; ++index) {
    64 const auto &tensor = context->model.outputs[index];
    65 if (tensor.type != ONNX_FLOAT32 || tensor.bytes / sizeof(float) > 4096 - values) {
    66 std::snprintf(error, capacity, "LOOM requires bounded float32 feature outputs");
    67 return false;
    68 }
    69 values += tensor.bytes / sizeof(float);
    70 }
    71 if (options.frame != LOOM_RETURN_NONE && options.frame != LOOM_RETURN_SIMULATED &&
    72 (context->model.output_count != 1 || (values != 2 && values != 3))) {
    73 std::snprintf(error, capacity, "Declared LOOM return vectors require one float32 output with two or three values");
    74 return false;
    75 }
    76 return true;
    77}

    79/** @brief Embed only in an exclusively owned private JPEG so published or hard-linked references stay immutable. */
    80bool embed_onnx_features(const inference_context *context, const char *image, uint32_t sequence)
    81{
    82 /* The worker serializes callbacks; reusable bounded EXIF storage avoids a large per-request stack frame. */
    83 static exif_document document;
    84 static unsigned char payload[EXIF_MAX_PAYLOAD], record[LOOM_ONNX_FEATURE_CAPACITY];
    85 const size_t record_size = loom_onnx_features_pack(record, sizeof(record), &context->model,
    86 context->capture_identity ? context->frame_sequence : sequence, context->sensor_timestamp_ns);
    87 if (!record_size) return false;
    88 exif_document_init(&document);
    89 size_t payload_size = 0;
    90 if (exif_read_jpeg(image, payload, sizeof(payload), &payload_size) == 0) {
    91 exif_view view;
    92 if (exif_parse(&view, payload, payload_size) || image_exif_import(&document, &view)) return false;
    93 } else if (errno != ENODATA) return false;
    94 unsigned char *destination = exif_document_reserve(&document, EXIF_CAMERA, LOOM_ONNX_FEATURES_TAG,
    95 EXIF_UNDEFINED, static_cast<uint32_t>(record_size));
    96 if (!destination) return false;
    97 for (size_t index = 0; index < record_size; ++index) destination[index] = record[index];
    98 if (exif_serialize(&document, payload, sizeof(payload), &payload_size)) return false;
    99 FILE *output = std::fopen(image, "r+b");
    100 if (!output) return false;
    101 struct stat info{};
    102 bool valid = !fstat(fileno(output), &info) && S_ISREG(info.st_mode) && info.st_nlink == 1 &&
    103 exif_update_jpeg(output, payload, payload_size) == 0 && !std::fflush(output) && !fsync(fileno(output));
    104 if (std::fclose(output)) valid = false;
    105 return valid;
    106}

    108/** @brief Retain exact model/preprocessing identity with features; ordinary offline analysis never mutates its input. */
    109bool write_onnx_features(void *configuration, const inference_context *context,
    110 const char *image, uint32_t sequence, char *error, size_t capacity)
    111{
    112 const auto &options = *static_cast<const loom_return_options *>(configuration);
    113 const auto &model = context->model;
    114 size_t total = 0;
    115 for (size_t slot = 0; slot < model.output_count; ++slot) {
    116 if (model.outputs[slot].type != ONNX_FLOAT32 || model.outputs[slot].bytes / sizeof(float) > 4096 - total) {
    117 std::snprintf(error, capacity, "LOOM features require at most 4096 float32 output values");
    118 return false;
    119 }
    120 total += model.outputs[slot].bytes / sizeof(float);
    121 const auto *values = static_cast<const float *>(model.outputs[slot].data);
    122 for (size_t index = 0; index < model.outputs[slot].bytes / sizeof(float); ++index)
    123 if (!std::isfinite(values[index])) { std::snprintf(error, capacity, "Nonfinite LOOM feature"); return false; }
    124 }
    125 if (context->private_image && !embed_onnx_features(context, image, sequence)) {
    126 std::snprintf(error, capacity, "Cannot embed model-identified LOOM feature evidence");
    127 return false;
    128 }
    129 char filename[INFERENCE_PATH_CAPACITY], digest[65];
    130 const size_t length = std::strlen(image);
    131 if (length < 4 || std::strcmp(image + length - 4, ".jpg") || !inference_file_sha256(image, digest)) return false;
    132 const int count = std::snprintf(filename, sizeof(filename), "%.*s.loom.ini", static_cast<int>(length - 4), image);
    133 if (count < 0 || static_cast<size_t>(count) >= sizeof(filename)) return false;
    134 FILE *output = std::fopen(filename, "wx");
    135 if (!output) { std::snprintf(error, capacity, "Cannot exclusively create LOOM features: %s", filename); return false; }
    136 std::fprintf(output, "schema=LOOM_ONNX_FEATURES_V1\nsequence=%u\nimage_sha256=%s\n"
    137 "model_sha256=%s\nmodel_descriptor_sha256=%s\noutput_count=%zu\n",
    138 sequence, digest, model.graph_sha256, model.descriptor_sha256, model.output_count);
    139 if (options.frame == LOOM_RETURN_NORTH_EAST || options.frame == LOOM_RETURN_BODY_FORWARD_RIGHT) {
    140 const auto *values = static_cast<const float *>(model.outputs[0].data);
    141 std::fprintf(output, "return_vector_frame=%s\nreturn_first=%.9g\nreturn_second=%.9g\nreturn_confidence=%.9g\n",
    142 options.frame == LOOM_RETURN_NORTH_EAST ? "north_east" : "body_forward_right",
    143 static_cast<double>(values[0]), static_cast<double>(values[1]),
    144 model.outputs[0].bytes == 3 * sizeof(float) ? static_cast<double>(values[2]) : 1.0);
    145 } else if (options.frame == LOOM_RETURN_SIMULATED) {
    146 std::fprintf(output, "return_vector_frame=simulated\n");
    147 }
    148 for (size_t slot = 0; slot < model.output_count; ++slot) {
    149 const auto &tensor = model.outputs[slot];
    150 std::fprintf(output, "output.%zu.name=%s\noutput.%zu.shape=", slot, tensor.name, slot);
    151 for (size_t axis = 0; axis < tensor.rank; ++axis)
    152 std::fprintf(output, "%s%lld", axis ? "," : "", static_cast<long long>(tensor.dimensions[axis]));
    153 std::fprintf(output, "\noutput.%zu.values=", slot);
    154 const auto *values = static_cast<const float *>(tensor.data);
    155 for (size_t index = 0; index < tensor.bytes / sizeof(float); ++index)
    156 std::fprintf(output, "%s%.9g", index ? "," : "", static_cast<double>(values[index]));
    157 std::fputc('\n', output);
    158 }
    159 bool written = !std::ferror(output) && std::fflush(output) == 0 && fsync(fileno(output)) == 0;
    160 if (std::fclose(output)) written = false;
    161 if (!written) { unlink(filename); std::snprintf(error, capacity, "Cannot publish LOOM features"); }
    162 return written;
    163}

    176/** @brief Serialize only this retained exposure's feature evidence.
    177 * @param frame Immutable view whose lease remains alive during serialization.
    178 * @param filename Exclusive-create analysis path supplied by the publication owner.
    179 * @return True when the core publishes complete feature evidence.
    180 * @details Keep camera request identity separate from the FC photo number;
    181 * downstream code must establish their association rather than guessing it. */
    182bool write_features(const imx500::FrameView &frame, exif_document *document)
    183{
    184 if (!features_ready(frame)) return true;
    185 const size_t size = loom_features_size(frame.tensor.size());
    186 auto *record = exif_document_reserve(document, EXIF_CAMERA, LOOM_FEATURES_TAG, EXIF_UNDEFINED,
    187 static_cast<uint32_t>(size));
    188 return record && loom_features_pack(record, size, frame.tensor.data(), frame.tensor.size(),
    189 frame.tensor_info.data(), frame.tensor_info.size(),
    190 frame.sequence, frame.sensor_timestamp_ns);
    191}
    192#endif

    226 || std::strcmp(argv[1], "--simulation-mode") == 0

    243 && std::strcmp(argv[1], "--serve") != 0
    244 && std::strcmp(argv[1], "--capture") != 0)) {
    245 std::fprintf(stderr,
    246 "Usage: loomcam --probe nn_loom.rpk\n"
    247 " loomcam --serve nn_loom.rpk WIDTH HEIGHT\n"
    248 " loomcam --capture nn_loom.rpk /absolute/photo.jpg\n"
    249 " loomcam --image FILE --sequence N [--reference-dir DIR] [--model FILE]\n"
    250 " loomcam --self-test | --version\n");
    251 return 2;
    252 }
    253 return imx500::run_worker(argc, argv, processor, "loomcam");
    254#endif
