    102/** @brief Read the capture-server watchdog clock.
    103 * @return Monotonic milliseconds, or zero on failure.
    104 * @details Keep supervision independent of system time corrections. */
    105static uint64_t monotonic_ms(void)
    106{
    107 struct timespec now;
    108 return clock_gettime(CLOCK_MONOTONIC, &now) == 0
    109 ? (uint64_t)now.tv_sec * 1000U + (uint64_t)now.tv_nsec / 1000000U : 0;
    110}

    112/** @brief Wait up to a bounded time for the capture child to exit.
    113 * @param pid Owned positive child PID.
    114 * @param[out] status Wait status, set if this call reaps the child.
    115 * @param timeout_ms Nonnegative budget; zero checks without sleeping.
    116 * @return One if reaped or no longer a child, zero on timeout, or -1 on error.
    117 * @details Use nonblocking waits so a child stuck in a device call does not
    118 * indefinitely block its supervisor. */
    119static int wait_child(pid_t pid, int *status, int timeout_ms)
    120{
    121 const uint64_t started_ms = monotonic_ms();
    122 if (started_ms == 0) return -1;
    123 for (;;) {
    124 pid_t result = waitpid(pid, status, WNOHANG);
    125 if (result == pid || (result < 0 && errno == ECHILD)) return 1;
    126 if (result < 0 && errno != EINTR) return -1;
    127 uint64_t now_ms = monotonic_ms();
    128 if (now_ms == 0 || now_ms - started_ms >= (uint64_t)timeout_ms) return 0;
    129 struct timespec pause = {.tv_sec = 0, .tv_nsec = CHILD_WAIT_INTERVAL_MS * 1000000L};
    130 while (nanosleep(&pause, &pause) != 0 && errno == EINTR) {}
    131 }
    132}

    134/** @brief Stop an owned capture child, escalating only after its grace period.
    135 * @param pid Owned positive child PID.
    136 * @param[out] status Wait status, set if the child is reaped here.
    137 * @return One if ownership ended, zero if still waiting after escalation, or -1 on error.
    138 * @details Reap an already exited child first. Give normal teardown a chance
    139 * to release the camera before sending SIGKILL. */
    140static int terminate_child(pid_t pid, int *status)
    141{
    142 int result = wait_child(pid, status, 0);
    143 if (result != 0) return result;
    144 if (kill(pid, SIGTERM) != 0 && errno != ESRCH) return -1;
    145 result = wait_child(pid, status, CHILD_STOP_GRACE_MS);
    146 if (result != 0) return result;
    147 if (kill(pid, SIGKILL) != 0 && errno != ESRCH) return -1;
    148 return wait_child(pid, status, CHILD_STOP_GRACE_MS);
    149}

    151/** @brief Open a close-on-exec pipe for child protocol setup.
    152 * @param[out] descriptors Read and write descriptors; valid only on success.
    153 * @return Zero on success, or -1 with errno on failure.
    154 * @details Keep both descriptors above stderr so child's descriptor
    155 * redirection cannot alias an endpoint when a standard descriptor was closed. */
    156static int create_cloexec_pipe(int descriptors[2])
    157{
    158 if (pipe(descriptors) != 0) return -1;
    159 for (size_t index = 0; index < 2; ++index) {
    160 if (descriptors[index] <= STDERR_FILENO) {
    161 const int replacement = fcntl(descriptors[index], F_DUPFD_CLOEXEC, STDERR_FILENO + 1);
    162 if (replacement < 0) {
    163 const int saved_errno = errno;
    164 close(descriptors[0]);
    165 close(descriptors[1]);
    166 errno = saved_errno;
    167 return -1;
    168 }
    169 close(descriptors[index]);
    170 descriptors[index] = replacement;
    171 }
    172 int flags = fcntl(descriptors[index], F_GETFD);
    173 if (flags < 0 || fcntl(descriptors[index], F_SETFD, flags | FD_CLOEXEC) != 0) {
    174 int saved_errno = errno;
    175 close(descriptors[0]);
    176 close(descriptors[1]);
    177 errno = saved_errno;
    178 return -1;
    179 }
    180 }
    181 return 0;
    182}

    207/** @brief Parse one server timing payload after its protocol prefix.
    208 * @param text Space-separated unsigned timing fields.
    209 * @param timing Receives validated timing evidence.
    210 * @param callback True for callback sequence/drop fields, false for polling timing.
    211 * @return 1 for valid evidence, 0 for malformed or implausible evidence.
    212 * @details Parsing is intentionally strict: timing metadata enriches EXIF but must
    213 * never be accepted merely because a server process printed a vaguely similar line. */
    214static int parse_capture_timing(const char *text, struct capture_timing *timing, bool callback)
    215{
    216 if (!text || !timing) return 0;
    217 uint64_t values[4] = {0};
    218 const size_t count = callback ? 4 : 2;
    219 for (size_t index = 0; index < count; ++index) {
    220 char *end;
    221 if (*text < '0' || *text > '9') return 0;
    222 errno = 0;
    223 unsigned long long value = strtoull(text, &end, 10);
    224 if (errno != 0 || value > UINT64_MAX) return 0;
    225 values[index] = value;
    226 if (index + 1 == count) {
    227 if (*end != '\0') return 0;
    228 } else {
    229 if (*end != ' ') return 0;
    230 text = end + 1;
    231 }
    232 }
    233 struct capture_timing parsed = {
    234 .request_monotonic_us = values[0], .arrival_monotonic_us = values[1],
    235 .callback_sequence = values[2], .callback_drops = values[3], .callback_arrival = callback
    236 };
    237 if (!capture_timing_valid(&parsed)) return 0;
    238 *timing = parsed;
    239 return 1;
    240}

    247/** @brief Read server diagnostic lines until an expected terminal reply arrives.
    248 * @param expected Success line required to complete the operation.
    249 * @param timeout_ms Whole-operation deadline.
    250 * @param keep_running Optional cancellation flag; checked between bounded polls
    251 * so stopping USHER need not wait for a thermal camera's full startup deadline.
    252 * @return 0 on the expected reply, -1 for cancellation, timeout, EOF, or server error.
    253 * @details Informational timing lines are parsed before the terminal reply. The buffer
    254 * avoids a syscall per character while retaining line framing. One absolute deadline
    255 * prevents a noisy or malicious child from extending an operation indefinitely. */
    256static int read_server_status(const char *expected, int timeout_ms,
    257 const volatile sig_atomic_t *keep_running)
    258{
    259 char line[256] = {0};
    260 size_t line_size = 0;
    261 const uint64_t started_ms = monotonic_ms();
    262 if (started_ms == 0) return -1;
    263
    264 for (;;) {
    265 if (!capture_running(keep_running)) return -1;
    266 if (status_buffer_pos == status_buffer_size) {
    267 if (capture_server_output < 0) return -1;
    268 uint64_t now_ms = monotonic_ms();
    269 if (now_ms == 0 || now_ms - started_ms >= (uint64_t)timeout_ms) return -1;
    270 int remaining_ms = timeout_ms - (int)(now_ms - started_ms);
    271 int poll_timeout_ms = keep_running != NULL && remaining_ms > 100 ? 100 : remaining_ms;
    272 struct pollfd descriptor = {capture_server_output, POLLIN, 0};
    273 int poll_result = poll(&descriptor, 1, poll_timeout_ms);
    274 if (poll_result < 0 && errno == EINTR) continue;
    275 if (poll_result == 0) continue;
    276 if (poll_result < 0
    277 || ((descriptor.revents & POLLIN) == 0
    278 && (descriptor.revents & (POLLERR | POLLHUP | POLLNVAL)) != 0)) {
    279 return -1;
    280 }
    281 ssize_t bytes_read = read(capture_server_output, status_buffer, sizeof(status_buffer));
    282 if (bytes_read < 0 && (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK)) continue;
    283 if (bytes_read <= 0) {
    284 return -1;
    285 }
    286 status_buffer_size = (size_t)bytes_read;
    287 status_buffer_pos = 0;
    288 }
    289 char character = status_buffer[status_buffer_pos++];
    290 if (character != '\n' && ((unsigned char)character < 32 || character == 127 || line_size + 1 >= sizeof(line))) return -1;
    291 if (character != '\n' && line_size + 1 < sizeof(line)) {
    292 line[line_size++] = character;
    293 continue;
    294 }
    295
    296 line[line_size] = '\0';
    297 double delay;
    298 char trailing;
    299 if (strcmp(expected, "LWIR_SERVER_OK") == 0 && !capture_timing_valid(&capture_times)
    300 && sscanf(line, "LWIR_SERVER_DELAY %lf %c", &delay, &trailing) == 1
    301 && isfinite(delay) && delay >= 0 && delay <= 20) {
    302 capture_delay_s = delay;
    303 }
    304 const char timing_prefix[] = "LWIR_SERVER_TIMING ";
    305 if (strcmp(expected, "LWIR_SERVER_OK") == 0
    306 && !capture_times.callback_arrival
    307 && line_size >= sizeof(timing_prefix)
    308 && strncmp(line, timing_prefix, sizeof(timing_prefix) - 1) == 0
    309 && parse_capture_timing(line + sizeof(timing_prefix) - 1, &capture_times, false)) {
    310 capture_delay_s = (capture_times.arrival_monotonic_us - capture_times.request_monotonic_us) / 1e6;
    311 }
    312 const char callback_prefix[] = "LWIR_SERVER_CALLBACK_TIMING ";
    313 if (strcmp(expected, "LWIR_SERVER_OK") == 0
    314 && line_size >= sizeof(callback_prefix)
    315 && strncmp(line, callback_prefix, sizeof(callback_prefix) - 1) == 0
    316 && parse_capture_timing(line + sizeof(callback_prefix) - 1, &capture_times, true)) {
    317 capture_delay_s = (capture_times.arrival_monotonic_us - capture_times.request_monotonic_us) / 1e6;
    318 }
    319 if (strcmp(line, expected) == 0) {
    320 return 0;
    321 }
    322 if (strncmp(line, "LWIR_SERVER_ERROR", 17) == 0) {
    323 fprintf(stderr, "LWIR_CAM_PIPE:\t%s\n", line);
    324 return -1;
    325 }
    326 line_size = 0;
    327 }
    328}

    330/** @brief Close pipes and reap or terminate the owned capture server.
    331 * @details Buffered response state is reset before the next spawn. Closing stdin asks
    332 * cooperative server versions to exit; SIGTERM is used only when it remains alive. */
    333static void stop_capture_server(void)
    334{
    335 status_buffer_size = status_buffer_pos = 0;
    336 if (capture_server_input >= 0) {
    337 close(capture_server_input);
    338 capture_server_input = -1;
    339 }
    340 if (capture_server_output >= 0) {
    341 close(capture_server_output);
    342 capture_server_output = -1;
    343 }
    344 if (capture_server_pid < 0) {
    345 return;
    346 }
    347
    348 int status = 0;
    349 if (terminate_child(capture_server_pid, &status) <= 0) {
    350 fprintf(stderr, "LWIR_CAM_PIPE:\tcapture server pid %ld could not be reaped\n",
    351 (long)capture_server_pid);
    352 return;
    353 }
    354 capture_server_pid = -1;
    355}

    357/** Send one line request to the running server and wait for its reply. On any
    358 * failure (broken pipe, dead process, or a timed-out/erroring reply) the server
    359 * state is fully reset so a caller's retry starts with a clean respawn instead of
    360 * repeatedly hitting the same stale, already-dead connection. */
    361/** @brief Send one newline-terminated server request and require an OK reply.
    362 * @return 0 on end-to-end success or -1 after fully resetting failed server state. */
    363static int send_request_and_wait(const char *request, size_t request_size, int timeout_ms, const char *expected)
    364{
    365 const uint64_t started_ms = monotonic_ms();
    366 if (!started_ms || timeout_ms <= 0) { stop_capture_server(); return -1; }
    367 size_t sent = 0;
    368 while (sent < request_size) {
    369 if (!capture_running(running)) { stop_capture_server(); return -1; }
    370 const uint64_t now_ms = monotonic_ms();
    371 if (!now_ms || now_ms < started_ms || now_ms - started_ms >= (uint64_t)timeout_ms) {
    372 stop_capture_server(); return -1;
    373 }
    374 ssize_t written = write(capture_server_input, request + sent, request_size - sent);
    375 if (written > 0) {
    376 sent += (size_t)written;
    377 } else if (written < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
    378 struct pollfd writable = {.fd = capture_server_input, .events = POLLOUT};
    379 const int remaining = timeout_ms - (int)(now_ms - started_ms);
    380 const int ready = poll(&writable, 1, running && remaining > 100 ? 100 : remaining);
    381 if (ready < 0 && errno == EINTR) continue;
    382 if (ready == 0 || (ready > 0 && !(writable.revents & (POLLERR | POLLHUP | POLLNVAL)))) continue;
    383 stop_capture_server();
    384 return -1;
    385 } else if (written == 0 || errno != EINTR) {
    386 if (!written) errno = EIO;
    387 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to send request: %s\n", strerror(errno));
    388 stop_capture_server();
    389 return -1;
    390 }
    391 }
    392 const uint64_t elapsed_ms = monotonic_ms() - started_ms;
    393 if (elapsed_ms >= (uint64_t)timeout_ms || read_server_status(expected, timeout_ms - (int)elapsed_ms, running) != 0) {
    394 stop_capture_server();
    395 return -1;
    396 }
    397 return 0;
    398}

    400/** @brief Start the persistent thermal capture subprocess.
    401 * @param unused Reserved backend argument.
    402 * @param keep_running Optional shutdown flag used during the readiness wait.
    403 * @return Zero after readiness, or -1 on ownership, setup, or protocol failure.
    404 * @details Resolve any old owner before launching. Connect separate command
    405 * and response pipes and require the startup acknowledgement before capture. */
    406static int start_capture_server(const char *unused,
    407 const volatile sig_atomic_t *keep_running)
    408{
    409 (void)unused;
    410 if (capture_server_pid >= 0) {
    411 stop_capture_server();
    412 if (capture_server_pid >= 0) return -1;
    413 }
    414 char resolved_dir[PATH_MAX];
    415 const char *photo_dir = usher_resolve_path(photo_directory, resolved_dir, sizeof(resolved_dir));
    416
    417 if (usher_ensure_directory(photo_dir) != 0) {
    418 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to create photo directory %s: %s\n",
    419 photo_dir, strerror(errno));
    420 return -1;
    421 }
    422
    423 struct stat directory_status;
    424 if (stat(photo_dir, &directory_status) != 0
    425 || !S_ISDIR(directory_status.st_mode)
    426 || access(photo_dir, W_OK) != 0) {
    427 fprintf(stderr, "LWIR_CAM_PIPE:\tphoto directory is not writable: %s\n",
    428 photo_dir);
    429 return -1;
    430 }
    431 if (access(capture_command, X_OK) != 0) {
    432 fprintf(stderr, "LWIR_CAM_PIPE:\tcapture command is not executable: %s: %s\n",
    433 capture_command, strerror(errno));
    434 return -1;
    435 }
    436
    437 int command_pipe[2];
    438 int response_pipe[2];
    439 if (create_cloexec_pipe(command_pipe) != 0) {
    440 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to create capture-server pipes: %s\n",
    441 strerror(errno));
    442 return -1;
    443 }
    444 if (create_cloexec_pipe(response_pipe) != 0) {
    445 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to create capture-server pipes: %s\n",
    446 strerror(errno));
    447 close(command_pipe[0]);
    448 close(command_pipe[1]);
    449 return -1;
    450 }
    451
    452 posix_spawn_file_actions_t actions;
    453 int action_result = posix_spawn_file_actions_init(&actions);
    454 bool actions_initialized = action_result == 0;
    455 if (action_result == 0) action_result = posix_spawn_file_actions_adddup2(&actions, command_pipe[0], STDIN_FILENO);
    456 if (action_result == 0) action_result = posix_spawn_file_actions_adddup2(&actions, response_pipe[1], STDOUT_FILENO);
    457 if (action_result == 0) action_result = posix_spawn_file_actions_addclose(&actions, command_pipe[1]);
    458 if (action_result == 0) action_result = posix_spawn_file_actions_addclose(&actions, response_pipe[0]);
    459 if (action_result != 0) {
    460 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to configure capture server: %s\n",
    461 strerror(action_result));
    462 if (actions_initialized) posix_spawn_file_actions_destroy(&actions);
    463 close(command_pipe[0]);
    464 close(command_pipe[1]);
    465 close(response_pipe[0]);
    466 close(response_pipe[1]);
    467 return -1;
    468 }

    485 int spawn_result = posix_spawn(&capture_server_pid, capture_command,
    486 &actions, NULL, arguments, environ);
    487 posix_spawn_file_actions_destroy(&actions);
    488 close(command_pipe[0]);
    489 close(response_pipe[1]);
    490 if (spawn_result != 0) {
    491 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to start %s: %s\n",
    492 capture_command, strerror(spawn_result));
    493 close(command_pipe[1]);
    494 close(response_pipe[0]);
    495 capture_server_pid = -1;
    496 return -1;
    497 }
    498 capture_server_input = command_pipe[1];
    499 capture_server_output = response_pipe[0];
    500 const int endpoints[] = {capture_server_input, capture_server_output};
    501 for (size_t index = 0; index < sizeof(endpoints) / sizeof(endpoints[0]); ++index) {
    502 const int flags = fcntl(endpoints[index], F_GETFL);
    503 if (flags < 0 || fcntl(endpoints[index], F_SETFL, flags | O_NONBLOCK) < 0) {
    504 stop_capture_server();
    505 return -1;
    506 }
    507 }
    508 if (read_server_status("LWIR_SERVER_READY", LWIR_SERVER_START_TIMEOUT_MS, keep_running) != 0) {
    509 fprintf(stderr, "LWIR_CAM_PIPE:\tcapture server did not become ready\n");
    510 stop_capture_server();
    511 return -1;
    512 }
    513
    514 printf("LWIR_CAM_PIPE:\tcapture command: %s\n", capture_command);
    515 printf("LWIR_CAM_PIPE:\tphoto directory: %s\n", photo_dir);
    516 printf("LWIR_CAM_PIPE:\tpersistent capture server ready\n");
    517 return 0;
    518}

    653int lwir_cam_pipe_shoot(char *filename, size_t filename_size, int image_number)
    654{
    655 capture_delay_s = -1;
    656 capture_times = (struct capture_timing) {0};
    657 if (filename == NULL || filename_size == 0 || image_number < 0) {
    658 return -1;
    659 }
    660
    661 char resolved_dir[PATH_MAX];
    662 const char *photo_dir = usher_resolve_path(photo_directory, resolved_dir, sizeof(resolved_dir));
    663
    664 int length = snprintf(filename, filename_size, "%s/l%06d.jpg",
    665 photo_dir, image_number);
    666 if (length < 0 || (size_t)length >= filename_size) {
    667 filename[0] = '\0';
    668 return -1;
    669 }
    670
    671 struct stat existing;
    672 if (lstat(filename, &existing) == 0 || errno != ENOENT) {
    673 fprintf(stderr, "LWIR_CAM_PIPE:\tfailed to replace %s: %s\n", filename, strerror(errno));
    674 filename[0] = '\0';
    675 return -1;
    676 }
    677
    678 if (capture_server_pid < 0 || capture_server_input < 0 || capture_server_output < 0) {
    679 filename[0] = '\0';
    680 return -1;
    681 }
    682 char request[PATH_MAX + 2];
    683 int request_size = snprintf(request, sizeof(request), "%s\n", filename);
    684 if (request_size < 0 || (size_t)request_size >= sizeof(request)) {
    685 filename[0] = '\0';
    686 return -1;
    687 }
    688 if (send_request_and_wait(request, (size_t)request_size, LWIR_REQUEST_TIMEOUT_MS, "LWIR_SERVER_OK") != 0) {
    689 fprintf(stderr, "LWIR_CAM_PIPE:\tcapture server failed to save %s\n", filename);
    690 unlink(filename);
    691 filename[0] = '\0';
    692 return -1;
    693 }
    694
    695 struct stat image_status;
    696 if (stat(filename, &image_status) != 0 || !S_ISREG(image_status.st_mode)
    697 || image_status.st_size == 0) {
    698 fprintf(stderr, "LWIR_CAM_PIPE:\t%s did not create a valid image at %s\n",
    699 capture_command, filename);
    700 unlink(filename);
    701 filename[0] = '\0';
    702 return -1;
    703 }
    704 return 0;
    705}
