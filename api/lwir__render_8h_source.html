    18#ifndef LWIR_RENDER_H
    19#define LWIR_RENDER_H

    21/** @file
    22 * @brief Display rendering of Tiny1-C frames into JPEG-ready Y/Cb/Cr scanlines.
    23 * @details Rendering is strictly a display concern: it reads an immutable combined frame
    24 * (see lwir_frame.h) and never reorders, rewrites or calibrates the stored raw thermal
    25 * bytes, so measurements and saved data stay in native coordinates.
    26 *
    27 * Workflow per frame: lwir_render_prepare() validates options and resolves one display
    28 * window (struct lwir_render_range); lwir_render_dimensions() gives the oriented size;
    29 * lwir_render_row() is then called once per display row to fill a caller-owned buffer
    30 * of `display_width * 3` bytes, so no full RGB or rotated frame is ever materialized.
    31 *
    32 * Views: LWIR_SENSOR passes the camera's own YUYV image through; LWIR_FIXED maps an
    33 * operator raw window; LWIR_PERCENTILE derives the window from the frame's histogram.
    34 * Thermal views map the window through a palette, flag below/above/invalid samples with
    35 * distinct colours, and may draw a legend strip. An optional measured correction profile
    36 * applies scale/offset, a support range and bad-pixel replacement for display only.
    37 *
    38 * Orientation is applied as clockwise rotation, then mirror, then flip; rows are produced
    39 * by one inverse transform per scanline and a signed native stride.
    40 *
    41 * SIMD: NEON (ARM64) and SSSE3 (x86-64, runtime-detected) accelerate +/-1-stride rows and
    42 * produce the same bytes as the scalar code, which handles every remaining case.
    43 */
    44#include "lwir_frame.h"
    45

    50/** @brief Source of display intensity. */
    51enum lwir_view {
    52 LWIR_SENSOR, /**< Camera's own YUYV image; no thermal window, profile or denoise. */
    53 LWIR_FIXED, /**< Fixed raw Kelvin*64 window from lwir_render_options low/high. */
    54 LWIR_PERCENTILE /**< Window from per-frame low/high percentiles of valid samples. */
    55};
    56/** @brief Thermal palette; ignored by LWIR_SENSOR. */
    57enum lwir_palette {
    58 LWIR_WHITE_HOT, /**< Greyscale, warmer is brighter. */
    59 LWIR_BLACK_HOT, /**< Greyscale, warmer is darker. */
    60 LWIR_IRON /**< Six-anchor black-purple-red-orange-yellow ramp (scalar path only). */
    61};

    63/** @brief Capacity of the sorted bad-pixel list in struct lwir_profile. */
    64#define LWIR_MAX_BAD_PIXELS 128U
    65/** @brief Measured display-correction profile for one camera, gain label and geometry.
    66 * @details Applied only when rendering; raw frames and measurements are unaffected.
    67 * Corrected value = round(raw * scale_ppm / 1e6) + offset_mk * 64 / 1000 (truncated),
    68 * in the same Kelvin*64 units as raw samples.
    69 */
    70struct lwir_profile {
    71 char serial[64]; /**< Device serial (or explicit replay identity) the profile applies to. */
    72 char gain[32]; /**< Operator gain label the profile was measured with. */
    73 char sha256[65]; /**< Hex digest text identifying the profile source (63 chars + NUL). */
    74 unsigned width; /**< Native width the bad-pixel indices refer to. */
    75 unsigned height; /**< Native height the bad-pixel indices refer to. */
    76 unsigned scale_ppm; /**< Gain in parts per million; renderer accepts 500000..1500000. */
    77 int offset_mk; /**< Offset in millikelvin; renderer accepts -50000..50000. */
    78 uint16_t range_min; /**< Lowest supported raw code (non-zero); smaller samples render invalid. */
    79 uint16_t range_max; /**< Highest supported raw code (< 65535); larger samples render invalid. */
    80 unsigned bad_count; /**< Used entries of bad_pixels. */
    81 /** Native raster indices (row * width + column), strictly ascending for binary search. */
    82 uint32_t bad_pixels[LWIR_MAX_BAD_PIXELS];
    83};

    102/** @brief Display policy for one rendering request; never affects stored samples. */
    103struct lwir_render_options {
    104 enum lwir_view view;
    105 enum lwir_palette palette;
    106 /** LWIR_FIXED window in corrected Kelvin*64 units; requires 0 <= low < high <= 65535. */
    107 int32_t low;
    108 int32_t high;
    109 /** LWIR_PERCENTILE bounds, 0..100 with low_percentile < high_percentile. */
    110 unsigned low_percentile;
    111 unsigned high_percentile;
    112 /** Draw a 16-row gradient and degC label strip; thermal views, display >= 112x24 only. */
    113 bool legend;
    114 /* Display only: clockwise rotation, then left/right mirror, then up/down flip. */
    115 unsigned rotation;
    116 bool mirror;
    117 bool flip;
    118 /** Edge-preserving 3x3 mean threshold in raw units (0 disables, max 640 = 10 K). */
    119 unsigned denoise_raw;
    120 /** JPEG quality 1..100; zero preserves the library's quality-100 default. */
    121 unsigned jpeg_quality;
    122 /** Optional measured correction, borrowed; must match the native geometry. */
    123 const struct lwir_profile *profile;
    124};

    126/** @brief Resolved display window and diagnostic counts from lwir_render_prepare(). */
    127struct lwir_render_range {
    128 int32_t low; /**< Window start in corrected Kelvin*64 units (0 for LWIR_SENSOR). */
    129 int32_t high; /**< Window end, always > low (255 for LWIR_SENSOR). */
    130 size_t below; /**< Valid samples below the window (rendered as the below flag colour). */
    131 size_t above; /**< Valid samples above the window (rendered as the above flag colour). */
    132 size_t invalid; /**< Endpoint, bad-without-neighbours or out-of-profile-support samples. */
    133};

    135/** @brief Resolve one deterministic display window while retaining original samples.
    136 * @details LWIR_SENSOR returns {0, 255} without reading samples. LWIR_FIXED copies the
    137 * requested window. LWIR_PERCENTILE ranks valid display samples in a 65536-bin histogram.
    138 * All thermal views then count invalid/below/above samples for diagnostics.
    139 * @param frame Immutable combined frame, or image-only YUYV for LWIR_SENSOR.
    140 * @param bytes Exact size.
    141 * @param width Native width.
    142 * @param height Native height.
    143 * @param options Display-only policy.
    144 * @param range Resolved bounds/clipping counts, valid only on success.
    145 * @return True for a supported rendering request. On failure errno is EINVAL for invalid
    146 * arguments/options/profile/legend geometry, or ENODATA when a percentile view finds no
    147 * valid sample.
    148 * @warning Uses a static histogram: not re-entrant or thread-safe; call serially.
    149 */
    150bool lwir_render_prepare(const uint8_t *frame, size_t bytes, unsigned width, unsigned height,
    151 const struct lwir_render_options *options, struct lwir_render_range *range);
    152/** @brief Derive display dimensions; quarter-turns do not change thermal geometry.
    153 * @param width Native width.
    154 * @param height Native height.

    174/** @brief Emit one bounded Y/Cb/Cr scanline, without a full RGB or rotated frame.
    175 * @param frame Immutable combined frame, or image-only YUYV for LWIR_SENSOR.
