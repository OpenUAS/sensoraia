    38/** RAII guard for the C optimizer and its separately allocated boundary transform. */
    39struct Optimizer {
    40 cmaes_t state{}; /**< Optimizer-owned population/covariance resources. */
    41 cmaes_boundary_transformation_t bounds{}; /**< Bounded-coordinate transform, separate from optimizer state. */
    42 bool initialized = false; /**< Exit only after successful final initialization. */
    43 bool bounds_initialized = false; /**< Transform cleanup remains valid if optimizer setup fails. */
    44 /**
    45 * @brief Release initialized C resources in reverse dependency order.
    46 * @details Partial initialization is possible when the C API throws through
    47 * an error path, so each resource has an independent readiness flag.
    48 */
    49 ~Optimizer() {
    50 if (initialized) cmaes_exit(&state);
    51 if (bounds_initialized) cmaes_boundary_transformation_exit(&bounds);
    52 }
    53};
    54}
    55std::vector<Point> refine(const Grid& grid, const Vector& mean, const Matrix& covariance,
    56 const Point& start, const std::vector<Point>& initial, const PlannerOptions& options) {
    57 if (initial.empty() || (options.keep_planning && !options.keep_planning())) return initial;
    58 // Excluding fixed Z is more than an optimization: it prevents numerical
    59 // search from wasting evaluations on a coordinate that cannot change.
    60 const int coordinates = options.zmin == options.zmax ? 2 : 3;
    61 const int dimension = static_cast<int>(coordinates * initial.size());
    62 std::vector<double> lower(dimension), upper(dimension), deviations(dimension), latent(dimension), bounded(dimension);
    63 auto best = initial;
    64 for (std::size_t waypoint = 0; waypoint < initial.size(); ++waypoint) {
    65 best[waypoint] = clip_point(grid, initial[waypoint], options);
    66 for (int coordinate = 0; coordinate < coordinates; ++coordinate) {
    67 const auto index = coordinates * waypoint + coordinate;
    68 lower[index] = coordinate == 0 ? grid.xmin + options.margin_cells * grid.step : coordinate == 1 ? grid.ymin + options.margin_cells * grid.step : options.zmin;
    69 upper[index] = coordinate == 0 ? grid.xmax - options.margin_cells * grid.step : coordinate == 1 ? grid.ymax - options.margin_cells * grid.step : options.zmax;
    70 deviations[index] = coordinate == 2 ? options.step_z : options.step_xy;
    71 bounded[index] = best[waypoint][coordinate];
    72 }
    73 }
    74 Optimizer optimizer;
    75 cmaes_boundary_transformation_init(&optimizer.bounds, lower.data(), upper.data(), lower.size());
    76 optimizer.bounds_initialized = true;
    77 // CMA-ES samples an unbounded Gaussian. Inverse-transform the warm start so
    78 // the corresponding bounded point remains the center of its first search.
    79 cmaes_boundary_transformation_inverse(&optimizer.bounds, bounded.data(), latent.data(), latent.size());
    80 char silent_filename[] = "non";
    81 cmaes_init_para(&optimizer.state, dimension, latent.data(), deviations.data(), options.seed, options.popsize, silent_filename);
    82 optimizer.state.sp.filename = silent_filename;
    83 double* fitness = cmaes_init_final(&optimizer.state);
    84 optimizer.state.sp.filename = nullptr;
    85 optimizer.initialized = true;
    86 const auto mask = importance_mask(mean, covariance, options);
    87 double best_value = trajectory_objective(grid, covariance, mask, start, best, options);
    88 int evaluations = 0;
    89 // Consume complete populations only: the C library updates its covariance
    90 // from a generation as a unit, so a partial final batch is not meaningful.
    91 for (int generation = 0; generation < options.maxiter && evaluations + options.popsize <= options.maxfevals; ++generation) {
    92 double* const* population = cmaes_SamplePopulation(&optimizer.state);
    93 for (int member = 0; member < options.popsize; ++member) {
    94 if (options.keep_planning && !options.keep_planning()) return best;
    95 cmaes_boundary_transformation(&optimizer.bounds, population[member], bounded.data(), bounded.size());
    96 // Start from the best full route and replace only searched axes;
    97 // in fixed-altitude mode this preserves every waypoint's Z exactly.
    98 auto proposal = best;
    99 for (std::size_t waypoint = 0; waypoint < proposal.size(); ++waypoint) {
    100 for (int coordinate = 0; coordinate < coordinates; ++coordinate)
    101 proposal[waypoint][coordinate] = bounded[coordinates * waypoint + coordinate];
    102 }
    103 fitness[member] = trajectory_objective(grid, covariance, mask, start, proposal, options);
    104 if (!std::isfinite(fitness[member])) throw std::runtime_error("Nonfinite CMA-ES objective");
    105 if (fitness[member] < best_value) { best_value = fitness[member]; best = std::move(proposal); }
    106 }
    107 evaluations += options.popsize;
    108 cmaes_UpdateDistribution(&optimizer.state, fitness);
    109 if (cmaes_TestForTermination(&optimizer.state)) break;
    110 }
    111 return best;
    112}

    114std::vector<Point> plan(const Grid& grid, const Vector& mean, const Matrix& covariance,
    115 const Point& start, const PlannerOptions& options) {
    116 auto initial = grid_search(grid, mean, covariance, start, options);
    117 return options.refine ? refine(grid, mean, covariance, start, initial, options) : initial;
