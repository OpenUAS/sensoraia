    28/** @brief Optional token-bound evidence mapping a controller pose interval to MORA time. */
    29struct pose_clock_evidence {
    30 union usher_clock_request_union token; /**< Request token joining this pose to one clock exchange. */
    31 struct clock_interval sample_time; /**< Conservative monotonic interval covering the FC sample. */
    32 uint64_t probe_sent_us; /**< Host monotonic send time for the matching request. */
    33 uint64_t probe_received_us; /**< Host monotonic receive time for the matching response. */
    34 uint32_t probe_receive_fc_us; /**< Flight-controller receive timestamp from the exchange. */
    35 uint32_t probe_transmit_fc_us; /**< Flight-controller transmit timestamp from the exchange. */
    36 bool mapped; /**< True only when token and interval passed clock-alignment validation. */
    37};

    39/** @brief Snapshot counters for accepted, flushed, dropped, and rejected pose records. */
    40struct pose_log_stats {
    41 uint64_t accepted; /**< Records admitted to the in-memory queue. */
    42 uint64_t synced; /**< Records in batches successfully fdatasync'ed. */
    43 uint64_t dropped; /**< Valid records discarded due to lock/capacity/I/O pressure. */
    44 uint64_t rejected; /**< Invalid payloads or clock evidence not admitted. */
    45 int error; /**< First writer/startup errno-style failure, or zero. */
    46};

    48/** @brief Start a durable asynchronous CSV session, or disable logging when directory is empty.
    49 * @param directory Writable destination directory, optionally using `~` shorthand.
    50 * @return 0 on active/disabled success, otherwise -1. */
    51int pose_log_start(const char *directory);
    52/** @brief Validate and enqueue a raw pose payload without clock mapping evidence.
    53 * @return 1 when queued, 0 when intentionally dropped due to contention/capacity, or -1 when invalid. */
    54int pose_log_record(const uint8_t *payload, size_t length, uint64_t receive_monotonic_us);
    55/** @brief Validate and enqueue a pose payload with optional mapped timing provenance.
    56 * @details Trylock protects capture latency: logging is best-effort evidence, never a reason
    57 * to block the real-time UART path. */
    58int pose_log_record_clocked(const uint8_t *payload, size_t length, uint64_t receive_monotonic_us,
    59 const struct pose_clock_evidence *evidence);
    60/** @brief Return a synchronized snapshot of logging health and loss counters. */
    61struct pose_log_stats pose_log_status(void);
    62/** @brief Flush queued records, join the writer, and close the CSV file.
    63 * @return 0 after durable orderly shutdown, or -1 if a pending I/O error exists. */
    64int pose_log_stop(void);
