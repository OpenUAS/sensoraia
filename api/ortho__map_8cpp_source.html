    54std::vector<std::string> csvFields(const std::string& line) {
    55 std::vector<std::string> result;
    56 std::string value;
    57 bool quoted = false, closed = false;
    58 for (std::size_t i = 0; i < line.size(); ++i) {
    59 const char c = line[i];
    60 if (quoted) {
    61 if (c == '"' && i + 1 < line.size() && line[i + 1] == '"') { value += '"'; ++i; }
    62 else if (c == '"') { quoted = false; closed = true; }
    63 else value += c;
    64 } else if (c == ',') { result.push_back(value); value.clear(); closed = false; }
    65 else if (c == '"' && value.empty() && !closed) quoted = true;
    66 else if (closed || c == '"') throw std::runtime_error("Malformed source CSV quoting");
    67 else if (c != '\r') value += c;
    68 }
    69 if (quoted) throw std::runtime_error("Unclosed source CSV quote");
    70 result.push_back(value);
    71 return result;

    73std::vector<OrthoSource> readSources(const fsPath& path, const fsPath& sourceDirectory = {}) {
    74 std::ifstream input(path);
    75 if (!input) throw std::runtime_error("Cannot open orthorectification source CSV: " + path.u8string());
    76 std::string line;
    77 if (!std::getline(input, line)) throw std::runtime_error("Empty source CSV");
    78 const auto header = csvFields(line);
    79 std::map<std::string, std::size_t> columns;
    80 for (std::size_t i = 0; i < header.size(); ++i)
    81 if (!columns.emplace(header[i], i).second) throw std::runtime_error("Duplicate source CSV column");
    82 const bool haveUtc = columns.count("capture_utc") != 0;
    83 for (const auto& group : {std::vector<const char*>{"fx_px", "fy_px", "cx_px", "cy_px"},
    84 std::vector<const char*>{"lens_k1", "lens_k2", "lens_blue_scale", "lens_red_scale"}}) {
    85 const auto present = std::count_if(group.begin(), group.end(),
    86 [&](const char* key) { return columns.count(key) != 0; });
    87 if (present != 0 && present != static_cast<std::ptrdiff_t>(group.size()))
    88 throw std::runtime_error("Source CSV has incomplete optical calibration columns");
    89 }
    90 if ((columns.count("utc_absolute_seconds") != 0) != haveUtc ||
    91 (columns.count("utc_read_qpc_100ns") != 0) != haveUtc)
    92 throw std::runtime_error("Source CSV has incomplete capture UTC columns");
    93 if ((columns.count("tile_row") != 0) != (columns.count("tile_column") != 0))
    94 throw std::runtime_error("Source CSV has incomplete capture grid columns");
    95 std::vector<OrthoSource> sources;
    96 std::map<fsPath, bool> paths;
    97 // One fixed camera/altitude/time session keeps the intrinsic and map scale valid for every row.
    98 while (std::getline(input, line)) {
    99 const auto fields = csvFields(line);
    100 if (fields.size() != header.size()) throw std::runtime_error("Truncated/malformed source CSV row");
    101 const auto text = [&](const char* key) -> const std::string& {
    102 const auto it = columns.find(key);
    103 if (it == columns.end()) throw std::runtime_error(std::string("Source CSV lacks ") + key);
    104 return fields[it->second];
    105 };
    106 const auto get = [&](const char* key) { return number(text(key), key); };
    107 if (text("trigger") != "box_tile") continue;
    108 const fsPath name = std::filesystem::u8path(text("filename"));
    109 if (name.empty() || name.has_parent_path() || name.extension() != ".jpg")
    110 throw std::runtime_error("Source CSV filename must be a local JPEG basename");
    111 OrthoSource s;
    112 s.path = (sourceDirectory.empty() ? path.parent_path() : sourceDirectory) / name;
    113 if (!paths.emplace(s.path, true).second) throw std::runtime_error("Duplicate source photo");
    114 s.latitude = get("observed_latitude"); s.longitude = get("observed_longitude");
    115 s.altitude = get("observed_altitude_m");
    116 s.pitch = get("observed_pitch_deg"); s.bank = get("observed_bank_deg");
    117 s.heading = get("observed_heading_deg"); s.fov = get("sdk_fov_deg");
    118 range(s.latitude, -80, 84, "source latitude"); range(s.longitude, -180, 180, "source longitude");
    119 range(s.fov, 10, 120, "source vertical FOV");
    120 if (text("projection") != "perspective" || !text("map_epsg").empty() ||
    121 std::abs(s.pitch - 90) > 0.00002 || std::abs(std::remainder(s.heading - s.bank, 360.0)) > 0.001 ||
    122 get("commanded_pitch_deg") != 90 || get("commanded_bank_deg") != 0 ||
    123 get("commanded_heading_deg") != 0)
    124 throw std::runtime_error("Orthorectification requires unrotated, confirmed true-north nadir source photos");
    125 const auto integer = [&](const char* key, double low, double high) {
    126 const double n = get(key); range(n, low, high, key);
    127 if (std::floor(n) != n) throw std::runtime_error(std::string("Noninteger ") + key);
    128 return n;
    129 };
    130 s.width = static_cast<int>(integer("width", 64, 8192));
    131 s.height = static_cast<int>(integer("height", 64, 8192));
    132 if (columns.count("tile_row"))
    133 s.captureCell = std::make_pair(static_cast<int>(integer("tile_row", 0, 100000)),
    134 static_cast<int>(integer("tile_column", 0, 100000)));
    135 const double sw = integer("source_width", 64, 16384);
    136 const double sh = integer("source_height", 64, 16384);
    137 const double cw = get("crop_width"), ch = get("crop_height");
    138 const double x = get("crop_x"), y = get("crop_y");
    139 if (cw <= 0 || ch <= 0 || x < 0 || y < 0 || x + cw > sw + 1e-6 || y + ch > sh + 1e-6 ||
    140 std::abs(cw / s.width - ch / s.height) > 1e-6)
    141 throw std::runtime_error("Invalid source crop/intrinsics");
    142 s.focal = sh / (2 * std::tan(s.fov * pi / 360)) * s.width / cw;
    143 s.cx = (sw / 2 - x) * s.width / cw - 0.5;
    144 s.cy = (sh / 2 - y) * s.height / ch - 0.5;
    145 if (columns.count("lens_profile")) {
    146 const auto& profile = text("lens_profile");
    147 if (profile != "simconnect" && profile != "raspberry_pi_ai")
    148 throw std::runtime_error("Unsupported source lens profile");
    149 s.distorted = profile == "raspberry_pi_ai";
    150 }
    151 if (columns.count("start_ground_m") && !text("start_ground_m").empty())
    152 s.referencePlane = get("start_ground_m");
    153 if (columns.count("fx_px")) {
    154 const double fx = get("fx_px"), fy = get("fy_px");
    155 if (std::abs(fx - s.focal) > .01 || std::abs(fy - fx) > .01 ||
    156 std::abs(get("cx_px") - s.cx) > .01 || std::abs(get("cy_px") - s.cy) > .01)
    157 throw std::runtime_error("Source saved-image intrinsics disagree with capture framing");
    158 }
    159 if (columns.count("lens_k1")) {
    160 s.lens = {get("lens_k1"), get("lens_k2"), get("lens_blue_scale"), get("lens_red_scale")};
    161 range(s.lens.k1, -.05, 0, "source lens_k1"); range(s.lens.k2, 0, .01, "source lens_k2");
    162 range(s.lens.blueScale, .999, 1.001, "source lens_blue_scale");
    163 range(s.lens.redScale, .999, 1.001, "source lens_red_scale");
    164 if (!s.distorted && (s.lens.k1 != 0 || s.lens.k2 != 0 ||
    165 s.lens.blueScale != 1 || s.lens.redScale != 1))
    166 throw std::runtime_error("Rectilinear source has nonidentity lens effects");
    167 }
    168 const auto timestamp = [&](const char* key) {
    169 const auto n = integer(key, 1, 9e15);
    170 return static_cast<long long>(n);
    171 };
    172 s.poseTime = timestamp("pose_received_qpc_100ns"); s.frameTime = timestamp("frame_qpc_100ns");
    173 const auto requested = timestamp("request_qpc_100ns");
    174 if (s.poseTime > s.frameTime || requested > s.frameTime || s.frameTime - s.poseTime > 10000000)
    175 throw std::runtime_error("Source camera/frame timestamps are missing, stale or reversed");
    176 if (haveUtc && !text("capture_utc").empty()) {
    177 s.captureUtc = parseUtcDateTime(text("capture_utc"));
    178 s.utcAbsoluteSeconds = get("utc_absolute_seconds");
    179 s.utcRead100ns = timestamp("utc_read_qpc_100ns");
    180 if (!captureUtcSampleValid(*s.captureUtc, s.utcAbsoluteSeconds, s.utcRead100ns, s.frameTime))
    181 throw std::runtime_error("Source capture UTC readback is stale, reversed or outside the locked second");
    182 }
    183 if (haveUtc && !s.captureUtc &&
    184 (!text("utc_absolute_seconds").empty() || !text("utc_read_qpc_100ns").empty()))
    185 throw std::runtime_error("Source capture UTC evidence lacks its target date/time");
    186 if (!std::filesystem::is_regular_file(s.path)) throw std::runtime_error("Missing source JPEG: " + s.path.u8string());
    187 if (!sources.empty() && (std::abs(s.altitude - sources[0].altitude) > 0.01 ||
    188 s.width != sources[0].width || s.height != sources[0].height ||
    189 std::abs(s.focal / sources[0].focal - 1) > 1e-6 ||
    190 std::abs(s.cx - sources[0].cx) > .01 || std::abs(s.cy - sources[0].cy) > .01 ||
    191 s.distorted != sources[0].distorted ||
    192 (s.distorted && (s.lens.k1 != sources[0].lens.k1 || s.lens.k2 != sources[0].lens.k2 ||
    193 s.lens.blueScale != sources[0].lens.blueScale || s.lens.redScale != sources[0].lens.redScale))))
    194 throw std::runtime_error("Source set changed altitude, lens or viewport; use one fixed-MSL capture session");
    195 if (!sources.empty() && (s.captureUtc.has_value() != sources[0].captureUtc.has_value() ||
    196 (s.captureUtc && s.captureUtc->text != sources[0].captureUtc->text)))
    197 throw std::runtime_error("Source set changed capture UTC; use one fixed-time capture session");
    198 sources.push_back(s);
    199 }
    200 if (input.bad()) throw std::runtime_error("Cannot read source CSV");
    201 if (sources.empty()) throw std::runtime_error("No box photographs in source CSV");
    202 return sources;

    226struct Photo {
    227 OrthoSource info;
    228 int validInset = 0;
    229 double east = 0, north = 0, c = 1, s = 0, scale = 1;
    230 std::vector<unsigned char> bgra;
    231 fsPath raster;
    232 void load(int maxSide = 0, RectificationCache* rectification = nullptr) {
    233 JpegStream decoder;
    234 auto image = decoder.read(info.path, info.width, info.height, info.distorted ? 0 : maxSide);
    235 info.width = image.width; info.height = image.height;
    236 info.focal *= image.scale;
    237 info.cx = (info.cx + .5) * image.scale - .5;
    238 info.cy = (info.cy + .5) * image.scale - .5;
    239 const auto count = static_cast<std::size_t>(info.width) * info.height;
    240 bgra = std::move(image.pixels);
    241 if (info.distorted) {
    242 if (rectification && !rectification->prepare(info, count)) rectification = nullptr;
    243 cv::Mat input(info.height, info.width, CV_8UC4, bgra.data());
    244 std::vector<cv::Mat> channels;
    245 cv::split(input, channels);
    246 const double scales[] = {info.lens.blueScale, 1, info.lens.redScale, 1};
    247 const cv::Matx33d ideal(info.focal, 0, info.cx, 0, info.focal, info.cy, 0, 0, 1);
    248 for (int channel = 0; channel < 4; ++channel) {
    249 const double focal = info.focal * scales[channel];
    250 const cv::Matx33d sensor(focal, 0, info.cx, 0, focal, info.cy, 0, 0, 1);
    251 cv::Mat mx, my, corrected;
    252 int channelInset = 0;
    253 if (rectification) {
    254 if (rectification->mapX[channel].empty()) {
    255 for (int previous = 0; previous < channel; ++previous) {
    256 if (scales[previous] == scales[channel]) {
    257 rectification->mapX[channel] = rectification->mapX[previous];
    258 rectification->mapY[channel] = rectification->mapY[previous];
    259 rectification->inset[channel] = rectification->inset[previous];
    260 break;
    261 }
    262 }
    263 }
    264 mx = rectification->mapX[channel]; my = rectification->mapY[channel];
    265 channelInset = rectification->inset[channel];
    266 }
    267 if (mx.empty()) {
    268 cv::initUndistortRectifyMap(sensor, cv::Vec<double, 5>(info.lens.k1, info.lens.k2, 0, 0, 0),
    269 cv::noArray(), ideal, cv::Size(info.width, info.height), CV_32FC1, mx, my);
    270 for (int y = 0; y < info.height; ++y) for (int x = 0; x < info.width; ++x) {
    271 const double u = mx.at<float>(y, x), v = my.at<float>(y, x);
    272 if (!std::isfinite(u) || !std::isfinite(v))
    273 throw std::runtime_error("Nonfinite rectified camera ray");
    274 if (u < 0 || v < 0 || u > info.width - 1 || v > info.height - 1)
    275 channelInset = std::max(channelInset,
    276 std::min({x, y, info.width - 1 - x, info.height - 1 - y}) + 1);
    277 }
    278 if (rectification) {
    279#if defined(__x86_64__)
    280 cv::convertMaps(mx, my, rectification->mapX[channel], rectification->mapY[channel],
    281 CV_16SC2, false);
    282 mx = rectification->mapX[channel]; my = rectification->mapY[channel];
    283#else
    284 rectification->mapX[channel] = mx; rectification->mapY[channel] = my;
    285#endif
    286 rectification->inset[channel] = channelInset;
    287 }
    288 }
    289 validInset = std::max(validInset, channelInset);
    290 cv::remap(channels[channel], corrected, mx, my, cv::INTER_LINEAR, cv::BORDER_CONSTANT);
    291 channels[channel] = corrected;
    292 }
    293 cv::Mat corrected;
    294 cv::merge(channels, corrected);
    295 if (!corrected.isContinuous() || corrected.total() * corrected.elemSize() != bgra.size())
    296 throw std::runtime_error("Rectified photo storage differs from the decoded image");
    297 for (std::size_t offset = 0; offset < bgra.size(); ++offset)
    298 bgra[offset] = corrected.data[offset];
    299 if (2 * validInset > std::min(info.width, info.height) - 64)
    300 throw std::runtime_error("Lens rectification leaves too few fully supported image pixels");
    301 }
    302 }
    303 bool project(double e, double n, double z, double& x, double& y, int border = 5) const {
    304 border += validInset;
    305 const double depth = info.altitude - z;
    306 if (depth <= 5) return false;
    307 const double de = (e - east) / scale, dn = (n - north) / scale;
    308 if (info.camera) {
    309 const auto pixel = info.camera->groundToPixel(depth) * cv::Vec3d(c * de + s * dn, -s * de + c * dn, 1);
    310 if (pixel[2] <= 1e-9) return false;
    311 x = pixel[0] / pixel[2]; y = pixel[1] / pixel[2];
    312 return std::isfinite(x) && std::isfinite(y) && x >= border && y >= border &&
    313 x < info.width - border - 1 && y < info.height - border - 1;
    314 }
    315 x = info.cx + info.focal * (c * de + s * dn) / depth;
    316 y = info.cy - info.focal * (-s * de + c * dn) / depth;
    317 return x >= border && y >= border && x < info.width - border - 1 && y < info.height - border - 1;
    318 }
    319 void color(double x, double y, unsigned char* pixel) const {
    320 const int ix = static_cast<int>(x), iy = static_cast<int>(y);
    321 const double dx = x - ix, dy = y - iy;
    322 const auto i = (static_cast<std::size_t>(iy) * info.width + ix) * 4;

    326 dx * bgra[i + info.width * 4 + k + 4])), 0L, 255L));
    327 pixel[3] = 255;
    328 }

    334OrthoLayout orthoLayout(const MapConfig& config) {
    335 validateJpegSize(config.jpegWidth, config.jpegHeight);
    336 if (config.jpegWidth > 4096 || config.jpegHeight > 4096)
    337 throw std::runtime_error("Orthorectified tile dimensions must be at most 4096");
    338 range(config.mapPixelSize, .05, 2, "map_pixel_size_m");
    339 const MapProjection projection((config.start.lat + config.end.lat) / 2,
    340 (config.start.lon + config.end.lon) / 2);
    341 const auto corner = projection.forward(config.start.lat, config.start.lon);
    342 OrthoLayout result{corner.first, corner.second, corner.first, corner.second, 0, 0, projection.epsg()};
    343 for (const auto lat : {config.start.lat, config.end.lat}) for (const auto lon : {config.start.lon, config.end.lon}) {
    344 const auto p = projection.forward(lat, lon);
    345 result.west = std::min(result.west, p.first); result.east = std::max(result.east, p.first);
    346 result.north = std::max(result.north, p.second); result.south = std::min(result.south, p.second);
    347 }
    348 const double columns = std::ceil((result.east - result.west) / (config.jpegWidth * config.mapPixelSize));
    349 const double rows = std::ceil((result.north - result.south) / (config.jpegHeight * config.mapPixelSize));
    350 if (!std::isfinite(columns) || !std::isfinite(rows) || columns < 1 || rows < 1 || columns * rows > 100000)
    351 throw std::runtime_error("Invalid/excessive orthophoto tile count");
    352 result.columns = static_cast<std::size_t>(columns); result.rows = static_cast<std::size_t>(rows);
    353 result.east = result.west + columns * config.jpegWidth * config.mapPixelSize;
    354 result.south = result.north - rows * config.jpegHeight * config.mapPixelSize;
    355 return result;
    356}

    358/** @brief Fill unsupported areas using a global push-pull colour pyramid.
    359 * @param premultiplied Four-channel float overview with colour weighted by alpha.
    360 * @param workers Worker count for row processing.
    361 * @param stopped Cooperative cancellation predicate.
    362 * @return Completed four-channel float overview with alpha set to one.
    363 * @details Downsample weighted colour and support together; pull coarser
    364 * estimates into gaps without introducing tile-local fill boundaries.
    365 * @throws std::runtime_error On invalid input, absent support, or cancellation. */
    366static cv::Mat completeMosaicPyramid(const cv::Mat& premultiplied, unsigned workers,
    367 const std::function<bool()>& stopped) {
    368 if (premultiplied.empty() || premultiplied.type() != CV_32FC4)
    369 throw std::runtime_error("Mosaic completion requires premultiplied float colour/alpha");
    370 // Push/pull from one global pyramid so completion crosses source gaps, not tile boundaries.
    371 std::vector<cv::Mat> levels{premultiplied.clone()};
    372 while (levels.back().cols > 1 || levels.back().rows > 1) {
    373 if (stopped()) throw std::runtime_error("Mosaic completion canceled");
    374 cv::Mat reduced;
    375 cv::pyrDown(levels.back(), reduced);
    376 levels.push_back(std::move(reduced));
    377 }
    378 auto& root = levels.back().at<cv::Vec4f>(0, 0);
    379 if (!std::isfinite(root[3]) || root[3] <= 0)
    380 throw std::runtime_error("Cannot complete a mosaic without any source colour");
    381 for (int channel = 0; channel < 3; ++channel) root[channel] /= root[3];
    382 root[3] = 1;
    383 for (std::size_t level = levels.size() - 1; level > 0; --level) {
    384 cv::Mat expanded;
    385 cv::pyrUp(levels[level], expanded, levels[level - 1].size());
    386 auto& destination = levels[level - 1];
    387 parallelRows(destination.rows, [&](int row) {
    388 if (stopped()) throw std::runtime_error("Mosaic completion canceled");
    389 auto* values = destination.ptr<cv::Vec4f>(row);
    390 const auto* coarse = expanded.ptr<cv::Vec4f>(row);
    391 for (int column = 0; column < destination.cols; ++column) {
    392 const float alpha = std::clamp(values[column][3], 0.0f, 1.0f);
    393 for (int channel = 0; channel < 3; ++channel)
    394 values[column][channel] += (1 - alpha) * coarse[column][channel];
    395 values[column][3] = 1;
    396 }
    397 }, workers);
    398 }
    399 return levels.front();
    400}

    516 found->second.touched = ++sourcePhotoClock;

    520/** @brief Prepare disk-backed rasters for low-memory rendering.
    521 * @param stopped Cooperative cancellation predicate.
    522 * @details Identify each raster with its input digest so reusable prepared
    523 * data cannot quietly stand in for changed source imagery.
    524 * @throws std::runtime_error On cancellation or invalid source data or storage. */
    525void Orthorectifier::Impl::prepareDiskSources(const std::function<bool()>& stopped) {
    526 if (diskSourcesPrepared) return;
    527 if (config.workingDirectory.empty()) throw std::runtime_error("Low-memory processing requires a working directory");
    528 const auto directory = config.preparedSourceDirectory.empty() ? config.workingDirectory / "sources" :
    529 config.preparedSourceDirectory;
    530 std::filesystem::create_directories(directory);
    531 if (sources.size() > 4096) throw std::runtime_error("Low-memory source count exceeds 4096");
    532 sourceRasters.resize(sources.size());
    533 for (std::size_t index = 0; index < sources.size(); ++index) {
    534 if (stopped()) throw std::runtime_error("Source preparation canceled");
    535 auto& info = sources[index];
    536 const auto sourceSize = std::filesystem::file_size(info.path);
    537 if (!sourceSize || sourceSize > 128ull * 1024 * 1024)
    538 throw std::runtime_error("Invalid streamed JPEG file size");
    539 const auto sourceHash = fileDigest(info.path, stopped);
    540 std::ostringstream identity;
    541 identity.imbue(std::locale::classic());
    542 identity << std::setprecision(17) << sourceHash << ' ' << info.width << ' ' << info.height << ' '
    543 << info.focal << ' ' << info.cx << ' ' << info.cy << ' ' << info.distorted << ' '
    544 << info.lens.k1 << ' ' << info.lens.k2 << ' ' << info.lens.blueScale << ' ' << info.lens.redScale;
    545 if (info.camera) identity << " fy=" << info.focalY;
    546 const auto cacheIdentity = "PANGEA_PREPARED_SOURCE_V1\n" + identity.str();
    547 const auto key = config.preparedSourceDirectory.empty() ? std::to_string(index) :
    548 dataDigest(cacheIdentity.data(), cacheIdentity.size());
    549 const auto path = directory / (key + ".bgra");
    550 const auto metadata = path.string() + ".verified";
    551 int inset = 0;
    552 if (std::filesystem::exists(metadata)) {
    553 std::ifstream record(metadata);
    554 std::string saved, digest, trailing;
    555 std::getline(record, saved);
    556 if (saved != identity.str() || !(record >> inset >> digest) || (record >> trailing) ||
    557 inset < 0 || 2 * inset > std::min(info.width, info.height) - 64 || fileDigest(path, stopped) != digest)
    558 throw std::runtime_error("Source checkpoint identity or checksum mismatch; originals retained");
    559 DiskRaster checked(path, info.width - 2 * inset, info.height - 2 * inset);
    560 } else {
    561 storageHeadroom(directory, static_cast<std::uint64_t>(info.width) * info.height * (info.distorted ? 12 : 4));
    562 const auto decoded = info.distorted ? directory / (key + ".decoded") : path;
    563 { JpegStream decoder; decoder.decode(info.path, decoded, info.width, info.height, stopped); }
    564 if (info.distorted) {
    565 const DiskRaster raster(decoded, info.width, info.height);
    566 const auto correctedPath = directory / (key + ".corrected");
    567 std::ofstream output(correctedPath, std::ios::binary | std::ios::trunc);
    568 const double scales[] = {info.lens.blueScale, 1, info.lens.redScale, 1};
    569 for (int firstRow = 0; firstRow < info.height; firstRow += 32) {
    570 if (stopped()) throw std::runtime_error("Strip rectification canceled");
    571 const int rows = std::min(32, info.height - firstRow);
    572 cv::Mat stripe(rows, info.width, CV_8UC4);
    573 for (int firstColumn = 0; firstColumn < info.width; firstColumn += 256) {
    574 const int columns = std::min(256, info.width - firstColumn);
    575 for (int channel = 0; channel < 4; ++channel) {
    576 const double focalY = info.focalY > 0 ? info.focalY : info.focal;
    577 const cv::Matx33d sensor(info.focal * scales[channel], 0, info.cx,
    578 0, focalY * scales[channel], info.cy, 0, 0, 1);
    579 const cv::Matx33d ideal(info.focal, 0, info.cx - firstColumn,
    580 0, focalY, info.cy - firstRow, 0, 0, 1);
    581 cv::Mat mx, my;
    582 cv::initUndistortRectifyMap(sensor, cv::Vec<double, 5>(info.lens.k1, info.lens.k2, 0, 0, 0),
    583 cv::noArray(), ideal, {columns, rows}, CV_32FC1, mx, my);
    584 double minX = info.width - 1, maxX = 0, minY = info.height - 1, maxY = 0;
    585 for (int row = 0; row < rows; ++row) for (int column = 0; column < columns; ++column) {
    586 const double sx = mx.at<float>(row, column), sy = my.at<float>(row, column);
    587 if (!std::isfinite(sx) || !std::isfinite(sy))
    588 throw std::runtime_error("Nonfinite rectified camera ray");
    589 if (sx < 0 || sy < 0 || sx > info.width - 1 || sy > info.height - 1)
    590 inset = std::max(inset, std::min({firstColumn + column, firstRow + row,
    591 info.width - 1 - firstColumn - column, info.height - 1 - firstRow - row}) + 1);
    592 minX = std::min(minX, std::clamp(sx, 0.0, double(info.width - 1)));
    593 maxX = std::max(maxX, std::clamp(sx, 0.0, double(info.width - 1)));
    594 minY = std::min(minY, std::clamp(sy, 0.0, double(info.height - 1)));
    595 maxY = std::max(maxY, std::clamp(sy, 0.0, double(info.height - 1)));
    596 }
    597 const int left = std::max(0, static_cast<int>(std::floor(minX)) - 1);
    598 const int top = std::max(0, static_cast<int>(std::floor(minY)) - 1);
    599 const int right = std::min(info.width, static_cast<int>(std::ceil(maxX)) + 2);
    600 const int bottom = std::min(info.height, static_cast<int>(std::ceil(maxY)) + 2);
    601 const auto input = raster.read({left, top, right - left, bottom - top});
    602 cv::Mat plane, corrected;
    603 cv::extractChannel(input, plane, channel);
    604 mx -= left; my -= top;
    605 cv::remap(plane, corrected, mx, my, cv::INTER_LINEAR, cv::BORDER_CONSTANT);
    606 for (int row = 0; row < rows; ++row) for (int column = 0; column < columns; ++column)
    607 stripe.at<cv::Vec4b>(row, firstColumn + column)[channel] = corrected.at<unsigned char>(row, column);
    608 }
    609 }
    610 output.write(reinterpret_cast<const char*>(stripe.data), static_cast<std::streamsize>(stripe.total() * 4));
    611 }
    612 output.close();
    613 if (!output || 2 * inset > std::min(info.width, info.height) - 64)
    614 throw std::runtime_error("Rectification write failed or unsupported camera crop");
    615 const DiskRaster corrected(correctedPath, info.width, info.height);
    616 const auto temporary = path.string() + ".partial";
    617 std::ofstream cropped(temporary, std::ios::binary | std::ios::trunc);
    618 for (int row = inset; row < info.height - inset; ++row) {
    619 if (stopped()) throw std::runtime_error("Source crop canceled");
    620 const auto pixels = corrected.read({inset, row, info.width - 2 * inset, 1});
    621 cropped.write(reinterpret_cast<const char*>(pixels.data), pixels.cols * 4);
    622 }
    623 cropped.close();
    624 if (!cropped) throw std::runtime_error("Cannot write rectified raster");
    625 publishWorkingFile(temporary, path);
    626 std::filesystem::remove(decoded); std::filesystem::remove(correctedPath);
    627 }
    628 if (fileDigest(info.path, stopped) != sourceHash)
    629 throw std::runtime_error("Photo changed during preparation; no source checkpoint committed");
    630 saveWorkingText(metadata, identity.str() + "\n" + std::to_string(inset) + " " + fileDigest(path, stopped) + "\n");
    631 }
    632 sourceRasters[index] = path;
    633 info.width -= 2 * inset; info.height -= 2 * inset; info.cx -= inset; info.cy -= inset;
    634 if (info.camera) info.camera->crop(inset);
    635 info.distorted = false;
    636 }
    637 diskSourcesPrepared = true;
    638}

    676/** @brief Bound the projected source footprint on a mosaic grid.
    677 * @param source Source pose and camera geometry.
    678 * @param scale Number of base pixels per grid pixel.
    679 * @param seamPlan True when computing the seam-planning footprint.
    680 * @return Pixel rectangle in the selected mosaic grid.
    681 * @details Project image corners to the common reference plane so the
    682 * renderer can avoid processing unrelated regions. */
    683cv::Rect Orthorectifier::Impl::mosaicRectangle(const OrthoSource& source, int scale, bool seamPlan) const {
    684 const auto position = projection.forward(source.latitude, source.longitude);
    685 const double heading = projection.convergence(source.latitude, source.longitude) * pi / 180;
    686 const double c = std::cos(heading), s = std::sin(heading);
    687 const double ray = (source.altitude - config.referencePlane) / source.focal *
    688 projection.scale(source.latitude, source.longitude);
    689 const double pixel = config.mapPixelSize * scale;
    690 double minX = std::numeric_limits<double>::infinity(), minY = minX;
    691 double maxX = -minX, maxY = -minX;
    692 for (const double u : {1.0, double(source.width - 2)}) for (const double v : {1.0, double(source.height - 2)}) {
    693 const auto local = source.camera ? source.camera->ground(u, v, source.altitude - config.referencePlane) :
    694 cv::Point2d((u - source.cx) * ray, -(v - source.cy) * ray);
    695 const double unit = source.camera ? projection.scale(source.latitude, source.longitude) : 1;
    696 const double x = (position.first + unit * (c * local.x - s * local.y) - west) / pixel;
    697 const double y = (north - position.second - unit * (s * local.x + c * local.y)) / pixel;
    698 minX = std::min(minX, x); maxX = std::max(maxX, x);
    699 minY = std::min(minY, y); maxY = std::max(maxY, y);
    700 }
    701 const int pad = (64 + scale - 1) / scale + 2;
    702 double endX = std::ceil(double(cols) * config.jpegWidth / scale) + pad;
    703 double endY = std::ceil(double(rows) * config.jpegHeight / scale) + pad;
    704 if (seamPlan) {
    705 double actualEast = west, actualSouth = north;
    706 for (const auto latitude : {config.start.lat, config.end.lat})
    707 for (const auto longitude : {config.start.lon, config.end.lon}) {
    708 const auto corner = projection.forward(latitude, longitude);
    709 actualEast = std::max(actualEast, corner.first);
    710 actualSouth = std::min(actualSouth, corner.second);
    711 }
    712 endX = std::ceil((actualEast - west) / pixel) + pad;
    713 endY = std::ceil((north - actualSouth) / pixel) + pad;
    714 }
    715 const double x0 = std::max(std::floor(minX), -double(pad)), y0 = std::max(std::floor(minY), -double(pad));
    716 const double x1 = std::min(std::ceil(maxX), endX), y1 = std::min(std::ceil(maxY), endY);
    717 if (x1 <= x0 || y1 <= y0) return {};
    718 if (std::max({std::abs(x0), std::abs(y0), std::abs(x1), std::abs(y1)}) > 1e8)
    719 throw std::runtime_error("Great mosaic extent exceeds supported grid coordinates");
    720 return {static_cast<int>(x0), static_cast<int>(y0), static_cast<int>(x1 - x0), static_cast<int>(y1 - y0)};
    721}

    723/** @brief Resample one source over a mosaic rectangle.
    724 * @param photo Projection-ready source with decoded or disk-backed pixels.
    725 * @param rectangle Output window in mosaic-grid pixels.
    726 * @param scale Number of base pixels per grid pixel.
    727 * @param[out] image Warped source colour.
    728 * @param[out] mask Support for the warped colour.
    729 * @return Transform mapping local destination coordinates into the source.
    730 * @details Evaluate the mapping at pixel centres; use the calibrated camera
    731 * when present and process in strips to bound source access. */
    732cv::Matx33d Orthorectifier::Impl::mosaicWarp(const Photo& photo, const cv::Rect& rectangle, int scale,
    733 cv::Mat& image, cv::Mat& mask) const {
    734 const double pixel = config.mapPixelSize * scale;
    735 const double e = west + (rectangle.x + .5) * pixel, n = north - (rectangle.y + .5) * pixel;
    736 double u = 0, v = 0;
    737 if (!photo.info.camera) photo.project(e, n, config.referencePlane, u, v, 1);
    738 const double factor = pixel * photo.info.focal / ((photo.info.altitude - config.referencePlane) * photo.scale);
    739 cv::Matx33d transform(factor * photo.c, -factor * photo.s, u, factor * photo.s, factor * photo.c, v, 0, 0, 1);
    740 if (photo.info.camera) {
    741 const double de = (e - photo.east) / photo.scale, dn = (n - photo.north) / photo.scale;
    742 const double step = pixel / photo.scale;
    743 transform = photo.info.camera->groundToPixel(photo.info.altitude - config.referencePlane) *
    744 cv::Matx33d(step * photo.c, -step * photo.s, photo.c * de + photo.s * dn,
    745 -step * photo.s, -step * photo.c, -photo.s * de + photo.c * dn, 0, 0, 1);
    746 }
    747 const cv::Mat source = photo.raster.empty() ? cv::Mat(photo.info.height, photo.info.width, CV_8UC4,
    748 const_cast<unsigned char*>(photo.bgra.data())) : cv::Mat();
    749 cv::Mat warped;
    750 warped.create(rectangle.size(), CV_8UC4);
    751 constexpr int stripRows = 32;
    752 parallelRows((rectangle.height + stripRows - 1) / stripRows, [&](int strip) {
    753 const int first = strip * stripRows, last = std::min(rectangle.height, first + stripRows);
    754 auto local = transform;
    755 for (int axis = 0; axis < 3; ++axis) local(axis, 2) += local(axis, 1) * first;
    756 cv::Mat destination = warped.rowRange(first, last);
    757 const auto warp = [&](auto&& self, cv::Mat output, cv::Matx33d mapping) -> void {
    758 cv::Mat input = source;
    759 if (!photo.raster.empty()) {
    760 double minX = photo.info.width - 1, maxX = 0, minY = photo.info.height - 1, maxY = 0;
    761 double minDepth = std::numeric_limits<double>::infinity(), maxDepth = -minDepth;
    762 for (int row : {0, output.rows - 1}) for (int column : {0, output.cols - 1}) {
    763 const double depth = mapping(2, 0) * column + mapping(2, 1) * row + mapping(2, 2);
    764 minDepth = std::min(minDepth, depth); maxDepth = std::max(maxDepth, depth);
    765 const double denominator = std::abs(depth) > 1e-12 ? depth : 1e-12;
    766 const double sx = std::clamp((mapping(0, 0) * column + mapping(0, 1) * row + mapping(0, 2)) / denominator, 0.0, double(photo.info.width - 1));
    767 const double sy = std::clamp((mapping(1, 0) * column + mapping(1, 1) * row + mapping(1, 2)) / denominator, 0.0, double(photo.info.height - 1));
    768 minX = std::min(minX, sx); maxX = std::max(maxX, sx); minY = std::min(minY, sy); maxY = std::max(maxY, sy);
    769 }
    770 if (maxDepth <= 1e-9) { output.setTo(cv::Scalar(0)); return; }
    771 // Across the projective horizon, corner extrema cannot safely bound the source read.
    772 if (minDepth <= 1e-9) { minX = minY = 0; maxX = photo.info.width - 1; maxY = photo.info.height - 1; }
    773 const int left = std::max(0, static_cast<int>(std::floor(minX)) - 2);
    774 const int top = std::max(0, static_cast<int>(std::floor(minY)) - 2);
    775 const int right = std::min(photo.info.width, static_cast<int>(std::ceil(maxX)) + 3);
    776 const int bottom = std::min(photo.info.height, static_cast<int>(std::ceil(maxY)) + 3);
    777 if (static_cast<std::uint64_t>(right - left) * (bottom - top) * 4 >
    778 static_cast<std::uint64_t>(config.mosaicPhotoCacheMiB) * 1024 * 1024) {
    779 // Split the destination before loading either source window; keep the warp origin aligned.
    780 const bool splitRows = output.rows >= output.cols && output.rows > 1;
    781 const int split = (splitRows ? output.rows : output.cols) / 2;
    782 if (!split) throw std::runtime_error("Single warp sample exceeds raster budget");
    783 const cv::Rect first(0, 0, splitRows ? output.cols : split, splitRows ? split : output.rows);
    784 const cv::Rect second(splitRows ? 0 : split, splitRows ? split : 0,
    785 splitRows ? output.cols : output.cols - split, splitRows ? output.rows - split : output.rows);
    786 self(self, output(first), mapping);
    787 for (int axis = 0; axis < 3; ++axis)
    788 mapping(axis, 2) += mapping(axis, 0) * second.x + mapping(axis, 1) * second.y;
    789 self(self, output(second), mapping);
    790 return;
    791 }
    792 input = DiskRaster(photo.raster, photo.info.width, photo.info.height).read({left, top, right - left, bottom - top});
    793 for (int axis = 0; axis < 3; ++axis) {
    794 mapping(0, axis) -= left * mapping(2, axis);
    795 mapping(1, axis) -= top * mapping(2, axis);
    796 }
    797 }
    798 if (photo.info.camera)
    799 cv::warpPerspective(input, output, mapping, output.size(), cv::INTER_LINEAR | cv::WARP_INVERSE_MAP,
    800 cv::BORDER_REPLICATE);
    801 else
    802 cv::warpAffine(input, output, cv::Matx23d(mapping(0, 0), mapping(0, 1), mapping(0, 2),
    803 mapping(1, 0), mapping(1, 1), mapping(1, 2)), output.size(),
    804 cv::INTER_LINEAR | cv::WARP_INVERSE_MAP, cv::BORDER_REPLICATE);
    805 };
    806 warp(warp, destination, local);
    807 }, config.mosaicThreads);
    808 cv::cvtColor(warped, image, cv::COLOR_BGRA2BGR);
    809 mask.create(rectangle.size(), CV_8U);
    810 const int border = 1 + photo.validInset;
    811 parallelRows(rectangle.height, [&](int y) {
    812 auto* row = mask.ptr<unsigned char>(y);
    813 for (int x = 0; x < rectangle.width; ++x) {
    814 const double depth = transform(2, 0) * x + transform(2, 1) * y + transform(2, 2);
    815 const double sx = (transform(0, 0) * x + transform(0, 1) * y + transform(0, 2)) / depth;
    816 const double sy = (transform(1, 0) * x + transform(1, 1) * y + transform(1, 2)) / depth;
    817 row[x] = depth > 1e-9 && sx >= border && sy >= border && sx < photo.info.width - border - 1 &&
    818 sy < photo.info.height - border - 1 ? 255 : 0;
    819 }
    820 }, config.mosaicThreads);
    821 return transform;
    822}

    824/** @brief Build the shared seam and exposure plan on first use.
    825 * @param stopped Cooperative cancellation predicate.
    826 * @details Coarsen the overview until the global planning workspace fits its
    827 * limits. Reuse that plan across windows so adjacent renders agree. */
    828void Orthorectifier::Impl::prepareMosaic(const std::function<bool()>& stopped) {
    829 if (mosaicPrepared) return;
    830 if (config.lowMemory) prepareDiskSources(stopped);
    831 const auto prepareStarted = std::chrono::steady_clock::now();
    832 seamScale = config.mosaicSeamScale;
    833 // Increase overview scale until source work and the global grid fit their memory budgets.
    834 for (;;) {
    835 std::size_t pixels = 0;
    836 for (const auto& source : sources) {
    837 const auto rectangle = mosaicRectangle(source, seamScale, true);
    838 pixels += static_cast<std::size_t>(rectangle.width) * rectangle.height;
    839 }
    840 const auto globalPixels = std::ceil(double(cols) * config.jpegWidth / seamScale) *
    841 std::ceil(double(rows) * config.jpegHeight / seamScale);
    842 if (pixels <= static_cast<std::size_t>(config.mosaicOverviewPixels) &&
    843 (!config.lowMemory || globalPixels <= config.mosaicOverviewPixels)) break;
    844 seamScale *= 2;
    845 if (seamScale > (config.lowMemory ? 8192 : 256))
    846 throw std::runtime_error("Mosaic overview exceeds its memory budget; split the map area");
    847 }
    848 std::cout << config.mode << ": camera-calibrated mosaic, " << config.mosaicSeamMethod << " seams at 1/" << seamScale
    849 << ", " << config.mosaicBlendBands << " blending bands; no stereo cost volume.\n";
    850 std::vector<cv::UMat> images, masks, originalMasks;
    851 std::vector<cv::Point> corners;
    852 std::vector<std::size_t> seamViews;
    853 std::vector<cv::Mat> overviewImages(sources.size());
    854 std::vector<cv::Mat> overviewMasks(sources.size());
    855 parallelRows(static_cast<int>(sources.size()), [&](int index) {
    856 if (stopped()) throw std::runtime_error("Mosaic overview canceled");
    857 const auto rectangle = mosaicRectangle(sources[index], seamScale, true);
    858 if (rectangle.empty()) return;
    859 Photo photo;
    860 photo.info = sources[index];
    861 const auto coordinate = projection.forward(photo.info.latitude, photo.info.longitude);
    862 photo.east = coordinate.first; photo.north = coordinate.second;
    863 const double heading = projection.convergence(photo.info.latitude, photo.info.longitude) * pi / 180;
    864 photo.c = std::cos(heading); photo.s = std::sin(heading);
    865 photo.scale = projection.scale(photo.info.latitude, photo.info.longitude);
    866 if (config.lowMemory) photo.raster = sourceRasters.at(index);
    867 else photo.load(config.mosaicOverviewSourceMaxSide);
    868 mosaicWarp(photo, rectangle, seamScale, overviewImages[index], overviewMasks[index]);
    869 }, config.mosaicThreads);
    870 // Seam ownership is global; per-output-tile seam finding would make adjacent tiles disagree.
    871 for (std::size_t i = 0; i < sources.size(); ++i) {
    872 if (stopped()) throw std::runtime_error("Great overview canceled");
    873 const auto rectangle = mosaicRectangle(sources[i], seamScale, true);
    874 if (mosaicRectangle(sources[i], seamScale).empty()) continue;
    875 if (mosaicViews.size() == 4096)
    876 throw std::runtime_error("Great seam plan exceeds 4096 overlapping sources; split the map area");
    877 // A coarse overview can miss a narrow valid footprint; retain its full-resolution colour fallback.
    878 mosaicViews.push_back({i, rectangle, {}, 1, mosaicRectangle(sources[i], 1)});
    879 if (rectangle.empty()) continue;
    880 cv::Mat image = overviewImages[i], mask = overviewMasks[i];
    881 if (cv::countNonZero(mask) == 0) continue;
    882 images.emplace_back(); masks.emplace_back();
    883 image.copyTo(images.back()); mask.copyTo(masks.back());
    884 if (config.mosaicFillGaps) { originalMasks.emplace_back(); mask.copyTo(originalMasks.back()); }
    885 corners.push_back(rectangle.tl());
    886 seamViews.push_back(mosaicViews.size() - 1);
    887 }
    888 if (mosaicViews.empty())
    889 throw std::runtime_error("Great has no source-photo coverage in the requested area");
    890 const auto exposureStarted = std::chrono::steady_clock::now();
    891 if (config.mosaicExposure && images.size() > 1) {
    892 cv::detail::GainCompensator exposure;
    893 std::vector<std::pair<cv::UMat, unsigned char>> tagged;
    894 for (const auto& mask : masks) tagged.emplace_back(mask, static_cast<unsigned char>(255));
    895 exposure.feed(corners, images, tagged);
    896 const auto gains = exposure.gains();
    897 if (gains.size() != images.size()) throw std::runtime_error("Great exposure solver returned wrong gain count");
    898 for (std::size_t i = 0; i < gains.size(); ++i) {
    899 if (!std::isfinite(gains[i]) || gains[i] <= 0)
    900 throw std::runtime_error("Great exposure solver returned invalid gain");
    901 mosaicViews[seamViews[i]].gain = std::clamp(gains[i], .8, 1.25);
    902 }
    903 }
    904 const auto seamsStarted = std::chrono::steady_clock::now();
    905 if (config.mosaicSeamMethod != "voronoi")
    906 for (std::size_t i = 0; i < images.size(); ++i)
    907 images[i].convertTo(images[i], CV_32F, mosaicViews[seamViews[i]].gain);
    908 if (stopped()) throw std::runtime_error("Great seam optimization canceled");
    909 if (!images.empty()) {
    910 if (config.mosaicSeamMethod == "dp") {
    911 cv::detail::DpSeamFinder finder(cv::detail::DpSeamFinder::COLOR_GRAD);
    912 finder.find(images, corners, masks);
    913 } else if (config.mosaicSeamMethod == "voronoi") {
    914 cv::detail::VoronoiSeamFinder finder;
    915 finder.find(images, corners, masks);
    916 } else {
    917 cv::detail::GraphCutSeamFinder finder(cv::detail::GraphCutSeamFinderBase::COST_COLOR_GRAD);
    918 finder.find(images, corners, masks);
    919 }
    920 }
    921 for (std::size_t i = 0; i < masks.size(); ++i) {
    922 cv::dilate(masks[i], mosaicViews[seamViews[i]].seam, cv::Mat(), cv::Point(-1, -1), 1);
    923 }
    924 if (config.mosaicFillGaps) prepareMosaicCompletion(images, originalMasks, seamViews, stopped);
    925 if (stopped()) throw std::runtime_error("Great seam optimization canceled");
    926 mosaicPrepared = true;
    927 const auto finished = std::chrono::steady_clock::now();
    928 const auto elapsed = [](auto first, auto last) { return std::chrono::duration<double>(last - first).count(); };
    929 std::cout << "TIMING mosaic_prepare photos_seconds=" << elapsed(prepareStarted, exposureStarted)
    930 << " exposure_seconds=" << elapsed(exposureStarted, seamsStarted)
    931 << " seams_seconds=" << elapsed(seamsStarted, finished)
    932 << " total_seconds=" << elapsed(prepareStarted, finished) << '\n';
    933 std::cout << config.mode << " overview ready: " << mosaicViews.size()
    934 << " source footprints; global seams reused for every output tile.\n";
    935}

    937/** @brief Build a shared fill overview while retaining observed support.
    938 * @param images Warped colour overviews.
    939 * @param originalMasks Matching support masks before seam selection.
    940 * @param seamViews Source indices corresponding to the overview entries.
    941 * @param stopped Cooperative cancellation predicate.
    942 * @details Keep the observed mask separate from inferred fill so later
    943 * renders need not report extrapolated colours as measured coverage. */
    944void Orthorectifier::Impl::prepareMosaicCompletion(const std::vector<cv::UMat>& images,
    945 const std::vector<cv::UMat>& originalMasks, const std::vector<std::size_t>& seamViews,
    946 const std::function<bool()>& stopped) {
    947 double actualEast = west, actualSouth = north;
    948 for (const auto latitude : {config.start.lat, config.end.lat})
    949 for (const auto longitude : {config.start.lon, config.end.lon}) {
    950 const auto corner = projection.forward(latitude, longitude);
    951 actualEast = std::max(actualEast, corner.first); actualSouth = std::min(actualSouth, corner.second);
    952 }
    953 const int width = std::max(1, static_cast<int>(std::ceil((actualEast - west) / (config.mapPixelSize * seamScale))));
    954 const int height = std::max(1, static_cast<int>(std::ceil((north - actualSouth) / (config.mapPixelSize * seamScale))));
    955 if (static_cast<std::size_t>(width) * height > static_cast<std::size_t>(config.lowMemory ? config.mosaicOverviewPixels : 8000000))
    956 throw std::runtime_error("Global completion overview exceeds 8 million pixels; increase mosaic.seam_scale");
    957 cv::Mat premultiplied(height, width, CV_32FC4, cv::Scalar(0));
    958 cv::Mat scores(height, width, CV_32F, cv::Scalar(std::numeric_limits<float>::infinity()));
    959 cv::Mat observed(height, width, CV_8U, cv::Scalar(0));
    960 for (std::size_t index = 0; index < images.size(); ++index) {
    961 if (stopped()) throw std::runtime_error("Global mosaic source fusion canceled");
    962 const auto& view = mosaicViews[seamViews[index]];
    963 const auto rectangle = view.rectangle & cv::Rect(0, 0, width, height);
    964 if (rectangle.empty()) continue;
    965 cv::Mat colour;
    966 images[index].convertTo(colour, CV_32F, config.mosaicSeamMethod == "voronoi" ? view.gain : 1);
    967 const auto original = originalMasks[index].getMat(cv::ACCESS_READ);
    968 for (int row = rectangle.y; row < rectangle.y + rectangle.height; ++row) {
    969 auto* destination = premultiplied.ptr<cv::Vec4f>(row);
    970 auto* best = scores.ptr<float>(row);
    971 auto* coverage = observed.ptr<unsigned char>(row);
    972 const int sourceRow = row - view.rectangle.y;
    973 const auto* pixels = colour.ptr<cv::Vec3f>(sourceRow);
    974 const auto* valid = original.ptr<unsigned char>(sourceRow);
    975 const auto* seam = view.seam.ptr<unsigned char>(sourceRow);
    976 for (int column = rectangle.x; column < rectangle.x + rectangle.width; ++column) {
    977 const int sourceColumn = column - view.rectangle.x;
    978 if (!valid[sourceColumn]) continue;
    979 const double dx = (sourceColumn + .5) / view.rectangle.width - .5;
    980 const double dy = (sourceRow + .5) / view.rectangle.height - .5;
    981 const float score = static_cast<float>(dx * dx + dy * dy + (seam[sourceColumn] ? 0 : 10));
    982 if (score >= best[column]) continue;
    983 best[column] = score; coverage[column] = 255;
    984 const auto& pixel = pixels[sourceColumn];
    985 destination[column] = cv::Vec4f(pixel[0], pixel[1], pixel[2], 1);
    986 }
    987 }
    988 }
    989 cv::distanceTransform(observed, mosaicSourceDistance, cv::DIST_L2, 3);
    990 completedMosaic = completeMosaicPyramid(premultiplied, config.mosaicThreads, stopped);
    991 std::cout << "Global source-derived completion ready: " << width << " x " << height
    992 << "; completed pixels are tracked separately from observed photo colour.\n";
    993}

    995/** @brief Render a mosaic window using the session-wide seam plan.
    996 * @param map Projected centre and georeference of the window.
    997 * @param width Output columns.
    998 * @param height Output rows.
    999 * @param stopped Cooperative cancellation predicate.
    1000 * @return Owned pixels, confidence, and rendering provenance.
    1001 * @details Combine planned contributions in the local window while tracking
    1002 * fallback colour and coverage independently of multiband blending. */
    1003OrthoImage Orthorectifier::Impl::renderMosaicRegion(const MapTile& map, int width, int height,
    1004 const std::function<bool()>& stopped) {
    1005 prepareMosaic(stopped);
    1006 OrthoImage result;
    1007 result.map = map; result.width = width; result.height = height; result.mode = config.mode;
    1008 result.referencePlane = config.referencePlane;
    1009 result.seamScale = seamScale; result.blendBands = config.mosaicBlendBands;
    1010 result.seamMethod = config.mosaicSeamMethod;
    1011 result.exposureCompensated = config.mosaicExposure;
    1012 const auto count = static_cast<std::size_t>(width) * height;
    1013 result.bgra.assign(count * 4, 0); result.confidence.assign(count, 0);
    1014 const cv::Rect region(static_cast<int>(std::llround((map.east - west) / config.mapPixelSize - width / 2.0)),
    1015 static_cast<int>(std::llround((north - map.north) / config.mapPixelSize - height / 2.0)),
    1016 width, height);
    1017 cv::Mat best(height, width, CV_32F, cv::Scalar(std::numeric_limits<float>::infinity()));
    1018 cv::Mat fallback(height, width, CV_8UC3, cv::Scalar(0));
    1019 cv::Mat coverage(height, width, CV_8U, cv::Scalar(0));
    1020 cv::detail::MultiBandBlender blender(false, config.mosaicBlendBands);
    1021 blender.prepare(cv::Rect(0, 0, width, height));
    1022 bool fed = false;
    1023 for (const auto& view : mosaicViews) {
    1024 if (stopped()) throw std::runtime_error("Great blending canceled");
    1025 const cv::Rect footprint = view.footprint;
    1026 if ((footprint & region).empty()) continue;
    1027 auto& photo = mosaicPhoto(view.source);
    1028 cv::Mat image, mask;
    1029 const auto transform = mosaicWarp(photo, region, 1, image, mask);
    1030 if (cv::countNonZero(mask) == 0) continue;
    1031 image.convertTo(image, CV_8U, view.gain);
    1032 result.sources.push_back(photo.info);
    1033 cv::Mat selected(height, width, CV_8U, cv::Scalar(0));
    1034 parallelRows(height, [&](int y) {
    1035 const auto* valid = mask.ptr<unsigned char>(y);
    1036 const auto* colours = image.ptr<cv::Vec3b>(y);
    1037 auto* scores = best.ptr<float>(y);
    1038 auto* backup = fallback.ptr<cv::Vec3b>(y);
    1039 auto* covered = coverage.ptr<unsigned char>(y);
    1040 auto* owned = selected.ptr<unsigned char>(y);
    1041 const double gy = (region.y + y + .5) / seamScale - .5 - view.rectangle.y;
    1042 const int sy = view.seam.empty() ? 0 :
    1043 std::clamp(static_cast<int>(std::lround(gy)), 0, view.seam.rows - 1);
    1044 for (int x = 0; x < width; ++x) {
    1045 if (!valid[x]) continue;
    1046 covered[x] = 255;
    1047 const double depth = transform(2, 0) * x + transform(2, 1) * y + transform(2, 2);
    1048 const double u = (transform(0, 0) * x + transform(0, 1) * y + transform(0, 2)) / depth;
    1049 const double v = (transform(1, 0) * x + transform(1, 1) * y + transform(1, 2)) / depth;
    1050 const double score = std::pow((u - photo.info.cx) / photo.info.width, 2) +
    1051 std::pow((v - photo.info.cy) / photo.info.height, 2);
    1052 if (score < scores[x]) { scores[x] = static_cast<float>(score); backup[x] = colours[x]; }
    1053 if (view.seam.empty()) continue;
    1054 const double gx = (region.x + x + .5) / seamScale - .5 - view.rectangle.x;
    1055 const int sx = std::clamp(static_cast<int>(std::lround(gx)), 0, view.seam.cols - 1);
    1056 if (view.seam.at<unsigned char>(sy, sx)) owned[x] = 255;
    1057 }
    1058 }, config.mosaicThreads);
    1059 if (cv::countNonZero(selected)) {
    1060 cv::Mat signedImage; image.convertTo(signedImage, CV_16S);
    1061 blender.feed(signedImage, selected, cv::Point(0, 0));
    1062 fed = true;
    1063 }
    1064 }
    1065 cv::Mat blended, blendedMask;
    1066 if (fed) blender.blend(blended, blendedMask);
    1067 std::atomic<std::size_t> reconstructed{0}, reconstructionBlend{0};
    1068 parallelRows(height, [&](int y) {
    1069 if (stopped()) throw std::runtime_error("Great mosaic finalization canceled");
    1070 std::size_t rowReconstructed = 0, rowBlended = 0;
    1071 for (int x = 0; x < width; ++x) {
    1072 const bool observed = coverage.at<unsigned char>(y, x) != 0;
    1073 if (!observed && completedMosaic.empty()) continue;
    1074 const auto index = static_cast<std::size_t>(y) * width + x;
    1075 auto* pixel = result.bgra.data() + index * 4;
    1076 if (observed && fed && blendedMask.at<unsigned char>(y, x)) {
    1077 const auto value = blended.at<cv::Vec3s>(y, x);
    1078 for (int channel = 0; channel < 3; ++channel) pixel[channel] = cv::saturate_cast<unsigned char>(value[channel]);
    1079 } else if (observed) {
    1080 const auto value = fallback.at<cv::Vec3b>(y, x);
    1081 for (int channel = 0; channel < 3; ++channel) pixel[channel] = value[channel];
    1082 }
    1083 pixel[3] = 255;
    1084 result.confidence[index] = observed ? 64 : 16;
    1085 if (!completedMosaic.empty()) {
    1086 const double sourceX = std::clamp((region.x + x + .5) / seamScale - .5,
    1087 0.0, double(completedMosaic.cols - 1));
    1088 const double sourceY = std::clamp((region.y + y + .5) / seamScale - .5,
    1089 0.0, double(completedMosaic.rows - 1));
    1090 const int left = static_cast<int>(sourceX), top = static_cast<int>(sourceY);
    1091 const int right = std::min(left + 1, completedMosaic.cols - 1);
    1092 const int bottom = std::min(top + 1, completedMosaic.rows - 1);
    1093 const double fractionX = sourceX - left, fractionY = sourceY - top;
    1094 const auto sample = [&](const cv::Mat& image, int channel) {
    1095 const int channels = image.channels();
    1096 const auto* first = image.ptr<float>(top); const auto* last = image.ptr<float>(bottom);
    1097 return (1 - fractionY) * ((1 - fractionX) * first[left * channels + channel] +
    1098 fractionX * first[right * channels + channel]) + fractionY *
    1099 ((1 - fractionX) * last[left * channels + channel] + fractionX * last[right * channels + channel]);
    1100 };
    1101 double weight = observed ? std::clamp(sample(mosaicSourceDistance, 0) / 2, 0.0, 1.0) : 0;
    1102 weight = weight * weight * (3 - 2 * weight);
    1103 if (weight < 1) {
    1104 for (int channel = 0; channel < 3; ++channel)
    1105 pixel[channel] = cv::saturate_cast<unsigned char>(weight * pixel[channel] +
    1106 (1 - weight) * sample(completedMosaic, channel));
    1107 if (observed) { result.confidence[index] = 48; ++rowBlended; }
    1108 else ++rowReconstructed;
    1109 }
    1110 }
    1111 }
    1112 reconstructed += rowReconstructed; reconstructionBlend += rowBlended;
    1113 }, config.mosaicThreads);
    1114 result.validFraction = completedMosaic.empty() ? static_cast<double>(cv::countNonZero(coverage)) / count : 1;
    1115 result.reconstructedFraction = static_cast<double>(reconstructed.load()) / count;
    1116 result.reconstructionBlendFraction = static_cast<double>(reconstructionBlend.load()) / count;
    1117 return result;
    1118}

    1121 const std::function<bool()>& stopped,
    1122 unsigned workers, unsigned memoryMiB) {
    1123 const auto started = std::chrono::steady_clock::now();
    1124 const auto sources = readSources(csv);
    1125 if (std::none_of(sources.begin(), sources.end(), [](const OrthoSource& source) { return source.distorted; }))
    1126 return csv;
    1127 if (std::filesystem::exists(directory)) throw std::runtime_error("Rectified working directory already exists");
    1128 std::filesystem::create_directories(directory);
    1129 std::ifstream input(csv);
    1130 std::string line;
    1131 std::getline(input, line);
    1132 const auto header = csvFields(line);
    1133 std::map<std::string, std::size_t> columns;
    1134 for (std::size_t i = 0; i < header.size(); ++i) columns.emplace(header[i], i);

    1136 std::ofstream output(result, std::ios::binary);
    1137 output << line << '\n';
    1138 std::uint64_t bytesPerWorker = 1;
    1139 for (const auto& source : sources) {
    1140 const auto required = static_cast<std::uint64_t>(source.width) * source.height * 128 +
    1141 std::filesystem::file_size(source.path) + 4 * 1024 * 1024;
    1142 bytesPerWorker = std::max(bytesPerWorker, required);
    1143 }
    1144 const auto memoryWorkers = std::max<std::uint64_t>(1,
    1145 static_cast<std::uint64_t>(memoryMiB) * 1024 * 1024 / bytesPerWorker);
    1146 workers = static_cast<unsigned>(std::min<std::uint64_t>({std::max(1u, workers),
    1147 std::max(1u, std::thread::hardware_concurrency()), sources.size(), memoryWorkers}));
    1148 std::vector<int> insets(sources.size());
    1149 parallelRows(static_cast<int>(workers), [&](int worker) {
    1150 JpegEncoder encoder;
    1151 RectificationCache rectification;
    1152 for (std::size_t index = static_cast<unsigned>(worker); index < sources.size(); index += workers) {

    1223 const std::function<bool()>& stopped) {

    1231 for (std::size_t i = 0; i < state.sources.size(); ++i) {
    1232 const auto p = state.projection.forward(state.sources[i].latitude, state.sources[i].longitude);
    1233 nearest.emplace_back(std::hypot(p.first - result.map.east, p.second - result.map.north), i);
    1234 }
    1235 std::sort(nearest.begin(), nearest.end());

    1237 for (const auto& location : nearest) {
    1238 if (photos.size() == 12) break;
    1239 const auto& info = state.sources[location.second];
    1240 const double maxFootprint = (info.altitude - config.referencePlane) / info.focal * std::hypot(info.width, info.height);
    1241 if (location.first > maxFootprint / 2 + std::hypot(tileWidth, tileHeight) / 2) break;

    1243 std::size_t loaded = 0;

    1246 throw std::runtime_error("Source view memory exceeds 512 MB; reduce source viewport dimensions");

    1248 photos.push_back(sourcePhoto(location.second));
    1249 }
    1250 const auto count = static_cast<std::size_t>(result.width) * result.height;
    1251 result.bgra.assign(count * 4, 0);
    1252 result.confidence.assign(count, 0);
    1253 const double left = map.east - (result.width - 1) * config.mapPixelSize / 2;

    1256 if (stopped()) throw std::runtime_error("Simple mosaic canceled");

    1258 const double e = left + x * config.mapPixelSize, n = top - y * config.mapPixelSize;
    1259 double best = std::numeric_limits<double>::infinity(), weights = 0;
    1260 std::array<double, 3> sum{};

    1264 const auto& photo = *owner;
    1265 double u, v;
    1266 if (!photo.project(e, n, config.referencePlane, u, v, 1)) continue;
    1267 const double score = std::pow((u - photo.info.cx) / photo.info.width, 2) +
    1268 std::pow((v - photo.info.cy) / photo.info.height, 2);
    1269 unsigned char colour[4]{};

    1271 if (config.blend == "feather") {
    1272 const double edge = std::min({u, v, photo.info.width - 1 - u, photo.info.height - 1 - v});
    1273 const double weight = edge * edge / (1 + score);
    1274 for (std::size_t channel = 0; channel < 3; ++channel) sum[channel] += colour[channel] * weight;
    1275 weights += weight;
    1276 } else if (score < best) {
    1277 best = score;
    1278 std::copy(colour, colour + 4, pixel);
    1279 }
    1280 }
    1281 if (weights > 0) {
    1282 for (std::size_t channel = 0; channel < 3; ++channel)
    1283 pixel[channel] = static_cast<unsigned char>(std::clamp(std::lround(sum[channel] / weights), 0L, 255L));
    1284 pixel[3] = 255;
    1285 }
    1286 if (pixel[3]) result.confidence[index] = 64;
    1287 }
    1288 });
    1289 result.validFraction = static_cast<double>(std::count(result.confidence.begin(), result.confidence.end(), 64)) / count;
    1290 return result;
    1291}
