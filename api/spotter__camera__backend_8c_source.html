    61/** @brief Clear the event hint while holding @c target_mutex.
    62 * @details The result queue, not the event counter, owns pending work. Clear
    63 * the counter only while producers are excluded and the queue is empty. */
    64static void drain_notification_locked(void)
    65{
    66 if (notification_fd < 0) return;
    67 eventfd_t value;
    68 int status;
    69 do {
    70 status = eventfd_read(notification_fd, &value);
    71 } while (status < 0 && errno == EINTR);
    72}
    73
    74/** @brief Wake the event loop after queuing a result under @c target_mutex.
    75 * @details An eventfd wake is a readability hint, not a second result queue;
    76 * a saturated counter already has the desired readable state. */
    77static void notify_locked(void)
    78{
    79 if (notification_fd < 0) return;
    80 const uint64_t increment = 1U;
    81 ssize_t written;
    82 do {
    83 written = write(notification_fd, &increment, sizeof(increment));
    84 } while (written < 0 && errno == EINTR);
    85}

    107/** @brief Reset target accumulation and prepare completion notification.
    108 * @return Zero with a usable event descriptor, or -1 on eventfd failure.
    109 * @details Reset shot ordering with the result ring so a prior stream's
    110 * observations cannot be mistaken for this stream's output. */
    111static int spotter_stream_init(void)
    112{
    113 pthread_mutex_lock(&target_mutex);
    114 spotter_targets_reset(&target_store);
    115 pending_head = 0U;
    116 pending_count = 0U;
    117 last_admitted_shot = 0U;
    118 if (notification_fd < 0) {
    119 notification_fd = eventfd(0, EFD_NONBLOCK | EFD_CLOEXEC);
    120 }
    121 const int result = notification_fd >= 0 ? 0 : -1;
    122 pthread_mutex_unlock(&target_mutex);
    123 return result;
    124}

    126/** @brief Discard pending target results and close their notification channel.
    127 * @details Serialize descriptor lifetime with producers so they cannot write
    128 * to a closed or subsequently reused descriptor. */
    129static void spotter_stream_drain(void)
    130{
    131 pthread_mutex_lock(&target_mutex);
    132 pending_head = 0U;
    133 pending_count = 0U;
    134 if (notification_fd >= 0) {
    135 close(notification_fd);
    136 notification_fd = -1;
    137 }
    138 pthread_mutex_unlock(&target_mutex);
    139}

    141/** @brief Get the stream's borrowed poll descriptor.
    142 * @return Event descriptor, or -1 if the stream has none.
    143 * @details The lock protects lookup, not the lifetime after return; the
    144 * event-loop owner coordinates polling with teardown. */
    145static int spotter_stream_completion_fd(void)
    146{
    147 pthread_mutex_lock(&target_mutex);
    148 const int fd = notification_fd;
    149 pthread_mutex_unlock(&target_mutex);
    150 return fd;
    151}

    153/** @brief Pop one queued Spottercam point.
    154 * @param[out] result Destination, unchanged if no point is available.
    155 * @return True when a point was copied.
    156 * @details Keep the event descriptor readable while more points remain, and
    157 * clear it under the producer lock once the last point is removed. */
    158static bool spotter_stream_poll_result(struct camera_point_result *result)
    159{
    160 if (!result) return false;
    161 bool available = false;
    162 pthread_mutex_lock(&target_mutex);
    163 if (pending_count > 0U) {
    164 *result = pending_points[pending_head];
    165 pending_head = (pending_head + 1U) % SPOTTER_POINT_QUEUE_CAPACITY;
    166 --pending_count;
    167 available = true;
    168 }
    169 if (pending_count == 0U) {
    170 drain_notification_locked();
    171 }
    172 pthread_mutex_unlock(&target_mutex);
    173 return available;
    174}

    176/** @brief Convert a saved image's detections into stream observations.
    177 * @param filename Published JPEG used to locate matching detector evidence.
    178 * @param shot Pose and photo number associated with that image.
    179 * @details Reject non-increasing shot IDs and invalid evidence before
    180 * updating the target store. Rescale intrinsics to the detection raster
    181 * before projecting; these are flat-ground estimates, not surveyed positions. */
    182static void spotter_image_saved(const char *filename, const union dc_shot_union *shot)
    183{
    184 if (!spotter_camera_backend_targets_enabled() || !filename || !shot || shot->data.nr <= 0) {
    185 return;
    186 }
    187 const uint32_t frame_number = (uint32_t)shot->data.nr;
    188 pthread_mutex_lock(&target_mutex);
    189 if (frame_number <= last_admitted_shot
    190 || !spotter_cam_pipe_read_detections(filename, &parsed_detections)
    191 || parsed_detections.count == 0U
    192 || parsed_detections.width < 2U || parsed_detections.width > 4096U
    193 || parsed_detections.height < 2U || parsed_detections.height > 4096U) {
    194 pthread_mutex_unlock(&target_mutex);
    195 return;
    196 }
    197 struct coverage_camera camera = target_profile.calibration;
    198 const double scale_x = (double)parsed_detections.width / (double)camera.width;
    199 const double scale_y = (double)parsed_detections.height / (double)camera.height;
    200 camera.fx *= scale_x;
    201 camera.fy *= scale_y;
    202 /* Preserve the principal point's pixel-center convention during resize. */
    203 camera.cx = (camera.cx + 0.5) * scale_x - 0.5;
    204 camera.cy = (camera.cy + 0.5) * scale_y - 0.5;
    205 camera.width = parsed_detections.width;
    206 camera.height = parsed_detections.height;
    207
    208 const double deg_per_rad = 57.29577951308232;
    209 const double rad_per_deg = 0.017453292519943295;
    210 const double earth_radius_m = 6378137.0;
    211 const double agl_m = shot->data.groundalt > 0 ? shot->data.groundalt / 256.0 : shot->data.alt / 1000.0;
    212 const struct coverage_pose pose = {
    213 .latitude_deg = shot->data.lat / 1e7,
    214 .longitude_deg = shot->data.lon / 1e7,
    215 .agl_m = agl_m,
    216 .roll_deg = shot->data.phi / 4096.0 * deg_per_rad,
    217 .pitch_deg = shot->data.theta / 4096.0 * deg_per_rad,
    218 .yaw_deg = shot->data.psi / 4096.0 * deg_per_rad,
    219 .uncertainty_m = 0.0
    220 };
    221 const double cos_lat = cos(pose.latitude_deg * rad_per_deg);
    222 if (!isfinite(cos_lat) || cos_lat <= 1e-6) {
    223 pthread_mutex_unlock(&target_mutex);
    224 return;
    225 }
    226
    227 for (unsigned index = 0U; index < parsed_detections.count; ++index) {
    228 const struct spotter_detection *box = &parsed_detections.objects[index];
    229 const double cx = box->precise_center ? box->center_x : box->x + 0.5 * box->width;
    230 const double cy = box->precise_center ? box->center_y : box->y + 0.5 * box->height;
    231 struct coverage_point offset;
    232 if (!coverage_project_pixel(&camera, &pose, cx, cy, &offset)) {
    233 continue;
    234 }
    235 const double target_lat = pose.latitude_deg + (offset.north / earth_radius_m) * deg_per_rad;
    236 const double target_lon = remainder(
    237 pose.longitude_deg + (offset.east / (earth_radius_m * cos_lat)) * deg_per_rad, 360.0);
    238 if (!isfinite(target_lat) || !isfinite(target_lon)
    239 || fabs(target_lat) > 90.0 || fabs(target_lon) > 180.0) {
    240 continue;
    241 }
    242 const long long lat_e7 = llround(target_lat * 1e7);
    243 const long long lon_e7 = llround(target_lon * 1e7);
    244 const long long ground_alt_mm = (long long)shot->data.alt - llround(agl_m * 1000.0);
    245 const long long score = llround(box->confidence * 1000.0);
    246 if ((lat_e7 == 0 && lon_e7 == 0) || score < 0 || score > 1000
    247 || ground_alt_mm < INT32_MIN || ground_alt_mm > INT32_MAX) {
    248 continue;
    249 }
    250 const struct spotter_target observation = {
    251 .latitude = (int32_t)lat_e7,
    252 .longitude = (int32_t)lon_e7,
    253 .altitude_mm = (int32_t)ground_alt_mm,
    254 .score = (uint16_t)score,
    255 .valid = true
    256 };
    257 const int slot = spotter_targets_offer(&target_store, &observation, NULL, NULL);
    258 if (slot < 0) {
    259 continue;
    260 }
    261 last_admitted_shot = frame_number;
    262 const struct camera_point_result update = {
    263 .session_id = frame_number,
    264 .valid = true,
    265 .lat = observation.latitude,
    266 .lon = observation.longitude,
    267 .alt_mm = observation.altitude_mm,
    268 .agl_mm = 0,
    269 .quality = observation.score,
    270 .metric = slot
    271 };
    272 if (pending_count < SPOTTER_POINT_QUEUE_CAPACITY) {
    273 const size_t tail = (pending_head + pending_count) % SPOTTER_POINT_QUEUE_CAPACITY;
    274 pending_points[tail] = update;
    275 ++pending_count;
    276 } else {
    277 pending_points[pending_head] = update;
    278 pending_head = (pending_head + 1U) % SPOTTER_POINT_QUEUE_CAPACITY;
    279 }
    280 notify_locked();
    281 printf("SPOTTER_TARGETS:\tshot %u assigned T%d lat=%.7f lon=%.7f score=%u\n",
    282 frame_number, slot + 1, observation.latitude / 1e7, observation.longitude / 1e7,
    283 (unsigned)observation.score);
    284 break;
    285 }
    286 pthread_mutex_unlock(&target_mutex);
    287}
