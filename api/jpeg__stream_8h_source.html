    19 * @file jpeg_stream.h
    20 * @brief Decode JPEGs with IJG scanlines into BGRA previews or disk-backed rasters.
    21 * @details Streaming one scanline at a time avoids retaining both the
    22 * compressed source and a full decoded frame in memory. The working raster is
    23 * an internal lossless intermediate. Its bounded raster path rejects
    24 * progressive JPEGs; preview decoding retains the codec's progressive support.

    114 /** @brief Read a JPEG preview without retaining the compressed file in memory.
    115 * @param source JPEG path with dimensions matching its capture metadata.
    116 * @param width Expected full-resolution width.
    117 * @param height Expected full-resolution height.
    118 * @param maxSide Preferred longest decoded side, or zero for full resolution.
    119 * @return Owned BGRA pixels and the decoder's intrinsic scaling ratio.
    120 * @throws std::runtime_error For damaged JPEGs, invalid dimensions or I/O errors. */
    121 Image read(const std::filesystem::path& source, int width, int height, int maxSide) {
    122 reset();
    123 const Operation operation{*this};
    124 const auto bytes = std::filesystem::file_size(source);
    125 if (!bytes || bytes > 128 * 1024 * 1024 || width <= 0 || height <= 0 ||
    126 width > 16384 || height > 16384 || maxSide < 0)
    127 throw std::runtime_error("Invalid source JPEG size or dimensions");
    128 input_ = fopen(source.c_str(), "rb");
    129 if (!input_) throw std::runtime_error("Cannot open source JPEG");
    130 decoder_.err = jpeg_std_error(&error_.manager);
    131 error_.manager.error_exit = failure;
    132 error_.emit = error_.manager.emit_message;
    133 error_.manager.emit_message = message;
    134 if (setjmp(error_.recovery)) throw std::runtime_error(std::string("JPEG decode: ") + error_.message);
    135 created_ = true; jpeg_create_decompress(&decoder_);
    136 jpeg_stdio_src(&decoder_, input_);
    137 jpeg_read_header(&decoder_, TRUE);
    138 if (decoder_.image_width != static_cast<unsigned>(width) ||
    139 decoder_.image_height != static_cast<unsigned>(height))
    140 throw std::runtime_error("Source JPEG dimensions disagree with capture CSV");
    141 decoder_.scale_denom = 8;
    142 for (unsigned numerator = 8; numerator > 0; --numerator) {
    143 decoder_.scale_num = numerator;
    144 jpeg_calc_output_dimensions(&decoder_);
    145 if (!maxSide || std::max(decoder_.output_width, decoder_.output_height) <= static_cast<unsigned>(maxSide) || numerator == 1)
    146 break;
    147 }
    148 decoder_.out_color_space = JCS_RGB;
    149 jpeg_start_decompress(&decoder_);
    150 image_.width = static_cast<int>(decoder_.output_width);
    151 image_.height = static_cast<int>(decoder_.output_height);
    152 image_.scale = double(decoder_.scale_num) / decoder_.scale_denom;
    153 image_.pixels.resize(static_cast<std::size_t>(image_.width) * image_.height * 4);
    154 row_.resize(static_cast<std::size_t>(image_.width) * 3);
    155 while (decoder_.output_scanline < decoder_.output_height) {
    156 const auto row = decoder_.output_scanline;
    157 JSAMPROW pixels = row_.data();
    158 if (jpeg_read_scanlines(&decoder_, &pixels, 1) != 1)
    159 throw std::runtime_error("Incomplete JPEG scanline");
    160 auto* destination = image_.pixels.data() + static_cast<std::size_t>(row) * image_.width * 4;
    161 for (int column = 0; column < image_.width; ++column) {
    162 destination[column * 4] = row_[column * 3 + 2];
    163 destination[column * 4 + 1] = row_[column * 3 + 1];
    164 destination[column * 4 + 2] = row_[column * 3];
    165 destination[column * 4 + 3] = 255;
    166 }
    167 }
    168 jpeg_finish_decompress(&decoder_);
    169 if (decoder_.err->num_warnings) throw std::runtime_error("Damaged source JPEG");
    170 return std::move(image_);
    171 }
    172 /**
    173 * @brief Stream a source JPEG into a new, atomically published raster file.
    174 * @param[in] source Input baseline JPEG path.
    175 * @param[in] target Destination raster path; existing files are not replaced.
    176 * @param[in] width Expected source width in pixels.
    177 * @param[in] height Expected source height in pixels.
    178 * @param[in] stopped Cooperative cancellation predicate.
    179 * @throws std::runtime_error For invalid dimensions, decode/write errors,
    180 * damaged JPEG data, or cancellation.
    181 */
    182 void decode(const std::filesystem::path& source, const std::filesystem::path& target,
    183 int width, int height, const std::function<bool()>& stopped) {
    184 reset();
    185 const Operation operation{*this};
    186 if (width <= 0 || height <= 0 || width > 8192 || height > 8192)
    187 throw std::runtime_error("Invalid streamed JPEG dimensions");
    188 if (std::filesystem::exists(target)) throw std::runtime_error("Working raster already exists");
    189 input_ = fopen(source.c_str(), "rb");
    190 if (!input_) throw std::runtime_error("Cannot open JPEG for streamed decoding");
    191 const auto temporary = target.string() + ".partial";
    192 output_ = std::make_unique<std::ofstream>(temporary, std::ios::binary | std::ios::trunc);
    193 if (!*output_) throw std::runtime_error("Cannot create working raster");
    194 row_.resize(static_cast<std::size_t>(width) * 4);
    195 decoder_.err = jpeg_std_error(&error_.manager);
    196 error_.manager.error_exit = failure;
    197 error_.emit = error_.manager.emit_message;
    198 error_.manager.emit_message = message;
    199 if (setjmp(error_.recovery)) throw std::runtime_error(std::string("JPEG decode: ") + error_.message);
    200 created_ = true; jpeg_create_decompress(&decoder_);
    201 jpeg_stdio_src(&decoder_, input_);
    202 jpeg_read_header(&decoder_, TRUE);
    203 if (decoder_.image_width != static_cast<unsigned>(width) ||
    204 decoder_.image_height != static_cast<unsigned>(height) || decoder_.progressive_mode)
    205 throw std::runtime_error("Streamed JPEG requires matching dimensions and baseline encoding");
    206 decoder_.out_color_space = JCS_RGB;
    207 jpeg_start_decompress(&decoder_);
    208 while (decoder_.output_scanline < decoder_.output_height) {
    209 if (stopped && stopped()) throw std::runtime_error("Streamed source preparation canceled");
    210 JSAMPROW pixels = row_.data();
    211 if (jpeg_read_scanlines(&decoder_, &pixels, 1) != 1)
    212 throw std::runtime_error("Incomplete JPEG scanline");
    213 for (std::size_t pixel = static_cast<std::size_t>(width); pixel > 0; --pixel) {
    214 const std::size_t source = (pixel - 1) * 3;
    215 const std::size_t destination = (pixel - 1) * 4;
    216 const unsigned char red = row_[source];
    217 const unsigned char green = row_[source + 1];
    218 const unsigned char blue = row_[source + 2];
    219 row_[destination] = blue;
    220 row_[destination + 1] = green;
    221 row_[destination + 2] = red;
    222 row_[destination + 3] = 255;
    223 }
    224 output_->write(reinterpret_cast<const char*>(row_.data()), static_cast<std::streamsize>(row_.size()));
    225 if (!*output_) throw std::runtime_error("Cannot write working raster");
    226 }
    227 jpeg_finish_decompress(&decoder_); output_->close();
    228 if (!*output_ || decoder_.err->num_warnings) throw std::runtime_error("Damaged JPEG or working raster write");
    229 publishWorkingFile(temporary, target, false);
    230 }
    231};
    232}
