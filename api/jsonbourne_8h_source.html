    18/**
    19 * @file jsonbourne.h
    20 * @brief JSON reader, writer and DOM utilities for Linux C11 with a C++11-compatible API.
    21 *
    22 * @details
    23 * Strict parsing validates RFC 8259 grammar and Unicode scalar values in UTF-8.
    24 * The input need not be terminated, aligned or padded and is never modified.
    25 * Decoded strings and unchanged numeric tokens are owned by the returned DOM.
    26 * Duplicate names and their order are preserved; jsonbourne_obj_getn_unique() rejects
    27 * ambiguous lookups. Embedded NULs are supported through explicit byte lengths.
    28 *
    29 * Parsing and cloning make one allocation, released with free() when the default
    30 * allocator is used. Custom allocators must provide malloc-compatible alignment
    31 * and own their corresponding release policy. jsonbourne_read_buf(), jsonbourne_read_measure(),
    32 * jsonbourne_read_validate() and the *_buf writers do not allocate. Input, DOM and output
    33 * buffers must not overlap and must remain stable throughout each operation.
    34 * Allocator callbacks must not mutate input or DOM data. Independent calls share
    35 * no mutable state and may run concurrently.
    36 *
    37 * JSONBOURNE_READER_DEPTH_LIMIT bounds reader nesting and fixed traversal storage.
    38 * JSONBOURNE_WRITER_DEPTH_LIMIT bounds writer/copy traversal and defaults to the
    39 * reader limit. Both are positive hard limits, initially 999; zero does not
    40 * mean unlimited. Embedded consumers should select smaller application-specific
    41 * limits when compiling jsonbourne.c. Runtime limits additionally bound input,
    42 * total DOM bytes, decoded string bytes and value count before allocation.
    43 * SIMD is automatic for baseline x86-64 SSE2 and AArch64 NEON; define
    44 * JSONBOURNE_DISABLE_SIMD=1 for a scalar build. Every vector load is within the
    45 * supplied input extent. Normal -O2/-O3 suffices; no fast-math or SSSE3 is needed.
    46 *
    47 * Writers validate accessible DOM structures, UTF-8, numbers, list lengths and
    48 * cycles. All output sizes INCLUDE the terminating NUL, matching the original
    49 * API. Numbers keep their decimal spelling and precision. Relaxed hexadecimal
    50 * tokens are converted exactly, up to JSONBOURNE_MAX_HEX_DIGITS significant hex
    51 * digits; larger conversions fail rather than overflow. The legacy relaxed
    52 * nonfinite writer policy maps NaN to 0 and infinities to signed DBL_MAX text.
    53 * Compile jsonbourne.c once with the desired build options and link its object.
    54 * Public names follow Yuan conventions, but the DOM layout, flag bits and
    55 * writer lengths are JSONbourne-specific rather than Yuan ABI compatible.
    56 * Pretty-print indentation/newlines must contain JSON whitespace only.
    57 *
    58 * @par Why tokens and lengths are preserved
    59 * Keeping decimal text avoids rounding large integers through binary floating
    60 * point. Explicit byte lengths distinguish embedded NULs from terminators, and
    61 * preserving duplicate keys lets the application decide whether ambiguity is
    62 * acceptable instead of silently discarding input.
    63 *
    64 * @pre Input, DOM nodes, linked entries and output storage must refer to valid,
    65 * accessible memory for their advertised lifetimes and extents. Structural
    66 * validation detects malformed lists and cycles, not arbitrary invalid pointers.
    67 * @note Select compile-time options when building jsonbourne.c; defining them
    68 * only in a client does not change the linked implementation's storage limits.
    69 * @see jsonbourne_read_buf(), jsonbourne_read_limits_default(), jsonbourne_version()
    70 */

    95#ifndef JSONBOURNE_READER_DEPTH_LIMIT
    96/**
    97 * @brief Maximum simultaneous open reader containers; must be positive.
    98 * @details A fixed stack makes nesting memory predictable without recursive
    99 * calls or growth allocations. Smaller application-specific bounds save stack
    100 * space even when a particular input has little nesting.
    101 * @see jsonbourne_read_limits::max_depth
    102 */
    103#define JSONBOURNE_READER_DEPTH_LIMIT 999
    104#endif
    105#ifndef JSONBOURNE_WRITER_DEPTH_LIMIT
    106/**
    107 * @brief Positive writer/copy nesting bound, defaulting to the reader bound.
    108 * @details Separate limits allow a program to inspect deeper input than it is
    109 * willing to traverse for output or copying, without unbounded stack growth.
    110 */
    111#define JSONBOURNE_WRITER_DEPTH_LIMIT JSONBOURNE_READER_DEPTH_LIMIT
    112#endif
    113#if JSONBOURNE_READER_DEPTH_LIMIT < 1 || JSONBOURNE_WRITER_DEPTH_LIMIT < 1
    114#error JSONbourne reader and writer depth limits must be positive
    115#endif
    116#ifndef JSONBOURNE_MAX_HEX_DIGITS
    117/**
    118 * @brief Maximum significant hexadecimal digits converted to decimal output.
    119 * @details Bounds exact conversion work and its fixed scratch array; accepted
    120 * settings are 16 through 16384. Leading zeroes do not consume this limit.

    122#define JSONBOURNE_MAX_HEX_DIGITS 1024
    123#endif
    124#if JSONBOURNE_MAX_HEX_DIGITS < 16 || JSONBOURNE_MAX_HEX_DIGITS > 16384
    125#error JSONBOURNE_MAX_HEX_DIGITS must be between 16 and 16384
    126#endif

    148/**
    149 * @brief Independent reader options and convenience presets.
    150 * @details Presets are unions of feature bits, not distinct parser modes.
    151 * Use the names rather than persisting raw masks so code can follow flag remaps.
    152 */
    153enum jsonbourne_read_flag_e {
    154 JSONBOURNE_READ_NOFLAG = 0, /**< Strict RFC 8259 input. */
    155 JSONBOURNE_READ_ALLOW_TRAILING_COMMAS = 0x0002, /**< Permit a comma before container closure. */
    156 JSONBOURNE_READ_ALLOW_UNQUOTED_KEY = 0x0004, /**< Permit nonempty ASCII identifier-like keys. */
    157 JSONBOURNE_READ_ALLOW_GLOBAL_OBJECT = 0x0008, /**< Permit an object without outer braces. */
    158 JSONBOURNE_READ_ALLOW_EQUALS = 0x0010, /**< Permit equals in place of a member colon. */
    159 JSONBOURNE_READ_ALLOW_NO_COMMAS = 0x0020, /**< Permit adjacent members/elements without commas. */
    160 JSONBOURNE_READ_ALLOW_COMMENTS = 0x0040, /**< Skip C-style block and line comments. */
    161 JSONBOURNE_READ_RESERVED = 0x0080, /**< Accepted reserved bit with no syntax effect. */
    162 JSONBOURNE_READ_TRACK_LOCATION = 0x0100, /**< Allocate location extensions for values and keys. */
    163 JSONBOURNE_READ_ALLOW_SINGLE_QUOTED_STR = 0x0200, /**< Permit single quotes and escaped apostrophes. */
    164 JSONBOURNE_READ_ALLOW_HEX_NUM = 0x0400, /**< Permit hexadecimal integer tokens. */
    165 JSONBOURNE_READ_ALLOW_PLUS_SIGN = 0x0800, /**< Permit an explicit positive number sign. */
    166 JSONBOURNE_READ_ALLOW_DECIMAL_POINT = 0x1000, /**< Permit .5 and 1. forms. */
    167 JSONBOURNE_READ_ALLOW_INF_AND_NAN = 0x2000, /**< Permit signed Infinity and NaN tokens. */
    168 JSONBOURNE_READ_ALLOW_MULTILINE_STR = 0x4000, /**< Permit raw CR/LF and escaped line continuations. */
    169 JSONBOURNE_READ_SIMPLIFIED = 0x003e, /**< Combine the five object/spacing conveniences. */
    170 JSONBOURNE_READ_JSON5 = 0x7e46 /**< JSON5-style conveniences, not full conformance. */
    171 /* bitmask combining nine parsing options
    172 0x0002 | 0x0004 | 0x0040 | 0x0200 | 0x0400 |
    173 0x0800 | 0x1000 | 0x2000 | 0x4000 = 0x7e46
    174 */
    175};
