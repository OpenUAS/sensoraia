    146/** Open only a bounded regular file; O_NONBLOCK prevents special-file stalls.
    147 * @param path Input path.
    148 * @param maximum Maximum permitted size in bytes.
    149 * @return Owning stream, or throws a descriptive error. */
    150File open_regular(const std::string& path, size_t maximum) {
    151 const int descriptor = open(path.c_str(), O_RDONLY | O_NONBLOCK | O_CLOEXEC);
    152 require(descriptor >= 0, "cannot open input: " + path);
    153 struct stat status{};
    154 if (fstat(descriptor, &status) != 0 || !S_ISREG(status.st_mode) ||
    155 status.st_size < 0 || static_cast<size_t>(status.st_size) > maximum) {
    156 close(descriptor);
    157 throw std::runtime_error("input must be a bounded regular file: " + path);
    158 }
    159 File input(fdopen(descriptor, "rb"));
    160 if (!input) {
    161 close(descriptor);
    162 throw std::runtime_error("cannot open input stream: " + path);
    163 }
    164 return input;
    165}

    199 {"vibration_phase_deg", 0.0},

    208 {"vignetting", 0.0},

    210 {"sensor_full_well_e", 0.0},
    211 {"sensor_read_noise_e", 0.0},
    212 {"isp_sharpen", 0.0}});

    214 profile.set("output_height", nullptr);

    294 number(profile["height_override_m"], "height_override_m", 0.1, 10000);

    296 number(profile["focal_length_mm"], "focal_length_mm", 0.1, 1000);

    304 for (size_t index = 0; index < profile[key].size(); ++index)
    305 number(profile[key].at(index), key, -limit, limit);
    306 }
    307 return profile;
    308}

    357/** Decode one regular JPEG after applying source/tile dimensions and memory
    358 * checks.
    359 * @param path JPEG path.
    360 * @param tile True for a cache tile (must be exactly 256x256); false for a
    361 * source frame.
    362 * @param budget Maximum permitted working bytes.
    363 * @param header Optional destination for encoder facts used by --measure.
    364 * @param decode_pixels False to validate header/dimensions/budget only (used
    365 * when rendering from a map cache without pinhole source sampling).
    366 * @return Contiguous RGB pixels, or empty when @p decode_pixels is false. */
    367cv::Mat read_jpeg(const std::string& path, bool tile, size_t budget,
    368 JpegHeader* header = nullptr, bool decode_pixels = true) {
    369 File input = open_regular(path, (tile ? 4 : 64) * mib);
    370 auto decoder = std::make_unique<Decoder>();
    371 decoder->state.err = jpeg_std_error(&decoder->error.manager);
    372 decoder->error.emit = decoder->error.manager.emit_message;
    373 decoder->error.manager.emit_message = jpeg_message;
    374 decoder->error.manager.error_exit = jpeg_failure;
    375 if (setjmp(decoder->error.recovery))
    376 throw std::runtime_error("invalid JPEG: " + path);
    377 decoder->created = true;
    378 jpeg_create_decompress(&decoder->state);
    379 jpeg_stdio_src(&decoder->state, input.get());
    380 require(jpeg_read_header(&decoder->state, TRUE) == JPEG_HEADER_OK,
    381 "invalid JPEG header");
    382 const size_t width = decoder->state.image_width,
    383 height = decoder->state.image_height;
    384 require(width >= 2 && height >= 2 && width <= maximum_image_dimension &&
    385 height <= maximum_image_dimension &&
    386 width * height <= maximum_mock_pixels &&
    387 (!tile || (width == 256 && height == 256)),
    388 "unsupported JPEG dimensions: " + path);
    389 require(32 * mib + width * height * 12 <= budget,
    390 "input exceeds memory budget before decode");
    391 if (header) {
    392 header->width = static_cast<int>(width);
    393 header->height = static_cast<int>(height);
    394 const JQUANT_TBL* table = decoder->state.quant_tbl_ptrs[0];
    395 for (size_t index = 0; table && index < header->luma_quant.size(); ++index)
    396 header->luma_quant[index] = table->quantval[index];
    397 if (decoder->state.num_components == 3 &&
    398 decoder->state.jpeg_color_space == JCS_YCbCr) {
    399 const auto* components = decoder->state.comp_info;
    400 if (components[1].h_samp_factor == 1 && components[1].v_samp_factor == 1 &&
    401 components[2].h_samp_factor == 1 && components[2].v_samp_factor == 1) {
    402 const int horizontal = components[0].h_samp_factor,
    403 vertical = components[0].v_samp_factor;
    404 header->subsampling = horizontal == 2 && vertical == 2 ? "4:2:0"
    405 : horizontal == 2 && vertical == 1 ? "4:2:2"
    406 : horizontal == 1 && vertical == 1 ? "4:4:4"
    407 : "other";
    408 }
    409 }
    410 }
    411 if (!decode_pixels) {
    412 require(decoder->error.manager.num_warnings == 0, "damaged JPEG: " + path);
    413 return {};
    414 }
    415 decoder->state.out_color_space = JCS_RGB;
    416 jpeg_start_decompress(&decoder->state);
    417 decoder->image.create(static_cast<int>(height), static_cast<int>(width),
    418 CV_8UC3);
    419 while (decoder->state.output_scanline < height) {
    420 JSAMPROW row = decoder->image.ptr(decoder->state.output_scanline);
    421 require(jpeg_read_scanlines(&decoder->state, &row, 1) == 1,
    422 "incomplete JPEG scanline");
    423 }
    424 jpeg_finish_decompress(&decoder->state);
    425 require(decoder->error.manager.num_warnings == 0, "damaged JPEG: " + path);
    426 return decoder->image;
    427}

    471/** Write a JPEG derivative with EXIF into a same-directory temporary then
    472 * rename it.
    473 * @param output Destination, which is never the clean input.
    474 * @param image RGB derivative.
    475 * @param quality Profile JPEG quality.
    476 * @param metadata EXIF payload to retain and extend. */
    477void write_jpeg(const std::string& output, const cv::Mat& image, int quality,
    478 const exif_document* metadata) {
    479 std::array<unsigned char, EXIF_MAX_PAYLOAD> payload{};
    480 size_t length = 0;
    481 require(exif_serialize(metadata, payload.data(), payload.size(), &length) == 0,
    482 "EXIF record exceeds JPEG APP1 capacity");
    483 TemporaryOutput temporary(output);
    484 auto encoder = std::make_unique<Encoder>();
    485 encoder->state.err = jpeg_std_error(&encoder->error.manager);
    486 encoder->error.manager.error_exit = jpeg_failure;
    487 if (setjmp(encoder->error.recovery))
    488 throw std::runtime_error("JPEG encoding failed");
    489 encoder->created = true;

    722const std::array<float, 256>& srgb_decoding() {
    723 static const std::array<float, 256> table = [] {
    724 std::array<float, 256> values{};
    725 for (size_t index = 0; index < values.size(); ++index) {
    726 const float encoded = static_cast<float>(index) / 255.0f;
    727 values[index] = encoded <= 0.04045f
    728 ? encoded / 12.92f
    729 : std::pow((encoded + 0.055f) / 1.055f, 2.4f);
    730 }
    731 return values;
    732 }();
    733 return table;

    738class TileCache {
    739 struct Entry {
    740 int64_t column, row;
    741 cv::Mat linear;
    742 };
    743 std::list<Entry> entries;
    744 std::vector<std::pair<int64_t, int64_t>> verified;
    745 std::string root;
    746 size_t budget;
    747
    748 public:
    749 /** @brief Configure a cache with no resident tiles.
    750 * @param path Root of the local map cache.
    751 * @param limit Working-byte budget passed to the JPEG decoder.
    752 * @details Load lazily so views that do not overlap need not share an
    753 * eagerly decoded mosaic. */
    754 TileCache(const std::string& path, size_t limit)
    755 : root(path), budget(limit) {}
    756 /** @brief Ensure a tile in the frame's bounding box exists and is valid
    757 * without re-decoding tiles already verified earlier in the same render. */
    758 void verify(int64_t column, int64_t row) {
    759 const int64_t wrapped = (column % (1 << 18) + (1 << 18)) % (1 << 18);
    760 for (const auto& seen : verified)
    761 if (seen.first == wrapped && seen.second == row) return;
    762 tile(column, row);
    763 }
    764 /** @brief Fetch a linearized float32 RGB tile from the local zoom-18 cache.
    765 * @param column XYZ column, which may be outside the world range before wrapping.
    766 * @param row Validated XYZ row; not wrapped.
    767 * @return Reference-counted 256x256 CV_32FC3 pixels; callers must treat them
    768 * as read-only.
    769 * @throws std::runtime_error If the required tile cannot be read or decoded.
    770 * @details Normalize the column before lookup so opposite antimeridian
    771 * representations share an entry. Decoding and sRGB-to-linear conversion
    772 * happen once per unique tile rather than per strip/channel. */
    773 cv::Mat tile(int64_t column, int64_t row) {
    774 column = (column % (1 << 18) + (1 << 18)) % (1 << 18);
    775 for (auto entry = entries.begin(); entry != entries.end(); ++entry) {
    776 if (entry->column == column && entry->row == row) {
    777 entries.splice(entries.begin(), entries, entry);
    778 return entries.front().linear;
    779 }
    780 }
    781 const cv::Mat encoded =
    782 read_jpeg(root + "/GOOGLE/18/" + std::to_string(column) + "/" +
    783 std::to_string(row) + ".jpeg",
    784 true, budget);
    785 cv::Mat linear(256, 256, CV_32FC3);
    786 const float* decode = srgb_decoding().data();
    787 const unsigned char* source = encoded.data;
    788 float* destination = linear.ptr<float>();
    789 for (int index = 0; index < 256 * 256 * 3; ++index)
    790 destination[index] = decode[source[index]];
    791 bool recorded = false;
    792 for (const auto& seen : verified)
    793 if (seen.first == column && seen.second == row) recorded = true;
    794 if (!recorded) verified.emplace_back(column, row);
    795 entries.push_front({column, row, linear});
    796 if (entries.size() > tile_cache_entries) entries.pop_back();
    797 return linear;
    798 }

    801/** Camera intrinsics and Brown/fisheye distortion coefficients for ray
    802 * generation. */
    803struct Lens {
    804 Json profile;
    805 double focal_x, focal_y;
    806 cv::Mat distortion;
    807 bool fisheye;
    808 /** @brief Build intrinsics for the selected distortion model.
    809 * @param width Output columns in pixels.
    810 * @param height Output rows in pixels.
    811 * @param values Validated model, field of view, and distortion coefficients.
    812 * @details Brown intrinsics use the two rectilinear fields of view. Fisheye
    813 * intrinsics derive a shared focal length from the distorted diagonal ray. */
    814 Lens(int width, int height, const Json& values)
    815 : profile(values), fisheye(values["model"] == "fisheye") {
    816 const double center_x = (width - 1) * 0.5, center_y = (height - 1) * 0.5;
    817 std::vector<double> coefficients =
    818 values["distortion"].get<std::vector<double>>();
    819 distortion = cv::Mat(coefficients).clone();
    820 if (fisheye) {
    821 const double angle = values["diagonal_fov_deg"].get<double>() * pi / 360;
    822 double polynomial = 1, power = angle * angle;
    823 for (double coefficient : coefficients) {
    824 polynomial += coefficient * power;
    825 power *= angle * angle;
    826 }
    827 focal_x = focal_y = std::hypot(center_x, center_y) / (angle * polynomial);
    828 } else {
    829 focal_x = center_x /
    830 std::tan(values["horizontal_fov_deg"].get<double>() * pi / 360);
    831 focal_y = center_y /
    832 std::tan(values["vertical_fov_deg"].get<double>() * pi / 360);
    833 }
    834 require(std::isfinite(focal_x) && focal_x > 0 && focal_y > 0,
    835 "invalid lens focal scale");
    836 }
    837 /** @brief Camera matrix for one channel focal scale. */
    838 cv::Matx33d intrinsic(int width, int height, double scale) const {
    839 return cv::Matx33d(focal_x * scale, 0, (width - 1) * 0.5, 0,
    840 focal_y * scale, (height - 1) * 0.5, 0, 0, 1);
    841 }
    842 /** Forward lens model: normalized rays to distorted pixel coordinates.
    843 * @param rays N x 1 CV_32FC2 normalized camera rays.
    844 * @param width Full image width.
    845 * @param height Full image height.
    846 * @param scale Per-channel focal scaling.
    847 * @return N x 1 CV_32FC2 pixel coordinates. */
    848 cv::Mat project(const cv::Mat& rays, int width, int height,
    849 double scale) const {
    850 const cv::Matx33d camera = intrinsic(width, height, scale);
    851 cv::Mat result;
    852 if (fisheye) {
    853 cv::fisheye::distortPoints(rays, result, camera, distortion);
    854 return result;
    855 }
    856 cv::Mat objects(rays.rows, 1, CV_32FC3);
    857 for (int index = 0; index < rays.rows; ++index) {
    858 const cv::Vec2f point = rays.at<cv::Vec2f>(index);
    859 objects.at<cv::Vec3f>(index) = cv::Vec3f(point[0], point[1], 1);
    860 }
    861 cv::projectPoints(objects, cv::Vec3d(), cv::Vec3d(), camera, distortion,
    862 result, cv::noArray());
    863 return result;
    864 }
    865 /** Exactly invert distortion for arbitrary pixels, then forward-project to
    866 * reject non-convergent rays.
    867 * @param pixels N x 1 CV_32FC2 pixel coordinates.
    868 * @param width Full image width.
    869 * @param height Full image height.
    870 * @param scale Per-channel focal scaling for chromatic-aberration simulation.
    871 * @return N x 1 normalized camera rays. */
    872 cv::Mat undistort(const cv::Mat& pixels, int width, int height,
    873 double scale) const {
    874 const cv::Matx33d intrinsic = this->intrinsic(width, height, scale);
    875 cv::Mat normalized, projected;
    876 auto project = [&](const cv::Mat& rays) {
    877 return this->project(rays, width, height, scale);
    878 };
    879 if (fisheye) {
    880 cv::fisheye::undistortPoints(pixels, normalized, intrinsic, distortion);
    881 } else {
    882 cv::undistortPoints(
    883 pixels, normalized, intrinsic, distortion, cv::noArray(),
    884 cv::noArray(),
    885 cv::TermCriteria(cv::TermCriteria::COUNT | cv::TermCriteria::EPS,
    886 lens_inverse_iterations, 1e-8));
    887 }
    888 projected = project(normalized);
    889 // OpenCV's fixed-point inverse can stop several hundredths of a pixel
    890 // short under strong distortion; Newton-polish it against the forward
    891 // model with a numeric Jacobian, keeping only improving passes.
    892 double residual = cv::checkRange(normalized) && cv::checkRange(projected)
    893 ? cv::norm(projected, pixels, cv::NORM_INF)
    894 : INFINITY;
    895 constexpr float delta = 1e-4f;
    896 for (int pass = 0; pass < lens_polish_passes &&
    897 residual > lens_polish_tolerance_px &&
    898 std::isfinite(residual);
    899 ++pass) {
    900 const cv::Mat shift_x = normalized + cv::Scalar(delta, 0),
    901 shift_y = normalized + cv::Scalar(0, delta);
    902 const cv::Mat moved_x = project(shift_x), moved_y = project(shift_y);
    903 cv::Mat candidate = normalized.clone();
    904 for (int index = 0; index < pixels.rows; ++index) {
    905 const cv::Vec2f base = projected.at<cv::Vec2f>(index);
    906 const cv::Vec2f column_x = (moved_x.at<cv::Vec2f>(index) - base) / delta;
    907 const cv::Vec2f column_y = (moved_y.at<cv::Vec2f>(index) - base) / delta;
    908 const cv::Vec2f error = pixels.at<cv::Vec2f>(index) - base;
    909 const float determinant =
    910 column_x[0] * column_y[1] - column_y[0] * column_x[1];
    911 if (!(std::abs(determinant) > 1e-12f)) continue;
    912 candidate.at<cv::Vec2f>(index) += cv::Vec2f(
    913 (column_y[1] * error[0] - column_y[0] * error[1]) / determinant,
    914 (column_x[0] * error[1] - column_x[1] * error[0]) / determinant);
    915 }
    916 const cv::Mat candidate_projected = project(candidate);
    917 if (!cv::checkRange(candidate) || !cv::checkRange(candidate_projected))
    918 break;
    919 const double candidate_residual =
    920 cv::norm(candidate_projected, pixels, cv::NORM_INF);
    921 if (candidate_residual >= residual) break;
    922 normalized = candidate;
    923 projected = candidate_projected;
    924 residual = candidate_residual;
    925 }
    926 require(cv::checkRange(normalized) && cv::checkRange(projected) &&
    927 cv::norm(projected, pixels, cv::NORM_INF) <=
    928 maximum_reprojection_error_pixels,
    929 "lens inversion did not converge");
    930 return normalized;
    931 }
    932 /** Normalized rays for one strip; see RayField for the accuracy contract.
    933 * @param width Full image width.
    934 * @param height Full image height.
    935 * @param first First row represented by this strip.
    936 * @param rows Strip row count.
    937 * @param scale Per-channel focal scaling.
    938 * @return rows x width CV_32FC2 normalized rays. */
    939 cv::Mat rays(int width, int height, int first, int rows, double scale) const;
    940};

    942/** Full-frame ray field: exact inverse distortion on a coarse node grid,
    943 * bilinear interpolation between nodes.
    944 * @details Brown and fisheye inversion is iterative and dominated rendering
    945 * time when evaluated per pixel and per colour. The distortion field is
    946 * smooth, so the node step is halved until every interpolated cell-centre
    947 * ray forward-projects within ray_interpolation_tolerance_px of its pixel.
    948 * Step 1 is the exact per-pixel inverse. */
    949class RayField {
    950 std::vector<float> node_x, node_y;
    951 cv::Mat nodes;
    952 std::vector<int> cell_x;
    953 std::vector<float> weight_x;
    954 int columns;
    955
    956 static std::vector<float> axis(int size, int step) {
    957 std::vector<float> values;
    958 for (int value = 0; value < size - 1; value += step) values.push_back(value);
    959 values.push_back(static_cast<float>(size - 1));
    960 return values;
    961 }
    962 static void locate(const std::vector<float>& axis, float value, int& cell,
    963 float& weight) {
    964 cell = static_cast<int>(
    965 std::upper_bound(axis.begin(), axis.end() - 1, value) - axis.begin()) - 1;
    966 cell = std::clamp(cell, 0, static_cast<int>(axis.size()) - 2);
    967 weight = (value - axis[cell]) / (axis[cell + 1] - axis[cell]);
    968 }
    969 // Nodes are unit view directions: bounded and smooth up to 90 degrees,
    970 // unlike tangent-plane rays, so wide fisheye corners interpolate well.
    971 cv::Vec2f blend(int cell_row, float weight_row, int cell_column,
    972 float weight_column) const {
    973 const cv::Vec3f* top = nodes.ptr<cv::Vec3f>(cell_row);
    974 const cv::Vec3f* bottom = nodes.ptr<cv::Vec3f>(cell_row + 1);
    975 const cv::Vec3f upper = top[cell_column] * (1 - weight_column) +
    976 top[cell_column + 1] * weight_column;
    977 const cv::Vec3f lower = bottom[cell_column] * (1 - weight_column) +
    978 bottom[cell_column + 1] * weight_column;
    979 const cv::Vec3f direction = upper * (1 - weight_row) + lower * weight_row;
    980 return cv::Vec2f(direction[0] / direction[2], direction[1] / direction[2]);
    981 }
    982
    983 public:
    984 int step = 1;
    985 double error_px = 0;
    986 /** @brief Build and verify the ray field for one channel focal scale.
    987 * @throws std::runtime_error If exact inversion fails at any node/check. */
    988 RayField(const Lens& lens, int width, int height, double scale,
    989 size_t budget = static_cast<size_t>(-1))
    990 : columns(width) {
    991 for (step = ray_grid_initial_step;; step /= 2) {
    992 node_x = axis(width, step);
    993 node_y = axis(height, step);
    994 // Include inverse-polishing temporaries before allocating a dense grid.
    995 require(node_x.size() * node_y.size() <= budget / 256,
    996 "ray grid exceeds memory budget");
    997 cv::Mat pixels(static_cast<int>(node_x.size() * node_y.size()), 1, CV_32FC2);
    998 for (size_t row = 0; row < node_y.size(); ++row)
    999 for (size_t column = 0; column < node_x.size(); ++column)
    1000 pixels.at<cv::Vec2f>(static_cast<int>(row * node_x.size() + column)) =
    1001 cv::Vec2f(node_x[column], node_y[row]);
    1002 const cv::Mat rays = lens.undistort(pixels, width, height, scale);
    1003 nodes.create(static_cast<int>(node_y.size()),
    1004 static_cast<int>(node_x.size()), CV_32FC3);
    1005 for (int index = 0; index < rays.rows; ++index) {
    1006 const cv::Vec2f ray = rays.at<cv::Vec2f>(index);
    1007 nodes.at<cv::Vec3f>(index / nodes.cols, index % nodes.cols) =
    1008 cv::Vec3f(ray[0], ray[1], 1) / std::sqrt(1 + ray.dot(ray));
    1009 }
    1010 error_px = 0;
    1011 if (step == 1) break;
    1012 const int cells_x = static_cast<int>(node_x.size()) - 1,
    1013 cells_y = static_cast<int>(node_y.size()) - 1;
    1014 cv::Mat centres(cells_x * cells_y, 1, CV_32FC2);
    1015 for (int row = 0; row < cells_y; ++row)
    1016 for (int column = 0; column < cells_x; ++column)
    1017 centres.at<cv::Vec2f>(row * cells_x + column) =
    1018 cv::Vec2f((node_x[column] + node_x[column + 1]) * 0.5f,
    1019 (node_y[row] + node_y[row + 1]) * 0.5f);
    1020 cv::Mat blended(centres.rows, 1, CV_32FC2);
    1021 for (int row = 0; row < cells_y; ++row)
    1022 for (int column = 0; column < cells_x; ++column)
    1023 blended.at<cv::Vec2f>(row * cells_x + column) =
    1024 blend(row, 0.5f, column, 0.5f);
    1025 const cv::Mat reprojected = lens.project(blended, width, height, scale);
    1026 error_px = cv::checkRange(reprojected)
    1027 ? cv::norm(reprojected, centres, cv::NORM_INF)
    1028 : INFINITY;
    1029 if (error_px <= ray_interpolation_tolerance_px) break;
    1030 }
    1031 cell_x.resize(width);
    1032 weight_x.resize(width);
    1033 for (int column = 0; column < width; ++column)
    1034 locate(node_x, static_cast<float>(column), cell_x[column], weight_x[column]);
    1035 }

    1047 void row_nodes_into(int full_row, float* rx, float* ry) const {

    1049 float weight_row;

    1098 std::array<double, 3> angles, rates;

    1102 * @param profile Validated lens motion, height, and shutter settings.
    1103 * @details Global shutter removes the row readout term. Reverse rolling

    1108 : profile["height_override_m"].get<double>();

    1110 shot["speed_m_s"].get<double>() * profile["motion_scale"].get<double>();

    1112 angles = {shot["roll_deg"].get<double>(), shot["pitch_deg"].get<double>(),
    1113 shot["yaw_deg"].get<double>()};

    1116 ? profile["readout_s"].get<double>()

    1126 * @details Fixed attitude is permitted; only motion over the exposure
    1127 * requires evaluating distinct sample times. */
    1128 bool stationary() const {
    1129 return speed == 0 && rates == std::array<double, 3>{0, 0, 0} &&
    1130 (vibration == 0 || frequency == 0);

    1132 /** Intersect each ray with a flat ground plane at its row/exposure time.
    1133 * @param rays Normalized strip rays.
    1134 * @param first First full-image row in @p rays.
    1135 * @param height Full image height used for rolling-shutter timing.
    1136 * @param sample Exposure-time offset from the center of the shot.
    1137 * @return East and north displacement matrices in meters. */
    1138 std::pair<cv::Mat, cv::Mat> ground(const cv::Mat& rays, int first, int height,
    1139 double sample) const {
    1140 cv::Mat east(rays.size(), CV_32F), north(rays.size(), CV_32F);
    1141 std::atomic<bool> horizon{false};
    1142 cv::parallel_for_(cv::Range(0, rays.rows), [&](const cv::Range& range) {
    1143 for (int row = range.start; row < range.end; ++row) {
    1144 const float instant =
    1145 static_cast<float>(
    1146 (static_cast<double>(first + row) / (height - 1) - 0.5) *
    1147 readout) +
    1148 static_cast<float>(sample);
    1149 const double cycle = 2 * pi * frequency * instant + phase;
    1150 const double roll =
    1151 (angles[0] + rates[0] * instant + vibration * std::sin(cycle)) * pi /
    1152 180;
    1153 const double pitch = (angles[1] + rates[1] * instant +
    1154 vibration * std::sin(cycle + pi / 2)) *
    1155 pi / 180;
    1156 const double yaw = (angles[2] + rates[2] * instant) * pi / 180;
    1157 const float cos_roll = std::cos(roll), sin_roll = std::sin(roll),
    1158 cos_pitch = std::cos(pitch), sin_pitch = std::sin(pitch);
    1159 const float cos_yaw = std::cos(yaw), sin_yaw = std::sin(yaw);
    1160 const float north_forward = cos_yaw * cos_pitch,
    1161 north_right =
    1162 cos_yaw * sin_pitch * sin_roll - sin_yaw * cos_roll;
    1163 const float north_down =
    1164 cos_yaw * sin_pitch * cos_roll + sin_yaw * sin_roll;
    1165 const float east_forward = sin_yaw * cos_pitch,
    1166 east_right =
    1167 sin_yaw * sin_pitch * sin_roll + cos_yaw * cos_roll;
    1168 const float east_down =
    1169 sin_yaw * sin_pitch * cos_roll - cos_yaw * sin_roll;
    1170 const float down_right = cos_pitch * sin_roll,
    1171 down_down = cos_pitch * cos_roll;
    1172 const float translation_north = speed * instant * std::cos(course),
    1173 translation_east = speed * instant * std::sin(course);
    1174 for (int column = 0; column < rays.cols; ++column) {
    1175 const cv::Vec2f point = rays.at<cv::Vec2f>(row, column);
    1176 const float forward = -point[1], right = point[0];
    1177 const float down =
    1178 -sin_pitch * forward + down_right * right + down_down;
    1179 if (!(down > 1e-4f)) {
    1180 horizon = true;
    1181 return;
    1182 }
    1183 const float scale = altitude / down;
    1184 north.at<float>(row, column) =
    1185 (north_forward * forward + north_right * right + north_down) *
    1186 scale +
    1187 translation_north;
    1188 east.at<float>(row, column) =
    1189 (east_forward * forward + east_right * right + east_down) * scale +
    1190 translation_east;
    1191 }
    1192 }
    1193 });
    1194 require(!horizon, "lens/exposure rays reach the horizon");
    1195 require(cv::checkRange(east) && cv::checkRange(north),
    1196 "non-finite ground projection");
    1197 return {east, north};
    1198 }
    1199};

    1212/** @brief Choose the fewest odd exposure samples that resolve motion blur.
    1213 * @param field Reference-channel ray field.
    1214 * @param pose Shot motion, vibration and shutter model.
    1215 * @param width Output width.
    1216 * @param height Output height.
    1217 * @param maximum Profile sample ceiling (odd).
    1218 * @return Sample count whose neighbouring-sample ground spacing is at most
    1219 * temporal_sample_spacing_px at every probe, plus the largest blur path.
    1220 * @details Probe the corners, edge centres and centre of the frame, follow
    1221 * each ray's ground intersection across the exposure (including vibration
    1222 * reversals) and convert path length to pixels with the local ground
    1223 * sampling distance. Sub-pixel blur therefore costs few samples, while
    1224 * long exposures keep every sample the profile allows. */
    1225TemporalSampling temporal_sampling(const RayField& field, const Pose& pose,
    1226 int width, int height, int maximum) {
    1227 if (pose.stationary()) return {1, 0};
    1228 const int steps = std::max(
    1229 32, static_cast<int>(std::ceil(
    1230 pose.exposure * (pose.vibration > 0 ? pose.frequency : 0) * 8)));
    1231 double blur = 0;
    1232 for (float y : {0.0f, (height - 1) * 0.5f, height - 2.0f}) {
    1233 for (float x : {0.0f, (width - 1) * 0.5f, width - 2.0f}) {
    1234 cv::Mat probe(1, 3, CV_32FC2);
    1235 probe.at<cv::Vec2f>(0, 0) = field.at(x, y);
    1236 probe.at<cv::Vec2f>(0, 1) = field.at(x + 1, y);
    1237 probe.at<cv::Vec2f>(0, 2) = field.at(x, y + 1);
    1238 const int row = static_cast<int>(y);
    1239 const auto centre = pose.ground(probe, row, height, 0);
    1240 const cv::Mat& east = centre.first;
    1241 const cv::Mat& north = centre.second;
    1242 const double gsd = std::min(
    1243 std::hypot(east.at<float>(0, 1) - east.at<float>(0, 0),
    1244 north.at<float>(0, 1) - north.at<float>(0, 0)),
    1245 std::hypot(east.at<float>(0, 2) - east.at<float>(0, 0),
    1246 north.at<float>(0, 2) - north.at<float>(0, 0)));
    1247 require(std::isfinite(gsd) && gsd > 0, "invalid ground sampling distance");
    1248 const cv::Mat ray = probe.colRange(0, 1);
    1249 auto previous = pose.ground(ray, row, height, -0.5 * pose.exposure);
    1250 double path = 0;
    1251 for (int step = 1; step <= steps; ++step) {
    1252 auto next = pose.ground(ray, row, height,
    1253 (static_cast<double>(step) / steps - 0.5) * pose.exposure);
    1254 path += std::hypot(next.first.at<float>(0, 0) - previous.first.at<float>(0, 0),
    1255 next.second.at<float>(0, 0) - previous.second.at<float>(0, 0));
    1256 previous = std::move(next);
    1257 }
    1258 blur = std::max(blur, path / gsd);
    1259 }
    1260 }
    1261 int samples = static_cast<int>(
    1262 std::min<double>(maximum, std::ceil(blur / temporal_sample_spacing_px))) + 1;
    1263 if (samples % 2 == 0) ++samples;
    1264 return {std::clamp(samples, 1, maximum), blur};
    1265}

    1267/** @brief Linear-light to 8-bit sRGB lookup, built once.
    1268 * @details 65,536 linear levels keep every round-trip of an 8-bit sRGB code
    1269 * exact while replacing a per-pixel pow() call. */
    1270const std::array<unsigned char, 65536>& srgb_encoding() {
    1271 static const std::array<unsigned char, 65536> table = [] {
    1272 auto encode_one = [](size_t index) -> unsigned char {
    1273 const double linear = index / 65535.0;
    1274 const double encoded = linear <= 0.0031308
    1275 ? 12.92 * linear
    1276 : 1.055 * std::pow(linear, 1 / 2.4) - 0.055;
    1277 return cv::saturate_cast<unsigned char>(encoded * 255);
    1278 };
    1279 std::array<unsigned char, 65536> values{};
    1280 size_t start = 0;
    1281 for (int code = 0; code < 255; ++code) {
    1282 const double midpoint = (code + 0.5) / 255.0;
    1283 const double approx_linear =
    1284 midpoint <= 0.04045 ? midpoint / 12.92
    1285 : std::pow((midpoint + 0.055) / 1.055, 2.4);
    1286 size_t boundary = std::clamp<size_t>(
    1287 static_cast<size_t>(approx_linear * 65535.0), start, values.size());
    1288 while (boundary > start && encode_one(boundary - 1) > code) --boundary;
    1289 while (boundary < values.size() && encode_one(boundary) <= code)
    1290 ++boundary;
    1291 for (size_t index = start; index < boundary; ++index)
    1292 values[index] = static_cast<unsigned char>(code);
    1293 start = boundary;
    1294 }
    1295 for (size_t index = start; index < values.size(); ++index)
    1296 values[index] = 255;
    1297 return values;
    1298 }();
    1299 return table;
    1300}

    1310/** @brief Parallel separable Gaussian blur on a contiguous CV_32FC3 frame. */
    1311CAMERA_LENS_SIMD void blur_linear_rgb(cv::Mat& frame, double sigma) {
    1312 const int radius = std::max(1, static_cast<int>(std::ceil(sigma * 3.0)));
    1313 const int kernel_size = 2 * radius + 1;
    1314 std::vector<float> weights(kernel_size);
    1315 float weight_sum = 0;
    1316 for (int index = -radius; index <= radius; ++index) {
    1317 weights[index + radius] = std::exp(
    1318 -0.5f * static_cast<float>(index * index) /
    1319 static_cast<float>(sigma * sigma));
    1320 weight_sum += weights[index + radius];
    1321 }
    1322 for (float& weight : weights) weight /= weight_sum;
    1323 const int width = frame.cols, height = frame.rows;
    1324 cv::Mat horizontal(height, width, CV_32FC3);
    1325 const float* kernel = weights.data();
    1326 const int left_limit = std::min(radius, width),
    1327 right_start = std::max(left_limit, width - radius);
    1328 cv::parallel_for_(cv::Range(0, height), [&](const cv::Range& range) {
    1329 for (int row = range.start; row < range.end; ++row) {

    1331 float* destination = horizontal.ptr<float>(row);
    1332 for (int column = 0; column < left_limit; ++column) {
    1333 float red = 0, green = 0, blue = 0;
    1334 for (int tap = -radius; tap <= radius; ++tap) {
    1335 const int sample = reflect_101(column + tap, width) * 3;
    1336 const float weight = kernel[tap + radius];
    1337 red += source[sample] * weight;
    1338 green += source[sample + 1] * weight;
    1339 blue += source[sample + 2] * weight;
    1340 }
    1341 destination[column * 3] = red;
    1342 destination[column * 3 + 1] = green;
    1343 destination[column * 3 + 2] = blue;
    1344 }
    1345 for (int column = left_limit; column < right_start; ++column) {
    1346 float red = 0, green = 0, blue = 0;
    1347 const float* window = source + (column - radius) * 3;
    1348 for (int tap = 0; tap < kernel_size; ++tap) {
    1349 const float weight = kernel[tap];
    1350 red += window[tap * 3] * weight;
    1351 green += window[tap * 3 + 1] * weight;
    1352 blue += window[tap * 3 + 2] * weight;
    1353 }
    1354 destination[column * 3] = red;
    1355 destination[column * 3 + 1] = green;
    1356 destination[column * 3 + 2] = blue;
    1357 }
    1358 for (int column = right_start; column < width; ++column) {
    1359 float red = 0, green = 0, blue = 0;
    1360 for (int tap = -radius; tap <= radius; ++tap) {
    1361 const int sample = reflect_101(column + tap, width) * 3;
    1362 const float weight = kernel[tap + radius];
    1363 red += source[sample] * weight;
    1364 green += source[sample + 1] * weight;
    1365 blue += source[sample + 2] * weight;
    1366 }
    1367 destination[column * 3] = red;
    1368 destination[column * 3 + 1] = green;
    1369 destination[column * 3 + 2] = blue;
    1370 }
    1371 }
    1372 });
    1373 cv::parallel_for_(cv::Range(0, height), [&](const cv::Range& range) {
    1374 for (int row = range.start; row < range.end; ++row) {
    1375 float* destination = frame.ptr<float>(row);
    1376 const int values = width * 3;
    1377 for (int index = 0; index < values; ++index) destination[index] = 0;
    1378 for (int tap = -radius; tap <= radius; ++tap) {
    1379 const float weight = kernel[tap + radius];

    1381 horizontal.ptr<float>(reflect_101(row + tap, height));

    1383 destination[index] += source[index] * weight;

    1386 });

    1389/** @brief Stateless zero-mean, unit-variance Irwin-Hall normal sample.
    1390 * @details Keyed directly by (row_seed, channel_index) so noise is strictly
    1391 * deterministic, unbiased, and bit-identical across any thread count. */
    1392inline float stateless_normal(uint64_t row_seed, int index) {
    1393 uint64_t state =
    1394 row_seed + static_cast<uint64_t>(index) * 0x9e3779b97f4a7c15ULL;
    1395 state = (state ^ (state >> 30)) * 0xbf58476d1ce4e5b9ULL;
    1396 state = (state ^ (state >> 27)) * 0x94d049bb133111ebULL;
    1397 const uint64_t first = state ^ (state >> 31);
    1398 uint64_t second = state * 0x2545f4914f6cdd1dULL;
    1399 second ^= second >> 29;
    1400 const uint32_t sum =
    1401 static_cast<uint32_t>(first & 1023ULL) +
    1402 static_cast<uint32_t>((first >> 10) & 1023ULL) +
    1403 static_cast<uint32_t>((first >> 20) & 1023ULL) +
    1404 static_cast<uint32_t>((first >> 30) & 1023ULL) +
    1405 static_cast<uint32_t>((first >> 40) & 1023ULL) +
    1406 static_cast<uint32_t>((first >> 50) & 1023ULL) +
    1407 static_cast<uint32_t>(second & 1023ULL) +
    1408 static_cast<uint32_t>((second >> 10) & 1023ULL) +
    1409 static_cast<uint32_t>((second >> 20) & 1023ULL) +
    1410 static_cast<uint32_t>((second >> 30) & 1023ULL) +
    1411 static_cast<uint32_t>((second >> 40) & 1023ULL) +
    1412 static_cast<uint32_t>((second >> 50) & 1023ULL);
    1413 return (static_cast<int>(sum) - 6138) * (1.0f / 1024.0f);
    1414}

    1450 * while leaving YCrCb chroma differences (R - Y, B - Y) unchanged, avoiding
    1451 * full-frame colour-space splits and merges. */
    1452CAMERA_LENS_SIMD void sharpen_luma_rgb(cv::Mat& output, double sharpen) {
    1453 const int width = output.cols, height = output.rows;
    1454 cv::Mat luma(height, width, CV_8U), horizontal(height, width, CV_16U);
    1455 static constexpr int weights[7] = {5, 55, 248, 408, 248, 55, 5};
    1456 const int left_limit = std::min(3, width),
    1457 right_start = std::max(left_limit, width - 3);
    1458 cv::parallel_for_(cv::Range(0, height), [&](const cv::Range& range) {
    1459 for (int row = range.start; row < range.end; ++row) {
    1460 const unsigned char* rgb = output.ptr(row);
    1461 unsigned char* y = luma.ptr(row);
    1462 for (int column = 0; column < width; ++column)
    1463 y[column] = static_cast<unsigned char>(
    1464 (77 * rgb[column * 3] + 150 * rgb[column * 3 + 1] +
    1465 29 * rgb[column * 3 + 2] + 128) >>
    1466 8);
    1467 uint16_t* filtered = horizontal.ptr<uint16_t>(row);
    1468 for (int column = 0; column < left_limit; ++column) {
    1469 int sum = 0;
    1470 for (int tap = -3; tap <= 3; ++tap)
    1471 sum += y[reflect_101(column + tap, width)] * weights[tap + 3];
    1472 filtered[column] = static_cast<uint16_t>((sum + 16) >> 5);
    1473 }
    1474 for (int column = left_limit; column < right_start; ++column) {
    1475 const int sum = y[column - 3] * 5 + y[column - 2] * 55 +
    1476 y[column - 1] * 248 + y[column] * 408 +
    1477 y[column + 1] * 248 + y[column + 2] * 55 +
    1478 y[column + 3] * 5;
    1479 filtered[column] = static_cast<uint16_t>((sum + 16) >> 5);
    1480 }
    1481 for (int column = right_start; column < width; ++column) {
    1482 int sum = 0;
    1483 for (int tap = -3; tap <= 3; ++tap)
    1484 sum += y[reflect_101(column + tap, width)] * weights[tap + 3];
    1485 filtered[column] = static_cast<uint16_t>((sum + 16) >> 5);
    1486 }
    1487 }
    1488 });
    1489 const int amount_q8 = static_cast<int>(std::nearbyint(sharpen * 256.0));
    1490 cv::parallel_for_(cv::Range(0, height), [&](const cv::Range& range) {
    1491 for (int row = range.start; row < range.end; ++row) {
    1492 const uint16_t* taps[7];
    1493 for (int tap = -3; tap <= 3; ++tap)
    1494 taps[tap + 3] = horizontal.ptr<uint16_t>(reflect_101(row + tap, height));
    1495 const unsigned char* y = luma.ptr(row);
    1496 unsigned char* rgb = output.ptr(row);
    1497 for (int column = 0; column < width; ++column) {
    1498 const int blurred_q5 =
    1499 (taps[0][column] * 5 + taps[1][column] * 55 +
    1500 taps[2][column] * 248 + taps[3][column] * 408 +
    1501 taps[4][column] * 248 + taps[5][column] * 55 +
    1502 taps[6][column] * 5 + 512) >>
    1503 10;
    1504 const int delta =
    1505 (((static_cast<int>(y[column]) << 5) - blurred_q5) * amount_q8 +
    1506 4096) >>
    1507 13;
    1508 rgb[column * 3] = cv::saturate_cast<unsigned char>(rgb[column * 3] + delta);
    1509 rgb[column * 3 + 1] =
    1510 cv::saturate_cast<unsigned char>(rgb[column * 3 + 1] + delta);
    1511 rgb[column * 3 + 2] =
    1512 cv::saturate_cast<unsigned char>(rgb[column * 3 + 2] + delta);
    1513 }
    1514 }
    1515 });
    1516}

    1579/** @brief Bilinearly sample one channel from a 2D pointer grid of 256x256
    1580 * linear float32 RGB tiles without assembling a strip mosaic. */
    1581inline float sample_tile_channel(const float* const* tile_ptrs, int grid_width,
    1582 int max_x, int max_y, float map_x, float map_y,
    1583 int channel) {
    1584 const int sx = std::clamp(static_cast<int>(std::floor(map_x * 32.0f + 0.5f)),
    1585 0, max_x * 32);
    1586 const int sy = std::clamp(static_cast<int>(std::floor(map_y * 32.0f + 0.5f)),
    1587 0, max_y * 32);
    1588 const int ix = sx >> 5, iy = sy >> 5;
    1589 const float fx = static_cast<float>(sx & 31) * (1.0f / 32.0f),
    1590 fy = static_cast<float>(sy & 31) * (1.0f / 32.0f);
    1591 const int tx = ix >> 8, ty = iy >> 8, lx = ix & 255, ly = iy & 255;
    1592 if (lx < 255 && ly < 255) {
    1593 const float* top =
    1594 tile_ptrs[ty * grid_width + tx] + (ly * 256 + lx) * 3 + channel;
    1595 const float* bottom = top + 256 * 3;
    1596 const float upper = top[0] + fx * (top[3] - top[0]);
    1597 const float lower = bottom[0] + fx * (bottom[3] - bottom[0]);
    1598 return upper + fy * (lower - upper);
    1599 }
    1600 const int ix1 = std::min(ix + 1, max_x), iy1 = std::min(iy + 1, max_y);
    1601 const float v00 = tile_ptrs[(iy >> 8) * grid_width + (ix >> 8)]
    1602 [((iy & 255) * 256 + (ix & 255)) * 3 + channel];
    1603 const float v01 = tile_ptrs[(iy >> 8) * grid_width + (ix1 >> 8)]
    1604 [((iy & 255) * 256 + (ix1 & 255)) * 3 + channel];
    1605 const float v10 = tile_ptrs[(iy1 >> 8) * grid_width + (ix >> 8)]
    1606 [((iy1 & 255) * 256 + (ix & 255)) * 3 + channel];
    1607 const float v11 = tile_ptrs[(iy1 >> 8) * grid_width + (ix1 >> 8)]
    1608 [((iy1 & 255) * 256 + (ix1 & 255)) * 3 + channel];
    1609 const float upper = v00 + fx * (v01 - v00);
    1610 const float lower = v10 + fx * (v11 - v10);
    1611 return upper + fy * (lower - upper);
    1612}

    1614/** @brief Bilinearly sample all 3 RGB channels at identical coordinates from a
    1615 * 2D pointer grid of 256x256 linear float32 tiles. */
    1616inline void sample_tile_rgb(const float* const* tile_ptrs, int grid_width,
    1617 int max_x, int max_y, float map_x, float map_y,
    1618 float& red, float& green, float& blue) {
    1619 const int sx = std::clamp(static_cast<int>(std::floor(map_x * 32.0f + 0.5f)),
    1620 0, max_x * 32);
    1621 const int sy = std::clamp(static_cast<int>(std::floor(map_y * 32.0f + 0.5f)),
    1622 0, max_y * 32);
    1623 const int ix = sx >> 5, iy = sy >> 5;
    1624 const float fx = static_cast<float>(sx & 31) * (1.0f / 32.0f),
    1625 fy = static_cast<float>(sy & 31) * (1.0f / 32.0f);
    1626 const int tx = ix >> 8, ty = iy >> 8, lx = ix & 255, ly = iy & 255;
    1627 if (lx < 255 && ly < 255) {
    1628 const float* top = tile_ptrs[ty * grid_width + tx] + (ly * 256 + lx) * 3;
    1629 const float* bottom = top + 256 * 3;
    1630 red += (top[0] + fx * (top[3] - top[0])) +
    1631 fy * ((bottom[0] + fx * (bottom[3] - bottom[0])) -
    1632 (top[0] + fx * (top[3] - top[0])));
    1633 green += (top[1] + fx * (top[4] - top[1])) +
    1634 fy * ((bottom[1] + fx * (bottom[4] - bottom[1])) -
    1635 (top[1] + fx * (top[4] - top[1])));
    1636 blue += (top[2] + fx * (top[5] - top[2])) +
    1637 fy * ((bottom[2] + fx * (bottom[5] - bottom[2])) -
    1638 (top[2] + fx * (top[5] - top[2])));
    1639 return;
    1640 }
    1641 const int ix1 = std::min(ix + 1, max_x), iy1 = std::min(iy + 1, max_y);
    1642 const float* p00 =
    1643 tile_ptrs[(iy >> 8) * grid_width + (ix >> 8)] +
    1644 ((iy & 255) * 256 + (ix & 255)) * 3;
    1645 const float* p01 =
    1646 tile_ptrs[(iy >> 8) * grid_width + (ix1 >> 8)] +
    1647 ((iy & 255) * 256 + (ix1 & 255)) * 3;
    1648 const float* p10 =
    1649 tile_ptrs[(iy1 >> 8) * grid_width + (ix >> 8)] +
    1650 ((iy1 & 255) * 256 + (ix & 255)) * 3;
    1651 const float* p11 =
    1652 tile_ptrs[(iy1 >> 8) * grid_width + (ix1 >> 8)] +
    1653 ((iy1 & 255) * 256 + (ix1 & 255)) * 3;
    1654 red += (p00[0] + fx * (p01[0] - p00[0])) +
    1655 fy * ((p10[0] + fx * (p11[0] - p10[0])) -
    1656 (p00[0] + fx * (p01[0] - p00[0])));
    1657 green += (p00[1] + fx * (p01[1] - p00[1])) +
    1658 fy * ((p10[1] + fx * (p11[1] - p10[1])) -
    1659 (p00[1] + fx * (p01[1] - p00[1])));
    1660 blue += (p00[2] + fx * (p01[2] - p00[2])) +
    1661 fy * ((p10[2] + fx * (p11[2] - p10[2])) -
    1662 (p00[2] + fx * (p01[2] - p00[2])));
    1663}

    1665/** @brief Bilinearly sample a linear CV_32FC3 pinhole source image matching
    1666 * OpenCV INTER_LINEAR + BORDER_CONSTANT(0). */
    1667inline float sample_pinhole_channel(const cv::Mat& source, float map_x,
    1668 float map_y, int channel) {
    1669 const int sx = static_cast<int>(std::floor(map_x * 32.0f + 0.5f)),
    1670 sy = static_cast<int>(std::floor(map_y * 32.0f + 0.5f));
    1671 const int ix = sx >> 5, iy = sy >> 5;
    1672 const float fx = static_cast<float>(sx & 31) * (1.0f / 32.0f),
    1673 fy = static_cast<float>(sy & 31) * (1.0f / 32.0f);
    1674 if (ix >= 0 && ix + 1 < source.cols && iy >= 0 && iy + 1 < source.rows) {
    1675 const float* top = source.ptr<float>(iy) + ix * 3 + channel;
    1676 const float* bottom = source.ptr<float>(iy + 1) + ix * 3 + channel;
    1677 const float upper = top[0] + fx * (top[3] - top[0]);
    1678 const float lower = bottom[0] + fx * (bottom[3] - bottom[0]);
    1679 return upper + fy * (lower - upper);
    1680 }
    1681 auto fetch = [&](int x, int y) -> float {
    1682 if (x < 0 || x >= source.cols || y < 0 || y >= source.rows) return 0.0f;
    1683 return source.ptr<float>(y)[x * 3 + channel];
    1684 };
    1685 const float v00 = fetch(ix, iy),
    1686 v01 = fx > 0.0f ? fetch(ix + 1, iy) : 0.0f,
    1687 v10 = fy > 0.0f ? fetch(ix, iy + 1) : 0.0f,
    1688 v11 = (fx > 0.0f && fy > 0.0f) ? fetch(ix + 1, iy + 1) : 0.0f;
    1689 const float upper = v00 + fx * (v01 - v00);
    1690 const float lower = v10 + fx * (v11 - v10);
    1691 return upper + fy * (lower - upper);
    1692}

    1694/** @brief SIMD-friendly strip ground-bounds and horizon check. */
    1695CAMERA_LENS_SIMD void compute_strip_bounds_rows(
    1696 const std::vector<std::unique_ptr<RayField>>& fields, int channels,
    1697 const RowSamplePose* poses, int samples, int first, int start_row,
    1698 int end_row, double* min_e, double* max_e, double* min_n, double* max_n,
    1699 std::atomic<bool>& horizon, std::atomic<bool>& non_finite) {
    1700 int max_nodes = 0;
    1701 for (int channel = 0; channel < channels; ++channel)
    1702 max_nodes = std::max(max_nodes, fields[channel]->node_columns());
    1703 std::vector<float> rx_row(max_nodes), ry_row(max_nodes);
    1704 for (int r = start_row; r < end_row; ++r) {
    1705 if (horizon.load(std::memory_order_relaxed) ||
    1706 non_finite.load(std::memory_order_relaxed))
    1707 return;
    1708 const int full_row = first + r;
    1709 const RowSamplePose* row_poses =
    1710 poses + static_cast<size_t>(full_row) * samples;
    1711 float re_min = INFINITY, re_max = -INFINITY, rn_min = INFINITY,
    1712 rn_max = -INFINITY;
    1713 for (int channel = 0; channel < channels; ++channel) {
    1714 const int count = fields[channel]->node_columns();
    1715 fields[channel]->row_nodes_into(full_row, rx_row.data(), ry_row.data());
    1716 const float* rx = rx_row.data();
    1717 const float* ry = ry_row.data();
    1718 for (int sample = 0; sample < samples; ++sample) {
    1719 const RowSamplePose& p = row_poses[sample];
    1720 float min_down = INFINITY;
    1721 for (int column = 0; column < count; ++column)
    1722 min_down =
    1723 std::min(min_down, p.down_x * rx[column] + p.down_y * ry[column] +
    1724 p.down_z);
    1725 if (!(min_down > 1e-4f)) {
    1726 horizon.store(true, std::memory_order_relaxed);
    1727 return;
    1728 }
    1729 for (int column = 0; column < count; ++column) {
    1730 const float scale =
    1731 1.0f / (p.down_x * rx[column] + p.down_y * ry[column] + p.down_z);
    1732 const float east =
    1733 (p.east_x * rx[column] + p.east_y * ry[column] + p.east_z) *
    1734 scale +
    1735 p.translation_east;
    1736 const float north =
    1737 (p.north_x * rx[column] + p.north_y * ry[column] + p.north_z) *
    1738 scale +
    1739 p.translation_north;
    1740 re_min = east < re_min ? east : re_min;
    1741 re_max = east > re_max ? east : re_max;
    1742 rn_min = north < rn_min ? north : rn_min;
    1743 rn_max = north > rn_max ? north : rn_max;
    1744 }
    1745 }
    1746 }
    1747 if (!std::isfinite(re_min) || !std::isfinite(re_max) ||
    1748 !std::isfinite(rn_min) || !std::isfinite(rn_max)) {
    1749 non_finite.store(true, std::memory_order_relaxed);
    1750 return;
    1751 }
    1752 min_e[r] = re_min;
    1753 max_e[r] = re_max;
    1754 min_n[r] = rn_min;
    1755 max_n[r] = rn_max;
    1756 }
    1757}

    1819 (p.east_x * rx + p.east_y * ry + p.east_z) * scale +

    1821 const float north =
    1822 (p.north_x * rx + p.north_y * ry + p.north_z) * scale +
