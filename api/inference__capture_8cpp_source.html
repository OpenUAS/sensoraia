    48/** @brief Copy only model-sized input while retaining the original frame's timing identity. */
    49bool prepare_pixels(const imx500::FrameView &view, const std::array<const uint8_t *, 3> &planes,
    50 unsigned stride, void *data)
    51{
    52 auto &preparation = *static_cast<Preparation *>(data);
    53 inference_image image{};
    54 image.width = view.image.width;
    55 image.height = view.image.height;
    56 image.format = INFERENCE_YUV420;
    57 for (size_t plane = 0; plane < 3; ++plane) {
    58 image.planes[plane] = planes[plane];
    59 image.strides[plane] = plane ? stride / 2 : stride;
    60 image.lengths[plane] = image.strides[plane] * (plane ? image.height / 2 : image.height);
    61 }
    62 preparation.context->frame_sequence = view.sequence;
    63 preparation.context->sensor_timestamp_ns = view.sensor_timestamp_ns;
    64 preparation.context->capture_identity = true;
    65 return inference_context_prepare_image(preparation.context, &image, preparation.error, preparation.capacity);
    66}

    71/** @brief Preserve fake-camera JPEG bytes without reencoding, refusing overwrite and changing input files. */
    72bool copy_picture(const char *source, const char *destination)
    73{
    74 const int input = open(source, O_RDONLY | O_CLOEXEC | O_NONBLOCK);
    75 if (input < 0) return false;
    76 struct stat before{}, after{};
    77 if (fstat(input, &before) || !S_ISREG(before.st_mode) || before.st_size <= 0 || before.st_size > 128 * 1024 * 1024) {
    78 close(input);
    79 return false;
    80 }
    81 const int output = open(destination, O_WRONLY | O_CREAT | O_EXCL | O_CLOEXEC, 0600);
    82 if (output < 0) { close(input); return false; }
    83 unsigned char buffer[16384];
    84 bool valid = true;
    85 off_t copied = 0;
    86 while (valid) {
    87 const ssize_t count = read(input, buffer, sizeof(buffer));
    88 if (count < 0 && errno == EINTR) continue;
    89 if (count < 0) valid = false;
    90 if (count <= 0) break;
    91 if (copied > before.st_size - count) { valid = false; break; }
    92 copied += count;
    93 size_t written = 0;
    94 while (written < static_cast<size_t>(count)) {
    95 const ssize_t bytes = write(output, buffer + written, static_cast<size_t>(count) - written);
    96 if (bytes < 0 && errno == EINTR) continue;
    97 if (bytes <= 0) { valid = false; break; }
    98 written += static_cast<size_t>(bytes);
    99 }
    100 }
    101 valid = valid && copied == before.st_size && !fstat(input, &after) && before.st_size == after.st_size &&
    102 before.st_mtim.tv_sec == after.st_mtim.tv_sec && before.st_mtim.tv_nsec == after.st_mtim.tv_nsec && !fsync(output);
    103 if (close(input)) valid = false;
    104 if (close(output)) valid = false;
    105 if (!valid) unlink(destination);
    106 return valid;
    107}
    108}

    133 if (settings.backend == INFERENCE_IMX500) {
    134 if (settings.source != INFERENCE_LIVE || !settings.model[0]) return 2;
    135 char *arguments[] = {argv[0], argv[1], settings.model,
    136 capture ? argv[4] : serve ? argv[4] : nullptr, serve ? argv[5] : nullptr, nullptr};

    138 }
    139 if (settings.backend != INFERENCE_ONNX) return 2;

    141 if (serve && (!ini_uint(argv[4], &width) || !ini_uint(argv[5], &height))) return 2;
    142 if (width > std::numeric_limits<unsigned>::max() || height > std::numeric_limits<unsigned>::max() ||

    144 try {
    145 ContextOwner owner;
    146 auto &context = owner.context;

    151 if (settings.source == INFERENCE_LIVE) {
    152 camera = std::make_unique<imx500::Camera>();
    153 camera->open(nullptr, static_cast<unsigned>(width), static_cast<unsigned>(height));
    154 {
    155 auto frame = camera->capture();
    156 if (!camera->with_pixels(frame, prepare_pixels, &preparation)) throw std::runtime_error(error);
    157 }
    158 } else if (!settings.input[0] || !inference_jpeg_prepare(&context, settings.input, error, sizeof(error))) {
    159 throw std::runtime_error(settings.input[0] ? error : "Picture source requires input in settings");
    160 }
    161 if (!inference_context_run(&context, error, sizeof(error))) throw std::runtime_error(error);
    162 if (probe) { std::puts("camera and ONNX model ready"); return 0; }
    163 const auto shoot = [&](const char *filename, uint32_t sequence) {
    164 struct stat existing{};
    165 if (!filename || lstat(filename, &existing) == 0 || errno != ENOENT) return false;
    166 const std::string staged = std::string(filename) + ".onnx.pending.jpg";
    167 if (staged.size() >= INFERENCE_PATH_CAPACITY) return false;
    168 bool saved = false, prepared = false;
    169 if (camera) {
    170 auto frame = camera->capture();
    171 prepared = camera->with_pixels(frame, prepare_pixels, &preparation);
    172 const imx500::FrameProcessor photograph{nullptr, nullptr, nullptr, retain_pixels};
    173 saved = camera->save(frame, staged.c_str(), photograph);
    174 } else {
    175 saved = copy_picture(settings.input, staged.c_str());
    176 prepared = saved && inference_jpeg_prepare(&context, staged.c_str(), error, sizeof(error));
    177 }
    178 if (!saved) return false;
    179 /* The staged image is still exclusively producer-owned, so LOOM may embed feature evidence here. */
    180 context.private_image = true;
    181 const bool analyzed = prepared && inference_context_run(&context, error, sizeof(error)) &&
    182 processor.write(processor.context, &context, staged.c_str(), sequence, error, sizeof(error));
    183 if (!analyzed)
    184 std::fprintf(stderr, "%s: analysis_failed: %s\n", processor.application, error);
    185 std::string staged_result, published_result;
    186 bool result_published = false;
    187 if (analyzed && processor.result_suffix) {
    188 staged_result = staged.substr(0, staged.size() - 4) + processor.result_suffix;
    189 const size_t length = std::strlen(filename);
    190 if (length < 4 || std::strcmp(filename + length - 4, ".jpg")) {
    191 unlink(staged_result.c_str());
    192 unlink(staged.c_str());
    193 return false;
    194 }
    195 published_result = std::string(filename, length - 4) + processor.result_suffix;
    196 result_published = link(staged_result.c_str(), published_result.c_str()) == 0;
    197 if (!result_published) {
    198 unlink(staged_result.c_str());
    199 unlink(staged.c_str());
    200 return false;
    201 }
    202 }
    203 const bool published = link(staged.c_str(), filename) == 0;
    204 /* Remove only names created by this attempt; never erase a pre-existing image or analysis result. */
    205 if (!published && result_published) unlink(published_result.c_str());
    206 if (!staged_result.empty()) unlink(staged_result.c_str());
    207 unlink(staged.c_str());
    208 if (!published) return false;
    209 return true;
    210 };
    211 if (capture) return shoot(argv[4], 1) ? 0 : 1;
    212 std::puts("READY");
    213 if (std::fflush(stdout)) return 1;
    214 char command[INFERENCE_PATH_CAPACITY + 16];
    215 uint32_t sequence = 0;
    216 while (std::fgets(command, sizeof(command), stdin)) {
    217 const size_t length = std::strlen(command);
    218 if (!length || command[length - 1] != '\n' || std::strncmp(command, "SHOT ", 5) || sequence == UINT32_MAX) return 2;
    219 command[length - 1] = '\0';
    220 if (command[5] != '/' || std::strchr(command + 5, '\r')) return 2;
    221 std::puts(shoot(command + 5, ++sequence) ? "OK" : "ERROR");
    222 if (std::fflush(stdout)) return 1;
    223 }
    224 return std::ferror(stdin) ? 1 : 0;
    225 } catch (const std::exception &failure) {
    226 std::fprintf(stderr, "%s: %s\n", processor.application, failure.what());
    227 return 1;
    228 }
    229}
