    18/**
    19 * @file photo_exif.h
    20 * @brief Encode simulated-camera pose, optics, and uncertainty into EXIF.
    21 * @details Metadata distinguishes commanded pose from observed simulator pose,
    22 * and labels nominal/emulated camera settings rather than presenting them as
    23 * physical measurements. This is important because a complete EXIF record is
    24 * provenance, not proof of exposure synchronization or surveyed accuracy.

    49/** @brief Photo location with AMSL camera and ground heights in metres. */
    50struct PhotoPosition {
    51 double latitude, longitude, altitudeAmsl, groundAmsl;
    52 /** @brief Compute height above ground from the shared AMSL datum.
    53 * @return Camera-minus-ground height in metres. */
    54 double agl() const { return altitudeAmsl - groundAmsl; }
    55};

    57/**
    58 * @brief Camera and timing provenance required for a simulated-photo EXIF record.
    59 * @details Dimensions are pixels, timestamps are 100 ns monotonic counts, angles
    60 * are degrees except `fovRadians`, and the optional map records projected metres.
    61 */
    62struct PhotoCamera {
    63 double fovRadians = 0;
    64 double commandedHeading = 0;
    65 double pitch = 0, bank = 0, heading = 0;
    66 double observedLatitude = 0, observedLongitude = 0, observedAltitude = 0;
    67 long long observed100ns = 0;
    68 int sourceWidth = 0, sourceHeight = 0;
    69 double cropX = 0, cropY = 0, cropWidth = 0, cropHeight = 0;
    70 bool narrowAngle = false;
    71 bool piAiCamera = false;
    72 LensEffects lens;
    73 CameraExifSettings exifSettings;
    74 bool fixedStartAltitude = false;
    75 double startGroundMsl = 0;
    76 long long frame100ns = 0;
    77 std::optional<UtcDateTime> captureUtc;
    78 double utcAbsoluteSeconds = 0;
    79 long long utcRead100ns = 0;
    80 std::optional<MapTile> map;
    81 double imageRotationDegrees = 0, rotationBoundDegrees = 0;
    82 std::optional<unsigned long long> photoNumber;
    83};

    85/**
    86 * @brief Build bounded EXIF payloads for simulator-generated source photos.
    87 * @details The output buffer is owned by this object and is replaced by each
    88 * call to @ref build. Callers must consume or copy the payload before building
    89 * another record with the same instance.
    90 */
    91class PhotoExif {
    92protected:
    93 exif_document document_{};
    94 std::array<unsigned char, EXIF_MAX_PAYLOAD> payload_{};
    95
    96 /** @brief Convert a C EXIF status into the mapper's error path.
    97 * @param result Zero on success, nonzero on failure.
    98 * @param operation Context for the error.
    99 * @throws std::runtime_error With the current errno diagnostic on failure. */
    100 static void check(int result, const char* operation) {
    101 if (result != 0)
    102 throw std::runtime_error(std::string(operation) + ": " +
    103 std::generic_category().message(errno));
    104 }
    105 /** @brief Allocate one field in the bounded EXIF document.
    106 * @param directory Target EXIF directory.
    107 * @param tag Numeric field identifier.
    108 * @param format Element encoding.
    109 * @param count Number of elements, not the byte count.
    110 * @return Borrowed writable field storage.
    111 * @throws std::runtime_error If the document cannot reserve the field. */
    112 unsigned char* reserve(exif_directory directory, std::uint16_t tag,
    113 exif_format format, std::uint32_t count) {
    114 auto* value = exif_document_reserve(&document_, directory, tag, format, count);
    115 if (!value) check(-1, "Reserve photo EXIF field");
    116 return value;
    117 }
    118 /** @brief Encode a validated geographic coordinate as EXIF GPS rationals.
    119 * @param degrees Signed decimal degrees.
    120 * @param referenceTag Hemisphere tag ID.
    121 * @param valueTag Degrees/minutes/seconds tag ID.
    122 * @param positive Hemisphere for a nonnegative coordinate.
    123 * @param negative Hemisphere for a negative coordinate.
    124 * @throws std::runtime_error If either tag cannot be stored.
    125 * @details Round the total once so overflow from the fractional second
    126 * carries naturally into minutes and degrees. */
    127 void coordinate(double degrees, std::uint16_t referenceTag, std::uint16_t valueTag,
    128 char positive, char negative) {
    129 const char reference[] = {degrees < 0 ? negative : positive, '\0'};
    130 check(exif_document_ascii(&document_, EXIF_GPS, referenceTag, reference), "Set EXIF GPS hemisphere");
    131 // Round once in integer microseconds, carrying 60-second/minute boundaries.
    132 const auto total = static_cast<std::uint64_t>(std::llround(std::abs(degrees) * 3600 * 1000000));
    133 auto* value = reserve(EXIF_GPS, valueTag, EXIF_RATIONAL, 3);
    134 exif_encode_long(value, static_cast<std::uint32_t>(total / 3600000000ULL));
    135 exif_encode_long(value + 4, 1);
    136 exif_encode_long(value + 8, static_cast<std::uint32_t>(total / 60000000ULL % 60));
    137 exif_encode_long(value + 12, 1);
    138 exif_encode_long(value + 16, static_cast<std::uint32_t>(total % 60000000ULL));
    139 exif_encode_long(value + 20, 1000000);
    140 }
    141 /** @brief Write simulator-observed UTC into consistent EXIF and GPS fields.
    142 * @param camera Capture metadata with its optional UTC field engaged.
    143 * @throws std::runtime_error If the time sample is unqualified or a tag cannot be written.
    144 * @details Check frame/readback timing before publishing a wall-clock label.
    145 * Derive both tag families from the same UTC sample. */
    146 void captureTime(const PhotoCamera& camera) {
    147 const auto& utc = *camera.captureUtc;
    148 if (!captureUtcSampleValid(utc, camera.utcAbsoluteSeconds, camera.utcRead100ns, camera.frame100ns))
    149 throw std::runtime_error("Invalid verified capture UTC metadata");
    150 const auto date = utc.text.substr(0, 10);
    151 std::string exifDate = date;
    152 std::replace(exifDate.begin(), exifDate.end(), '-', ':');
    153 const auto timestamp = exifDate + " " + utc.text.substr(11, 5) + ":00";
    154 const auto milliseconds = static_cast<std::uint32_t>(
    155 std::floor((camera.utcAbsoluteSeconds - utc.absoluteSeconds()) * 1000));
    156 std::ostringstream subsecond;
    157 subsecond.imbue(std::locale::classic());
    158 subsecond << std::setfill('0') << std::setw(3) << milliseconds;
    159 check(exif_document_ascii(&document_, EXIF_PRIMARY, 0x0132, timestamp.c_str()), "Set EXIF DateTime");
    160 for (const auto tag : {0x9003, 0x9004})
    161 check(exif_document_ascii(&document_, EXIF_CAMERA, static_cast<std::uint16_t>(tag), timestamp.c_str()),
    162 "Set EXIF original/digitized time");
    163 for (const auto tag : {0x9010, 0x9011, 0x9012})
    164 check(exif_document_ascii(&document_, EXIF_CAMERA, static_cast<std::uint16_t>(tag), "+00:00"),
    165 "Set EXIF UTC offset");
    166 for (const auto tag : {0x9290, 0x9291, 0x9292})
    167 check(exif_document_ascii(&document_, EXIF_CAMERA, static_cast<std::uint16_t>(tag), subsecond.str().c_str()),
    168 "Set EXIF UTC subsecond");
    169 check(exif_document_ascii(&document_, EXIF_GPS, 0x001d, exifDate.c_str()), "Set GPS date");
    170 auto* time = reserve(EXIF_GPS, 0x0007, EXIF_RATIONAL, 3);
    171 exif_encode_long(time, static_cast<std::uint32_t>(utc.hour));
    172 exif_encode_long(time + 4, 1);
    173 exif_encode_long(time + 8, static_cast<std::uint32_t>(utc.minute));
    174 exif_encode_long(time + 12, 1);
    175 exif_encode_long(time + 16, milliseconds);
    176 exif_encode_long(time + 20, 1000);
    177 }
    178public:
    179 /** @brief Borrowed serialized EXIF bytes held in this object's workspace. */
    180 struct Payload {
    181 const unsigned char* data;
    182 std::size_t size;
    183 };
    184 /**
    185 * @brief Construct EXIF from validated photo position and optional camera evidence.
    186 * @param[in] position WGS-84 position with camera/ground AMSL heights in metres.
    187 * @param[in] width Encoded image width in pixels.
    188 * @param[in] height Encoded image height in pixels.
    189 * @param[in] camera Optional pose, optics, UTC-readback, and map provenance.
    190 * @return Serialized EXIF payload borrowed from this object.
    191 * @throws std::runtime_error If coordinates, optics, timestamps, or payload size
    192 * violate the supported metadata contract.
    193 */
    194 Payload build(const PhotoPosition& position, int width, int height,
    195 const std::optional<PhotoCamera>& camera = {}) {
    196 if (!std::isfinite(position.latitude) || std::abs(position.latitude) > 90 ||
    197 !std::isfinite(position.longitude) || std::abs(position.longitude) > 180 ||
    198 !std::isfinite(position.altitudeAmsl) || !std::isfinite(position.groundAmsl) ||
    199 !std::isfinite(position.agl()) ||
    200 std::abs(position.altitudeAmsl) >
    201 static_cast<double>(std::numeric_limits<std::uint32_t>::max()) / 1000 ||
    202 width < 1 || height < 1 || width > 16384 || height > 16384)
    203 throw std::runtime_error("Invalid position, heights or dimensions for photo EXIF");
    204 exif_document_init(&document_);
    205 check(exif_document_ascii(&document_, EXIF_PRIMARY, EXIF_SOFTWARE, "Tessavia / pangea_survey"),
    206 "Set EXIF software");
    207 check(exif_document_ascii(&document_, EXIF_PRIMARY, 0x010f, "Tessavia"),
    208 "Set virtual camera manufacturer");
    209 check(exif_document_ascii(&document_, EXIF_PRIMARY, 0x0110, "MSFS 2024 add-on camera"),
    210 "Set virtual camera model");
    211 auto* version = reserve(EXIF_GPS, EXIF_GPS_VERSION, EXIF_BYTE, 4);
    212 version[0] = 2;
    213 version[1] = 3;
    214 coordinate(position.latitude, EXIF_GPS_LATITUDE_REF, EXIF_GPS_LATITUDE, 'N', 'S');
    215 coordinate(position.longitude, EXIF_GPS_LONGITUDE_REF, EXIF_GPS_LONGITUDE, 'E', 'W');
    216 constexpr std::uint16_t gpsMapDatum = 0x0012;
    217 check(exif_document_ascii(&document_, EXIF_GPS, gpsMapDatum, "WGS-84"), "Set EXIF GPS datum");
    218 *reserve(EXIF_GPS, EXIF_GPS_ALTITUDE_REF, EXIF_BYTE, 1) = position.altitudeAmsl < 0 ? 1 : 0;
    219 const auto altitudeMm = static_cast<std::uint32_t>(std::llround(std::abs(position.altitudeAmsl) * 1000));
    220 check(exif_document_rational(&document_, EXIF_GPS, EXIF_GPS_ALTITUDE, altitudeMm, 1000),
    221 "Set EXIF AMSL altitude");
    222 exif_encode_long(reserve(EXIF_CAMERA, EXIF_PIXEL_X_DIMENSION, EXIF_LONG, 1),
    223 static_cast<std::uint32_t>(width));
    224 exif_encode_long(reserve(EXIF_CAMERA, EXIF_PIXEL_Y_DIMENSION, EXIF_LONG, 1),
    225 static_cast<std::uint32_t>(height));
    226
    227 std::ostringstream description, comment;
    228 description.imbue(std::locale::classic());
    229 comment.imbue(std::locale::classic());
    230 description << std::fixed << std::setprecision(6)
    231 << "Pangea simulated camera; AGL=" << position.agl()
    232 << (camera && camera->fixedStartAltitude ? " m (unverified estimate); AMSL=" : " m; AMSL=")
    233 << position.altitudeAmsl << " m";
    234 comment << std::fixed << std::setprecision(10)
    235 << "source=pangea_survey/MSFS2024; pose="
    236 << (camera && camera->piAiCamera ? "simconnect_observed_camera" : "simconnect_commanded_camera")
    237 << "; datum=WGS-84; latitude_deg=" << position.latitude
    238 << "; longitude_deg=" << position.longitude << std::setprecision(6)
    239 << "; agl_m=" << position.agl() << "; amsl_m=" << position.altitudeAmsl
    240 << "; ground_amsl_m=" << position.groundAmsl;
    241 if (camera && (camera->piAiCamera || camera->fixedStartAltitude)) {
    242 if (!std::isfinite(camera->startGroundMsl))
    243 throw std::runtime_error("Invalid initial ground reference in camera metadata");
    244 comment << "; agl_verified=" << (camera->fixedStartAltitude ? "false" : "true")
    245 << "; agl_basis=" << (camera->fixedStartAltitude ?
    246 "unverified_cached_sdk_ground_estimate" : "current_terrain_reference");
    247 if (camera->fixedStartAltitude)
    248 comment << "; start_ground_amsl_m=" << camera->startGroundMsl
    249 << "; reference_plane_height_m=" << position.altitudeAmsl - camera->startGroundMsl;
    250 }
    251 if (camera && camera->captureUtc) {
    252 captureTime(*camera);
    253 comment << "; capture_utc=" << camera->captureUtc->text
    254 << "; utc_absolute_seconds=" << std::setprecision(17) << camera->utcAbsoluteSeconds
    255 << "; utc_read_qpc_100ns=" << camera->utcRead100ns
    256 << "; utc_hold_method=repeated_simconnect_clock_resets"
    257 << "; utc_tolerance_seconds=1; utc_frame_synchronized=false"
    258 << "; exif_time_basis=verified_simulator_utc_readback_not_exposure_clock"
    259 << std::setprecision(6);
    260 }
    261 if (camera) {
    262 const auto& c = *camera;
    263 c.exifSettings.validate();
    264 const auto shutterUs = static_cast<std::uint32_t>(std::llround(c.exifSettings.shutterSeconds * 1000000));
    265 check(exif_document_rational(&document_, EXIF_CAMERA, EXIF_EXPOSURE_TIME, shutterUs, 1000000),
    266 "Set nominal simulated exposure time");
    267 exif_encode_short(reserve(EXIF_CAMERA, EXIF_ISO_SPEED, EXIF_SHORT, 1), c.exifSettings.iso);
    268 if (!c.exifSettings.hardwareIdentifier.empty())
    269 check(exif_document_ascii(&document_, EXIF_CAMERA, EXIF_BODY_SERIAL_NUMBER,
    270 c.exifSettings.hardwareIdentifier.c_str()),
    271 "Set INI camera hardware identifier");
    272 comment << "; camera.exposure_basis=nominal_emulation_not_measured"
    273 << "; camera.shutter_speed_seconds=" << shutterUs / 1000000.0
    274 << "; camera.iso=" << c.exifSettings.iso
    275 << "; camera.hardware_identifier="
    276 << (c.exifSettings.hardwareIdentifier.empty() ? "undefined" : c.exifSettings.hardwareIdentifier);
    277 if (!c.exifSettings.hardwareIdentifier.empty())
    278 comment << "; camera.hardware_identifier_basis=ini_setting";
    279 if (std::abs(position.groundAmsl) > static_cast<double>(std::numeric_limits<std::int32_t>::max()) / 256)
    280 throw std::runtime_error("Ground altitude exceeds Paparazzi EXIF fixed-point range");
    281 const double fields[] = {c.fovRadians, c.commandedHeading, c.pitch, c.bank, c.heading,
    282 c.observedLatitude, c.observedLongitude, c.observedAltitude,
    283 c.cropX, c.cropY, c.cropWidth, c.cropHeight, c.imageRotationDegrees, c.rotationBoundDegrees};
    284 if (!std::all_of(std::begin(fields), std::end(fields), [](double value) { return std::isfinite(value); }) ||
    285 c.fovRadians <= 0 || c.fovRadians >= 3.141592653589793 ||
    286 c.sourceWidth < 1 || c.sourceHeight < 1 || c.cropX < 0 || c.cropY < 0 ||
    287 c.cropWidth <= 0 || c.cropHeight <= 0 ||
    288 c.cropX + c.cropWidth > c.sourceWidth + 1e-6 ||
    289 c.cropY + c.cropHeight > c.sourceHeight + 1e-6 ||
    290 std::abs(c.observedLatitude) > 90 || std::abs(c.observedLongitude) > 180 ||
    291 c.observed100ns <= 0)
    292 throw std::runtime_error("Invalid camera lens/attitude metadata");
    293 const double scale = c.cropWidth / width;
    294 if (std::abs(c.cropHeight / height - scale) > 1e-6)
    295 throw std::runtime_error("Camera intrinsics require proportional image framing");
    296 const double radiansToDegrees = 180 / 3.14159265358979323846;
    297 const double sourceFocal = c.sourceHeight / (2 * std::tan(c.fovRadians / 2));
    298 const double focal = sourceFocal / scale;
    299 if (c.piAiCamera && (width != PiAiOptics::width || height != PiAiOptics::height ||
    300 std::abs(focal - PiAiOptics::focalPixels) > 0.01 || c.frame100ns <= 0))
    301 throw std::runtime_error("Pi AI camera frame does not match its nominal optical profile");
    302 const double equivalent35mm = 36 * focal / width;
    303 if (equivalent35mm > 65535)
    304 throw std::runtime_error("Virtual lens exceeds EXIF 35mm-equivalent range");
    305 check(exif_document_ascii(&document_, EXIF_PRIMARY, 0x0110, c.piAiCamera ?
    306 "Raspberry Pi AI Camera (nominal emulation)" : "MSFS 2024 add-on camera"),
    307 "Set virtual camera model");
    308 check(exif_document_ascii(&document_, EXIF_CAMERA, EXIF_LENS_MODEL,
    309 c.piAiCamera ? "Stock 4.74 mm f/1.79 (estimated lens effects v1)" :
    310 "SimConnect rectilinear virtual lens"),
    311 "Set virtual lens model");
    312 if (c.piAiCamera) {
    313 check(exif_document_ascii(&document_, EXIF_PRIMARY, 0x010f, "Raspberry Pi (emulated)"),
    314 "Set simulated camera manufacturer");
    315 check(exif_document_rational(&document_, EXIF_CAMERA, EXIF_FOCAL_LENGTH, 474, 100),
    316 "Set nominal physical lens focal length");
    317 check(exif_document_rational(&document_, EXIF_CAMERA, 0x829d, 179, 100),
    318 "Set nominal lens aperture");
    319 auto* lens = reserve(EXIF_CAMERA, 0xa432, EXIF_RATIONAL, 4);
    320 for (int i = 0; i < 4; ++i) {
    321 exif_encode_long(lens + i * 8, static_cast<std::uint32_t>(std::llround(
    322 (i < 2 ? PiAiOptics::focalMm : PiAiOptics::aperture) * 100)));
    323 exif_encode_long(lens + i * 8 + 4, 100);
    324 }
    325 check(exif_document_rational(&document_, EXIF_CAMERA, 0xa20e,
    326 static_cast<std::uint32_t>(width * 100000), 62868), "Set nominal sensor X resolution");
    327 check(exif_document_rational(&document_, EXIF_CAMERA, 0xa20f,
    328 static_cast<std::uint32_t>(height * 100000), 47120), "Set nominal sensor Y resolution");
    329 exif_encode_short(reserve(EXIF_CAMERA, 0xa210, EXIF_SHORT, 1), 3);
    330 }
    331 exif_encode_short(reserve(EXIF_CAMERA, 0xa405, EXIF_SHORT, 1),
    332 static_cast<std::uint16_t>(std::llround(equivalent35mm)));
    333 check(exif_document_ascii(&document_, EXIF_GPS, EXIF_GPS_DIRECTION_REF, "T"),
    334 "Set image direction reference");
    335 const double direction = std::fmod(std::fmod(c.commandedHeading + c.imageRotationDegrees, 360.0) + 360.0, 360.0);
    336 const double imageHeading = direction / radiansToDegrees;
    337 check(exif_document_rational(&document_, EXIF_GPS, EXIF_GPS_DIRECTION,
    338 static_cast<std::uint32_t>(std::llround(direction * 1000000)), 1000000),
    339 "Set image grid-up direction");
    340 check(exif_document_ascii(&document_, EXIF_GPS, EXIF_GPS_SPEED_REF, "K"), "Set GPS speed units");
    341 check(exif_document_rational(&document_, EXIF_GPS, EXIF_GPS_SPEED, 0, 1),
    342 "Set stationary virtual capture speed");
    343 comment << "; camera.projection=perspective; camera.mode="
    344 << (c.piAiCamera ? "raspberry_pi_ai_nominal" : (c.narrowAngle ? "narrow_angle_mapping" : "normal"))
    345 << "; camera.lens_model=" << (c.piAiCamera ? "brown_conrady_estimated_v1" : "ideal_rectilinear")
    346 << "; camera.physical_sensor_mm="
    347 << (c.piAiCamera ? "6.2868,4.712" : "undefined")
    348 << "; camera.fov_axis=vertical; camera.sdk_fov_deg=" << c.fovRadians * radiansToDegrees
    349 << "; camera.image_fov_x_deg=" << 2 * std::atan(width / (2 * focal)) * radiansToDegrees
    350 << "; camera.image_fov_y_deg=" << 2 * std::atan(height / (2 * focal)) * radiansToDegrees
    351 << "; camera.fx_px=" << focal << "; camera.fy_px=" << focal
    352 << "; camera.cx_px=" << (c.sourceWidth / 2.0 - c.cropX) / scale - 0.5
    353 << "; camera.cy_px=" << (c.sourceHeight / 2.0 - c.cropY) / scale - 0.5
    354 << "; camera.equivalent_35mm_focal_mm=" << equivalent35mm
    355 << "; camera.source_width=" << c.sourceWidth << "; camera.source_height=" << c.sourceHeight
    356 << "; camera.crop_x=" << c.cropX << "; camera.crop_y=" << c.cropY
    357 << "; camera.crop_width=" << c.cropWidth << "; camera.crop_height=" << c.cropHeight
    358 << "; camera.image_rotation_deg=" << c.imageRotationDegrees
    359 << "; camera.framing_rotation_bound_deg=" << c.rotationBoundDegrees
    360 << "; camera.intrinsics_basis=vertical_FOV_pinhole_model"
    361 << "; camera.attitude_convention=SDK_degrees_positive_pitch_down"
    362 << "; camera.commanded_pitch_deg=90.000000; camera.commanded_bank_deg=0.000000"
