    21/**
    22 * @file usher_protocol.c
    23 * @brief Incremental parser for USHER's UART framing protocol.
    24 * @details The UART transport is a byte stream, so this parser keeps state across
    25 * calls and accepts arbitrarily fragmented frames. It validates both Fletcher-style
    26 * checksum bytes before publishing a message. On malformed input it discards only
    27 * the current partial frame and resumes searching for the next start byte; this is
    28 * deliberately resilient to line noise and lost bytes.
    29 */

    44/**
    45 * @brief Consume one UART byte and advance a USHER transport parser.
    46 * The public parameter and lifetime contract is documented in usher_protocol.h.
    47 * @details Checksums cover the length, message ID, and payload bytes. A complete,
    48 * valid frame sets @c t->msg_received; the caller must clear that flag after
    49 * dispatching. Any framing or checksum error increments @c t->error and resets the
    50 * state machine, avoiding a poisoned partial frame affecting later traffic.
    51 * @warning @p t must remain valid and unchanged by another thread for the duration
    52 * of a frame; this parser intentionally contains no synchronization.
    53 */
    54void parse_usher(struct usher_transport *t, uint8_t c)
    55{
    56//printf("%02X %d %d\n",c, t->status, t->error);
    57
    58 switch (t->status) {
    59 case UNINIT:
    60 if (c == STX) {
    61 t->status++;
    62 }
    63 break;
    64 case GOT_STX:
    65 if (t->msg_received) {
    66 t->error++;
    67 goto error;
    68 }
    69 if (c < UsherSizeOf(0)) goto error;
    70 t->payload_len = c - 5;
    71 t->ck_a = t->ck_b = c;
    72 t->status++;
    73 t->payload_idx = 0;
    74 break;
    75 case GOT_LENGTH:
    76 t->msg_id = c;
    77 t->ck_a += c; t->ck_b += t->ck_a;
    78 t->status++;
    79 if (t->payload_len == 0) {
    80 t->status++;
    81 }
    82 break;
    83 case GOT_MSGID:
    84 t->payload[t->payload_idx] = c;
    85 t->ck_a += c; t->ck_b += t->ck_a;
    86 t->payload_idx++;
    87 if (t->payload_idx == t->payload_len) {
    88 t->status++;
    89 }
    90 break;
    91 case GOT_PAYLOAD:
    92 if (c != t->ck_a) {
    93 goto error;
    94 }
    95 t->status++;
    96 break;
    97 case GOT_CRC1:
    98 if (c != t->ck_b) {
    99 goto error;
    100 }
    101 t->msg_received = true;
    102 goto restart;
    103 default:
    104 goto error;
    105 }
    106 return;
    107error:
    108 t->error++;
    109restart:
    110 t->status = UNINIT;
