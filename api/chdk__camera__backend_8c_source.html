    56/** @brief Prepare capture storage and initialize the CHDK controller path.
    57 * @param source_image Unused fixture argument from the shared backend interface.
    58 * @return Zero for successful setup, or -1 on storage/controller setup failure.
    59 * @details Reject unusable output storage before advertising capture readiness. */
    60static int chdk_backend_init(const char *source_image)
    61{
    62 (void)source_image;
    63 char resolved[PATH_MAX];
    64 const char *directory = usher_resolve_path(photo_directory, resolved, sizeof(resolved));
    65 if (usher_ensure_directory(directory) != 0) return -1;
    66 struct stat status;
    67 if (stat(directory, &status) != 0 || !S_ISDIR(status.st_mode) || access(directory, W_OK) != 0) {
    68 fprintf(stderr, "USHER: CHDK photo directory is not writable: %s\n", directory);
    69 return -1;
    70 }
    71 return chdk_pipe_init();
    72}

    74/** @brief Capture one CHDK JPEG at the path derived from its shot number.
    75 * @param[out] filename Output path; cleared on failure when writable.
    76 * @param capacity Path buffer size including the terminator.
    77 * @param number Nonnegative photo number.
    78 * @return Zero for a completed capture, or -1 on invalid input or capture failure.
    79 * @details Pass the path to the controller only if it fits so truncated
    80 * names cannot misassociate a saved image with another shot. */
    81static int chdk_backend_shoot(char *filename, size_t capacity, int number)
    82{
    83 if (!filename || !capacity) return -1;
    84 filename[0] = '\0';
    85 if (number < 0) return -1;
    86 char resolved[PATH_MAX];
    87 const char *directory = usher_resolve_path(photo_directory, resolved, sizeof(resolved));
    88 int length = snprintf(filename, capacity, "%s/c%06d.jpg", directory, number);
    89 if (length < 0 || (size_t)length >= capacity || chdk_pipe_shoot(filename) != 0) {
    90 filename[0] = '\0';
    91 return -1;
    92 }
    93 return 0;
    94}
