    56/* Serial-owner state is separate from the mutex-protected worker mailbox. Do
    57 * not call start/stop/poll/configure concurrently from additional owner threads. */
    58static struct camera_backend_host ear_host;
    59static bool earcam_active, earcam_unavailable, earcam_session_active, earcam_simulated;
    60/**
    61 * @brief One reservation progressing through snapshot -> point -> evidence completion.
    62 * @details Start reserves a slot before collecting audio. Stop attaches an owned
    63 * immutable snapshot. Poll consumes the point, but the slot remains occupied
    64 * until evidence work finishes, preventing reuse while the worker still uses it.
    65 */
    66static struct ear_finalize_job {
    67 struct ear_session_snapshot *snapshot;
    68 struct ear_loudest_spot spot;
    69 uint32_t session_id;
    70 uint64_t order;
    71 bool in_use, taken, point_ready, evidence_done, simulated;
    72} ear_finalize_jobs[EAR_FINALIZE_JOB_COUNT];

    78static bool ear_finalize_started, ear_finalize_stopping;

    81/** @brief Guard finite values and integer conversion ranges before marking a wire point valid. */
    82static bool ear_point_valid(const struct ear_loudest_spot *spot)
    83{
    84 return spot && spot->valid && isfinite(spot->lat_deg) && isfinite(spot->lon_deg)
    85 && fabs(spot->lat_deg) <= EAR_LATITUDE_LIMIT_DEG && fabs(spot->lon_deg) <= EAR_LONGITUDE_LIMIT_DEG
    86 && (spot->lat_deg != 0 || spot->lon_deg != 0)
    87 && isfinite(spot->agl_m) && spot->agl_m >= 0 && spot->agl_m <= INT32_MAX / EAR_MILLIMETERS_PER_METER
    88 && isfinite(spot->alt_m) && spot->alt_m >= INT32_MIN / EAR_MILLIMETERS_PER_METER
    89 && spot->alt_m <= INT32_MAX / EAR_MILLIMETERS_PER_METER
    90 && isfinite(spot->confidence) && spot->confidence >= 0 && spot->confidence <= 1
    91 && isfinite(spot->level_db) && spot->level_db >= INT32_MIN / EAR_METRIC_UNITS_PER_DB
    92 && spot->level_db <= INT32_MAX / EAR_METRIC_UNITS_PER_DB;
    93}

    95/** @brief Notify the serial owner without blocking; mailbox flags remain authoritative if full. */
    96static void wake_ear_result_owner(void)
    97{
    98 const char notification = 1;
    99 ssize_t result;
    100 do { result = write(ear_finalize_wake[1], &notification, sizeof(notification)); } while (result < 0 && errno == EINTR);
    101 if (result < 0 && errno != EAGAIN && errno != EWOULDBLOCK)
    102 fprintf(stderr, "USHER:\tEAR completion wake failed: %s\n", strerror(errno));
    103}

    105/**
    106 * @brief Drain stopped sessions in admission order until shutdown has no work left.
    107 * @details The point mailbox is published before potentially slow CSV/JPEG/EXIF
    108 * and optional SODA submission. Never perform UART sends or live pose lookups here.
    109 */
    110static void *finalize_ear_sessions(void *unused)
    111{
    112 (void)unused;
    113 for (;;) {
    114 pthread_mutex_lock(&ear_finalize_mutex);
    115 struct ear_finalize_job *job = NULL;
    116 for (;;) {
    117 for (size_t slot = 0; slot < EAR_FINALIZE_JOB_COUNT; ++slot) {
    118 struct ear_finalize_job *candidate = &ear_finalize_jobs[slot];
    119 if (candidate->in_use && candidate->snapshot && !candidate->taken
    120 && (!job || candidate->order < job->order)) job = candidate;
    121 }
    122 if (job || ear_finalize_stopping) break;
    123 pthread_cond_wait(&ear_finalize_available, &ear_finalize_mutex);
    124 }
    125 if (!job) { pthread_mutex_unlock(&ear_finalize_mutex); return NULL; }
    126 job->taken = true;
    127 pthread_mutex_unlock(&ear_finalize_mutex);
    128 struct ear_loudest_spot spot = {0};
    129 if (ear_cam_pipe_snapshot_solve(job->snapshot, &spot) != 0) spot.valid = false;
    130 spot.valid = ear_point_valid(&spot);
    131 pthread_mutex_lock(&ear_finalize_mutex);
    132 job->spot = spot;
    133 job->point_ready = true;
    134 pthread_mutex_unlock(&ear_finalize_mutex);
    135 wake_ear_result_owner();
    136
    137 char sound_picture[EAR_EVIDENCE_PATH_CAPACITY];
    138 if (ear_cam_pipe_snapshot_render(job->snapshot, job->session_id, &spot,
    139 sound_picture, sizeof(sound_picture), job->simulated) == 0) {
    140 union dc_shot_union picture_shot = {0};
    141 picture_shot.data.nr = ear_cam_pipe_snapshot_number(job->snapshot);
    142 if (spot.valid) {
    143 picture_shot.data.lat = (int32_t)llround(spot.lat_deg * EAR_COORDINATE_UNITS_PER_DEG);
    144 picture_shot.data.lon = (int32_t)llround(spot.lon_deg * EAR_COORDINATE_UNITS_PER_DEG);
    145 picture_shot.data.alt = (int32_t)llround(spot.alt_m * EAR_MILLIMETERS_PER_METER);
    146 picture_shot.data.groundalt = (int32_t)llround(spot.agl_m * EAR_SHOT_AGL_UNITS_PER_METER);
    147 }
    148 if (image_exif_write(sound_picture, &picture_shot) == 0) {
    149 printf("USHER:\tEAR session %u sound picture %s\n", job->session_id, sound_picture);
    150 printf("Photo take %d\n", picture_shot.data.nr);
    151 if (ear_host.queue_analysis) ear_host.queue_analysis(sound_picture, &picture_shot, USHER_CAMERA_EARCAM);
    152 } else fprintf(stderr, "USHER:\tEAR session %u EXIF failed; image retained: %s\n", job->session_id, sound_picture);
    153 } else fprintf(stderr, "USHER:\tEAR session %u sound-picture rendering failed\n", job->session_id);
    154 ear_cam_pipe_snapshot_release(job->snapshot);
    155 pthread_mutex_lock(&ear_finalize_mutex);
    156 job->evidence_done = true;
    157 pthread_mutex_unlock(&ear_finalize_mutex);
    158 wake_ear_result_owner();
    159 }
    160}

    162/**
    163 * @brief Consume one completion on the serial owner and convert physical units once.
    164 * @return True when result receives a correlated valid or invalid point.
    165 * @details Wake bytes may coalesce; scan mailbox flags even after draining the
    166 * socket. USHER retains responsibility for token correlation and actual UART delivery.
    167 */
    168static bool ear_poll_result(struct camera_point_result *result)
    169{
    170 if (!result) return false;
    171 if (ear_finalize_wake[0] >= 0) {
    172 char notifications[EAR_COMPLETION_DRAIN_BYTES];
    173 ssize_t count;
    174 do { count = read(ear_finalize_wake[0], notifications, sizeof(notifications)); }
    175 while (count > 0 || (count < 0 && errno == EINTR));
    176 }
    177 for (size_t slot = 0; slot < EAR_FINALIZE_JOB_COUNT; ++slot) {
    178 pthread_mutex_lock(&ear_finalize_mutex);
    179 struct ear_finalize_job *job = &ear_finalize_jobs[slot];
    180 const bool ready = job->in_use && job->point_ready;
    181 const uint32_t session_id = job->session_id;
    182 const struct ear_loudest_spot spot = job->spot;
    183 job->point_ready = false;
    184 if (job->in_use && job->evidence_done) *job = (struct ear_finalize_job) {0};
    185 pthread_mutex_unlock(&ear_finalize_mutex);
    186 if (!ready) continue;
    187 ear_cam_pipe_log_result(session_id, &spot);
    188 printf("USHER:\tEAR session %u result valid=%d lat=%.7f lon=%.7f conf=%.3f samples=%u/%u\n",
    189 session_id, spot.valid, spot.lat_deg, spot.lon_deg, spot.confidence, spot.used_count, spot.sample_count);
    190 *result = (struct camera_point_result) {.session_id = session_id, .valid = ear_point_valid(&spot)};
    191 if (result->valid) {
    192 result->lat = (int32_t)llround(spot.lat_deg * EAR_COORDINATE_UNITS_PER_DEG);
    193 result->lon = (int32_t)llround(spot.lon_deg * EAR_COORDINATE_UNITS_PER_DEG);
    194 result->alt_mm = (int32_t)llround(spot.alt_m * EAR_MILLIMETERS_PER_METER);
    195 result->agl_mm = (int32_t)llround(spot.agl_m * EAR_MILLIMETERS_PER_METER);
    196 result->quality = (int32_t)llround(spot.confidence * EAR_QUALITY_UNITS_PER_SCORE);
    197 result->metric = (int32_t)llround(spot.level_db * EAR_METRIC_UNITS_PER_DB);
    198 }
    199 return true;
    200 }
    201 return false;
    202}

    204/**
    205 * @brief Reserve both finalizer capacity and adapter storage before listening.
    206 * @return False for zero token, shutdown, exhausted capacity or worker/buffer failure.
    207 * @details Replacing an unfinished active session releases only its reservation;
    208 * stopped snapshots stay admitted. Generic USHER routing handles duplicate tokens.
    209 */
    210static bool ear_start(uint32_t session_id)
    211{
    212 if (!session_id) return false;
    213 earcam_session_active = false;
    214 if (!earcam_active || !ear_host.running || !capture_running(ear_host.running)) return false;
    215 pthread_mutex_lock(&ear_finalize_mutex);
    216 if (current_ear_job) { *current_ear_job = (struct ear_finalize_job) {0}; current_ear_job = NULL; }
    217 if (ear_finalize_stopping || ear_finalize_order == UINT64_MAX) { pthread_mutex_unlock(&ear_finalize_mutex); return false; }
    218 if (!ear_finalize_started) {
    219 if (socketpair(AF_UNIX, SOCK_STREAM | SOCK_NONBLOCK | SOCK_CLOEXEC, 0, ear_finalize_wake) != 0) {
    220 ear_finalize_wake[0] = ear_finalize_wake[1] = -1;
    221 pthread_mutex_unlock(&ear_finalize_mutex);
    222 fprintf(stderr, "USHER:\tEAR finalizer completion channel unavailable: %s\n", strerror(errno));
    223 return false;
    224 }
    225 int result = pthread_create(&ear_finalize_thread, NULL, finalize_ear_sessions, NULL);
    226 if (result != 0) {
    227 close(ear_finalize_wake[0]); close(ear_finalize_wake[1]);
    228 ear_finalize_wake[0] = ear_finalize_wake[1] = -1;
    229 pthread_mutex_unlock(&ear_finalize_mutex);
    230 fprintf(stderr, "USHER:\tEAR finalizer unavailable: %s\n", strerror(result));
    231 return false;
    232 }
    233 ear_finalize_started = true;
    234 }
    235 for (size_t slot = 0; slot < EAR_FINALIZE_JOB_COUNT; ++slot) {
    236 if (!ear_finalize_jobs[slot].in_use) {
    237 current_ear_job = &ear_finalize_jobs[slot];
    238 *current_ear_job = (struct ear_finalize_job) {
    239 .session_id = session_id,
    240 .order = ++ear_finalize_order, .in_use = true, .simulated = earcam_simulated
    241 };
    242 break;
    243 }
    244 }
    245 pthread_mutex_unlock(&ear_finalize_mutex);
    246 if (!current_ear_job) return false;
    247 if (ear_cam_pipe_begin_session(session_id)) { earcam_session_active = true; return true; }
    248 pthread_mutex_lock(&ear_finalize_mutex);
    249 *current_ear_job = (struct ear_finalize_job) {0}; current_ear_job = NULL;
    250 pthread_mutex_unlock(&ear_finalize_mutex);
    251 return false;
    252}

    254/**
    255 * @brief Validate the active token before a zero-copy snapshot handoff.
    256 * @return True only when an immutable session was queued for the worker.
    257 * @details Failed detach releases the reservation; a wrong token cannot stop listening.
    258 */
    259static bool ear_stop(uint32_t session_id)
    260{
    261 const bool finalize = earcam_active && earcam_session_active && current_ear_job
    262 && current_ear_job->session_id == session_id;
    263 if (!finalize) return false;
    264 earcam_session_active = false;
    265 struct ear_session_snapshot *snapshot = ear_cam_pipe_detach_session();
    266 pthread_mutex_lock(&ear_finalize_mutex);
    267 if (snapshot) current_ear_job->snapshot = snapshot;
    268 else *current_ear_job = (struct ear_finalize_job) {0};
    269 current_ear_job = NULL;
    270 pthread_cond_signal(&ear_finalize_available);
    271 pthread_mutex_unlock(&ear_finalize_mutex);
    272 return snapshot != NULL;
    273}

    275/**
    276 * @brief Close admission and join all admitted stopped-session evidence work.
    277 * @details Completed point mailboxes remain pollable after the wake channel closes.
    278 * This is orderly draining, not a hard filesystem-I/O deadline or power-loss guarantee.
    279 */
    280static void ear_drain(void)
    281{
    282 earcam_session_active = false;
    283 pthread_mutex_lock(&ear_finalize_mutex);
    284 ear_finalize_stopping = true;
    285 pthread_cond_signal(&ear_finalize_available);
    286 pthread_mutex_unlock(&ear_finalize_mutex);
    287 if (ear_finalize_started) pthread_join(ear_finalize_thread, NULL);
    288 ear_finalize_started = false;
    289 for (size_t index = 0; index < EAR_COMPLETION_ENDPOINT_COUNT; ++index) {
    290 if (ear_finalize_wake[index] >= 0) close(ear_finalize_wake[index]);
    291 ear_finalize_wake[index] = -1;
    292 }
    293}

    306/** @brief Initialize supervision, not microphone readiness; hard setup failure stays latched. */
    307static int ear_init(const char *source)
    308{
    309 if (earcam_active) return 0;
    310 if (earcam_unavailable) return -1;
    311 if (ear_cam_pipe_init(source) != 0) { earcam_unavailable = true; return -1; }

    316/** @brief Drain before tearing down audio/storage, then reset quiescent finalizer state. */
    317static void ear_deinit(void)
    318{
    319 ear_drain();
    320 struct camera_point_result discarded;
    321 while (ear_poll_result(&discarded)) {}
    322 ear_cam_pipe_deinit();
    323 pthread_mutex_lock(&ear_finalize_mutex);
    324 for (size_t slot = 0; slot < EAR_FINALIZE_JOB_COUNT; ++slot) ear_finalize_jobs[slot] = (struct ear_finalize_job) {0};
    325 current_ear_job = NULL;
    326 ear_finalize_order = 0;
    327 ear_finalize_stopping = false;
    328 pthread_mutex_unlock(&ear_finalize_mutex);
    329 earcam_active = earcam_session_active = false;
    330}
