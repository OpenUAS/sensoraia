    18/**
    19 * @file camera_control.c
    20 * @brief Serialize camera-control payloads and track bounded FC-side tickets.
    21 * @details The request counter advances only after the caller confirms that the
    22 * entire UART frame was admitted. This avoids phantom pending controls when a
    23 * frame is rejected locally and prevents unsafe replay after ambiguous delivery.
    24 */
    25#include "camera_control.h"

    27bool usher_control_put_u32(uint8_t *bytes, size_t size, size_t offset, uint32_t value)
    28{
    29 if (!bytes || offset > size || size - offset < 4) return false;
    30 for (size_t index = 0; index < 4; ++index) bytes[offset + index] = (uint8_t)(value >> (8 * index));
    31 return true;
    32}
    33
    34bool usher_control_get_u32(const uint8_t *bytes, size_t size, size_t offset, uint32_t *value)
    35{
    36 if (!bytes || !value || offset > size || size - offset < 4) return false;
    37 *value = 0;
    38 for (size_t index = 0; index < 4; ++index) *value |= (uint32_t)bytes[offset + index] << (8 * index);
    39 return true;
    40}

    42bool usher_control_value_valid(uint8_t command, uint8_t value)
    43{
    44 return command == CAM_CMD_ZOOM_ABSOLUTE || (command == CAM_CMD_LENS_RETRACT && value == 0)
    45 || (command == CAM_CMD_KEEP_AWAKE && value <= 1);
    46}

    48bool usher_control_encode_request(uint8_t *bytes, size_t size, const struct usher_control_request *request)
    49{
    50 if (!bytes || !request || size != USHER_CONTROL_COMMAND_SIZE || !request->camera_mask
    51 || !request->session || !request->request_id || !request->command) return false;
    52 bytes[0] = USHER_CONTROL_VERSION;
    53 bytes[1] = request->camera_mask;
    54 usher_control_put_u32(bytes, size, 2, request->session);
    55 usher_control_put_u32(bytes, size, 6, request->request_id);
    56 bytes[10] = request->command;
    57 bytes[11] = request->value;
    58 return true;
    59}

    61bool usher_control_decode_request(const uint8_t *bytes, size_t size, struct usher_control_request *request)
    62{
    63 if (!bytes || !request || size != USHER_CONTROL_COMMAND_SIZE || bytes[0] != USHER_CONTROL_VERSION
    64 || !bytes[1] || !bytes[10]) return false;
    65 *request = (struct usher_control_request){.camera_mask = bytes[1], .command = bytes[10], .value = bytes[11]};
    66 usher_control_get_u32(bytes, size, 2, &request->session);
    67 usher_control_get_u32(bytes, size, 6, &request->request_id);
    68 return request->session && request->request_id;
    69}

    71bool usher_control_encode_result(uint8_t *bytes, size_t size, const struct usher_control_result *result)
    72{
    73 if (!bytes || !result || size != USHER_CONTROL_RESULT_SIZE || result->camera_id < 1 || result->camera_id > 8
    74 || !result->request.session || !result->request.request_id || !result->request.command
    75 || result->status < CAM_CONTROL_APPLIED || result->status > CAM_CONTROL_UNKNOWN
    76 || result->evidence > CAM_CONTROL_EVIDENCE_PLAYBACK) return false;
    77 bytes[0] = USHER_CONTROL_VERSION;
    78 bytes[1] = result->camera_id;
    79 usher_control_put_u32(bytes, size, 2, result->request.session);
    80 usher_control_put_u32(bytes, size, 6, result->request.request_id);
    81 bytes[10] = result->request.command;
    82 bytes[11] = result->observed;
    83 bytes[12] = result->status;
    84 bytes[13] = result->evidence;
    85 return true;
    86}

    88bool usher_control_decode_result(const uint8_t *bytes, size_t size, struct usher_control_result *result)
    89{
    90 if (!bytes || !result || size != USHER_CONTROL_RESULT_SIZE || bytes[0] != USHER_CONTROL_VERSION
    91 || bytes[1] < 1 || bytes[1] > 8 || !bytes[10]
    92 || bytes[12] < CAM_CONTROL_APPLIED || bytes[12] > CAM_CONTROL_UNKNOWN
    93 || bytes[13] > CAM_CONTROL_EVIDENCE_PLAYBACK) return false;
    94 *result = (struct usher_control_result){.camera_id = bytes[1], .status = bytes[12],
    95 .observed = bytes[11], .evidence = bytes[13], .request = {.command = bytes[10]}};
    96 usher_control_get_u32(bytes, size, 2, &result->request.session);
    97 usher_control_get_u32(bytes, size, 6, &result->request.request_id);
    98 return result->request.session && result->request.request_id;
    99}

    101void usher_control_client_reset(struct usher_control_client *client, uint32_t nonce)
    102{
    103 client->session = 0;
    104 client->negotiating_session = 0;
    105 client->confirmed = false;
    106 client->negotiating = false;
    107 client->nonce = nonce ? nonce : 1;
    108 /* Preserve the request counter but make prior-session results explicitly unknown. */
    109 for (size_t index = 0; index < 8; ++index) {
    110 client->capabilities[index] = 0;
    111 if (client->slots[index].request_id) client->slots[index].status = CAM_CONTROL_UNKNOWN;
    112 }
    113}

    115bool usher_control_client_hello(struct usher_control_client *client, const uint8_t *bytes, size_t size, uint32_t now_us)
    116{
    117 uint32_t nonce, session;
    118 if (!client->negotiating || !bytes || size != USHER_CONTROL_HELLO_SIZE || bytes[0] != USHER_CONTROL_VERSION
    119 || !usher_control_get_u32(bytes, size, 1, &nonce) || nonce != client->nonce
    120 || !usher_control_get_u32(bytes, size, 5, &session) || !session) return false;
    121 if (client->session != session) {
    122 for (size_t index = 0; index < 8; ++index)
    123 if (client->slots[index].request_id) client->slots[index].status = CAM_CONTROL_UNKNOWN;
    124 }
    125 for (size_t index = 0; index < 8; ++index)
    126 usher_control_get_u32(bytes, size, 9 + 4 * index, &client->capabilities[index]);
    127 client->confirmed = client->session == session && client->negotiating_session == session;
    128 client->session = session;
    129 client->hello_us = now_us;
    130 return true;
    131}

    133bool usher_control_client_ready(const struct usher_control_client *client, uint32_t now_us)
    134{
    135 return client->session && client->confirmed && (uint32_t)(now_us - client->hello_us) < USHER_CONTROL_LEASE_US;
    136}

    138void usher_control_client_expire(struct usher_control_client *client, uint32_t now_us)
    139{
    140 for (size_t index = 0; index < 8; ++index) {
    141 struct usher_control_slot *slot = &client->slots[index];
    142 if (slot->status == CAM_CONTROL_PENDING && (uint32_t)(now_us - slot->sent_us) >= USHER_CONTROL_TIMEOUT_US)
    143 slot->status = CAM_CONTROL_UNKNOWN;
    144 }
    145}

    147bool usher_control_client_prepare(struct usher_control_client *client, uint8_t mask, uint8_t command,
    148 uint8_t value, uint32_t now_us, struct usher_control_request *request)
    149{
    150 usher_control_client_expire(client, now_us);
    151 if (!request || !mask || !usher_control_value_valid(command, value)
    152 || !usher_control_client_ready(client, now_us) || client->next_request_id == UINT32_MAX) return false;
    153 for (size_t index = 0; index < 8; ++index)
    154 if ((mask & (1U << index)) && client->slots[index].status == CAM_CONTROL_PENDING) return false;
    155 *request = (struct usher_control_request){.session = client->session, .request_id = client->next_request_id + 1,
    156 .camera_mask = mask, .command = command, .value = value};
    157 return true;
    158}

    160void usher_control_client_commit(struct usher_control_client *client, const struct usher_control_request *request,
    161 uint32_t now_us)
    162{
    163 /* This is called after full-frame admission; prepare alone must not consume an ID. */
    164 client->next_request_id = request->request_id;
    165 for (size_t index = 0; index < 8; ++index) {
    166 if (!(request->camera_mask & (1U << index))) continue;
    167 client->slots[index] = (struct usher_control_slot){.request_id = request->request_id, .sent_us = now_us,
    168 .camera_mask = request->camera_mask, .command = request->command, .value = request->value,
    169 .status = CAM_CONTROL_PENDING};
    170 }
    171}

    173bool usher_control_client_receive(struct usher_control_client *client, const uint8_t *bytes, size_t size,
    174 uint32_t now_us)
    175{
    176 struct usher_control_result result;
    177 usher_control_client_expire(client, now_us);
    178 if (!usher_control_client_ready(client, now_us) || !usher_control_decode_result(bytes, size, &result)
    179 || result.request.session != client->session) return false;
    180 struct usher_control_slot *slot = &client->slots[result.camera_id - 1];
    181 if (slot->status != CAM_CONTROL_PENDING || slot->request_id != result.request.request_id
    182 || slot->command != result.request.command) return false;
    183 if (result.status == CAM_CONTROL_APPLIED
    184 && ((slot->command == CAM_CMD_LENS_RETRACT && result.evidence != CAM_CONTROL_EVIDENCE_PLAYBACK)
    185 || (slot->command != CAM_CMD_LENS_RETRACT && result.evidence != CAM_CONTROL_EVIDENCE_VALUE))) return false;
    186 if (result.status != CAM_CONTROL_APPLIED && result.evidence != CAM_CONTROL_EVIDENCE_NONE) return false;
    187 if (result.status == CAM_CONTROL_APPLIED
    188 && ((slot->command == CAM_CMD_LENS_RETRACT && result.observed)
    189 || (slot->command == CAM_CMD_KEEP_AWAKE && result.observed > 1))) return false;
    190 slot->status = result.status;
    191 slot->observed = result.observed;
    192 slot->evidence = result.evidence;
    193 return true;
    194}

    196uint8_t usher_control_client_camera_result(const struct usher_control_client *client, uint32_t ticket, uint8_t camera_id)
    197{
    198 if (!ticket) return CAM_CONTROL_NOT_SENT;
    199 if (camera_id < 1 || camera_id > 8 || client->slots[camera_id - 1].request_id != ticket) return CAM_CONTROL_UNKNOWN;
    200 return client->slots[camera_id - 1].status;
    201}

    203uint8_t usher_control_client_result(const struct usher_control_client *client, uint32_t ticket)
    204{
    205 if (!ticket) return CAM_CONTROL_NOT_SENT;
    206 uint8_t mask = 0, status = 0;
    207 bool mixed = false, pending = false;
    208 for (size_t index = 0; index < 8; ++index)
    209 if (client->slots[index].request_id == ticket) { mask = client->slots[index].camera_mask; break; }
    210 if (!mask) return CAM_CONTROL_UNKNOWN;
    211 /* Unknown evidence dominates; different terminal outcomes remain PARTIAL. */
    212 for (size_t index = 0; index < 8; ++index) {
    213 if (!(mask & (1U << index))) continue;
    214 const uint8_t current = usher_control_client_camera_result(client, ticket, (uint8_t)(index + 1));
    215 if (current == CAM_CONTROL_UNKNOWN) return current;
    216 if (current == CAM_CONTROL_PENDING) { pending = true; continue; }
    217 if (status && status != current) mixed = true;
    218 status = current;
    219 }
    220 return pending ? CAM_CONTROL_PENDING : mixed ? CAM_CONTROL_PARTIAL : status;
    221}
