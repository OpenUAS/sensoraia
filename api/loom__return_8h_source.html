    30/** @brief Local ground point in metres east/north of the reference origin. */
    31struct loom_return_point { double east, north; };
    32
    33/** @brief Wire metric values distinguishing active vector steering from arrival. */
    34enum loom_return_metric {
    35 LOOM_METRIC_NONE = 0,
    36 LOOM_METRIC_STEERING = 1,
    37 LOOM_METRIC_ARRIVED = 2,
    38 LOOM_METRIC_BARO_MAGNETO = 3
    39};

    41/** @brief Limits applied before a neural direction may become a target. */
    42struct loom_return_config {
    43 double home_radius_m; /**< Destination arrival / protection radius in metres. */
    44 double lookahead_m; /**< Carrot step distance ahead of the vehicle along a unit vector, metres. */
    45 double maximum_step_m; /**< Terminal per-observation target refinement bound, metres. */
    46 double minimum_confidence; /**< Required model confidence in 0..1. */
    47 double maximum_tilt_deg; /**< Maximum absolute roll or pitch accepted for a photo, degrees. */
    48};
    49
    50/** @brief Caller-owned sequence and smoothing state; first observation seeds target. */
    51struct loom_return_state {
    52 struct loom_return_point target;
    53 uint32_t sequence, random_state;
    54};

    56/** @brief Explicit model-vector coordinates; embeddings are never guessed to be bearings. */
    57enum loom_return_frame {
    58 LOOM_RETURN_NONE,
    59 LOOM_RETURN_NORTH_EAST,
    60 LOOM_RETURN_BODY_FORWARD_RIGHT,
    61 LOOM_RETURN_SIMULATED
    62};
    63
    64/** @brief Module-owned inference settings, separate from universal runtime settings. */
    65struct loom_return_options {
    66 struct loom_return_config limits;
    67 enum loom_return_frame frame;
    68 double home_latitude, home_longitude; /**< Optional live-sortie destination, geographic degrees. */
    69 unsigned seen;
    70};

    108/** @brief Convert an earth-frame direction into a HOME-bounded target.
    109 * @param config Valid policy limits.
    110 * @param position Vehicle east/north relative to HOME, metres.
    111 * @param direction Earth-frame east/north components; magnitude is not distance.

    113 * @param target Written only for an accepted, nonzero direction toward HOME.

    120/** @brief Admit a fresh bounded target and limit its movement.
