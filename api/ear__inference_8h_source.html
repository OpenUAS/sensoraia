    35/** @brief Select the explicit detector and qualify an ONNX PCM contract before capture starts.
    36 * @param settings EARcam application settings path.
    37 * @param rate Exact capture sample rate; no implicit resampling is performed.
    38 * @param enabled Set true for ONNX, false for a successfully selected native detector.
    39 * @param error Writable failure diagnostic.
    40 * @param capacity Available bytes in error.
    41 * @return True for an accepted native or ONNX selection; false never means native fallback.
    42 * @pre No inference thread or previously configured model is active. */
    43bool ear_inference_configure(const char *settings, unsigned rate, bool *enabled,
    44 char *error, size_t capacity);
    45/** @brief Start the sole classifier thread after successful ONNX configuration.
    46 * @param error Writable failure diagnostic.
    47 * @param capacity Available bytes in error.
    48 * @return True after thread creation, not proof of a classified acoustic event. */
    49bool ear_inference_start(char *error, size_t capacity);
    50/** @brief Supply capture samples without waiting for model execution or a busy model mailbox.
    51 * @param samples Borrowed signed mono PCM; no pointer survives this call.
    52 * @param count Number of samples in this continuous capture chunk.
    53 * @param end_us Capture end of the chunk in CLOCK_MONOTONIC microseconds, or zero when unknown.
    54 * @param error_us Bound on capture-window timing uncertainty, not inference duration.
    55 * @details Complete windows may replace pending work; overload loses evidence rather than blocking ALSA.
    56 * @pre One capture producer owns push/discontinuity calls until stop begins. */
    57void ear_inference_push(const int16_t *samples, size_t count, uint64_t end_us, uint64_t error_us);
    58/** @brief Consume at most one completed original-window result without blocking capture.
    59 * @param result Writable destination, unchanged when no result is available.
    60 * @return True for a newly consumed completion; its window time is not its inference completion time. */
    61bool ear_inference_take(struct ear_inference_result *result);
    62/** @brief Invalidate queued/completed evidence after capture continuity breaks, including in-flight generations. */
    63void ear_inference_discontinuity(void);
    64/** @brief Request termination, join the classifier and release its model; repeat calls are harmless.
    65 * @pre The capture producer has stopped supplying new chunks. */
    66void ear_inference_stop(void);
    67/** @brief Classify exact whole windows from a regular little-endian signed-16 mono PCM file.
    68 * @param filename Immutable replay input; no hardware or simulated exposure timing is invented.
    69 * @return Zero after complete replay, otherwise one for invalid input or model failure.
    70 * @pre The model is configured but its capture worker is not running. */
    71int ear_inference_file(const char *filename);
