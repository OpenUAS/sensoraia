    150class DesktopPool {
    151 std::vector<std::thread> workers;
    152 std::mutex mutex;
    153 std::condition_variable ready, completed;
    154 std::function<void(size_t)> task;
    155 size_t next_job = 0, job_count = 0, active = 0;
    156 size_t peak_active = 0;
    157 bool shutdown = false;
    158 std::exception_ptr failure;
    159 std::atomic<bool> cancel{false};
    160
    161 /** @brief Claim and execute jobs until the coordinator shuts down the pool.
    162 * @details Execute callbacks outside the mutex. On failure, stop assigning
    163 * jobs but let in-flight callbacks drain before the batch returns. */
    164 void work() {
    165 std::unique_lock<std::mutex> lock(mutex);
    166 for (;;) {
    167 ready.wait(lock, [this] { return shutdown || next_job < job_count; });
    168 if (shutdown) return;
    169 const size_t job = next_job++;
    170 ++active;
    171 peak_active = std::max(peak_active, active);
    172 lock.unlock();
    173 try {
    174 task(job);
    175 } catch (...) {
    176 lock.lock();
    177 if (!failure) failure = std::current_exception();
    178 cancel = true;
    179 next_job = job_count;
    180 lock.unlock();
    181 }
    182 lock.lock();
    183 --active;
    184 if (active == 0 && next_job == job_count) completed.notify_one();
    185 }
    186 }
    187
    188 public:
    189 /** @brief Start the desktop worker pool.
    190 * @param count Number of worker threads.
    191 * @details Block termination signals while spawning so workers inherit
    192 * the mask and shutdown remains the coordinator's responsibility. */
    193 explicit DesktopPool(size_t count) {
    194 sigset_t blocked, previous;
    195 sigemptyset(&blocked);
    196 sigaddset(&blocked, SIGINT);
    197 sigaddset(&blocked, SIGTERM);
    198 check(pthread_sigmask(SIG_BLOCK, &blocked, &previous) == 0,
    199 "cannot mask desktop worker signals");
    200 try {
    201 for (size_t index = 0; index < count; ++index)
    202 workers.emplace_back([this] { work(); });
    203 check(pthread_sigmask(SIG_SETMASK, &previous, nullptr) == 0,
    204 "cannot restore coordinator signals");
    205 } catch (...) {
    206 {
    207 std::lock_guard<std::mutex> lock(mutex);
    208 shutdown = true;
    209 }
    210 ready.notify_all();
    211 for (auto& worker : workers) worker.join();
    212 pthread_sigmask(SIG_SETMASK, &previous, nullptr);
    213 throw;
    214 }
    215 }
    216 /** @brief Wake and join the workers before destroying shared batch state.
    217 * @pre No coordinator call to @ref run is in progress. */
    218 ~DesktopPool() {
    219 {
    220 std::lock_guard<std::mutex> lock(mutex);
    221 shutdown = true;
    222 }
    223 ready.notify_all();
    224 for (auto& worker : workers) worker.join();
    225 }
    226 /** @brief Report the fixed worker capacity.
    227 * @return Number of threads, whether idle or active. */
    228 size_t size() const { return workers.size(); }
    229 /** @brief Read the lifetime concurrency high-water mark.
    230 * @return Greatest number of simultaneously active jobs. */
    231 size_t peak_workers() {
    232 std::lock_guard<std::mutex> lock(mutex);
    233 return peak_active;
    234 }
    235 /** @brief Check whether the current batch has failed or been interrupted.
    236 * @return True when workers should stop at their next checkpoint. */
    237 bool cancelled() const { return cancel.load(); }
    238
    239 /** @brief Run indexed work and wait for every active callback to finish.
    240 * @param count Number of jobs.
    241 * @param operation Worker callback; invocations may overlap.
    242 * @param checkpoint Coordinator callback for resource and stop checks.
    243 * @details Poll the checkpoint while waiting. Preserve the first failure
    244 * and rethrow it only after callbacks no longer borrow batch state.
    245 * @pre One coordinator calls this method; it is not reentrant. */
    246 void run(size_t count, std::function<void(size_t)> operation,
    247 const std::function<void()>& checkpoint) {
    248 std::unique_lock<std::mutex> lock(mutex);
    249 task = std::move(operation);
    250 job_count = count;
    251 next_job = 0;
    252 failure = nullptr;
    253 cancel = false;
    254 ready.notify_all();
    255 while (active != 0 || next_job != job_count) {
    256 completed.wait_for(lock, std::chrono::milliseconds(20));
    257 lock.unlock();
    258 try {
    259 checkpoint();
    260 } catch (...) {
    261 lock.lock();
    262 if (!failure) failure = std::current_exception();
    263 cancel = true;
    264 next_job = job_count;
    265 lock.unlock();
    266 }
    267 lock.lock();
    268 }
    269 task = {};
    270 if (failure) std::rethrow_exception(failure);
    271 }
    272};

    275class Resources {
    276 double last_cpu = seconds(CLOCK_PROCESS_CPUTIME_ID);
    277 double last_wall = seconds(CLOCK_MONOTONIC);
    278 const double started_cpu = last_cpu;
    279 const double started_wall = last_wall;
    280 uint64_t previous_total = 0, previous_idle = 0;
    281 double idle_fraction = 0;
    282 double sampled_at = 0;
    283 uint64_t reserve; /**< RAM left for Linux, camera buffers and concurrent
    284 services. */
    285 double duty; /**< Maximum fraction of one CPU core used over each checkpoint
    286 interval. */
    287 fs::path pause_file; /**< Operator-controlled file whose existence suspends
    288 processing. */
    289 bool desktop_mode = false;
    290 uint64_t address_limit = memory_limit;
    291 std::unique_ptr<DesktopPool> pool;
    292 public:
    293 /** @brief Establish mapping limits for onboard or desktop execution.
    294 * @param reserve_mb Required available memory reserve in MiB.
    295 * @param cpu_percent Onboard CPU budget as a percentage of one core.
    296 * @param pause Optional path whose existence pauses work.
    297 * @param desktop Whether to provision desktop workers.
    298 * @details Keep the onboard policy conservative; desktop concurrency is
    299 * capped by both available memory and CPU affinity. */
    300 Resources(uint64_t reserve_mb, int cpu_percent, fs::path pause = {},
    301 bool desktop = false)
    302 : reserve(reserve_mb * mib),
    303 duty(cpu_percent / 100.0),
    304 pause_file(std::move(pause)), desktop_mode(desktop) {
    305 check(reserve_mb >= 192 && reserve_mb <= 4096 && cpu_percent >= 1 &&
    306 cpu_percent <= 20,
    307 "reserve must be 192..4096 MiB; CPU duty must be 1..20 percent of "
    308 "one core");
    309 rlimit address{memory_limit, memory_limit}, core{0, 0}, files{64, 64};
    310 if (desktop_mode) {
    311 check(proc_kib("/proc/meminfo", "MemTotal:") >= 2 * 1024 * mib,
    312 "--desktop requires at least 2 GiB RAM; MORA must use default mode");
    313 check(getrlimit(RLIMIT_AS, &address) == 0,
    314 "cannot read inherited address-space limit");
    315 address_limit = address.rlim_cur == RLIM_INFINITY ? 0 : address.rlim_cur;
    316 duty = 1;
    317 files = {256, 256};
    318 } else {
    319 check(setrlimit(RLIMIT_AS, &address) == 0,
    320 "cannot install resource limits");
    321 }
    322 check(setrlimit(RLIMIT_CORE, &core) == 0 &&
    323 setrlimit(RLIMIT_NOFILE, &files) == 0,
    324 "cannot install resource limits");
    325 sched_param parameters{};
    326 if (!desktop_mode) {
    327 check(setpriority(PRIO_PROCESS, 0, 19) == 0 &&
    328 sched_setscheduler(0, SCHED_IDLE, &parameters) == 0,
    329 "cannot enable idle CPU scheduling");
    330 check(syscall(SYS_ioprio_set, IOPRIO_WHO_PROCESS, 0,
    331 IOPRIO_PRIO_VALUE(IOPRIO_CLASS_IDLE, 0)) == 0,
    332 "cannot enable idle I/O scheduling");
    333 }
    334 check(prctl(PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0) == 0,
    335 "cannot disable privilege escalation");
    336 cv::setNumThreads(1);
    337 if (desktop_mode) {
    338 cpu_set_t affinity;
    339 CPU_ZERO(&affinity);
    340 check(sched_getaffinity(0, sizeof(affinity), &affinity) == 0,
    341 "desktop CPU affinity unavailable");
    342 const uint64_t available_memory =
    343 proc_kib("/proc/meminfo", "MemAvailable:");
    344 check(available_memory > reserve + memory_limit + 16 * mib,
    345 "insufficient RAM headroom for desktop workers");
    346 const uint64_t memory_workers =
    347 (available_memory - reserve - memory_limit) / (16 * mib);
    348 const size_t count = std::min<uint64_t>(
    349 std::min(CPU_COUNT(&affinity), 64), memory_workers);
    350 check(count > 0, "no desktop worker capacity");
    351 pool = std::make_unique<DesktopPool>(count);
    352 }
    353 }

    374 /** @brief Estimate how many photos can be prepared in one desktop batch.
    375 * @param photo_budget_mib Nonzero estimated peak memory per photo in MiB.
    376 * @return Batch size from one photo through the worker count.
    377 * @details Preserve reserve and worker overhead before dividing the remaining
    378 * headroom. The minimum allows serial preparation, not a memory guarantee.
    379 * @pre Desktop worker pool is initialized. */
    380 size_t photo_batch_capacity(uint64_t photo_budget_mib = 128) const {
    381 const uint64_t available_memory =
    382 proc_kib("/proc/meminfo", "MemAvailable:");
    383 const uint64_t overhead = reserve + memory_limit + pool->size() * 16 * mib;
    384 if (available_memory <= overhead) return 1;
    385 return std::max<size_t>(1, std::min<uint64_t>(
    386 pool->size(), (available_memory - overhead) / (photo_budget_mib * mib)));
    387 }

    389 /** Report whether system headroom currently meets the configured reserve and
    390 * duty budget. */
    391 bool available() {
    392 if (!pause_file.empty()) {
    393 struct stat status{};
    394 if (lstat(pause_file.c_str(), &status) == 0 || errno != ENOENT)
    395 return false;
    396 }
    397 const double now = seconds(CLOCK_MONOTONIC);
    398 if (desktop_mode)
    399 return proc_kib("/proc/meminfo", "MemAvailable:") >= reserve;
    400 // Sample procfs every quarter-second rather than once per pixel, row, or
    401 // tile.
    402 if (now - sampled_at >= 0.25) {
    403 std::ifstream input("/proc/stat");
    404 std::string label;
    405 std::array<uint64_t, 8> ticks{};
    406 input >> label;
    407 uint64_t total = 0;
    408 for (auto& value : ticks) {
    409 input >> value;
    410 total += value;
    411 }
    412 check(input.good() && label == "cpu", "CPU headroom unavailable");
    413 if (previous_total && total > previous_total &&
    414 ticks[3] >= previous_idle) {
    415 idle_fraction = static_cast<double>(ticks[3] - previous_idle) /
    416 (total - previous_total);
    417 }
    418 previous_total = total;
    419 previous_idle = ticks[3];
    420 sampled_at = now;
    421 }
    422 const uint64_t resident = proc_kib("/proc/self/status", "VmRSS:");
    423 const uint64_t available_memory =
    424 proc_kib("/proc/meminfo", "MemAvailable:");
    425 // Require one quarter of system CPU to appear idle before this best-effort
    426 // worker proceeds.
    427 return resident > 0 &&
    428 available_memory >=
    429 reserve + memory_limit - std::min(resident, memory_limit) &&
    430 idle_fraction >= 0.25;
    431 }

    433 /** Bound CPU duty by sleeping and stop work when memory, CPU or pause gates
    434 * close.
    435 * @throws std::runtime_error on cancellation or unavailable required system
    436 * telemetry. */
    437 /** Sleep to enforce CPU duty and wait while system headroom or operator pause
    438 * is closed.
    439 * @throws std::runtime_error on process cancellation or unavailable
    440 * telemetry. */
    441 void checkpoint() {
    442 if (parent_pid && getppid() != parent_pid) stopping = 1;
    443 check(!stopping, "interrupted; disk checkpoint retained");
    444 const double now_cpu = seconds(CLOCK_PROCESS_CPUTIME_ID);
    445 const double now_wall = seconds(CLOCK_MONOTONIC);
    446 const double rest = desktop_mode ? 0 :
    447 (now_cpu - last_cpu) / duty - (now_wall - last_wall);
    448 const double deadline = now_wall + std::max(0.0, rest);
    449 while (!stopping && seconds(CLOCK_MONOTONIC) < deadline) {
    450 poll(nullptr, 0,
    451 static_cast<int>(std::clamp(
    452 std::ceil((deadline - seconds(CLOCK_MONOTONIC)) * 1000), 1.0,
    453 1000.0)));
    454 if (parent_pid && getppid() != parent_pid) stopping = 1;
    455 }
    456 while (!stopping && !available()) {
    457 poll(nullptr, 0, 1000);
    458 if (parent_pid && getppid() != parent_pid) stopping = 1;
    459 }
    460 check(!stopping, "interrupted; disk checkpoint retained");
    461 last_cpu = now_cpu;
    462 last_wall = seconds(CLOCK_MONOTONIC);
    463 }

    516/** Open a no-follow regular file and enforce its byte ceiling before decoding.
    517 * @param path Input path.
    518 * @param maximum Maximum file size in bytes.
    519 * @return Owning stream, or throws InvalidPhoto for unsafe/unusable input. */
    520File regular_file(const fs::path& path, uint64_t maximum) {
    521 int descriptor =
    522 open(path.c_str(), O_RDONLY | O_NONBLOCK | O_NOFOLLOW | O_CLOEXEC);
    523 check(descriptor >= 0, "cannot open regular input");
    524 struct stat status{};
    525 if (fstat(descriptor, &status) != 0 || !S_ISREG(status.st_mode) ||
    526 status.st_size < 0 || static_cast<uint64_t>(status.st_size) > maximum) {
    527 close(descriptor);
    528 throw InvalidPhoto("input exceeds regular-file limits");

    546/** Same-directory temporary file committed with fsync+rename for crash-safe
    547 * checkpoints. */
    548/** Same-directory temporary output committed with flush, fsync, rename, and
    549 * directory fsync. */
    550struct AtomicFile {
    551 fs::path target, temporary;
    552 File file;
    553 /** @param destination Final path; its parent directory must already exist. */
    554 explicit AtomicFile(const fs::path& destination) : target(destination) {
    555 std::string pattern = destination.string() + ".tmp-XXXXXX";
    556 const int descriptor = mkstemp(pattern.data());
    557 check(descriptor >= 0, "cannot create checkpoint temporary");
    558 temporary = pattern;
    559 file.reset(fdopen(descriptor, "wb"));
    560 if (!file) {
    561 close(descriptor);
    562 unlink(temporary.c_str());
    563 }
    564 check(bool(file), "cannot open checkpoint stream");
    565 }
    566 /** @brief Remove abandoned staging data without removing committed output. */
    567 ~AtomicFile() {
    568 if (!temporary.empty()) unlink(temporary.c_str());
    569 }
    570 /** Append exact bytes to the temporary output. */
    571 void write(const void* data, size_t length) {
    572 check(fwrite(data, 1, length, file.get()) == length,
    573 "checkpoint write failed");
    574 }
    575 /** @brief Append serialized text without adding a terminator or newline.
    576 * @param value Exact bytes to append. */
    577 void text(const std::string& value) { write(value.data(), value.size()); }
    578 /** Durably publish the complete temporary file at target. */
    579 void commit() {
    580 check(fflush(file.get()) == 0 && fsync(fileno(file.get())) == 0,
    581 "checkpoint flush failed");
    582 check(fclose(file.release()) == 0, "checkpoint close failed");
    583 check(rename(temporary.c_str(), target.c_str()) == 0,
    584 "checkpoint publish failed");
    585 temporary.clear();
    586 sync_directory(target.parent_path());
    587 }
    588};

    676 * order. */
    677cv::Matx33d rotation(double roll, double pitch, double yaw) {
    678 const cv::Matx33d roll_matrix(1, 0, 0, 0, cos(roll), -sin(roll), 0, sin(roll),
    679 cos(roll));

    711/** Inclusive slippy-map tile rectangle for accumulated mosaic coverage. */
    712struct Bounds {
    713 int left = 0, top = 0, right = -1, bottom = -1;
    714 /** Empty until an accepted source photo contributes map coverage. */
    715 bool empty() const { return right < left; }
    716 /** Serialize the inclusive tile rectangle into checkpoint JSON. */
    717 Json json() const { return Json::array({left, top, right, bottom}); }
    718 /** Parse bounds and enforce zoom-18 world and maximum map-side constraints.
    719 */
    720 static Bounds from(const Json& value, int max_map_side = 8192) {
    721 check(value.is_array() && value.size() == 4, "invalid map bounds");
    722 Bounds result{value[0].get<int>(), value[1].get<int>(), value[2].get<int>(),
    723 value[3].get<int>()};
    724 check(
    725 result.empty() ||
    726 (result.left >= 0 && result.top >= 0 && result.right < (1 << 18) &&
    727 result.bottom < (1 << 18) &&
    728 result.right - result.left < max_map_side / tile_size &&
    729 result.bottom - result.top < max_map_side / tile_size),
    730 "map exceeds configured pixel side or Mercator limits");
    731 return result;
    732 }
    733 /** @brief Extend the committed tile envelope without changing either input.
    734 * @param other New footprint in tile coordinates.
    735 * @param max_map_side Maximum allowed map side in pixels.
    736 * @return Bounding rectangle, not a claim that all enclosed tiles have coverage.
    737 * @details Reapply map limits to the union; individually acceptable
    738 * footprints can breach the limit when combined. */
    739 Bounds include(const Bounds& other, int max_map_side = 8192) const {
    740 if (empty()) return other;
    741 return from(Json::array({std::min(left, other.left), std::min(top, other.top),
    742 std::max(right, other.right),
    743 std::max(bottom, other.bottom)}), max_map_side);
    744 }

    747/** Decoded source frame, pixel-to-world homography, and intersected tile
    748 * bounds. */
    749struct Photo {
    750 cv::Mat image;
    751 cv::Matx33d to_world;
    752 Bounds bounds;
    753 /** Transform the four source-image corners into another image/tile plane. */
    754 std::array<cv::Point2f, 4> corners(const cv::Matx33d& transform) const {
    755 std::array<cv::Point2f, 4> result{};
    756 const std::array<cv::Point, 4> pixels{{{0, 0},
    757 {image.cols - 1, 0},
    758 {image.cols - 1, image.rows - 1},
    759 {0, image.rows - 1}}};
    760 for (size_t index = 0; index < pixels.size(); ++index) {
    761 const cv::Vec3d point =
    762 transform * cv::Vec3d(pixels[index].x, pixels[index].y, 1);
    763 result[index] = {float(point[0] / point[2]), float(point[1] / point[2])};
    764 }
    765 return result;
    766 }
    767 /** Convert source corners back to longitude/latitude for per-photo
    768 * provenance. */
    769 Json geographic_corners() const {
    770 Json result = Json::array();
    771 for (const cv::Point& pixel :
    772 std::array<cv::Point, 4>{{{0, 0},
    773 {image.cols - 1, 0},
    774 {image.cols - 1, image.rows - 1},
    775 {0, image.rows - 1}}}) {
    776 const cv::Vec3d point = to_world * cv::Vec3d(pixel.x, pixel.y, 1);
    777 const double world_x = point[0] / point[2], world_y = point[1] / point[2];
    778 result.push_back(Json::array({world_x / world_pixels * 360 - 180,
    779 atan(sinh(pi * (1 - 2 * world_y / world_pixels))) * 180 / pi}));
    780 }
    781 return result;
    782 }
    783 /** Validate source EXIF and derive a flat-ground source-pixel-to-world
    784 * transform. MIMIC uses embedded geometry; real JPEGs require explicit
    785 * intrinsics and mount angles so the worker never guesses a sensor model. */
    786 void geometry(const exif_view* metadata, int width, int height, const Json& camera,
    787 bool standard_profile) {
    788 Json provenance;
    789 const std::string document =
    790 exif_text(metadata, EXIF_PRIMARY, EXIF_DOCUMENT_NAME);
    791 standard_profile = standard_profile &&
    792 document.find("USHER_MOCK_MAP_V1") == std::string::npos;
    793 if (camera.is_null() && !standard_profile) {
    794 try {
    795 provenance = Json::parse(document);
    796 } catch (const Json::exception&) {
    797 throw InvalidPhoto(
    798 "missing MIMIC geometry; real JPEGs require --camera-profile");
    799 }
    800 const bool xyz_geometry = provenance.value("tile_scheme", "") == "XYZ";
    801 // Older MIMIC EXIF used GOOGLE as a cache-layout token, not imagery provenance.
    802 const bool legacy_cache_geometry = provenance.value("provider", "") == "GOOGLE";
    803 photo_check(
    804 provenance.is_object() &&
    805 provenance.value("schema", "") == "USHER_MOCK_MAP_V1" &&
    806 provenance.value("projection", "") == "pinhole_flat_ground" &&
    807 (xyz_geometry || legacy_cache_geometry) &&
    808 provenance.value("zoom", 0) == 18,
    809 "only clean USHER MIMIC pinhole photos are supported");
    810 } else if (!camera.is_null()) {
    811 photo_check(document.find("USHER_MOCK_MAP_V1") == std::string::npos,
    812 "mock photo in real-camera input");
    813 photo_check(width == camera.at("width") && height == camera.at("height"),
    814 "JPEG dimensions differ from real-camera profile");
    815 }
    816 photo_check(exif_text(metadata, EXIF_CAMERA, EXIF_USER_COMMENT)
    817 .find("USHER_LENS_V1") == std::string::npos,
    818 "lens derivative cannot be inverted without its shutter/lens "
    819 "model; use clean photo");
    820 exif_value orientation{};
    821 uint32_t orientation_value = 0;
    822 photo_check(!exif_get(metadata, EXIF_PRIMARY, EXIF_ORIENTATION, &orientation) ||
    823 (orientation.format == EXIF_SHORT && orientation.count == 1 &&
    824 exif_value_unsigned(&orientation, 0, &orientation_value) && orientation_value == 1),
    825 "unsupported EXIF orientation");
    826 std::istringstream description(
    827 exif_text(metadata, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION));
    828 std::string field;
    829 std::getline(description, field, ';');
    830 Json shot = Json::object();
    831 const bool flyeye = (standard_profile || !camera.is_null()) &&
    832 field.find("FlyEye simulated camera") == 0;
    833 if (flyeye) {
    834 const Json pose = Json::parse(
    835 exif_text(metadata, EXIF_CAMERA, EXIF_USER_COMMENT));
    836 photo_check(pose.value("source", "") == "FlyEye/MSFS2024" &&
    837 pose.value("datum", "") == "WGS-84",
    838 "unsupported FlyEye position metadata");
    839 shot.set("lat_deg", pose.at("latitude_deg"));
    840 shot.set("lon_deg", pose.at("longitude_deg"));
    841 shot.set("groundalt_m", pose.at("agl_m"));
    842 shot.set("roll_deg", 0);
    843 shot.set("pitch_deg", 0);
    844 shot.set("yaw_deg", 0);
    845 } else {
    846 photo_check(field.find("Paparazzi USHER ") == 0,
    847 "missing USHER shot metadata");
    848 std::map<std::string, std::string> fields;
    849 while (std::getline(description, field, ';')) {
    850 const size_t start = field.find_first_not_of(" \t\r\n"),
    851 end = field.find_last_not_of(" \t\r\n");
    852 photo_check(start != std::string::npos, "empty metadata field");
    853 field = field.substr(start, end - start + 1);
    854 const size_t separator = field.find('=');
    855 photo_check(separator != std::string::npos &&
    856 fields
    857 .emplace(field.substr(0, separator),
    858 field.substr(separator + 1))
    859 .second,
    860 "duplicate or invalid metadata field");
    861 }
    862 for (const char* key : {"lat_deg", "lon_deg", "roll_deg", "pitch_deg",
    863 "yaw_deg", "groundalt_m"}) {
    864 std::string selected = key;
    865 if (camera.is_null() &&
    866 (selected == "lat_deg" || selected == "lon_deg") &&
    867 fields.count("original_" + selected))
    868 selected = "original_" + selected;
    869 photo_check(fields.count(selected), "missing shot field");
    870 try {
    871 size_t consumed = 0;
    872 shot.set(key, std::stod(fields.at(selected), &consumed));
    873 photo_check(consumed == fields.at(selected).size(),
    874 "invalid shot number");
    875 } catch (const std::logic_error&) {
    876 throw InvalidPhoto("invalid shot number");
    877 }
    878 }
    879 }
    880 const double latitude = numeric(shot, "lat_deg", -85, 85) * pi / 180;
    881 const double longitude = numeric(shot, "lon_deg", -180, 180);
    882 const double altitude = numeric(shot, "groundalt_m", 0.1, 10000) /
    883 (projected_pixel_m * cos(latitude));
    884 const double roll = numeric(shot, "roll_deg", -180, 180) * pi / 180;
    885 const double pitch = numeric(shot, "pitch_deg", -180, 180) * pi / 180;
    886 const double yaw = numeric(shot, "yaw_deg", -360, 360) * pi / 180;
    887 cv::Matx33d rays, mount = cv::Matx33d::eye();
    888 if (camera.is_null()) {
    889 const double horizontal =
    890 standard_profile ? tan(60 * pi / 360) :
    891 tan(numeric(provenance, "horizontal_fov_deg", 1, 170) * pi / 360);
    892 const double vertical =
    893 standard_profile ? horizontal * (height - 1) / (width - 1) :
    894 tan(numeric(provenance, "vertical_fov_deg", 1, 170) * pi / 360);
    895 rays = cv::Matx33d(0, -2 * vertical / (height - 1), vertical,
    896 2 * horizontal / (width - 1), 0, -horizontal, 0, 0, 1);
    897 } else {
    898 const double fx = camera.at("fx").get<double>(), fy = camera.at("fy").get<double>(),
    899 cx = camera.at("cx").get<double>(), cy = camera.at("cy").get<double>();
    900 rays = cv::Matx33d(0, -1 / fy, cy / fy, 1 / fx, 0, -cx / fx, 0, 0, 1);
    901 mount = rotation(camera.at("mount_roll_deg").get<double>() * pi / 180,
    902 camera.at("mount_pitch_deg").get<double>() * pi / 180,
    903 camera.at("mount_yaw_deg").get<double>() * pi / 180);
    904 }
    905 const cv::Matx33d ground = rotation(roll, pitch, yaw) * mount * rays;
    906 const double center_x = (longitude + 180) / 360 * world_pixels;
    907 const double center_y =
    908 (1 - asinh(tan(latitude)) / pi) * 0.5 * world_pixels;
    909 to_world =
    910 cv::Matx33d(0, altitude, center_x, -altitude, 0, center_y, 0, 0, 1) *
    911 ground;
    912 update_bounds(width, height);
    913 }

    915 /** @brief Recompute the tile footprint after setting the image homography.
    916 * @param width Source columns in pixels.
    917 * @param height Source rows in pixels.
    918 * @details Check homogeneous depth before dividing. Reject horizon,
    919 * dateline and oversized footprints instead of creating unbounded tile work.
    920 * @throws InvalidPhoto If the candidate footprint violates a mapping limit. */
    921 void update_bounds(int width, int height) {
    922 double minimum_x = INFINITY, minimum_y = INFINITY, maximum_x = -INFINITY,
    923 maximum_y = -INFINITY;
    924 for (int row : {0, height - 1})
    925 for (int column : {0, width - 1}) {
    926 const cv::Vec3d point = to_world * cv::Vec3d(column, row, 1);
    927 photo_check(point[2] > 0.01, "footprint reaches the horizon");
    928 const double world_x = point[0] / point[2],
    929 world_y = point[1] / point[2];
    930 photo_check(std::isfinite(world_x) && std::isfinite(world_y) &&
    931 world_x >= 0 && world_y >= 0 &&
    932 world_x < world_pixels && world_y < world_pixels,
    933 "footprint crosses Mercator/dateline boundary");
    934 minimum_x = std::min(minimum_x, world_x);
    935 maximum_x = std::max(maximum_x, world_x);
    936 minimum_y = std::min(minimum_y, world_y);
    937 maximum_y = std::max(maximum_y, world_y);
    938 }
    939 bounds = {
    940 int(floor(minimum_x / tile_size)), int(floor(minimum_y / tile_size)),
    941 int(floor(maximum_x / tile_size)), int(floor(maximum_y / tile_size))};
    942 photo_check(bounds.right - bounds.left < 16 &&
    943 bounds.bottom - bounds.top < 16 &&
    944 (bounds.right - bounds.left + 1) *
    945 (bounds.bottom - bounds.top + 1) <=
    946 64,
    947 "photo footprint exceeds 64-tile work limit");
    948 }

    981/** @brief Admit and decode one source JPEG into a projection-ready photo.
    982 * @param path Source JPEG path.
    983 * @param resources Resource and cancellation policy.
    984 * @param camera Camera calibration/profile fields.
    985 * @param standard_profile Whether the standard camera profile is active.
    986 * @param parallel_decode Whether to use reusable per-worker decode state.
    987 * @return Decoded image with its transform and tile footprint.
    988 * @details Reuse decoder storage only within a worker thread; discard that
    989 * state after fatal libjpeg recovery rather than reusing a failed codec. */
    990Photo read_photo(const fs::path& path, Resources& resources,
    991 const Json& camera, bool standard_profile = false,
    992 bool parallel_decode = false) {
    993 File input = regular_file(path, 32 * mib);
    994 static thread_local std::unique_ptr<Decoder> cached_decoder;
    995 std::unique_ptr<Decoder> local_decoder;
    996 if (parallel_decode) {
    997 if (!cached_decoder) cached_decoder = std::make_unique<Decoder>();
    998 } else local_decoder = std::make_unique<Decoder>();
    999 Decoder* const volatile decoder = parallel_decode ? cached_decoder.get() : local_decoder.get();
    1000 decoder->state.err = jpeg_std_error(&decoder->error.manager);
    1001 decoder->error.manager.error_exit = jpeg_failure;
    1002 decoder->error.emit = decoder->error.manager.emit_message;
    1003 decoder->error.manager.emit_message = jpeg_message;
    1004 if (setjmp(decoder->error.recovery)) {
    1005 if (parallel_decode) cached_decoder.reset();
    1006 throw InvalidPhoto("invalid JPEG stream");
    1007 }
    1008 if (decoder->created) {
    1009 jpeg_abort_decompress(&decoder->state);
    1010 decoder->metadata = {};
    1011 decoder->photo = {};
    1012 } else {
    1013 decoder->created = true;
    1014 jpeg_create_decompress(&decoder->state);
    1015 }
    1016 jpeg_stdio_src(&decoder->state, input.get());
    1017 jpeg_save_markers(&decoder->state, JPEG_APP0 + 1, 65533);
    1018 photo_check(jpeg_read_header(&decoder->state, TRUE) == JPEG_HEADER_OK,
    1019 "invalid JPEG header");
    1020 const unsigned width = decoder->state.image_width,
    1021 height = decoder->state.image_height;
    1022 photo_check(width >= 2 && height >= 2 && width <= 4096 && height <= 4096 &&
    1023 uint64_t(width) * height <=
    1024 (camera.is_null() ? 8388608u : 16777216u) &&
    1025 !decoder->state.progressive_mode,
    1026 "unsupported JPEG size or progressive encoding");
    1027 for (jpeg_saved_marker_ptr marker = decoder->state.marker_list; marker;
    1028 marker = marker->next) {

    1063/** Per-tile four-channel sums (RGB weighted totals and accumulated feather
    1064 * weight). */
    1065/** Per-tile RGB weighted sums plus total feather weight, stored for resumable
    1066 * blending. */
    1067struct Tile {
    1068 uint64_t sequence = 0;
    1069 cv::Mat sums{tile_size, tile_size, CV_32FC4, cv::Scalar::all(0)};
    1070 /** Load and verify an existing tile; absent files represent zero coverage. */
    1071 void load(const fs::path& path) {
    1072 if (!fs::exists(path)) return;
    1073 File input = regular_file(path, sums.total() * sums.elemSize() + 512);
    1074 std::array<char, 512> header{};
    1075 check(fgets(header.data(), header.size(), input.get()) != nullptr,
    1076 "tile header missing");
    1077 Json record;
    1078 try {
    1079 record = Json::parse(header.data());
    1080 } catch (const Json::exception& error) {
    1081 throw std::runtime_error("invalid tile header " + path.string() + ": " + error.what());
    1082 }
    1083 check(record.value("schema", "") == "PANGEA_TILE_V1",
    1084 "invalid tile format");
    1085 sequence = record.at("sequence").get<uint64_t>();
    1086 check(sequence <= max_photos, "invalid tile sequence");
    1087 const size_t length = sums.total() * sums.elemSize();
    1088 check(fread(sums.data, 1, length, input.get()) == length &&
    1089 fgetc(input.get()) == EOF && !ferror(input.get()),
    1090 "invalid tile data length");
    1091 check(record.at("sha256") == digest(sums.data, length),
    1092 "tile checksum mismatch");
    1093 check(cv::checkRange(sums, true, nullptr, 0, 255.0 * max_photos + 1),
    1094 "invalid tile accumulation");
    1095 }
    1096 /** Atomically persist tile data and sequence so a pending photo can be
    1097 * replayed safely. */
    1098 void save(const fs::path& path) {
    1099 const size_t length = sums.total() * sums.elemSize();
    1100 AtomicFile output(path);
    1101 output.text(Json::object({{"schema", "PANGEA_TILE_V1"},
    1102 {"sequence", sequence},
    1103 {"sha256", digest(sums.data, length)}})
    1104 .dump() +
    1105 "\n");
    1106 output.write(sums.data, length);
    1107 output.commit();
    1108 }

    1130/** Blend one validated photo into intersecting tiles, checkpointing each tile
    1131 * atomically.
    1132 * @param photo Source pixels, projection, and affected tile bounds.
    1133 * @param root Tile/state directory.
    1134 * @param sequence Unique accepted-photo sequence used to prevent duplicate
    1135 * contribution.
    1136 * @param resources Cooperative CPU/RAM/pause policy.
    1137 * @param tile_column Independent destination tile column.
    1138 * @param tile_row Independent destination tile row. */
    1139void blend_tile(const Photo& photo, const fs::path& root, uint64_t sequence,
    1140 Resources& resources, int tile_column, int tile_row) {
    1141 resources.tile_checkpoint();
    1142 disk_headroom(root, 2 * mib);
    1143 const cv::Matx33d local = cv::Matx33d(1, 0, -tile_column * tile_size, 0,
    1144 1, -tile_row * tile_size, 0, 0, 1) *
    1145 photo.to_world;
    1146 const auto footprint = photo.corners(local);
    1147 const std::array<cv::Point2f, 4> rectangle{
    1148 {{0, 0},
    1149 {tile_size - 1, 0},
    1150 {tile_size - 1, tile_size - 1},
    1151 {0, tile_size - 1}}};
    1152 std::vector<cv::Point2f> intersection;
    1153 if (cv::intersectConvexConvex(footprint, rectangle, intersection) <= 0)
    1154 return;
    1155 const fs::path path = tile_path(root, tile_column, tile_row);
    1156 Tile tile;
    1157 tile.load(path);
    1158 check(tile.sequence <= sequence, "tile newer than pending photo");
    1159 if (tile.sequence == sequence) return;
    1160 bool contributed = false;
    1161 // Process 16-row strips so warp temporaries stay small on the 512 MiB
    1162 // board.
    1163 for (int start_row = 0; start_row < tile_size; start_row += 16) {
    1164 resources.tile_checkpoint();
    1165 const cv::Matx33d transform =
    1166 cv::Matx33d(1, 0, -tile_column * tile_size, 0, 1,
    1167 -tile_row * tile_size - start_row, 0, 0, 1) *
    1168 photo.to_world;
    1169 const cv::Matx33d inverse = transform.inv();
    1170 cv::Mat warped;
    1171 cv::warpPerspective(photo.image, warped, transform,
    1172 cv::Size(tile_size, 16), cv::INTER_LINEAR,
    1173 cv::BORDER_CONSTANT);
    1174 for (int row = 0; row < warped.rows; ++row) {
    1175 auto* sums = tile.sums.ptr<cv::Vec4f>(start_row + row);
    1176 const auto* colors = warped.ptr<cv::Vec3b>(row);
    1177 for (int column = 0; column < tile_size; ++column) {
    1178 const cv::Vec3d source = inverse * cv::Vec3d(column, row, 1);
    1179 if (source[2] <= 0) continue;
    1180 const double source_x = source[0] / source[2],
    1181 source_y = source[1] / source[2];
    1182 const double edge =
    1183 std::min({source_x, source_y, photo.image.cols - 1 - source_x,
    1184 photo.image.rows - 1 - source_y});
    1185 if (edge < 1) continue;
    1186 // Feather the source edge over 12.5% of its smaller dimension to
    1187 // soften seams.
    1188 const float weight = std::min(
    1189 1.0,
    1190 edge / (0.125 * std::min(photo.image.cols, photo.image.rows)));
    1191 contributed = true;
    1192 for (int channel = 0; channel < 3; ++channel)
    1193 sums[column][channel] += colors[column][channel] * weight;
    1194 sums[column][3] += weight;
    1195 }
    1196 }
    1197 }
    1198 if (contributed) {
    1199 tile.sequence = sequence;
    1200 tile.save(path);
    1201 }
    1202}

    1297 photo.image.cols - 1 - source_x, photo.image.rows - 1 - source_y});
    1298 if (edge < 1) continue;
    1299 weights[column] = std::min(1.0,
    1300 edge / (0.125 * std::min(photo.image.cols, photo.image.rows)));
    1301 contributed = true;
    1302 }
    1303 }
    1304 }
    1305 if (contributed) prepared.tiles.push_back(std::move(tile));
    1306 }
    1307 }

    1525 * publication. */

    1527 fs::path input, root;
    1528 Resources& resources;
    1529 Json state;
    1530 pangea_survey::StorageDescriptor output_lock, process_lock;

    1538 /** @brief Extract bounded SIFT features for a possible alignment reference.
    1539 * @param photo Source image and initial world transform.
    1540 * @param name Source name for the alignment record.
    1541 * @return Descriptors and keypoints expressed in original image pixels.
    1542 * @details Detect on reduced, border-masked imagery to limit feature work;
    1543 * restore original coordinates before fitting geometry. */
    1544 AlignmentReference features(const Photo& photo, const std::string& name) {
    1545 struct Workspace {
    1546 cv::Mat resized, gray, mask;
    1547 cv::Ptr<cv::SIFT> detector = cv::SIFT::create(4000);
    1548 };
    1549 static thread_local Workspace workspace;
    1550 const double scale = std::min(1.0, 1024.0 / std::max(photo.image.cols, photo.image.rows));
    1551 cv::resize(photo.image, workspace.resized, cv::Size(), scale, scale, cv::INTER_AREA);
    1552 cv::cvtColor(workspace.resized, workspace.gray, cv::COLOR_RGB2GRAY);
    1553 workspace.mask.create(workspace.gray.size(), CV_8UC1);
    1554 workspace.mask.setTo(0);
    1555 const int margin = std::max(1, int(40 * scale));
    1556 workspace.mask(cv::Rect(margin, 2 * margin, workspace.gray.cols - 2 * margin,
    1557 workspace.gray.rows - 3 * margin)).setTo(255);
    1558 AlignmentReference current;
    1559 current.width = photo.image.cols;
    1560 current.height = photo.image.rows;
    1561 current.name = name;
    1562 current.to_world = photo.to_world;
    1563 current.position_prior = photo.to_world;
    1564 workspace.detector->detectAndCompute(workspace.gray, workspace.mask,
    1565 current.keypoints, current.descriptors);
    1566 for (auto& point : current.keypoints) point.pt *= float(1 / scale);
    1567 return current;
    1568 }
    1569
    1570 /** @brief Try to refine a photo's georeferenced transform using recent photos.
    1571 * @param[in,out] photo Source whose transform may be updated.
    1572 * @param name Source name for evidence and reference tracking.
    1573 * @param[in,out] cached Optional precomputed features, moved from when used.
    1574 * @return Alignment decision and available supporting evidence.
    1575 * @details Use position priors to screen candidate references; keep the
    1576 * original projection if no feature match passes the geometric checks. */
    1577 Json align(Photo& photo, const std::string& name, ImageFeatures* cached = nullptr) {
    1578 Json result = Json::object({{"status", "gps_fallback"}});
    1579 const int margin = 40;
    1580 if (photo.image.cols <= 2 * margin || photo.image.rows <= 3 * margin)
    1581 return Json::object({{"status", "image_too_small"}});
    1582 AlignmentReference current = cached ? std::move(*cached) : features(photo, name);
    1583 size_t best_inliers = 0;
    1584 cv::Matx33d selected = photo.to_world;
    1585 const cv::Vec3d center_pixel((photo.image.cols - 1) / 2.0,
    1586 (photo.image.rows - 1) / 2.0, 1);
    1587 const cv::Vec3d predicted_center = photo.to_world * center_pixel;
    1588 struct Candidate {
    1589 size_t inliers = 0;
    1590 cv::Matx33d to_world;
    1591 Json evidence;
    1592 };
    1593 std::vector<Candidate> candidates(references.size());
    1594 resources.parallel_tiles(references.size(), [&](size_t candidate_index) {
    1595 const auto* reference = &references[references.size() - 1 - candidate_index];
    1596 if (current.descriptors.empty() || reference->descriptors.empty()) return;
    1597 const cv::Vec3d relative = reference->position_prior.inv() * predicted_center;
    1598 if (relative[2] <= 0 || relative[0] / relative[2] < -reference->width * 0.25 ||
    1599 relative[0] / relative[2] > reference->width * 1.25 ||
    1600 relative[1] / relative[2] < -reference->height * 0.25 ||
    1601 relative[1] / relative[2] > reference->height * 1.25) return;
    1602 resources.tile_checkpoint();
    1603 struct Workspace {
    1604 cv::BFMatcher matcher{cv::NORM_L2};
    1605 std::vector<std::vector<cv::DMatch>> matches;
    1606 std::vector<cv::Point2f> source, target, supported;
    1607 std::vector<double> errors;
    1608 cv::Mat inliers;
    1609 };
    1610 static thread_local Workspace workspace;
    1611 auto& matches = workspace.matches;
    1612 auto& source = workspace.source;
    1613 auto& target = workspace.target;
    1614 auto& supported = workspace.supported;
    1615 auto& errors = workspace.errors;
    1616 matches.clear(); source.clear(); target.clear();
    1617 supported.clear(); errors.clear();
    1618 workspace.matcher.knnMatch(current.descriptors, reference->descriptors, matches, 2);
    1619 for (const auto& pair : matches) {
    1620 if (pair.size() != 2 || pair[0].distance >= 0.7f * pair[1].distance) continue;
    1621 source.push_back(current.keypoints[pair[0].queryIdx].pt);
    1622 target.push_back(reference->keypoints[pair[0].trainIdx].pt);
    1623 }
    1624 if (source.size() < 30) return;
    1625 cv::Mat& inliers = workspace.inliers;
    1626 const cv::Mat affine = cv::estimateAffinePartial2D(
    1627 source, target, inliers, cv::RANSAC, 3, 2000, 0.995, 10);
    1628 const int count = cv::countNonZero(inliers);
    1629 if (affine.empty() || count < 30 || count < source.size() * 0.25 ||
    1630 count <= 0) return;
    1631 const double scale = std::hypot(affine.at<double>(0, 0), affine.at<double>(1, 0));
    1632 const double angle = std::atan2(affine.at<double>(1, 0),
    1633 affine.at<double>(0, 0)) * 180 / pi;
    1634 if (!std::isfinite(scale) || scale < 0.9 || scale > 1.1 || std::abs(angle) > 5) return;
    1635 const cv::Matx33d transform(
    1636 affine.at<double>(0, 0), affine.at<double>(0, 1), affine.at<double>(0, 2),
    1637 affine.at<double>(1, 0), affine.at<double>(1, 1), affine.at<double>(1, 2),
    1638 0, 0, 1);
    1639 for (size_t index = 0; index < source.size(); ++index) {
    1640 if (!inliers.at<uint8_t>(static_cast<int>(index))) continue;
    1641 const cv::Vec3d point = transform * cv::Vec3d(source[index].x, source[index].y, 1);
    1642 supported.push_back(source[index]);
    1643 errors.push_back(std::hypot(point[0] - target[index].x, point[1] - target[index].y));
    1644 }
    1645 const cv::Rect support = cv::boundingRect(supported);
    1646 if (support.width < photo.image.cols * 0.15 ||
    1647 support.height < photo.image.rows * 0.15) return;
    1648 std::sort(errors.begin(), errors.end());
    1649 if (errors[errors.size() / 2] > 2) return;
    1650 const cv::Matx33d candidate = reference->to_world * transform;
    1651 const cv::Vec3d candidate_center = candidate * center_pixel;
    1652 const double correction = std::hypot(candidate_center[0] - predicted_center[0],
    1653 candidate_center[1] - predicted_center[1]);
    1654 const cv::Vec3d matched_center = transform * center_pixel;
    1655 const double local_distance = std::hypot(
    1656 matched_center[0] - (reference->width - 1) / 2.0,
    1657 matched_center[1] - (reference->height - 1) / 2.0);
    1658 if (!std::isfinite(correction) ||
    1659 local_distance > std::hypot(reference->width, reference->height) * 1.25)
    1660 return;
    1661 Candidate& match = candidates[candidate_index];
    1662 match.inliers = count;
    1663 match.to_world = candidate;
    1664 match.evidence = Json::object({{"status", "registered"}, {"reference", reference->name},
    1665 {"inliers", count}, {"median_error_px", errors[errors.size() / 2]},
    1666 {"center_correction_map_px", correction}, {"relative_scale", scale}});
    1667 });
    1668 for (const auto& candidate : candidates) {
    1669 if (candidate.inliers <= best_inliers) continue;
    1670 best_inliers = candidate.inliers;
    1671 selected = candidate.to_world;
    1672 result = candidate.evidence;
    1673 }
    1674 photo.to_world = selected;
    1675 photo.update_bounds(photo.image.cols, photo.image.rows);
    1676 current.to_world = photo.to_world;
    1677 if (references.size() == 8) references.erase(references.begin());
    1678 references.push_back(std::move(current));
    1679 return result;
    1680 }

    1700 /** Validate one discovered source and checkpoint its blend as an idempotent
    1701 * job. */
    1702 void process(const Json& job, PreparedPhoto* prepared = nullptr,
    1703 std::exception_ptr prepared_error = nullptr) {
    1704 const fs::path source = input / job.at("name").get<std::string>();
    1705 check(source.parent_path() == input, "invalid source path");
    1706 if (!unchanged(source, job)) return;
    1707 Photo photo;
    1708 try {
    1709 try {
    1710 if (prepared) photo = std::move(prepared->photo);
    1711 else if (prepared_error) std::rethrow_exception(prepared_error);
    1712 else photo = read_photo(source, resources, state.value("camera", Json()),
    1713 state.value("standard_profile", false));
    1714 } catch (const InvalidPhoto& error) {
    1715 if (!allow_prompt || profile_asked ||
    1716 std::string(error.what()) !=
    1717 "missing MIMIC geometry; real JPEGs require --camera-profile")
    1718 throw;
    1719 profile_asked = true;
    1720 std::cerr << "PANGEA: " << error.what() << '\n'
    1721 << "Standard profile assumes 60-degree horizontal FOV, "
    1722 "centered square pixels and nadir mounting; FlyEye "
    1723 "images assume north-up, zero roll/pitch. "
    1724 "Map geometry is approximate, not calibrated.\n"
    1725 << "use standard profile Y/n: " << std::flush;
    1726 std::string answer;
    1727 if (!std::getline(std::cin, answer) ||
    1728 !(answer.empty() || answer == "Y" || answer == "y"))
    1729 throw;
    1730 state.set("standard_profile", true);
    1731 state.set("assumed_geometry", Json::object({
    1732 {"horizontal_fov_deg", 60}, {"mount", "nadir"},
    1733 {"flyeye_roll_deg", 0}, {"flyeye_pitch_deg", 0},
    1734 {"flyeye_yaw_deg", 0}, {"accuracy", "uncalibrated"}}));
    1735 save();
    1736 photo = read_photo(source, resources, Json(), true);
    1737 }
    1738 } catch (const std::exception& error) {
    1739 if (!unchanged(source, job)) return;
    1740 if (dynamic_cast<const InvalidPhoto*>(&error) == nullptr) throw;
    1741 check(state.at("pending").is_null(),
    1742 "pending photo no longer decodes; checkpoint retained");
    1743 write_json(
    1744 root / "done" / (job.at("identity").get<std::string>() + ".json"),
    1745 Json::object({{"status", "rejected"}, {"source", job}, {"reason", error.what()}}));
    1746 state.set("rejected", state.at("rejected").get<unsigned>() + 1);
    1747 save();
    1748 std::cerr << "PANGEA: skipped " << source.filename() << ": "
    1749 << error.what() << '\n';
    1750 return;
    1751 }
    1752 if (!unchanged(source, job)) return;
    1753 Json alignment;
    1754 if (align_images) {
    1755 if (!state.at("pending").is_null() &&
    1756 state.at("pending").contains("registered_to_world")) {
    1757 const Json saved = state.at("pending").at("registered_to_world");
    1758 const cv::Matx33d position_prior = photo.to_world;
    1759 check(saved.is_array() && saved.size() == 9, "invalid saved alignment");
    1760 for (int index = 0; index < 9; ++index) {
    1761 photo.to_world.val[index] = saved[index].get<double>();
    1762 check(std::isfinite(photo.to_world.val[index]), "nonfinite saved alignment");
    1763 }
    1764 photo.update_bounds(photo.image.cols, photo.image.rows);
    1765 alignment = state.at("pending").at("alignment");
    1766 if (photo.image.cols > 80 && photo.image.rows > 120) {
    1767 if (references.size() == 8) references.erase(references.begin());
    1768 auto reference = features(photo, job.at("name").get<std::string>());
    1769 reference.position_prior = position_prior;
    1770 references.push_back(std::move(reference));
    1771 }
    1772 } else alignment = align(photo, job.at("name").get<std::string>(),
    1773 prepared ? prepared->features.get() : nullptr);
    1774 if (prepared) {
    1775 prepared->photo = photo;
    1776 prepared->tiles.clear();
    1777 prepare_projection(*prepared, resources);
    1778 }
    1779 }
    1780 Bounds combined;
    1781 try {
    1782 combined = Bounds::from(state.at("bounds"), max_map_side)
    1783 .include(photo.bounds, max_map_side);
    1784 } catch (const std::exception&) {
    1785 throw std::runtime_error(
    1786 "map extent limit reached; resume with a larger --max-map-side "
    1787 "for standalone stitching, or use a new output for another area");
    1788 }
    1789 const uint64_t sequence = state.at("sequence").get<uint64_t>() + 1;
    1790 check(sequence <= max_photos, "10000-photo limit reached; start a new map");
    1791 disk_headroom(root, 64 * mib);
    1792 // Persist intent before touching tiles so a restart can replay partial tile
    1793 // updates.
    1794 if (state.at("pending").is_null()) {
    1795 state.set("pending", job);
    1796 if (align_images) {
    1797 Json transform = Json::array();
    1798 for (double value : photo.to_world.val) transform.push_back(value);
    1799 Json pending = state.at("pending");
    1800 pending.set("registered_to_world", transform);
    1801 pending.set("alignment", alignment);
    1802 state.set("pending", pending);
    1803 }
    1804 save();
    1805 }
    1806 if (prepared) blend_prepared(*prepared, root, sequence, resources);
    1807 else blend(photo, root, sequence, resources);
    1808 Json completed = Json::object({{"status", "assembled"},
    1809 {"source", job},
    1810 {"sequence", sequence},
    1811 {"corners_lon_lat", photo.geographic_corners()}});
    1812 if (align_images) completed.set("alignment", alignment);
    1813 write_json(root / "done" / (job.at("identity").get<std::string>() + ".json"), completed);
    1814 state.set("bounds", combined.json());
    1815 state.set("sequence", sequence);
    1816 state.set("pending", nullptr);
    1817 if (align_images) {
    1818 Json saved = Json::array();
    1819 for (const auto& reference : references) {
    1820 Json transform = Json::array();
    1821 for (double value : reference.to_world.val) transform.push_back(value);
    1822 saved.push_back(Json::object({{"name", reference.name},
    1823 {"identity", identity(input / reference.name)}, {"to_world", transform}}));
    1824 }
    1825 state.set("alignment_references", saved);
    1826 }
    1827 save();
    1828 std::cout << "PANGEA: assembled " << source.filename() << " (#" << sequence
    1829 << ")\n";
    1830 }

    1905 check(std::isfinite(photo.to_world.val[index]), "nonfinite reference transform");
    1906 }

    1914 /** Process stable jobs; in watch mode, defer changing files and export
    1915 * periodically. */
    1916 void run(bool watch) {
    1917 double exported_at = seconds(CLOCK_MONOTONIC);
    1918 const auto export_due = [&](bool force) {
    1919 if (state.at("exported") != state.at("sequence") &&
    1920 (force || (watch && seconds(CLOCK_MONOTONIC) - exported_at >= 10))) {
    1921 export_map(root, state, resources, max_map_side);
    1922 state.set("exported", state.at("sequence"));
    1923 save();
    1924 exported_at = seconds(CLOCK_MONOTONIC);
    1925 }
    1926 };
    1927 std::map<std::string, std::pair<std::string, double>> observed;
    1928 while (!stopping) {
    1929 resources.checkpoint();
    1930 if (!state.at("pending").is_null()) {
    1931 process(state.at("pending"));
    1932 export_due(false);
    1933 continue;
    1934 }
    1935 Directory directory(opendir(input.c_str()));
    1936 check(bool(directory), "input directory unavailable");
    1937 std::vector<Json> jobs;
    1938 jobs.reserve(64);
    1939 std::map<std::string, std::pair<std::string, double>> next_observed;
    1940 size_t entries = 0;
    1941 while (dirent* entry = readdir(directory.get())) {
    1942 // Check cancellation/headroom during very large directory scans without
    1943 // a syscall per entry.
    1944 if (++entries % 64 == 0) resources.checkpoint();
    1945 check(entries <= 100000, "input directory entry limit reached");
    1946 const std::string name = entry->d_name;
    1947 if (fs::path(name).extension() != ".jpg" ||
    1948 name.find(".lens.") != std::string::npos)
    1949 continue;
    1950 if (!state.value("camera", Json()).is_null() &&
    1951 name.rfind(
    1952 state.at("camera").at("filename_prefix").get<std::string>(),
    1953 0) != 0)
    1954 continue;
    1955 struct stat status{};
    1956 if (lstat((input / name).c_str(), &status) != 0 ||
    1957 !S_ISREG(status.st_mode) ||
    1958 status.st_mtim.tv_sec <
    1959 static_cast<time_t>(state.at("since").get<uint64_t>()))
    1960 continue;
    1961 const std::string key = identity(input / name, true);
    1962 if (key.empty()) continue;
    1963 if (fs::exists(root / "done" / (key + ".json"))) continue;
    1964 if (watch) {
    1965 const auto previous = observed.find(name);
    1966 const double now = seconds(CLOCK_MONOTONIC);
    1967 const double stable_since =
    1968 previous != observed.end() && previous->second.first == key
    1969 ? previous->second.second
    1970 : now;
    1971 next_observed.emplace(name, std::make_pair(key, stable_since));
    1972 // Require two stable seconds so a producer cannot expose a
    1973 // half-written JPEG.
    1974 if (now - stable_since < 2) {
    1975 if (next_observed.size() == 64) break;
    1976 continue;
    1977 }
    1978 }
    1979 jobs.push_back(Json::object({{"name", name}, {"identity", key}}));
    1980 if (jobs.size() == 64 || next_observed.size() == 64) break;
    1981 }
    1982 observed = std::move(next_observed);
    1983 if (!jobs.empty()) {
    1984 std::sort(jobs.begin(), jobs.end(),
    1985 [](const Json& first, const Json& second) {
    1986 return first.at("name") < second.at("name");
    1987 });
    1988 if (!resources.desktop()) {
    1989 for (const Json& job : jobs) {
    1990 check(state.at("sequence").get<unsigned>() +
    1991 state.at("rejected").get<unsigned>() < max_photos,
    1992 "photo record limit reached; start a new output");
    1993 process(job);
    1994 export_due(false);
    1995 }
    1996 } else {
    1997 for (size_t batch_start = 0; batch_start < jobs.size();) {
    1998 const size_t batch_size =
    1999 std::min(resources.photo_batch_capacity(align_images ? 256 : 128),
