    36enum class FrameStatus { ready, startup, failed };
    37/** @brief Map libcamera metadata to this worker's frame disposition.
    38 * @param status Driver-reported frame status.
    39 * @return Ready, startup, or failed.
    40 * @details Startup frames are not usable captures, but unlike failures
    41 * they are expected while the camera pipeline settles. */
    42constexpr FrameStatus frame_status(libcamera::FrameMetadata::Status status)
    43{
    44 return status == libcamera::FrameMetadata::FrameSuccess ? FrameStatus::ready
    45 : status == libcamera::FrameMetadata::FrameStartup ? FrameStatus::startup
    46 : FrameStatus::failed;
    47}
    48}

    57/** @brief Physical IMX500 array width, used to reject an unexpected sensor geometry. */
    58constexpr unsigned sensor_width = 4056;
    59/** @brief Physical IMX500 array height; model input dimensions are separate policy. */
    60constexpr unsigned sensor_height = 3040;
    61
    62/** @brief Check that capture dimensions meet the worker's raster policy.
    63 * @param width Requested columns.
    64 * @param height Requested rows.
    65 * @return True for an even raster within the physical array and minimum dimensions.
    66 * @details Check here before driver setup; this does not guarantee that the
    67 * driver will accept a specific format or stream configuration. */
    68constexpr bool valid_capture_dimensions(unsigned width, unsigned height)
    69{
    70 return width >= 16 && width <= sensor_width && height >= 2 && height <= sensor_height
    71 && width % 2 == 0 && height % 2 == 0;
    72}

    74/** @brief Borrowed same-exposure evidence, without object or navigation semantics.
    75 * @details Views avoid copying neural outputs into another queue. Keep the
    76 * associated lease alive while reading any tensor or interpreting its crop. */
    77struct FrameView {
    78 Span<const float> tensor; /**< Borrowed output values from the retained request. */
    79 Span<const uint8_t> tensor_info; /**< Matching driver shape metadata, not model semantics. */
    80 libcamera::Rectangle crop{}; /**< Sensor-space crop used by consumers such as box projection. */
    81 libcamera::Size image{}; /**< Configured JPEG dimensions, not the neural model input size. */
    82 uint64_t sequence = 0; /**< Camera request sequence; deliberately distinct from an FC shot number. */
    83 uint64_t sensor_timestamp_ns = 0; /**< Sensor timestamp in its camera clock domain; zero when absent. */
    84 int64_t exposure_us = -1; /**< Reported exposure duration; negative means unavailable. */
    85 int64_t frame_duration_us = -1; /**< Reported frame duration; negative means unavailable. */
    86 uint64_t monotonic_us = 0;
    87 uint64_t time_error_us = 100001;
    88 bool timing_sampled = false;
    89 bool crop_valid = false; /**< Distinguishes a reported crop from zero-initialized placeholder geometry. */
    90};

    93/** @brief Confirm that a frame carries no neural output.
    94 * @param frame Borrowed view of the completed request.
    95 * @return True only if both tensor data and shape metadata are absent.
    96 * @details Check both views so partially exposed inference metadata is not
    97 * mistaken for a deliberately photo-only request. */
    98inline bool photo_only_frame(const FrameView &frame)
    99{
    100 return frame.tensor.empty() && frame.tensor_info.empty();
    101}
    102}

    104/** @brief Stack-owned model-specific operations; no dynamic plugin allocation.
    105 * @details validate qualifies readiness. write validates and publishes its own
    106 * analysis file; false or an exception must leave no partial analysis file.
    107 * Keep this contract small: capture mechanics should not learn detector boxes,
    108 * LOOM feature semantics or route-selection policy. */
    109struct FrameProcessor {
    110 const char *sidecar_suffix; /**< Trusted application-owned suffix with a lifetime covering the call. */
    111 bool (*validate)(const FrameView &frame); /**< Qualify model-specific readiness without retaining borrowed data. */
    112 bool (*write)(const FrameView &frame, const char *filename); /**< Exclusive-create analysis publication; failure must clean up its own partial file. */
    113 bool (*embed)(const FrameView &frame, exif_document *document) = nullptr;
    114};

    116/** @brief Exactly-once release of a borrowed frame, including exception exits.
    117 * @details The capture owner must outlive its lease. Only one lease per Camera
    118 * may exist. Move transfers ownership; copies and move assignment are disabled. */
    119class FrameLease {
    120public:
    121 /** @brief Nonthrowing release callback; a destructor must never unwind through it. */
    122 using Release = void (*)(void *) noexcept;
    123
    124 /** @brief Associate borrowed evidence with exactly one release operation.
    125 * @param view Evidence whose backing storage is retained by @p owner.
    126 * @param owner Borrowed capture-owner context, alive until release completes.
    127 * @param release Callback to recycle the held request, or NULL for an unowned view.
    128 * @details Camera constructs leases before interpreting metadata so failures
    129 * during extraction also release the request. */
    130 FrameLease(FrameView view, void *owner, Release release) noexcept;
    131 /** @brief Release once on every scope exit, including a processing exception. */
    132 ~FrameLease();
    133 /** @brief Transfer release responsibility and empty the moved-from view.
    134 * @param other Lease to consume; it will no longer recycle the request. */
    135 FrameLease(FrameLease &&other) noexcept;
    136 /** @brief Copies are forbidden because two destructors cannot own one request. */
    137 FrameLease(const FrameLease &) = delete;
    138 /** @brief Copy assignment cannot duplicate capture ownership. */
    139 FrameLease &operator=(const FrameLease &) = delete;
    140 /** @brief Assignment is disabled; move construction is the supported transfer. */
    141 FrameLease &operator=(FrameLease &&) = delete;
    142
    143 /** @brief Read evidence only while this unmoved lease remains alive.
    144 * @return Borrowed immutable view; its tensor spans have the lease's lifetime. */
    145 const FrameView &view() const noexcept { return view_; }
    146
    147private:
    148 friend class Camera;
    149 FrameView view_; /**< Borrowed metadata, never a second image/tensor allocation. */
    150 void *owner_; /**< Context whose lifetime must exceed this lease. */
    151 Release release_; /**< Sole remaining release operation, NULL after a move. */
    152};

    154/** @brief One exclusive IMX500 owner with bounded reusable YUV420 buffers.
    155 * @pre Control operations and lease destruction are serialized by their owner;
    156 * the camera's callback thread is synchronized internally. */
    157class Camera {
    158public:
    159 /** @brief Construct unopened camera state; construction does not acquire the sensor. */
    160 Camera();
    161 /** @brief Stop streaming before freeing requests/buffers and releasing the camera.
    162 * @pre Every frame lease has already been destroyed. */
    163 ~Camera();
    164 /** @brief Camera owners cannot be copied. */
    165 Camera(const Camera &) = delete;
    166 /** @brief Assignment cannot replace an acquired sensor owner. */
    167 Camera &operator=(const Camera &) = delete;
    168
    169 /** @brief Acquire one eligible camera and configure bounded capture, uploading an RPK only when requested.
    170 * @param model Readable RPK for IMX500 inference, or NULL for plain pixels.
    171 * @param width Requested JPEG width, not model-input width.
    172 * @param height Requested JPEG height, not model-input height.
    173 * @param photo_only Select IMX500 image-only capture and leave frame duration to libcamera.
    174 * @throws std::runtime_error for unavailable/ambiguous hardware or rejected configuration.
    175 * @details Configuration adjustments that change our expected geometry, YUV420
    176 * layout or buffer budget are rejected rather than silently changing downstream assumptions. */
    177 void open(const char *model, unsigned width = 2028, unsigned height = 1520,
    178 bool photo_only = false);
    179 /** @brief Lease a fresh post-request exposure and its own tensor metadata.
    180 * @return Move-only lease; destroy it before the next capture or camera shutdown.
    181 * @throws std::runtime_error for missing startup, a still-live prior lease,
    182 * queue failure, unavailable clock or frame timeout.
    183 * @details Freshness uses the sensor/BOOTTIME domain, not JPEG download time.
    184 * Model-specific validation remains the processor's responsibility. */
    185 FrameLease capture();
    186 /** @brief Map a leased frame and publish its JPEG plus available evidence.
    187 * @param frame Lease obtained from this camera, still alive and unmoved.
    188 * @param filename New absolute JPEG destination; existing images are not overwritten.
    189 * @param processor Application-owned sidecar policy for this same exposure.
    190 * @return True after JPEG publication, false for rejected layout/path or write failure.
    191 * @details Mapping is scoped separately from the lease: CPU DMA ownership ends
    192 * before the request can be recycled. Missing analysis does not discard good pixels.
    193 * @warning Success is not proof of a useful neural match, geolocation accuracy,
    194 * complete multi-file crash atomicity or parent-directory power-loss durability. */
    195 bool save(const FrameLease &frame, const char *filename, const FrameProcessor &processor);
    196 /** @brief Grant scoped CPU pixel access without transferring DMA/frame ownership.
    197 * @param frame Live lease acquired from this camera.
    198 * @param visitor Callback consuming validated YUV420 planes and stride before mappings are released.
    199 * @param context Borrowed callback state, not retained by the camera.
    200 * @return Callback result, or false for invalid ownership/layout/visitor.
    201 * @pre The visitor must not retain pixel pointers or release the lease during the call.
    202 * @details Prepare a model-sized tensor here, then release the lease before expensive inference. */
    203 bool with_pixels(const FrameLease &frame,
    204 bool (*visitor)(const FrameView &, const std::array<const uint8_t *, 3> &,
    205 unsigned, void *), void *context);
    206
    207private:
    208 struct Impl; /**< Camera/request implementation hidden from model-specific callers. */
    209 std::unique_ptr<Impl> impl_; /**< One owned capture state; libcamera has its own allocation policy. */
    210};

    212/** @brief Encode validated YUV420 planes directly, without full-frame RGB staging.
    213 * @param path New output path; exclusive creation refuses overwrites.
    214 * @param planes Accessible Y, Cb and Cr plane starts, immutable during encoding.
    215 * @param width Positive even image width in pixels.
    216 * @param height Positive even image height in pixels.
    217 * @param stride Even luma byte stride, at least the width rounded up to 16; chroma uses half.
    218 * @return True after encoding, flush, file sync and close; false removes partial output.
    219 * @pre The caller has checked plane lengths and image/stride layout. This low-level
    220 * encoder does not accept arbitrary unchecked external plane pointers. */
    221bool encode_jpeg(const char *path, const std::array<const uint8_t *, 3> &planes,

    224/** @brief Publish caller-validated YUV420 planes and same-exposure evidence.
    225 * @param view Borrowed evidence belonging to the supplied pixels.
    226 * @param planes Accessible YUV420 planes retained by the caller.
    227 * @param stride Luma byte stride; chroma uses half this stride.
    228 * @param filename New absolute JPEG destination.
    229 * @param processor Application's trusted sidecar suffix and writer callbacks.
    230 * @return True when the JPEG is published, regardless of missing analysis;
    231 * false when publication fails. File/processor exceptions may also propagate.
    232 * @pre Plane spans cover the declared image/stride and remain immutable for this
    233 * call. Capture retains its frame lease while invoking this routine.
    234 * @details Analysis failure retains a usable photograph. Publication failure
    235 * removes successfully created sidecars from this operation, never an existing image.
    236 * @warning JPEG and sidecars are not one filesystem transaction. Consumers must
    237 * check file/exposure identity, and directory power-loss durability is not guaranteed. */
    238bool publish_frame(const FrameView &frame, const std::array<const uint8_t *, 3> &planes,
    239 unsigned stride, const char *filename, const FrameProcessor &processor);

    241/** @brief Run the shared probe/capture/READY-SHOT-OK worker with supplied model policy.
    242 * @param argc Argument count for a capture mode, excluding saved-photo matcher modes.
    243 * @param argv Mode, RPK and mode-specific dimensions/destination tokens.
    244 * @param processor Readiness and sidecar callbacks; never interpreted as navigation here.
    245 * @param application Diagnostic prefix selected by the independent entry point.
    246 * @return Zero for success, two for syntax/configuration misuse, one for runtime failure.
    247 * @details The worker announces READY after a fresh frame and, for detection,

    249 * Standard output is protocol-only while diagnostics go to standard error. */
    250int run_worker(int argc, char **argv, const FrameProcessor &processor, const char *application);
