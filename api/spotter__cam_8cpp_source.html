    54/** Physical IMX500 pixel-array dimensions expected by the model's coordinate
    55 * transform. */
    56constexpr unsigned int sensor_width = imx500::sensor_width;
    57/** @brief Sensor-space height, separate from model and encoded-image dimensions. */
    58constexpr unsigned int sensor_height = imx500::sensor_height;
    59/** Input dimensions of the object RPK's absolute-coordinate detection head. */
    60constexpr unsigned int model_input_width = 320;
    61/** @brief This detection head's input height; LOOM does not reuse its box semantics. */
    62constexpr unsigned int model_input_height = 320;
    63/** The model emits boxes, scores, classes, and a detection count in this order.
    64 */
    65constexpr uint32_t expected_output_tensor_count = 4;
    66constexpr uint32_t bbox_values_per_candidate = 4; /**< Left/top/right/bottom values in each candidate. */
    67constexpr uint32_t score_values_before_candidates = 4; /**< Scores follow all box coordinate values. */
    68constexpr uint32_t class_values_before_candidates = 5; /**< Classes follow coordinates and scores. */
    69constexpr uint32_t values_per_candidate_with_score_and_class = 6; /**< Count follows the three candidate arrays. */
    70/** Maximum candidate count permitted by the model's fixed tensor scratch bound.
    71 */
    72constexpr uint32_t maximum_tensor_elements = IMX500_TENSOR_ELEMENT_LIMIT;
    73/** Maximum tensor rank documented by libcamera's Raspberry Pi IMX500 controls.
    74 */
    75constexpr uint32_t maximum_tensor_dimensions = IMX500_TENSOR_MAX_RANK;
    76/** Metadata layout: fixed header, tensor count, then fixed-size tensor records.
    77 */
    78constexpr size_t tensor_info_header_size = IMX500_TENSOR_HEADER_BYTES;
    79constexpr size_t tensor_count_offset =
    80 IMX500_TENSOR_COUNT_OFFSET; /**< Four-byte tensor count follows the header. */
    81constexpr size_t first_tensor_record_offset =
    82 IMX500_TENSOR_FIRST_RECORD_OFFSET; /**< First record follows count. */
    83constexpr size_t tensor_record_size =
    84 IMX500_TENSOR_RECORD_BYTES; /**< Fixed fields followed by sixteen 16-bit dimensions. */
    85constexpr size_t tensor_record_dimensions_offset =
    86 IMX500_TENSOR_DIMENSIONS_OFFSET; /**< Length and rank occupy the first eight bytes. */
    87constexpr size_t tensor_record_dimension_bytes =
    88 IMX500_TENSOR_DIMENSION_BYTES; /**< Each shape entry is an unsigned 16-bit value. */
    89constexpr size_t tensor_record_rank_offset =
    90 IMX500_TENSOR_RANK_OFFSET; /**< Rank follows the four-byte element count. */
    91constexpr size_t tensor_record_length_offset =
    92 IMX500_TENSOR_LENGTH_OFFSET; /**< Element count starts each record. */
    93/** Full IMX500 metadata structure size (16 records, each with up to 16
    94 * dimensions). */
    95constexpr size_t tensor_info_size =
    96 IMX500_TENSOR_INFO_BYTES;
    97/** Current single-class model contract: class zero, confidence 0.6, at most
    98 * five boxes. */
    99constexpr size_t maximum_detections = 5;
    100constexpr uint32_t object_class_index = 0; /**< Class selected by this detector's current model policy. */
    101constexpr float object_confidence_threshold = 0.6f; /**< Reject low-confidence boxes without invalidating the whole head. */
    102/** @brief One accepted object in final still-image coordinates.
    103 * @details The crop projection has already happened; consumers must not apply
    104 * the sensor-to-JPEG transform to these pixel coordinates a second time. */

    109 /** @var spottercam::BasicDetection::y
    110 * @brief Top edge in JPEG pixels. */
    111 /** @var spottercam::BasicDetection::width
    112 * @brief Positive accepted box width in JPEG pixels. */
    113 int x = 0, y = 0, width = 0, height = 0; /**< Positive accepted box height in JPEG pixels. */

    126/** Decode one little-endian 32-bit integer from the driver's tensor-info byte
    127 * array.
    128 * @param data At least four readable bytes.
    129 * @return The unsigned value represented by @p data[0..3]. */
    130uint32_t read_u32(const uint8_t* data) {
    131 return uint32_t(data[0]) | uint32_t(data[1]) << 8 | uint32_t(data[2]) << 16 |
    132 uint32_t(data[3]) << 24;
    133}

    135/** Validate the model's four output tensors and project object boxes into the
    136 * JPEG.
    137 * @param tensor Interleaved boxes followed by scores, classes, and the
    138 * detection count.
    139 * @param info Driver CnnOutputTensorInfo bytes for this same camera request.
    140 * @param crop Sensor crop reported in that request's ScalerCrop metadata.
    141 * @param image Configured still-stream dimensions.
    142 * @return Invalid by default; valid with zero or more boxes only after full
    143 * validation.
    144 * @details This is intentionally model-specific. Raw neural arrays cannot be
    145 * treated as boxes until shape, integral count and coordinate assumptions hold. */
    146BasicResult decode(imx500::Span<const float> tensor,
    147 imx500::Span<const uint8_t> info,
    148 libcamera::Rectangle crop, libcamera::Size image) {
    149 BasicResult result;
    150 if (!info.data() || !tensor.data() || info.size() < tensor_info_size ||
    151 tensor.size() > maximum_tensor_elements ||
    152 read_u32(info.data() + tensor_count_offset) !=
    153 expected_output_tensor_count ||
    154 crop.x < 0 || crop.y < 0 || !crop.width || !crop.height ||
    155 crop.width > sensor_width || crop.height > sensor_height ||
    156 uint64_t(crop.x) + crop.width > sensor_width ||
    157 uint64_t(crop.y) + crop.height > sensor_height || !image.width ||
    158 !image.height || image.width > sensor_width ||
    159 image.height > sensor_height)
    160 return result;
    161 std::array<uint32_t, expected_output_tensor_count> lengths{};
    162 /* Validate every shape before indexing flattened values. Divide before
    163 * multiplying so a corrupt extent cannot overflow its bounded element budget. */
    164 for (size_t index = 0; index < lengths.size(); ++index) {
    165 const uint8_t* record =
    166 info.data() + first_tensor_record_offset + index * tensor_record_size;
    167 lengths[index] = read_u32(record + tensor_record_length_offset);
    168 uint32_t dimensions = read_u32(record + tensor_record_rank_offset);
    169 if (dimensions == 0 || dimensions > maximum_tensor_dimensions ||
    170 lengths[index] > maximum_tensor_elements)
    171 return result;
    172 uint32_t product = 1;
    173 for (uint32_t dimension = 0; dimension < dimensions; ++dimension) {
    174 size_t dimension_offset = tensor_record_dimensions_offset +
    175 dimension * tensor_record_dimension_bytes;
    176 uint32_t size = uint32_t(record[dimension_offset]) |
    177 uint32_t(record[dimension_offset + 1]) << 8;
    178 if (size == 0 || product > maximum_tensor_elements / size) return result;
    179 product *= size;
    180 }
    181 if (product != lengths[index]) return result;
    182 }
    183 uint32_t slots = lengths[0] / bbox_values_per_candidate;
    184 if (slots == 0 || lengths[0] % bbox_values_per_candidate ||
    185 lengths[1] != slots || lengths[2] != slots || lengths[3] != 1 ||
    186 tensor.size() != values_per_candidate_with_score_and_class * slots + 1)
    187 return result;
    188 // Output layout stores four box coordinates, then scores, classes, and one
    189 // total count.
    190 float count = tensor[values_per_candidate_with_score_and_class * slots];
    191 if (!std::isfinite(count) || count < 0 || count > slots ||
    192 std::floor(count) != count)
    193 return result;
    194 result.valid = true;
    195 /* Invalid individual candidates are omitted; a valid head with no accepted
    196 * candidates still produces a valid empty result, not analysis_failed. */
    197 for (size_t index = 0; index < static_cast<size_t>(count); ++index) {
    198 size_t box_offset = bbox_values_per_candidate * index;
    199 float score = tensor[score_values_before_candidates * slots + index];
    200 float category = tensor[class_values_before_candidates * slots + index];
    201 float left = tensor[box_offset], top = tensor[box_offset + 1];
    202 float right = tensor[box_offset + 2], bottom = tensor[box_offset + 3];
    203 if (category != object_class_index || !std::isfinite(score) ||
    204 score < object_confidence_threshold || score > 1 ||
    205 !std::isfinite(left) || !std::isfinite(top) || !std::isfinite(right) ||
    206 !std::isfinite(bottom) || left < 0 || top < 0 || right <= left ||
    207 bottom <= top || right > model_input_width ||
    208 bottom > model_input_height)
    209 continue;
    210 size_t position = 0;
    211 while (position < result.count && result.objects[position].confidence >= score) ++position;
    212 if (position == maximum_detections) continue;
    213 // Normalize model coordinates, map to sensor pixels, remove ScalerCrop
    214 // origin, then scale to the JPEG.
    215 auto project = [](float coordinate, unsigned int sensor, int start,
    216 unsigned int extent, unsigned int output) {
    217 double offset = std::clamp(
    218 double(coordinate) / model_input_width * (sensor - 1) - start, 0.0,
    219 double(extent));
    220 return int(std::lround(offset * output / extent));
    221 };
    222 int x0 = project(left, sensor_width, crop.x, crop.width, image.width);
    223 int x1 = project(right, sensor_width, crop.x, crop.width, image.width);
    224 int y0 = project(top, sensor_height, crop.y, crop.height, image.height);
    225 int y1 = project(bottom, sensor_height, crop.y, crop.height, image.height);
    226 if (x1 <= x0 || y1 <= y0) continue;
    227 if (result.count < maximum_detections) ++result.count;
    228 for (size_t moved = result.count - 1; moved > position; --moved)
    229 result.objects[moved] = result.objects[moved - 1];
    230 result.objects[position] = {score, x0, y0, x1 - x0, y1 - y0};
    231 }
    232 return result;
    233}

    236/** @brief Interpret only the detector model's outputs from this retained exposure.
    237 * @param frame Borrowed same-exposure tensors, crop and configured image size.
    238 * @return Invalid without crop evidence, otherwise the validated detector result.
    239 * @details A guessed full-sensor crop could put correct boxes in the wrong JPEG
    240 * position, so missing geometry is a failed analysis rather than a default. */
    241BasicResult frame_result(const imx500::FrameView &frame)

    257/** @brief Publish detector results, never LOOM features or navigation evidence.
    258 * @param frame Borrowed exposure whose image/tensor identity is retained by a lease.
    259 * @param filename New analysis destination selected by the camera publication owner.
    260 * @return True after exclusive-create, flush, sync and close; false retains no partial output.
    261 * @details Readiness can describe an earlier warm-up frame. Decode this frame
    262 * again rather than reusing a previous result for newly captured pixels. */
    263bool write_detections(const imx500::FrameView &frame, const char *filename)
    264{
    265 const BasicResult result = frame_result(frame);
    266 if (!result.valid) {
    267 return false;
    268 }
    269 FILE *output = std::fopen(filename, "wx");
    270 if (!output) {
    271 return false;
    272 }
    273 std::fprintf(output,
    274 "schema=USHER_OBJECTS_V2\nsequence=%llu\nsensor_timestamp_ns=%llu\n"
    275 "image_width=%u\nimage_height=%u\ncount=%zu\n",
    276 static_cast<unsigned long long>(frame.sequence),
    277 static_cast<unsigned long long>(frame.sensor_timestamp_ns),
    278 frame.image.width, frame.image.height, result.count);
    279 for (size_t index = 0; index < result.count; ++index) {
    280 const auto &box = result.objects[index];
    281 std::fprintf(output,
    282 "detection.%zu.name=vehicle\ndetection.%zu.confidence=%.6f\n"
    283 "detection.%zu.x=%d\ndetection.%zu.y=%d\ndetection.%zu.width=%d\ndetection.%zu.height=%d\n",
    284 index, index, double(box.confidence), index, box.x, index, box.y, index, box.width, index, box.height);
    285 }
    286 bool written = std::fflush(output) == 0 && !std::ferror(output) && fsync(fileno(output)) == 0;
    287 if (std::fclose(output) != 0) {
    288 written = false;
    289 }
    290 if (!written) {
    291 unlink(filename);
    292 }
    293 return written;
    294}

    309/** @brief Reject duplicate assignment controls rather than letting INI field order decide detection policy. */
    310bool detector_setting(void *context, const char *key, const char *value)
    311{
    312 auto &options = *static_cast<OnnxDetectorOptions *>(context);
    313 unsigned bit;
    314 if (!std::strcmp(key, "score_threshold")) bit = 1;
    315 else if (!std::strcmp(key, "class_index")) bit = 2;
    316 else if (!std::strcmp(key, "max_detections")) bit = 4;
    317 else return false;
    318 if (options.seen & bit) return false;
    319 options.seen |= bit;
    320 if (bit == 1) return ini_real(value, &options.threshold) && options.threshold >= 0 && options.threshold <= 1;
    321 uint64_t number = 0;
    322 if (!ini_uint(value, &number) || number > (bit == 2 ? 65535U : 64U) || (bit == 4 && !number)) return false;
    323 if (bit == 2) options.category = static_cast<unsigned>(number);
    324 else options.limit = static_cast<unsigned>(number);
    325 return true;
    326}

    328/** @brief Read the declared numeric representation without assuming all exporters use float class/count tensors. */
    329bool tensor_number(const onnx_tensor &tensor, size_t index, double &number)
    330{
    331 const size_t element = tensor.type == ONNX_INT64 ? 8 : 4;
    332 if (index >= tensor.bytes / element) return false;
    333 if (tensor.type == ONNX_FLOAT32) number = static_cast<const float *>(tensor.data)[index];
    334 else if (tensor.type == ONNX_INT64) number = static_cast<double>(static_cast<const int64_t *>(tensor.data)[index]);
    335 else if (tensor.type == ONNX_INT32) number = static_cast<const int32_t *>(tensor.data)[index];
    336 else return false;
    337 return std::isfinite(number);
    338}

    340/** @brief Qualify output layout and label policy before readiness, not on the first aircraft request. */
    341bool validate_onnx_detector(void *context, const inference_context *network, char *error, size_t capacity)
    342{
    343 const auto &options = *static_cast<OnnxDetectorOptions *>(context);
    344 const auto &model = network->model;
    345 char label[64];
    346 if (model.input_count != 1 || !model.image.channels ||
    347 !inference_model_label(&model, options.category, label, sizeof(label))) {
    348 std::snprintf(error, capacity, "Invalid spotter_boxes_xyxy_v1 tensor or label contract");
    349 return false;
    350 }
    351 if (model.output_count == 1) {
    352 const auto &output = model.outputs[0];
    353 const int64_t channels = output.rank == 3 ? output.dimensions[1] : 0;
    354 const int64_t slots = output.rank == 3 ? output.dimensions[2] : 0;
    355 if (output.type != ONNX_FLOAT32 || output.rank != 3 || output.dimensions[0] != 1 ||
    356 channels < 5 || channels > 68 || slots <= 0 || slots > 8400 ||
    357 options.category >= static_cast<unsigned>(channels - 4) ||
    358 output.bytes != static_cast<size_t>(channels) * static_cast<size_t>(slots) * sizeof(float)) {
    359 std::snprintf(error, capacity, "Invalid spotter_boxes_xyxy_v1 tensor or label contract");
    360 return false;
    361 }
    362 return true;
    363 }
    364 const auto integer_size = [](const onnx_tensor &tensor) -> size_t {
    365 return tensor.type == ONNX_INT64 ? 8 : tensor.type == ONNX_INT32 || tensor.type == ONNX_FLOAT32 ? 4 : 0;
    366 };
    367 const size_t slots = model.outputs[0].bytes / (4 * sizeof(float));
    368 const size_t classes = integer_size(model.outputs[2]), count = integer_size(model.outputs[3]);
    369 if (model.output_count != 4 ||
    370 model.outputs[0].type != ONNX_FLOAT32 || model.outputs[0].bytes % (4 * sizeof(float)) ||
    371 !slots || slots > 4096 || model.outputs[1].type != ONNX_FLOAT32 || model.outputs[1].bytes != slots * sizeof(float) ||
    372 !classes || model.outputs[2].bytes != slots * classes || !count || model.outputs[3].bytes != count) {
    373 std::snprintf(error, capacity, "Invalid spotter_boxes_xyxy_v1 tensor or label contract");
    374 return false;
    375 }
    376 return true;
    377}

    379struct DetectionCandidate {
    380 double score;
    381 int x, y, width, height;
    382 double center_x, center_y;

    385static double box_iou(const DetectionCandidate &first, const DetectionCandidate &second)
    386{
    387 const int left = std::max(first.x, second.x);
    388 const int top = std::max(first.y, second.y);
    389 const int right = std::min(first.x + first.width, second.x + second.width);
    390 const int bottom = std::min(first.y + first.height, second.y + second.height);
    391 if (right <= left || bottom <= top) return 0.0;
    392 const double intersection = static_cast<double>(right - left) * static_cast<double>(bottom - top);
    393 const double first_area = static_cast<double>(first.width) * static_cast<double>(first.height);
    394 const double second_area = static_cast<double>(second.width) * static_cast<double>(second.height);
    395 const double union_area = first_area + second_area - intersection;
    396 return union_area > 0.0 ? intersection / union_area : 0.0;
    397}

    399static void insert_detection(DetectionCandidate (&detections)[64], size_t &accepted,
    400 unsigned limit, const DetectionCandidate &candidate, bool suppress_overlap)
    401{
    402 if (suppress_overlap) {
    403 for (size_t index = 0; index < accepted; ++index) {
    404 if (detections[index].score >= candidate.score && box_iou(detections[index], candidate) > 0.45) return;
    405 }
    406 size_t retained = 0;
    407 for (size_t index = 0; index < accepted; ++index) {
    408 if (detections[index].score >= candidate.score || box_iou(detections[index], candidate) <= 0.45) {
    409 detections[retained++] = detections[index];
    410 }
    411 }
    412 accepted = retained;
    413 }
    414 size_t position = 0;
    415 while (position < accepted && detections[position].score >= candidate.score) ++position;
    416 if (position >= limit) return;
    417 if (accepted < limit) ++accepted;
    418 for (size_t moved = accepted - 1; moved > position; --moved) detections[moved] = detections[moved - 1];
    419 detections[position] = candidate;
    420}

    422/** @brief Rank bounded boxes and publish source/model identity so saved-picture results cannot attach to another JPEG. */
    423bool write_onnx_detections(void *context, const inference_context *network,
    424 const char *image, uint32_t sequence, char *error, size_t capacity)
    425{
    426 const auto &options = *static_cast<OnnxDetectorOptions *>(context);
    427 const auto &model = network->model;
    428 char label[64];
    429 if (!inference_model_label(&model, options.category, label, sizeof(label))) return false;
    430 DetectionCandidate detections[64]{};
    431 size_t accepted = 0;
    432 if (model.input_count == 1 && model.output_count == 1 && model.image.channels &&
