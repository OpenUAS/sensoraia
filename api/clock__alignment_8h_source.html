    21/**
    22 * @file clock_alignment.h
    23 * @brief Bounded mapping between flight-controller microseconds and USHER monotonic time.
    24 * @details The protocol exchanges a random token and controller receive/transmit times.
    25 * The resulting interval is deliberately bounded, short-lived, and token-associated so
    26 * stale replies or a wrapped 32-bit controller clock cannot geotag audio with false
    27 * precision.
    28 */

    60/** @brief Register a newly transmitted clock probe.
    61 * @param clock Mutable synchronization state.
    62 * @param token Nonzero random token carried on the wire.
    63 * @param sent_us USHER monotonic timestamp at transmission.
    64 * @return True if the probe is accepted as pending.
    65 * @details Tokens must not repeat the pending or last accepted probe; this prevents a
    66 * delayed reply from being mistaken for fresh timing evidence. */
    67bool clock_alignment_request(struct clock_alignment *clock, union usher_clock_request_union token, uint64_t sent_us);
    68/** @brief Validate a flight-controller clock reply and establish a mapping anchor.
    69 * @param clock Mutable synchronization state.
    70 * @param payload Exact reply payload bytes.
    71 * @param length Payload length.
    72 * @param received_us USHER monotonic receive timestamp.
    73 * @return True only for a matching, bounded-latency reply.
    74 * @details Rather than guessing a one-way delay, this stores an interval spanning the
    75 * possible controller transmit instant, including bounded service time and clock drift. */
    76bool clock_alignment_reply(struct clock_alignment *clock, const uint8_t *payload, size_t length, uint64_t received_us);
    77/** @brief Project a controller timestamp into a conservative USHER time interval.
    78 * @param clock Accepted alignment state.
    79 * @param token Token proving the pose belongs to this probe epoch.
    80 * @param fc_us 32-bit controller microsecond timestamp.
    81 * @param now_us Current USHER monotonic time, used to reject stale/future projections.
    82 * @param interval Receives inclusive projected bounds.
    83 * @return True when @p interval contains a plausible mapping.
    84 * @warning The result is an interval, not a claimed exact timestamp. */
    85bool clock_alignment_map(const struct clock_alignment *clock, union usher_clock_request_union token,
    86 uint32_t fc_us, uint64_t now_us, struct clock_interval *interval);
