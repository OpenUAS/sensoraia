    73public:
    75 explicit Vector(std::size_t count = 0, double value = 0.0)
    76 : dipla::aigenlib::Vector(checked_size(count), value) {}
    77
    78 Vector(dipla::aigenlib::Vector values) : dipla::aigenlib::Vector(std::move(values)) { checked_size(dipla::aigenlib::Vector::size()); }
    80 int size() const noexcept { return static_cast<int>(dipla::aigenlib::Vector::size()); }
    82 int rows() const noexcept { return size(); }
    84 int cols() const noexcept { return 1; }
    86 double operator()(std::size_t row, std::size_t column) const {
    87 if (column != 0) throw std::out_of_range("Vector column must be zero");
    88 return at(row);
    89 }
    90
    91 static Vector Zero(std::size_t count) { return Vector(count); }
    93 static Vector Constant(std::size_t count, double value) { return Vector(count, value); }
    95 Vector eval() const { return *this; }
    97 bool allFinite() const {
    98 return std::all_of(begin(), end(), [](double value) { return std::isfinite(value); });
    99 }
    100
    101 double sum() const { return std::accumulate(begin(), end(), 0.0); }
    103 double norm() const {
    104 double result = 0.0;
    105 for (const double value : *this) result = std::hypot(result, value);
    106 return result;
    107 }
    108
    109 Vector& operator+=(const Vector& other) {
    110 if (size() != other.size()) throw std::invalid_argument("Vector dimensions differ");
    111 for (int index = 0; index < size(); ++index) (*this)[index] += other[index];
    112 return *this;
    113 }
    114
    115 bool isApprox(const Vector& other, double tolerance) const {
    116 if (size() != other.size() || !allFinite() || !other.allFinite() || !(tolerance >= 0.0)) return false;
    117 double error = 0.0;
    118 for (int index = 0; index < size(); ++index) error = std::hypot(error, (*this)[index] - other[index]);
    119 return error <= tolerance * std::min(norm(), other.norm());
    120 }
    121private:
    123 static std::size_t checked_size(std::size_t count) {
    124 if (count > dipla::aigenlib::max_extent) throw std::length_error("Planner vector exceeds grid bound");
    125 return count;
    126 }
    127};

    166template <class Number>
    167class Matrix4 {
    168 static_assert(std::is_floating_point_v<Number>, "Matrix4 needs floating-point scalars");
    169public:
    171 using Scalar = Number;
    173 constexpr int rows() const noexcept { return 4; }
    175 constexpr int cols() const noexcept { return 4; }
    177 constexpr int size() const noexcept { return 16; }
    179 Scalar& operator()(int row, int column) { return values_[offset(row, column)]; }
    181 Scalar operator()(int row, int column) const { return values_[offset(row, column)]; }
    183 static Matrix4 Zero() { return {}; }
    185 static Matrix4 Constant(Scalar value) {
    186 Matrix4 result;
    187 result.values_.fill(value);
    188 return result;
    189 }
    190
    191 static Matrix4 Identity() {
    192 Matrix4 result;
    193 for (int index = 0; index < 4; ++index) result(index, index) = Scalar(1);
    194 return result;
    195 }
    196
    197 bool allFinite() const {
    198 return std::all_of(values_.begin(), values_.end(), [](Scalar value) { return std::isfinite(value); });
    199 }
    200
    201 Scalar trace() const {
    202 Scalar result = 0;
    203 for (int index = 0; index < 4; ++index) result += (*this)(index, index);
    204 return result;
    205 }
    206
    207 Matrix4 transpose() const {
    208 Matrix4 result;
    209 for (int row = 0; row < 4; ++row) {
    210 for (int column = 0; column < 4; ++column) result(row, column) = (*this)(column, row);
    211 }
    212 return result;
    213 }
    214
    215 Matrix4 operator*(const Matrix4& right) const {
    216 Matrix4 result;
    217 for (int column = 0; column < 4; ++column) {
    218 for (int inner = 0; inner < 4; ++inner) {
    219 const Scalar factor = right(inner, column);
    220 for (int row = 0; row < 4; ++row) result(row, column) += (*this)(row, inner) * factor;
    221 }
    222 }
    223 return result;
    224 }
    225private:
    227 std::array<Scalar, 16> values_{};
    229 static std::size_t offset(int row, int column) {
    230 if (row < 0 || row >= 4 || column < 0 || column >= 4) throw std::out_of_range("Matrix4 index");
    231 return static_cast<std::size_t>(column * 4 + row);
    232 }
    233};
