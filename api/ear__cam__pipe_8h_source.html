    59/**
    60 * @brief Begin/reset active collection without solving or waiting for log I/O.
    61 * @param session_id Token retained in audit rows; backend admission validates it before this call.
    62 * @return False when no active or free buffer exists; true after resetting active counts.
    63 * @details Does not evict worker-owned snapshots. The active buffer is serial-owner-only.
    64 */
    65bool ear_cam_pipe_begin_session(uint32_t session_id);
    66/**
    67 * @brief Best-effort enqueue of a correlated result row on the serial owner.
    68 * @param session_id Token identifying the completed snapshot, not necessarily the current session.
    69 * @param result Borrowed estimate, copied into a row before return; NULL is ignored.
    70 * @details Only the log worker accesses its FILE. A dropped audit row is not a lost point completion.
    71 */
    72void ear_cam_pipe_log_result(uint32_t session_id, const struct ear_loudest_spot *result);
    73/**
    74 * @brief Transfer active storage to an immutable worker snapshot without copying samples.
    75 * @return Owned snapshot, or NULL if no active buffer exists.
    76 * @details Serial-owner-only. Pending pose windows are resolved/fallback-committed
    77 * before freezing count; active collection stays disabled until the next successful begin.
    78 * @warning Caller must eventually release exactly once, after all readers stop using it.
    79 */
    80struct ear_session_snapshot *ear_cam_pipe_detach_session(void);
    81/**
    82 * @brief Solve immutable detached evidence without touching current-session state.
    83 * @param snapshot Borrowed owned snapshot, which must remain unreleased throughout this call.
    84 * @param result Required estimate destination, cleared before invoking the core.
    85 * @return Core solver status: 0 for an estimate, -1 for invalid/insufficient evidence or arguments.
    86 * @details Worker-safe; shared core workspace serializes concurrent solves. No ownership transfer.
    87 */
    88int ear_cam_pipe_snapshot_solve(const struct ear_session_snapshot *snapshot, struct ear_loudest_spot *result);
    89/**
    90 * @brief Read the last request number from immutable evidence.

    95/**
    96 * @brief Render detached evidence using collision-safe session-tagged JPEG/CSV names.
    97 * @param snapshot Borrowed owned snapshot; remains immutable and unreleased during I/O.
    98 * @param session_id Required nonzero correlation token for file naming and CSV provenance.
    99 * @param result Required borrowed estimate; an invalid estimate may still produce evidence.
    100 * @param filename Destination path buffer; cleared by the renderer on handled creation failure.
    101 * @param filename_size Capacity of filename including its terminating NUL.
    102 * @param simulated Select the simulated-image prefix, not a different rendering algorithm.
    103 * @return 0 for decorated JPEG success, -1 for invalid arguments or primary rendering failure.
    104 * @details Worker-safe. CSV, undecorated field and .geo errors are reported separately;
    105 * JPEG success does not guarantee all evidence products or power-loss durability.
    106 */

    110/**
    111 * @brief Return detached storage to admission after all snapshot access has ended.

    162/**
    163 * @brief Associate the newest sound window with one flight-controller shot pose.
    164 * @param shot Flight pose and shot number received from USHER transport.
    165 * @return 0 when evidence is committed or admitted to the bounded pose queue;
    166 * -1 for invalid input, unavailable/stale data or duplicate timed windows.
    167 * @details Timed windows await pose evidence in a fixed queue for at most 200 ms;
    168 * unresolved or legacy measurements retain their labelled request-pose fallback.
    169 * Only the newest live measurement is read at each request, not every audio window.
    170 * Capacity pressure retains the newest half of active samples, never a worker snapshot.
    171 * Server recovery and CSV I/O run independently. This call never waits for them.
    172 */
    173int ear_cam_pipe_record(const union dc_shot_union *shot);
    174
    175/**
    176 * @brief Fuse the session into its final loudest-spot result and clear stored samples.
    177 * @param result Destination for the solved or invalid result.
    178 * @return Core solver status: 0 for an estimate, -1 for insufficient evidence or invalid arguments.
    179 * @details Finalization consumes the session so the next survey cannot accidentally mix old
    180 * and new flight legs. This legacy synchronous call solves committed samples and
    181 * joins the log writer; it is not suitable for a latency-sensitive UART callback.
    182 * Resolve pending poses first (ear_cam_pipe_solve() forces their fallback), and render
    183 * before clearing samples. Prefer detach/solve/render/release for asynchronous sessions.
    184 */
    185int ear_cam_pipe_finish(struct ear_loudest_spot *result);
    186
    187/**
    188 * @brief Solve the accumulated session without clearing it.
    189 * @param result Destination for the current estimate.
    190 * @return Core solver status: 0 for an estimate, -1 for insufficient evidence or invalid arguments.
    191 * @details Use for refinement decisions. Unlike @c ear_cam_pipe_finish(), subsequent records
    192 * remain part of the same survey session. Serial-owner-only; pending poses are
    193 * forced to their bounded fallback before this synchronous solve.
    194 */
    195int ear_cam_pipe_solve(struct ear_loudest_spot *result);

    197/**
    198 * @brief Render the current session as a north-up acoustic evidence JPEG.
    199 * @param result Solved loudest-spot estimate to mark on the image.
    200 * @param filename Destination buffer receiving the JPEG path.
    201 * @param filename_size Capacity of @p filename.
    202 * @param simulated True when rendering NPS/desk simulated data.
    203 * @return 0 on success; -1 if inputs, session data, or output creation fail.
    204 * @details Must be called before @c ear_cam_pipe_finish() clears the session. The JPEG and
    205 * sidecar preserve the sampled field and final estimate for post-flight review.
    206 */
    207int ear_cam_pipe_render(const struct ear_loudest_spot *result, char *filename, size_t filename_size,
    208 bool simulated);
