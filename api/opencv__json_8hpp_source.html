    18/**
    19 * @file opencv_json.hpp
    20 * @brief Header-only C++17 value adapter around OpenCV's JSON persistence layer.
    21 * @details OpenCV remains the parser and ordinary writer. Normalization carries
    22 * null, booleans and 64-bit integers through reserved string markers so those
    23 * types are not lost in FileStorage's representation. Nodes are then detached
    24 * from FileStorage into a heap-backed tree usable by existing C++ consumers.
    25 *
    26 * @par Ownership and mutation
    27 * Ordinary Value copies recursively clone their trees. Child lookups and items()
    28 * initially return shared child handles instead, allowing a selected child's
    29 * mutators to affect its parent. Copying such a handle creates an independent
    30 * tree; assigning a new Value to the handle rebinds the handle, not the parent slot.
    31 *
    32 * @par Deliberate limits of this adapter
    33 * This is not JSONbourne's bounded arena API: strings, maps, vectors and shared
    34 * nodes allocate, while cloning, conversion and serialization recurse. No fixed
    35 * nesting or allocation budget is imposed here. Objects use unique sorted map
    36 * keys rather than retaining duplicate members or their original source order.
    37 * @pre Application string/key data must not use the reserved
    38 * __USHER_OPENCV_JSON_ namespace, which belongs to the transport representation.
    39 * @note OpenCV acceptance and this adapter's normalization determine accepted
    40 * input; this wrapper is not an independent strict RFC 8259 validator.
    41 * @see usher_cv_json::Value::parse(), usher_cv_json::Value::dump_sorted()
    42 */

    72/**
    73 * @brief Heap-backed tagged storage independent of a FileStorage lifetime.
    74 * @details Separate signed, unsigned and real fields avoid forcing 64-bit
    75 * integers through a double. Only the field selected by kind is semantically
    76 * active; mutators need not clear storage belonging to previously selected kinds.
    77 */
    78struct Node {
    79 /** @brief Active payload category; boolean is distinct from integer storage. */
    80 enum class Kind {
    81 Null, /**< JSON null with no active scalar payload. */
    82 Bool, /**< Native boolean. */
    83 Signed, /**< Signed 64-bit integer. */
    84 Unsigned, /**< Unsigned 64-bit integer. */
    85 Real, /**< Double-precision real value. */
    86 String, /**< Owned byte string. */
    87 Array, /**< Ordered child-node sequence. */
    88 Object /**< Unique members in std::map key order. */
    89 };
    90 Kind kind = Kind::Null; /**< Tag governing interpretation and serialization. */
    91 bool boolean = false; /**< Bool payload. */
    92 int64_t signed_number = 0; /**< Signed payload. */
    93 uint64_t unsigned_number = 0; /**< Unsigned payload. */
    94 double real_number = 0; /**< Real payload; construction alone does not reject nonfinite values. */
    95 std::string string; /**< String payload with std::string byte-length semantics. */
    96 std::vector<std::shared_ptr<Node>> array; /**< Array children in insertion order. */
    97 std::map<std::string, std::shared_ptr<Node>> object; /**< Unique keys with shared child ownership. */
    98};

    100/**
    101 * @brief C++ value interface with explicit conversions and OpenCV JSON I/O.
    102 * @details Copy construction/assignment are deep; lookup-produced handles share
    103 * a selected node until copied or rebound. Conversion failures throw rather than
    104 * silently turning missing, differently typed or out-of-range data into defaults.
    105 * @note Allocation failures may additionally propagate standard-library exceptions.
    106 */
    107class Value {
    108public:
    109 /** @brief Adapter errors exposed independently of OpenCV's exception type. */
    110 class exception : public std::runtime_error {
    111 public:
    112 /** @brief Reuse runtime_error constructors to retain descriptive error text. */
    113 using std::runtime_error::runtime_error;

    115 /** @brief One named value supplied to an object factory. */
    116 using Field = std::pair<std::string, Value>;

    118 /** @brief Allocate a fresh null node instead of using a shared mutable singleton. */
    119 Value() : node_(std::make_shared<Node>()) {}
    120 /** @brief Construct null from the C++ nullptr tag. */
    121 Value(std::nullptr_t) : Value() {}
    122 /**
    123 * @brief Construct a boolean without confusing it with an integral payload.
    124 * @param[in] value Boolean to store.
    125 */
    126 Value(bool value) : Value() {
    127 node_->kind = Node::Kind::Bool;
    128 node_->boolean = value;
    129 }
    130 /**
    131 * @brief Copy a terminated string; a null pointer selects an empty string.
    132 * @param[in] value C string, or nullptr; embedded NULs require the std::string overload.
    133 */
    134 Value(const char *value) : Value(std::string(value ? value : "")) {}
    135 /**
    136 * @brief Own a complete byte string rather than borrowing external storage.
    137 * @param[in] value String copied with its explicit byte length.
    138 */
    139 Value(const std::string &value) : Value() {
    140 node_->kind = Node::Kind::String;
    141 node_->string = value;
    142 }

    144 /**
    145 * @brief Store native integral input in its signed or unsigned 64-bit category.
    146 * @tparam Number Integral source type other than bool.
    147 * @param[in] value Number converted to the corresponding 64-bit storage type.
    148 * @note bool is excluded so overload resolution preserves boolean semantics.
    149 */
    150 template <typename Number,
    151 typename = std::enable_if_t<std::is_integral_v<Number> &&
    152 !std::is_same_v<Number, bool>>>
    153 Value(Number value) : Value() {
    154 if constexpr (std::is_signed_v<Number>) {
    155 node_->kind = Node::Kind::Signed;
    156 node_->signed_number = static_cast<int64_t>(value);
    157 } else {
    158 node_->kind = Node::Kind::Unsigned;
    159 node_->unsigned_number = static_cast<uint64_t>(value);
    160 }
    161 }
    162 /**
    163 * @brief Store floating-point input as a double, not an integer-shaped value.
    164 * @tparam Number Floating-point source type.
    165 * @param[in] value Number converted to double; this constructor does not check finiteness.
    166 */
    167 template <typename Number,
    168 typename = std::enable_if_t<std::is_floating_point_v<Number>>,
    169 typename = void>
    170 Value(Number value) : Value() {
    171 node_->kind = Node::Kind::Real;
    172 node_->real_number = static_cast<double>(value);
    173 }

    175 /**
    176 * @brief Recursively detach a copy so later mutation does not alias the source.
    177 * @param[in] other Accessible source value, not a moved-from handle.
    178 */
    179 Value(const Value &other) : node_(clone(other.node_)) {}
    180 /** @brief Transfer node ownership; only destruction/reassignment is assumed for the source. */
    181 Value(Value &&) noexcept = default;
    182 /**
    183 * @brief Deep-copy and rebind this handle without modifying any former parent slot.
    184 * @param[in] other Source value; self-assignment is permitted.
    185 * @return This handle after assignment.
    186 */
    187 Value &operator=(const Value &other) {
    188 if (this != &other)
    189 node_ = clone(other.node_);
    190 return *this;
    191 }
    192 /**
    193 * @brief Move/rebind ownership without recursively cloning nodes.
    194 * @return This handle; the moved-from source must be reassigned before inspection.
    195 */
    196 Value &operator=(Value &&) noexcept = default;

    198 /** @brief Make an empty object. @return Independently owned object value. */
    199 static Value object() {
    200 Value value;
    201 value.node_->kind = Node::Kind::Object;

    204 /**
    205 * @brief Build an object by cloning each supplied member value.
    206 * @param[in] fields Named values; later repeated keys replace earlier members.
    207 * @return Object in map key order rather than initializer/source order.

    210 Value value = object();
    211 for (const auto &field : fields)

    233 /**
    234 * @brief Copy a byte range and delegate parsing to the string overload.
    235 * @tparam Iterator Iterator type accepted by std::string's range constructor.
    236 * @param[in] first Beginning of the input range.
    237 * @param[in] last One-past-end iterator.
    238 * @return Tree detached from OpenCV's temporary storage.
    239 * @throws exception If normalization or OpenCV input processing fails.
    240 */
    241 template <typename Iterator>
    242 static Value parse(Iterator first, Iterator last) {
    243 return parse(std::string(first, last));
    244 }
    245 /**
    246 * @brief Parse terminated input; use a range/std::string for an explicit extent.
    247 * @param[in] input Terminated bytes; nullptr selects an empty input string.
    248 * @return Parsed tree, independent of the input buffer.
    249 * @throws exception If the selected input cannot be read.
    250 */
    251 static Value parse(const char *input) {
    252 return parse(std::string(input ? input : ""));
    253 }
    254 /**
    255 * @brief Normalize compatibility tokens, parse with OpenCV and detach the tree.
    256 * @param[in] input JSON input whose original bytes remain unchanged.
    257 * @return Independently owned tree with restored bool/null/integer categories.
    258 * @throws exception For rejected sentinels/escapes, conversion or OpenCV parse errors.
    259 * @note cv::Exception is translated at this boundary so callers can use the
    260 * adapter's error type without depending on OpenCV's parser diagnostics API.
    261 */
    262 static Value parse(const std::string &input) {
    263 const std::string normalized = normalize_for_opencv(input);
    264 try {
    265 cv::FileStorage storage(normalized, cv::FileStorage::READ |
    266 cv::FileStorage::MEMORY |
    267 cv::FileStorage::FORMAT_JSON);
    268 if (!storage.isOpened())
    269 throw exception("OpenCV FileStorage could not open JSON input");
    270 return from_node(storage.root());
    271 } catch (const cv::Exception &error) {
    272 throw exception(std::string("invalid OpenCV JSON: ") + error.what());
    273 }
    274 }

    276 /** @brief Inspect the null tag. @return True only for a null node. */
    277 bool is_null() const { return node_->kind == Node::Kind::Null; }
    278 /** @brief Inspect the object tag. @return True only for an object node. */
    279 bool is_object() const { return node_->kind == Node::Kind::Object; }
    280 /** @brief Inspect the array tag. @return True only for an array node. */
    281 bool is_array() const { return node_->kind == Node::Kind::Array; }
    282 /** @brief Inspect the string tag. @return True only for a string node. */
    283 bool is_string() const { return node_->kind == Node::Kind::String; }
    284 /** @brief Test all three numeric categories without treating bool as numeric. @return Numeric tag status. */
    285 bool is_number() const {
    286 return node_->kind == Node::Kind::Signed ||
    287 node_->kind == Node::Kind::Unsigned ||
    288 node_->kind == Node::Kind::Real;
    289 }
    290 /** @brief Test signed/unsigned storage without accepting a real such as 1.0. @return Integer tag status. */
    291 bool is_number_integer() const {
    292 return node_->kind == Node::Kind::Signed ||
    293 node_->kind == Node::Kind::Unsigned;
    294 }
    295 /** @brief Inspect unsigned storage rather than testing a number's sign. @return Unsigned tag status. */
    296 bool is_number_unsigned() const {
    297 return node_->kind == Node::Kind::Unsigned;
    298 }
    299 /**
    300 * @brief Check member presence without interpreting a null value as missing.
    301 * @param[in] key Non-null terminated member name.
    302 * @return True for an existing object member; false for nonobjects/missing keys.
    303 */
    304 bool contains(const char *key) const {
    305 return is_object() && node_->object.find(key) != node_->object.end();
    306 }

    308 * @brief Check a member name through the C-string lookup interface.

    312 bool contains(const std::string &key) const { return contains(key.c_str()); }

    314 * @brief Count container entries, not string bytes or scalar payload size.
    315 * @return Array length, object member count, or zero for a scalar/null.

    317 size_t size() const {
    318 return is_array() ? node_->array.size()
    319 : is_object() ? node_->object.size()
    320 : 0;

    323 /**
    324 * @brief Look up a shared child handle, using null for a missing member.
    325 * @param[in] key Non-null terminated name.
    326 * @return Handle initially sharing the child, or a fresh null value if absent.
    327 * @throws exception If this value is not an object; lookup does not insert.
    328 */
    329 Value operator[](const char *key) const { return lookup(key, false); }
    330 /**
    331 * @brief Perform optional lookup through the terminated-name overload.
    332 * @param[in] key Name; lookup stops at its first embedded NUL.
    333 * @return Shared child handle or a fresh null for a missing member.
    334 * @throws exception If this value is not an object.
    335 */
    336 Value operator[](const std::string &key) const {
    337 return lookup(key.c_str(), false);
    338 }
    339 /**
    340 * @brief Require an object member instead of substituting null.
    341 * @param[in] key Non-null terminated member name.
    342 * @return Handle initially sharing the selected child node.
    343 * @throws exception For a nonobject or missing key.
    344 */
    345 Value at(const char *key) const { return lookup(key, true); }
    346 /**
    347 * @brief Require a member through the terminated-name interface.
    348 * @param[in] key Name; an embedded NUL terminates lookup.
    349 * @return Shared child handle.
    350 * @throws exception For a nonobject or missing key.
    351 */
    352 Value at(const std::string &key) const { return at(key.c_str()); }
    353 /**
    354 * @brief Check signed array indices before conversion to an unsigned extent.
    355 * @param[in] index Zero-based array position.
    356 * @return Shared child handle.
    357 * @throws exception For a negative/out-of-range index or nonarray value.
    358 */
    359 Value operator[](int index) const { return at(index); }
    360 /**
    361 * @brief Select an array element with the same bounds policy as at().
    362 * @param[in] index Zero-based array position.
    363 * @return Shared child handle.
    364 * @throws exception For a nonarray or out-of-range position.

    366 Value operator[](size_t index) const { return at(index); }
    367 /**
    368 * @brief Reject a negative position before any size_t cast can wrap it.
    369 * @param[in] index Signed zero-based array position.
    370 * @return Shared child handle.
    371 * @throws exception For invalid array access.

    374 if (index < 0)
    375 throw exception("JSON array index out of range");
    376 return at(static_cast<size_t>(index));
    377 }
    378 /**
    379 * @brief Check both the array kind and its actual vector extent.
    380 * @param[in] index Zero-based array position.
    381 * @return Handle sharing the selected node, not a recursive copy.
    382 * @throws exception For a nonarray or out-of-range position.
    383 */
    384 Value at(size_t index) const {
    385 if (!is_array() || index >= node_->array.size())
    386 throw exception("JSON array index out of range");
    387 return Value(node_->array[index]);
    388 }

    390 /**
    391 * @brief Convert explicitly while preserving the distinction between types.
    392 * @tparam Type Value, string, bool, arithmetic type, std::array or std::vector;
    393 * container elements are converted recursively using their value_type.
    394 * @return Converted value; requesting Value makes a deep copy.
    395 * @throws exception For incompatible tags, nonfinite/out-of-range numeric
    396 * conversion or a fixed-size array whose length does not match.
    397 * @note Integer targets reject real-tagged inputs even if mathematically
    398 * integral. Floating conversions use double and do not promise integer precision.
    399 * Unsupported target types produce a compile-time error, not a runtime fallback.
    400 */
    401 template <typename Type> Type get() const {
    402 if constexpr (std::is_same_v<Type, Value>)
    403 return *this;
    404 else if constexpr (std::is_same_v<Type, std::string>) {
    405 if (!is_string())
    406 throw exception("JSON value is not a string");
    407 return node_->string;
    408 } else if constexpr (std::is_same_v<Type, bool>) {
    409 if (node_->kind != Node::Kind::Bool)
    410 throw exception("JSON value is not a boolean");
    411 return node_->boolean;
    412 } else if constexpr (std::is_floating_point_v<Type>) {
    413 if (!is_number())
    414 throw exception("JSON value is not numeric");
    415 const double number = node_->kind == Node::Kind::Real ? node_->real_number
    416 : node_->kind == Node::Kind::Signed
    417 ? static_cast<double>(node_->signed_number)
    418 : static_cast<double>(node_->unsigned_number);
    419 if (!std::isfinite(number) ||
    420 number < -std::numeric_limits<Type>::max() ||
    421 number > std::numeric_limits<Type>::max())
    422 throw exception("JSON number is out of range");
    423 return static_cast<Type>(number);
    424 } else if constexpr (std::is_integral_v<Type> && std::is_signed_v<Type>) {
    425 if (!is_number_integer())
    426 throw exception("JSON value is not an integer");
    427 const long double number =
    428 node_->kind == Node::Kind::Signed
    429 ? static_cast<long double>(node_->signed_number)
    430 : static_cast<long double>(node_->unsigned_number);
    431 if (number < std::numeric_limits<Type>::min() ||
    432 number > std::numeric_limits<Type>::max())
    433 throw exception("JSON integer is out of range");
    434 return static_cast<Type>(number);
    435 } else if constexpr (std::is_integral_v<Type> && std::is_unsigned_v<Type>) {
    436 if (!is_number_integer())
    437 throw exception("JSON value is not an integer");
    438 const long double number =
    439 node_->kind == Node::Kind::Signed
    440 ? static_cast<long double>(node_->signed_number)
    441 : static_cast<long double>(node_->unsigned_number);
    442 if (number < 0 || number > std::numeric_limits<Type>::max())
    443 throw exception("JSON integer is out of range");
    444 return static_cast<Type>(number);
    445 } else if constexpr (is_std_array<Type>::value) {
    446 if (!is_array() || size() != std::tuple_size_v<Type>)
    447 throw exception("JSON array has unexpected length");
    448 Type result{};
    449 for (size_t index = 0; index < result.size(); ++index)
    450 result[index] = at(index).template get<typename Type::value_type>();
    451 return result;
    452 } else if constexpr (is_std_vector<Type>::value) {
    453 if (!is_array())
    454 throw exception("JSON value is not an array");
    455 Type result;
    456 result.reserve(size());
    457 for (size_t index = 0; index < size(); ++index)
    458 result.push_back(at(index).template get<typename Type::value_type>());
    459 return result;
    460 } else
    461 static_assert(!sizeof(Type), "unsupported JSON conversion type");
    462 }

    464 /**
    465 * @brief Use a default only for absence, never to hide an invalid present value.
    466 * @tparam Type Conversion/default type supported by get().
    467 * @param[in] key Non-null terminated member name.
    468 * @param[in] fallback Result when no member exists (including on a nonobject).
    469 * @return Converted member or fallback.
    470 * @throws exception If a present member cannot be converted to Type.
    471 */
    472 template <typename Type> Type value(const char *key, Type fallback) const {
    473 return contains(key) ? at(key).template get<Type>() : std::move(fallback);
    474 }
    475 /**
    476 * @brief Provide a string-valued default rather than returning a borrowed pointer.
    477 * @param[in] key Non-null terminated member name.
    478 * @param[in] fallback Non-null terminated default string.
    479 * @return Owned member/default string.
    480 * @throws exception If a present member is not a string.
    481 */
    482 std::string value(const char *key, const char *fallback) const {
    483 return contains(key) ? at(key).get<std::string>() : std::string(fallback);
    484 }

    486 /**
    487 * @brief Snapshot object names and shared child handles in map order.
    488 * @return Newly allocated member vector; child nodes initially remain shared.
    489 * @throws exception If this value is not an object.
    490 * @note The vector is detached from later map insertion/removal, but a returned
    491 * child's in-place mutation can still affect the source node it shares.
    492 */
    493 std::vector<std::pair<std::string, Value>> items() const {
    494 if (!is_object())
    495 throw exception("JSON value is not an object");
    496 std::vector<std::pair<std::string, Value>> result;
    497 result.reserve(node_->object.size());
    498 for (const auto &item : node_->object)
    499 result.emplace_back(item.first, Value(item.second));
    500 return result;
    501 }

    503 /**
    504 * @brief Clone a member value, replace any matching key and select object kind.
    505 * @param[in] key Owned member name copied into the map.
    506 * @param[in] value Source value cloned rather than linked into this tree.
    507 * @throws exception If a directly supplied string uses a reserved bridge marker.
    508 * @note Existing other-kind payload storage is not cleared by this operation.
    509 */
    510 void set(const std::string &key, const Value &value) {
    511 require_unreserved(value);
    512 node_->object[key] = clone(value.node_);
    513 node_->kind = Node::Kind::Object;
    514 }

    568 /**
    569 * @brief Produce compact map-ordered output without FileStorage formatting.
    570 * @return Owned text with non-ASCII strings expressed as Unicode escapes.
    571 * @throws exception For malformed/truncated UTF-8 sequence shapes or failed number formatting.
    572 * @pre Use finite real values and valid Unicode strings when strict JSON is needed.
    573 * @note This is deterministic map-key ordering, not RFC 8785 canonicalization.
    574 * UTF-8 checks cover lead/continuation shape and truncation, not every scalar
    575 * validity rule; real-tagged integral spellings keep a .0 suffix for round trips.
    576 */
    577 std::string dump_sorted() const {
    578 std::string output;
    579 const auto append_string = [&output](const std::string &text) {
    580 static constexpr char hex[] = "0123456789abcdef";
    581 output.push_back('"');
    582 for (size_t index = 0; index < text.size();) {
    583 const unsigned char first = static_cast<unsigned char>(text[index]);
    584 if (first == '"' || first == '\\') {
    585 output.push_back('\\');
    586 output.push_back(static_cast<char>(first));
    587 ++index;
    588 } else if (first < 0x20) {
    589 const char *escape = first == '\b' ? "\\b"
    590 : first == '\f' ? "\\f"
    591 : first == '\n' ? "\\n"
    592 : first == '\r' ? "\\r"
    593 : first == '\t' ? "\\t"
    594 : nullptr;
    595 if (escape)
    596 output += escape;
    597 else {
    598 output += "\\u00";
    599 output.push_back(hex[first >> 4]);
    600 output.push_back(hex[first & 0x0f]);
    601 }
    602 ++index;
    603 } else if (first < 0x80) {
    604 output.push_back(static_cast<char>(first));
    605 ++index;
    606 } else {
    607 uint32_t codepoint = 0;
    608 size_t count = 0;
    609 if ((first & 0xe0) == 0xc0) {
    610 codepoint = first & 0x1f;
    611 count = 2;
    612 } else if ((first & 0xf0) == 0xe0) {
    613 codepoint = first & 0x0f;
    614 count = 3;
    615 } else if ((first & 0xf8) == 0xf0) {
    616 codepoint = first & 0x07;
    617 count = 4;
    618 } else
    619 throw exception("invalid UTF-8 in JSON string");
    620 if (index + count > text.size())
    621 throw exception("truncated UTF-8 in JSON string");
    622 for (size_t offset = 1; offset < count; ++offset) {
    623 const unsigned char next =
    624 static_cast<unsigned char>(text[index + offset]);
    625 if ((next & 0xc0) != 0x80)
    626 throw exception("invalid UTF-8 in JSON string");
    627 codepoint = (codepoint << 6) | (next & 0x3f);
    628 }
    629 const auto append_hex = [&output](uint16_t unit) {
    630 output += "\\u";
    631 output.push_back(hex[(unit >> 12) & 0xf]);
    632 output.push_back(hex[(unit >> 8) & 0xf]);
    633 output.push_back(hex[(unit >> 4) & 0xf]);
    634 output.push_back(hex[unit & 0xf]);
    635 };
    636 /* Supplementary scalars need UTF-16 pairs because JSON escapes encode 16-bit units. */
    637 if (codepoint <= 0xffff)
    638 append_hex(static_cast<uint16_t>(codepoint));
    639 else {
    640 codepoint -= 0x10000;
    641 append_hex(static_cast<uint16_t>(0xd800 + (codepoint >> 10)));
    642 append_hex(static_cast<uint16_t>(0xdc00 + (codepoint & 0x3ff)));
    643 }
    644 index += count;
    645 }
    646 }
    647 output.push_back('"');
    648 };
    649 const auto append = [&](auto &&self,
    650 const std::shared_ptr<Node> &node) -> void {
    651 switch (node->kind) {
    652 case Node::Kind::Null:
    653 output += "null";
    654 break;
    655 case Node::Kind::Bool:
    656 output += node->boolean ? "true" : "false";
    657 break;
    658 case Node::Kind::Signed:
    659 output += std::to_string(node->signed_number);
    660 break;
    661 case Node::Kind::Unsigned:
    662 output += std::to_string(node->unsigned_number);
    663 break;
    664 case Node::Kind::Real: {
    665 char buffer[64];
    666 const auto result =
    667 std::to_chars(buffer, buffer + sizeof(buffer), node->real_number,
    668 std::chars_format::general);
    669 if (result.ec != std::errc{})
    670 throw exception("cannot format JSON number");
    671 const std::string formatted(buffer, result.ptr);
    672 output += formatted;
    673 /* Preserve the Real category instead of reparsing an integer-shaped spelling as an integer. */
    674 if (formatted.find_first_of(".eE") == std::string::npos)
    675 output += ".0";
    676 break;
    677 }
    678 case Node::Kind::String:
    679 append_string(node->string);
    680 break;
    681 case Node::Kind::Array:
    682 output.push_back('[');
    683 for (size_t index = 0; index < node->array.size(); ++index) {
    684 if (index)
    685 output.push_back(',');
    686 self(self, node->array[index]);
    687 }
    688 output.push_back(']');
    689 break;
    690 case Node::Kind::Object: {
    691 output.push_back('{');
    692 bool first = true;
    693 for (const auto &item : node->object) {
    694 if (!first)
    695 output.push_back(',');
    696 first = false;
    697 append_string(item.first);
    698 output.push_back(':');
    699 self(self, item.second);
    700 }
    701 output.push_back('}');
    702 break;
    703 }
    704 }
    705 };
    706 append(append, node_);
    707 return output;
    708 }

    836inline std::shared_ptr<Node> Value::clone(const std::shared_ptr<Node> &source) {
    837 auto result = std::make_shared<Node>(*source);
    838 /* Copy scalar/string storage first, then replace copied child pointers with detached trees. */
    839 result->array.clear();
    840 result->object.clear();
    841 for (const auto &item : source->array)
    842 result->array.push_back(clone(item));
    843 for (const auto &item : source->object)
    844 result->object.emplace(item.first, clone(item.second));
    845 return result;
    846}

    848inline bool Value::equal_nodes(const std::shared_ptr<Node> &left,
    849 const std::shared_ptr<Node> &right) {
    850 const auto integer = [](Node::Kind kind) {
    851 return kind == Node::Kind::Signed || kind == Node::Kind::Unsigned;
    852 };
    853 const auto numeric = [&](Node::Kind kind) {
    854 return integer(kind) || kind == Node::Kind::Real;
    855 };
    856 if (numeric(left->kind) && numeric(right->kind)) {
    857 /* Negative signed values must never wrap into an apparently equal unsigned magnitude. */
    858 if (left->kind == Node::Kind::Signed && right->kind == Node::Kind::Unsigned)
    859 return left->signed_number >= 0 &&
    860 static_cast<uint64_t>(left->signed_number) ==
    861 right->unsigned_number;
    862 if (left->kind == Node::Kind::Unsigned && right->kind == Node::Kind::Signed)
    863 return right->signed_number >= 0 &&
    864 left->unsigned_number ==
    865 static_cast<uint64_t>(right->signed_number);
    866 const long double left_number =
    867 left->kind == Node::Kind::Signed
    868 ? static_cast<long double>(left->signed_number)
    869 : left->kind == Node::Kind::Unsigned
    870 ? static_cast<long double>(left->unsigned_number)
    871 : static_cast<long double>(left->real_number);
    872 const long double right_number =
    873 right->kind == Node::Kind::Signed
    874 ? static_cast<long double>(right->signed_number)
    875 : right->kind == Node::Kind::Unsigned
    876 ? static_cast<long double>(right->unsigned_number)
    877 : static_cast<long double>(right->real_number);
    878 return left_number == right_number;
    879 }
    880 if (left->kind != right->kind)
    881 return false;
    882 switch (left->kind) {
    883 case Node::Kind::Null:
    884 return true;
    885 case Node::Kind::Bool:
    886 return left->boolean == right->boolean;
    887 case Node::Kind::Signed:
    888 return left->signed_number == right->signed_number;
    889 case Node::Kind::Unsigned:
    890 return left->unsigned_number == right->unsigned_number;
    891 case Node::Kind::Real:
    892 return left->real_number == right->real_number;
    893 case Node::Kind::String:
    894 return left->string == right->string;
    895 case Node::Kind::Array:
    896 if (left->array.size() != right->array.size())
    897 return false;
    898 for (size_t i = 0; i < left->array.size(); ++i)
    899 if (!equal_nodes(left->array[i], right->array[i]))
    900 return false;
    901 return true;
    902 case Node::Kind::Object:
    903 if (left->object.size() != right->object.size())
    904 return false;
    905 for (const auto &item : left->object) {
    906 auto found = right->object.find(item.first);
    907 if (found == right->object.end() ||
    908 !equal_nodes(item.second, found->second))
    909 return false;
    910 }
    911 return true;
    912 }
    913 return false;
    914}

    934inline std::string Value::normalize_for_opencv(const std::string &input) {
    935 /* Raw marker rejection reserves an internal transport namespace, not a user data format. */
    936 if (input.find("__USHER_OPENCV_JSON_") != std::string::npos)
    937 throw exception("JSON input uses a reserved OpenCV bridge token");
    938 std::string output;
    939 output.reserve(input.size() + 32);
    940 bool in_string = false;
    941 const auto hex_value = [](char character) -> int {
    942 if (character >= '0' && character <= '9')
    943 return character - '0';
    944 if (character >= 'a' && character <= 'f')
    945 return character - 'a' + 10;
    946 if (character >= 'A' && character <= 'F')
    947 return character - 'A' + 10;
    948 return -1;
    949 };
    950 const auto unicode_escape = [&](size_t offset,
    951 uint32_t *codepoint) -> size_t {
    952 if (offset + 4 > input.size())
    953 throw exception("truncated JSON Unicode escape");
    954 uint32_t value = 0;
    955 for (size_t digit = 0; digit < 4; ++digit) {
    956 const int nibble = hex_value(input[offset + digit]);
    957 if (nibble < 0)
    958 throw exception("invalid JSON Unicode escape");
    959 value = (value << 4) | static_cast<uint32_t>(nibble);
    960 }
    961 *codepoint = value;
    962 return offset + 4;
    963 };
    964 const auto append_utf8 = [&output](uint32_t codepoint) {
    965 if (codepoint <= 0x7f)
    966 output.push_back(static_cast<char>(codepoint));
    967 else if (codepoint <= 0x7ff) {
    968 output.push_back(static_cast<char>(0xc0 | (codepoint >> 6)));
    969 output.push_back(static_cast<char>(0x80 | (codepoint & 0x3f)));
    970 } else if (codepoint <= 0xffff) {
    971 output.push_back(static_cast<char>(0xe0 | (codepoint >> 12)));
    972 output.push_back(static_cast<char>(0x80 | ((codepoint >> 6) & 0x3f)));
    973 output.push_back(static_cast<char>(0x80 | (codepoint & 0x3f)));
    974 } else {
    975 output.push_back(static_cast<char>(0xf0 | (codepoint >> 18)));
    976 output.push_back(static_cast<char>(0x80 | ((codepoint >> 12) & 0x3f)));
    977 output.push_back(static_cast<char>(0x80 | ((codepoint >> 6) & 0x3f)));
    978 output.push_back(static_cast<char>(0x80 | (codepoint & 0x3f)));
    979 }
    980 };
    981 for (size_t i = 0; i < input.size();) {
    982 const char c = input[i];
    983 if (in_string) {
    984 if (c == '"') {
    985 in_string = false;
    986 output.push_back(c);
    987 ++i;
    988 continue;
    989 }
    990 if (c == '\\') {
    991 if (i + 1 >= input.size())
    992 throw exception("truncated JSON string escape");
    993 const char escaped = input[i + 1];
    994 if (escaped == '/') {
    995 output.push_back('/');
    996 i += 2;
    997 continue;
    998 }
    999 if (escaped == 'u') {
    1000 uint32_t codepoint = 0;
    1001 size_t next = unicode_escape(i + 2, &codepoint);
    1002 /* Combine UTF-16 escapes before UTF-8 encoding; isolated halves are not scalars. */
    1003 if (codepoint >= 0xd800 && codepoint <= 0xdbff) {
    1004 if (next + 6 > input.size() || input[next] != '\\' ||
    1005 input[next + 1] != 'u')
    1006 throw exception("unpaired JSON high surrogate");
    1007 uint32_t low = 0;
    1008 next = unicode_escape(next + 2, &low);
    1009 if (low < 0xdc00 || low > 0xdfff)
    1010 throw exception("invalid JSON surrogate pair");
    1011 codepoint = 0x10000 + ((codepoint - 0xd800) << 10) + (low - 0xdc00);
    1012 } else if (codepoint >= 0xdc00 && codepoint <= 0xdfff) {
    1013 throw exception("unpaired JSON low surrogate");
    1014 }
    1015 append_utf8(codepoint);
    1016 i = next;
    1017 continue;
    1018 }
    1019 if (std::string_view("\\\"nrtbf").find(escaped) ==
    1020 std::string_view::npos)
    1021 throw exception("invalid JSON string escape");
    1022 output.push_back(c);
    1023 output.push_back(escaped);
    1024 i += 2;
    1025 continue;
    1026 }
    1027 output.push_back(c);
    1028 ++i;
    1029 continue;
    1030 }
    1031 if (c == '"') {
    1032 in_string = true;
    1033 output.push_back(c);
    1034 ++i;
    1035 continue;
    1036 }
    1037 const auto token = [&](const char *text, const char *replacement) {
    1038 const size_t length = std::char_traits<char>::length(text);
    1039 const bool left =
    1040 i == 0 || !(std::isalnum(static_cast<unsigned char>(input[i - 1])) ||
    1041 input[i - 1] == '_');
    1042 const size_t end = i + length;
    1043 const bool right =
    1044 end >= input.size() ||
    1045 !(std::isalnum(static_cast<unsigned char>(input[end])) ||
    1046 input[end] == '_');
    1047 if (!left || !right || input.compare(i, length, text) != 0)
    1048 return false;
    1049 output += "\"";
    1050 output += replacement;
    1051 output += "\"";
    1052 i = end;
    1053 return true;
    1054 };
    1055 /* Only outside-string tokens are rewritten; quoted ordinary words keep their meaning. */
    1056 if (token("null", "__USHER_OPENCV_JSON_NULL__") ||
    1057 token("true", "__USHER_OPENCV_JSON_TRUE__") ||
    1058 token("false", "__USHER_OPENCV_JSON_FALSE__"))
    1059 continue;
    1060 if (c == '-' || std::isdigit(static_cast<unsigned char>(c))) {
    1061 size_t end = i;
    1062 while (end < input.size() &&
    1063 (std::isdigit(static_cast<unsigned char>(input[end])) ||
    1064 input[end] == '-' || input[end] == '+' || input[end] == '.' ||
    1065 input[end] == 'e' || input[end] == 'E'))
    1066 ++end;
    1067 const std::string number = input.substr(i, end - i);
    1068 if (number.find_first_of(".eE") == std::string::npos) {
    1069 /* Carry integer digits as text so FileStorage cannot narrow them before from_chars. */
    1070 const char *prefix = !number.empty() && number[0] == '-'
    1071 ? "\"__USHER_OPENCV_JSON_INT__"
    1072 : "\"__USHER_OPENCV_JSON_UINT__";
    1073 output += prefix;
    1074 output += number;
    1075 output += "__\"";
    1076 i = end;
    1077 continue;
    1078 }
    1079 output += number;
    1080 i = end;
    1081 continue;
    1082 }
    1083 output.push_back(c);
    1084 ++i;
    1085 }
    1086 return output;
    1087}

    1203inline std::string Value::write_opencv(const Value &value, int indent) {
    1204 cv::FileStorage storage("", cv::FileStorage::WRITE | cv::FileStorage::MEMORY |
    1205 cv::FileStorage::FORMAT_JSON);
    1206 if (!storage.isOpened())
    1207 throw exception("cannot open OpenCV JSON writer");
    1208 if (value.is_object()) {
    1209 for (const auto &item : value.node_->object)
    1210 write_node(storage, item.first, Value(item.second), true);
    1211 } else {
    1212 write_node(storage, "", value, false);
    1213 }
    1214 std::string output = storage.releaseAndGetString();
    1215 /* Restore markers after serialization, when OpenCV has finished quoting transport strings. */
    1216 const auto replace_token = [&output](const std::string &marker,
    1217 const std::string &replacement) {
    1218 const std::string quoted = "\"" + marker + "\"";
    1219 size_t position = 0;
    1220 while ((position = output.find(quoted, position)) != std::string::npos) {
    1221 output.replace(position, quoted.size(), replacement);
    1222 position += replacement.size();
    1223 }
    1224 };
    1225 replace_token("__USHER_OPENCV_JSON_NULL__", "null");
    1226 replace_token("__USHER_OPENCV_JSON_TRUE__", "true");
    1227 replace_token("__USHER_OPENCV_JSON_FALSE__", "false");
    1228 const auto replace_integer = [&output](const char *prefix) {
    1229 size_t position = 0;
    1230 while ((position = output.find(prefix, position)) != std::string::npos) {
    1231 const size_t start = position;
    1232 const size_t digits = position + std::char_traits<char>::length(prefix);
    1233 const size_t end = output.find("__\"", digits);
    1234 if (end == std::string::npos)
    1235 break;
    1236 const size_t quoted_start = start - 1;
    1237 output.replace(quoted_start, end + 3 - quoted_start,
    1238 output.substr(digits, end - digits));
    1239 position = quoted_start + 1;
    1240 }
    1241 };
    1242 replace_integer("__USHER_OPENCV_JSON_INT__");
    1243 replace_integer("__USHER_OPENCV_JSON_UINT__");
    1244 if (indent >= 0)
    1245 return output;
    1246 std::string compact;
    1247 compact.reserve(output.size());
    1248 /* Whitespace inside quoted strings is data; track escaped quotes before removing trivia. */
    1249 bool in_string = false, escaped = false;
    1250 for (char character : output) {
    1251 if (in_string) {
    1252 compact.push_back(character);
    1253 if (escaped)
    1254 escaped = false;
    1255 else if (character == '\\')
    1256 escaped = true;
    1257 else if (character == '"')
    1258 in_string = false;
    1259 } else if (character == '"') {
    1260 in_string = true;
    1261 compact.push_back(character);
    1262 } else if (character != ' ' && character != '\t' && character != '\n' &&
    1263 character != '\r') {
    1264 compact.push_back(character);
    1265 }
    1266 }
    1267 return compact;
    1268}
