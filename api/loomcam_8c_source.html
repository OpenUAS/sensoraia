    74/** @brief Read one bounded observation field from an already-fetched EXIF description. */
    75static bool description_lookup(const struct exif_value *description, const char *key,
    76 char value[64], bool *found)
    77{
    78 *found = false;
    79 if (!description || description->format != EXIF_ASCII || !description->data) return false;
    80 const size_t length = strlen(key);
    81 char selected[64];
    82 size_t selected_length = 0;
    83 for (size_t index = 0; index + length < description->size; ++index) {
    84 if (index && description->data[index - 1] != ' ' && description->data[index - 1] != ';') continue;
    85 if (memcmp(description->data + index, key, length)) continue;
    86 if (*found) return false;
    87 size_t used = 0, cursor = index + length;
    88 while (cursor < description->size && description->data[cursor] && description->data[cursor] != ';' &&
    89 description->data[cursor] != ' ') {
    90 if (used + 1 == 64) return false;
    91 selected[used++] = (char)description->data[cursor++];
    92 }
    93 if (!used) return false;
    94 selected[used] = '\0';
    95 selected_length = used;
    96 *found = true;
    97 }
    98 if (*found) {
    99 for (size_t index = 0; index <= selected_length; ++index) value[index] = selected[index];
    100 }
    101 return true;
    102}

    104/** @brief Require a uniquely identified flight-description value.
    105 * @param description EXIF description to scan.
    106 * @param key Exact field prefix including the equals sign.
    107 * @param[out] value Buffer for up to 63 characters and the terminator.
    108 * @return True only if a well-formed matching field was found.
    109 * @details Use the duplicate-aware lookup even when absence is an error. */
    110static bool description_field(const struct exif_value *description, const char *key, char value[64])
    111{
    112 bool found = false;
    113 return description_lookup(description, key, value, &found) && found;
    114}

    116/** @brief Read a required finite numeric metadata field.
    117 * @param description EXIF description to scan.
    118 * @param key Exact field prefix including the equals sign.
    119 * @param[out] number Parsed value, written on success.
    120 * @return True for a present, fully parsed number.
    121 * @details Reject missing and malformed fields alike when no default is allowed. */
    122static bool description_number(const struct exif_value *description, const char *key, double *number)
    123{
    124 char value[64];
    125 return description_field(description, key, value) && ini_real(value, number);
    126}

    128/** @brief Read numeric metadata with a default for absence only.
    129 * @param description EXIF description to scan.
    130 * @param key Exact field prefix including the equals sign.
    131 * @param default_value Result if the key is absent.
    132 * @param[out] number Parsed or default value on success.
    133 * @return False for malformed or duplicate evidence, even if a default exists.
    134 * @details Keep corrupt evidence distinct from older records missing an optional field. */
    135static bool description_optional_number(const struct exif_value *description, const char *key,
    136 double default_value, double *number)
    137{
    138 char value[64];
    139 bool found = false;
    140 if (!description_lookup(description, key, value, &found)) return false;
    141 if (!found) {
    142 *number = default_value;
    143 return true;
    144 }
    145 return ini_real(value, number);
    146}

    148/** @brief Read the graph and descriptor identities declared by the model metadata.
    149 * @param settings Selected model and descriptor paths.
    150 * @param[out] graph Buffer for the declared graph digest.
    151 * @param[out] descriptor Buffer for the descriptor digest.
    152 * @return True after parsing the bounded descriptor and checking model readability.
    153 * @details Read descriptor identity without creating a runtime session. This
    154 * helper does not hash the graph file itself. */
    155static bool read_onnx_descriptor_identity(const struct inference_settings *settings,
    156 char graph[65], char descriptor[65])
    157{
    158 struct stat model_status;
    159 if (!settings || !settings->model[0] || !settings->descriptor[0] ||
    160 stat(settings->model, &model_status) != 0 || !S_ISREG(model_status.st_mode) ||
    161 model_status.st_size <= 0 || access(settings->model, R_OK) != 0) return false;
    162 static char contents[16385];
    163 static struct inference_model model;
    164 char error[256];
    165 size_t length = 0;
    166 if (!ini_read_file(settings->descriptor, contents, sizeof(contents), &length) ||
    167 !inference_model_parse(&model, contents, length, error, sizeof(error))) return false;
    168 for (size_t index = 0; index <= 64; ++index) {
    169 graph[index] = model.graph_sha256[index];
    170 descriptor[index] = model.descriptor_sha256[index];
    171 }
    172 return true;
    173}

    175/** @brief Decode an unsigned field from LOOM's little-endian record.
    176 * @param bytes At least eight readable bytes.
    177 * @return Decoded unsigned value.
    178 * @details Assemble bytes rather than reading a native word at a potentially
    179 * unaligned record offset. */
    180static uint64_t feature_u64(const unsigned char *bytes)
    181{
    182 uint64_t number = 0;
    183 for (unsigned index = 0; index < 8; ++index) number |= (uint64_t)bytes[index] << (8 * index);
    184 return number;
    185}

    187/** @brief Generate bounded LOOM feature evidence for an existing JPEG.
    188 * @param settings ONNX model assignment and runtime configuration.
    189 * @param options Contract that governs acceptable output shape.
    190 * @param image Input JPEG path.
    191 * @param sequence Frame identity to embed.
    192 * @param timestamp_ns Sensor timestamp in nanoseconds to embed.
    193 * @param[out] record Destination for packed feature evidence.
    194 * @param capacity Available bytes in @p record.
    195 * @param[out] record_size Packed length; use only on success.
    196 * @return True after successful inference and packing.
    197 * @details Validate tensor shape before inference, enforce the stricter
    198 * direction-vector shape when required, and serialize before closing the session.
    199 * Calls must not overlap because the context is shared. */
    200static bool run_onnx_in_memory(const struct inference_settings *settings,
    201 const struct loom_return_options *options,
    202 const char *image, uint64_t sequence, uint64_t timestamp_ns,
    203 unsigned char *record, size_t capacity, size_t *record_size)
    204{
    205 static struct inference_context context;
    206 char error[512];
    207 context = (struct inference_context){0};
    208 if (!inference_context_open(&context, settings, "loom_features_v1", error, sizeof(error))) return false;
    209 size_t values = 0;
    210 bool valid_shape = context.model.input_count == 1 && context.model.image.channels != 0;
    211 for (size_t index = 0; valid_shape && index < context.model.output_count; ++index) {
    212 const struct onnx_tensor *tensor = &context.model.outputs[index];
    213 if (tensor->type != ONNX_FLOAT32 || tensor->bytes / sizeof(float) > 4096 - values) valid_shape = false;
    214 else values += tensor->bytes / sizeof(float);
    215 }
    216 if (options->frame != LOOM_RETURN_NONE && options->frame != LOOM_RETURN_SIMULATED &&
    217 (context.model.output_count != 1 || (values != 2 && values != 3))) valid_shape = false;
    218 bool ok = valid_shape &&
    219 inference_jpeg_prepare(&context, image, error, sizeof(error)) &&
    220 inference_context_run(&context, error, sizeof(error));
    221 if (ok) {
    222 *record_size = loom_onnx_features_pack(record, capacity, &context.model, sequence, timestamp_ns);
    223 ok = *record_size != 0;
    224 }
    225 inference_context_close(&context);
    226 return ok;
    227}

    229/**
    230 * @brief Interpret retained same-photo neural vectors under the declared return contract.
    231 * @param frame Current saved AIcam photo and reference/model configuration.
    232 * @param point Required destination in camera-point wire units.
    233 * @return False when model/pose/identity/direction validation rejects the photo.
    234 * @warning Never replace no-match with the current pose, home position or guessed
    235 * direction: those would look like valid navigation evidence on the FC.
    236 * @details Generic embeddings are never guessed to be return directions. Only
    237 * the PC simulation harness may turn them into an explicitly marked mock vector.
    238 * The persistent owner subsequently limits movement and suppresses stale phases.
    239 */
    240static bool match_previous_photos(const struct loom_frame *frame, struct camera_point_result *point)
    241{
    242 *point = (struct camera_point_result) {.session_id = frame->sequence};
    243 struct inference_settings settings;
    244 struct loom_return_options options;
    245 if (!return_settings(frame->settings, &settings, &options) || options.frame == LOOM_RETURN_NONE) return false;
    246 static unsigned char payload[EXIF_MAX_PAYLOAD], information[IMX500_TENSOR_INFO_BYTES];
    247 static unsigned char onnx_record[LOOM_ONNX_FEATURE_CAPACITY];
    248 static float features[4096];
    249 size_t size = 0;
    250 struct exif_view view;
    251 struct exif_value description;
    252 if (exif_read_jpeg(frame->image, payload, sizeof(payload), &size) || exif_parse(&view, payload, size) ||
    253 !exif_get(&view, EXIF_PRIMARY, EXIF_IMAGE_DESCRIPTION, &description) ||
    254 description.format != EXIF_ASCII || !description.data)
    255 return false;
    256 char aligned_value[64];
    257 bool aligned_found = false;
    258 if (!description_lookup(&description, "observation_pose_aligned=", aligned_value, &aligned_found)) return false;
    259 if (aligned_found && strcmp(aligned_value, "true") && strcmp(aligned_value, "false")) return false;
    260 const bool aligned = aligned_found && !strcmp(aligned_value, "true");
    261 const unsigned home_fields = (1U << 5) | (1U << 6);
    262 const bool home_configured = (options.seen & home_fields) == home_fields;
    263 double latitude = 0, longitude = 0, altitude = 0, agl = 0, roll = 0, pitch = 0, yaw = 0;
    264 if (aligned) {
    265 if (!description_number(&description, "observation_lat_deg=", &latitude) || fabs(latitude) > 85 ||
    266 !description_number(&description, "observation_lon_deg=", &longitude) || fabs(longitude) > 180 ||
    267 !description_number(&description, "observation_alt_m=", &altitude) || fabs(altitude) > 2000000 ||
    268 !description_number(&description, "observation_agl_m=", &agl) || agl < 0 || agl > 10000 ||
    269 !description_optional_number(&description, "observation_roll_deg=", 0, &roll) ||
    270 !description_optional_number(&description, "observation_pitch_deg=", 0, &pitch) ||
    271 !description_number(&description, "observation_yaw_deg=", &yaw)) return false;
    272 } else {
    273 if (!description_optional_number(&description, "lat_deg=",
    274 home_configured ? options.home_latitude : NAN, &latitude) ||
    275 !isfinite(latitude) || fabs(latitude) > 85 ||
    276 !description_optional_number(&description, "lon_deg=",
    277 home_configured ? options.home_longitude : NAN, &longitude) ||
    278 !isfinite(longitude) || fabs(longitude) > 180 ||
    279 !description_number(&description, "alt_m=", &altitude) || fabs(altitude) > 2000000 ||
    280 !description_number(&description, "groundalt_m=", &agl) || agl < 0 || agl > 10000 ||
    281 !description_optional_number(&description, "roll_deg=", 0, &roll) ||
    282 !description_optional_number(&description, "pitch_deg=", 0, &pitch) ||
    283 !description_number(&description, "yaw_deg=", &yaw)) return false;
    284 if (!latitude && !longitude && home_configured) {
    285 latitude = options.home_latitude;
    286 longitude = options.home_longitude;
    287 }
    288 }
    289 if (!isfinite(roll) || !isfinite(pitch) || !isfinite(yaw) ||
    290 fabs(roll) > options.limits.maximum_tilt_deg ||
    291 fabs(pitch) > options.limits.maximum_tilt_deg ||
    292 fabs(yaw) > 36000) return false;
    293 char identity[64];
    294 uint64_t observed_sequence = 0, observed_time = 0, shot_number = 0;
    295 if (!description_field(&description, "observation_sequence=", identity) || !ini_uint(identity, &observed_sequence) ||
    296 !description_field(&description, "sensor_timestamp_ns=", identity) || !ini_uint(identity, &observed_time))
    297 return false;
    298 if (observed_sequence) {
    299 if (observed_sequence != frame->sequence) return false;
    300 } else if (!description_field(&description, "nr=", identity) || !ini_uint(identity, &shot_number) ||
    301 shot_number != frame->sequence) {
    302 return false;
    303 }
    304 float vector[3];
    305 uint64_t feature_sequence = 0, feature_time = 0;
    306 struct exif_value record;
    307 if (settings.backend == INFERENCE_ONNX) {
    308 char graph[65], descriptor[65];
    309 if (!read_onnx_descriptor_identity(&settings, graph, descriptor)) return false;
    310 if (!exif_get(&view, EXIF_CAMERA, LOOM_ONNX_FEATURES_TAG, &record)) {
    311 size_t packed_size = 0;
    312 if (!run_onnx_in_memory(&settings, &options, frame->image,
    313 observed_sequence ? observed_sequence : frame->sequence,
    314 observed_time, onnx_record, sizeof(onnx_record), &packed_size))
    315 return false;
    316 record.data = onnx_record;
    317 record.size = packed_size;
    318 }
    319 if (options.frame == LOOM_RETURN_SIMULATED) {
    320 if (!loom_onnx_features_valid(record.data, record.size, graph, descriptor)) return false;
    321 } else if (!loom_onnx_return_vector(record.data, record.size, graph, descriptor, vector)) {
    322 return false;
    323 }
    324 feature_sequence = feature_u64(record.data + 136);
    325 feature_time = feature_u64(record.data + 144);
    326 } else if (settings.backend == INFERENCE_IMX500) {
    327 size_t count = 0;
    328 uint64_t sequence, sensor_timestamp_ns;
    329 if (!exif_get(&view, EXIF_CAMERA, LOOM_FEATURES_TAG, &record) ||
    330 !loom_features_unpack(record.data, record.size, features, 4096, information, sizeof(information),
    331 &count, &sequence, &sensor_timestamp_ns) ||
    332 !count || (options.frame != LOOM_RETURN_SIMULATED && count != 2 && count != 3) ||
    333 !sequence || !sensor_timestamp_ns) return false;
    334 if (options.frame != LOOM_RETURN_SIMULATED) {
    335 vector[0] = features[0]; vector[1] = features[1]; vector[2] = count == 3 ? features[2] : 1;
    336 }
    337 feature_sequence = sequence;
    338 feature_time = sensor_timestamp_ns;
    339 } else return false;
    340 if (feature_time != observed_time ||
    341 feature_sequence != (observed_sequence ? observed_sequence : frame->sequence))
    342 return false;
    343 const bool use_home_position = (aligned || options.frame == LOOM_RETURN_SIMULATED) && home_configured;
    344 const double origin_latitude = use_home_position ? options.home_latitude : latitude;
    345 const double origin_longitude = use_home_position ? options.home_longitude : longitude;
    346 const double radians = 0.017453292519943295, earth = 6378137;
    347 const double scale = cos(origin_latitude * radians);
    348 const struct loom_return_point position = use_home_position ? (struct loom_return_point){
    349 earth * scale * remainder(longitude - origin_longitude, 360) * radians,
    350 earth * (latitude - origin_latitude) * radians
    351 } : (struct loom_return_point){0, 0};
    352 if (options.frame == LOOM_RETURN_SIMULATED) {
    353 uint64_t seed = UINT64_C(1469598103934665603) ^ frame->sequence;
    354 for (size_t index = 0; index < record.size; ++index)
    355 seed = (seed ^ record.data[index]) * UINT64_C(1099511628211);
    356 loom_return_simulated_vector(seed, position, yaw, vector);
    357 }
    358 struct loom_return_point direction, target;
    359 if (!loom_return_vector_attitude(options.frame, vector[0], vector[1], roll, pitch, yaw,
    360 options.limits.maximum_tilt_deg, &direction) ||
    361 !loom_return_direction(&options.limits, position, direction, vector[2], &target)) return false;
    362 const double target_latitude = origin_latitude + target.north / earth / radians;
    363 const double target_longitude = remainder(origin_longitude + target.east / (earth * scale) / radians, 360);
    364 if (!isfinite(target_latitude) || !isfinite(target_longitude) ||
    365 fabs(target_latitude) > 85 || fabs(target_longitude) > 180 ||
    366 (!llround(target_latitude * 1e7) && !llround(target_longitude * 1e7))) return false;
    367 const double step_m = hypot(target.east - position.east, target.north - position.north);
    368 const bool arrived = step_m <= options.limits.maximum_step_m;
    369 *point = (struct camera_point_result){.session_id = frame->sequence, .valid = true,
    370 .lat = (int32_t)llround(target_latitude * 1e7), .lon = (int32_t)llround(target_longitude * 1e7),
    371 .alt_mm = (int32_t)llround(altitude * 1000), .agl_mm = (int32_t)llround(agl * 1000),
    372 .quality = (int32_t)llround(vector[2] * 1000),
    373 .metric = arrived ? LOOM_METRIC_ARRIVED
    374 : (use_home_position ? LOOM_METRIC_STEERING : LOOM_METRIC_BARO_MAGNETO)};
    375 return true;
    376}

    378/**
    379 * @brief Parse a nonzero correlation sequence without truncation or signed input.
    380 * @param text Entire decimal token; no whitespace, sign or suffix is accepted.
    381 * @param[out] sequence Destination, written only on successful validation.
    382 * @return True for a positive uint32_t photo number; false otherwise.
    383 */
    384static bool parse_sequence(const char *text, uint32_t *sequence)
    385{
    386 if (!text || *text < '0' || *text > '9') return false;
    387 char *end = NULL;
    388 errno = 0;
    389 const unsigned long value = strtoul(text, &end, LOOM_DECIMAL_RADIX);
    390 if (errno || *end || !value || value > UINT32_MAX) return false;
    391 *sequence = (uint32_t)value;
    392 return true;
    393}

    395/** @brief Reuse common path/INI policy without opening a model or camera. */
    396static bool return_settings(const char *path, struct inference_settings *settings,
    397 struct loom_return_options *options)
    398{
    399 if (path && !strcmp(path, "-")) path = NULL;
    400 char discovered[INFERENCE_PATH_CAPACITY], error[2048];
    401 inference_settings_init(settings, INFERENCE_AUTO);
    402 loom_return_options_init(options);
    403 if (!path) {
    404 if (!inference_settings_find("loomcam", "loom_cam", discovered, sizeof(discovered))) return true;
    405 path = discovered;
    406 }
    407 if (inference_settings_read(settings, path, loom_return_setting, options, error, sizeof(error))) {
    408 if (loom_return_options_valid(options)) return true;
    409 fprintf(stderr, "loomcam: inconsistent return limits\n");
    410 return false;
    411 }
    412 fprintf(stderr, "loomcam: %s\n", error);
    413 return false;
    414}

    416/** @brief Describe whether PC imagery can run a selected ONNX graph. */
    417static int simulation_mode(int argc, char **argv)
    418{
    419 const char *path = NULL;
    420 bool strict = false;
    421 for (int index = 2; index < argc; ++index) {
    422 if (!strcmp(argv[index], "--strict") && !strict) strict = true;
    423 else if (!strcmp(argv[index], "--settings") && !path && ++index < argc) path = argv[index];
    424 else return 2;
    425 }
    426 struct inference_settings settings;

    451/** @brief Split one bounded ASCII command, rejecting binary/trailing/oversized input. */
    452static int return_line(char *line, size_t capacity, unsigned timeout_ms)
    453{
    454 struct timespec started;
    455 if (timeout_ms && clock_gettime(CLOCK_MONOTONIC, &started)) return -1;
    456 size_t length = 0;
    457 for (;;) {
    458 if (timeout_ms) {
    459 struct timespec now;
    460 if (clock_gettime(CLOCK_MONOTONIC, &now)) return -1;
    461 const int64_t elapsed = (int64_t)(now.tv_sec - started.tv_sec) * 1000 +
    462 (now.tv_nsec - started.tv_nsec) / 1000000;
    463 if (elapsed < 0 || elapsed >= timeout_ms) return -1;
    464 struct pollfd input = {.fd = STDIN_FILENO, .events = POLLIN};
    465 const int ready = poll(&input, 1, (int)(timeout_ms - elapsed));
    466 if (ready < 0 && errno == EINTR) continue;
    467 if (ready <= 0) return -1;
    468 }
    469 const int character = fgetc(stdin);
    470 if (character == EOF) return length ? -1 : 0;
    471 if (character == '\n') { line[length] = '\0'; return length ? 1 : -1; }
    472 if (character < 32 || character >= 127 || length + 1 >= capacity) return -1;
    473 line[length++] = (char)character;
    474 }
    475}

    477/** @brief Keep bounded T1 state separate from the waypoint transport transaction. */
    478static int return_server(int argc, char **argv)
    479{
    480 const char *path = NULL;
    481 bool simulation = false;
    482 uint32_t seed = 19;
    483 for (int index = 2; index < argc; ++index) {
    484 if (!strcmp(argv[index], "--simulation") && !simulation) simulation = true;
    485 else if (!strcmp(argv[index], "--settings") && !path && ++index < argc) path = argv[index];
    486 else if (!strcmp(argv[index], "--seed") && ++index < argc && parse_sequence(argv[index], &seed)) {}
    487 else return 2;
    488 }
    489 struct inference_settings settings;
    490 struct loom_return_options options;
    491 if (!return_settings(path, &settings, &options)) return 1;
    492 struct loom_return_state state = {.random_state = seed};
    493 uint32_t previous = 0;
    494 if (setvbuf(stdin, NULL, _IONBF, 0)) return 1;
    495 puts("READY LOOM_RETURN_V1");
    496 if (fflush(stdout)) return 1;
    497 char line[256];
    498 int line_status;
    499 while ((line_status = return_line(line, sizeof(line), 0)) > 0) {
    500 char *fields[10];
    501 unsigned count = 1;
    502 fields[0] = line;
    503 for (char *cursor = line; *cursor; ++cursor) {
    504 if (*cursor != ' ') continue;
    505 *cursor = '\0';
    506 if (count == 10 || !cursor[1] || cursor[1] == ' ') return 2;
    507 fields[count++] = cursor + 1;
    508 }
    509 uint32_t sequence;
    510 if (count < 2 || !parse_sequence(fields[1], &sequence) || sequence <= previous) return 2;
    511 previous = sequence;
    512 const bool simulated_request = !strcmp(fields[0], "SIM");
    513 const bool mocked_observation = !strcmp(fields[0], "SIMOBS");
    514 const bool model_observation = !strcmp(fields[0], "OBS");
    515 if ((!simulated_request && !mocked_observation && !model_observation) ||
    516 (model_observation && count != 8 && count != 10) || (mocked_observation && (!simulation || count != 7)) ||
    517 (simulated_request && (count != 2 || !simulation))) return 2;
    518 double values[8] = {0};
    519 if (model_observation || mocked_observation) {
    520 const unsigned value_count = model_observation ? count - 2 : 5;
    521 for (unsigned index = 0; index < value_count; ++index)
    522 if (!ini_real(fields[index + 2], &values[index])) return 2;
    523 }
    524 struct loom_return_state proposed = state;
    525 struct loom_return_point target, direction;
    526 bool accepted_vector = false;
    527 if (model_observation) {
    528 accepted_vector = loom_return_vector_attitude(options.frame, values[2], values[3],
    529 values[6], values[7], values[5],
    530 options.limits.maximum_tilt_deg, &direction) &&
    531 loom_return_direction(&options.limits, (struct loom_return_point){values[0], values[1]},
    532 direction, values[4], &target);
    533 } else if (mocked_observation) {
    534 direction = (struct loom_return_point){values[2], values[3]};
    535 accepted_vector = loom_return_direction(&options.limits,
    536 (struct loom_return_point){values[0], values[1]}, direction, values[4], &target);
    537 }
    538 accepted_vector = accepted_vector &&
    539 loom_return_update(&proposed, &options.limits, sequence, target, &target);
    540 const bool simulated_fallback = !accepted_vector && !mocked_observation && simulation &&
    541 loom_return_simulate(&proposed, &options.limits, sequence, &target);
    542 if (!accepted_vector && !simulated_fallback) {
    543 printf("SKIP %" PRIu32 " invalid_direction\n", sequence);
    544 if (fflush(stdout)) return 1;
    545 continue;
    546 }
    547 printf("MOVE %" PRIu32 " %.9f %.9f %s\n", sequence, target.east, target.north,
    548 accepted_vector ? (mocked_observation ? "simulated" : "neural") : "simulated");
    549 if (fflush(stdout)) return 1;
    550 char expected[64];
    551 snprintf(expected, sizeof(expected), "ACK %" PRIu32, sequence);
    552 if (return_line(line, sizeof(line), LOOM_CONFIRMATION_TIMEOUT_MS) != 1) return 1;
    553 if (!strcmp(line, expected)) {
    554 state = proposed;
    555 printf("OK %" PRIu32 "\n", sequence);
    556 } else printf("SKIP %" PRIu32 " unconfirmed\n", sequence);
    557 if (fflush(stdout)) return 1;
    558 }
    559 return line_status < 0 || ferror(stdin) ? 2 : 0;
    560}
