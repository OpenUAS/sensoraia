    19 * @file image_framing.h
    20 * @brief Crop, resample, and rotate BGRA images into the requested camera frame.

    22 * the exact crop geometry lets callers derive matching intrinsics. A single

    24 * second full image allocation when the input can be viewed directly.

    82 std::uint32_t packed = 0;
    83 for (std::size_t channel = 0; channel < sizeof(packed); ++channel)
    84 packed |= static_cast<std::uint32_t>(pixel[channel]) << (channel * 8);
    85 const auto bytes = _mm_cvtsi32_si128(static_cast<int>(packed));
    86 const auto zero = _mm_setzero_si128();

    90 static void writeRow(unsigned char* destination, int width, SamplePixel&& sample) {
    91 const auto rounding = _mm_set1_ps(0.5f);
    92 int x = 0;
    93 for (; x + 1 < width; x += 2) {
    94 const auto first = _mm_cvttps_epi32(_mm_add_ps(sample(x), rounding));
    95 const auto second = _mm_cvttps_epi32(_mm_add_ps(sample(x + 1), rounding));
    96 const auto words = _mm_packs_epi32(first, second);
    97 const auto bytes = _mm_packus_epi16(words, words);
    98 const auto packed = static_cast<std::uint64_t>(_mm_cvtsi128_si64(bytes));
    99 for (std::size_t channel = 0; channel < sizeof(packed); ++channel)
    100 destination[x * 4 + channel] = static_cast<unsigned char>(packed >> (channel * 8));
    101 }
    102 if (x < width) {
    103 const auto value = _mm_cvttps_epi32(_mm_add_ps(sample(x), rounding));
    104 const auto words = _mm_packs_epi32(value, value);
    105 const auto packed = static_cast<std::uint32_t>(_mm_cvtsi128_si32(_mm_packus_epi16(words, words)));
    106 for (std::size_t channel = 0; channel < sizeof(packed); ++channel)
    107 destination[x * 4 + channel] = static_cast<unsigned char>(packed >> (channel * 8));
    108 }

    188 /** @brief Write one sampled row to the byte raster.
    189 * @tparam SamplePixel Callable yielding four lanes for an output column.
    190 * @param[out] destination Row storage of at least four bytes per pixel.
    191 * @param width Output column count.
    192 * @param sample Interpolating sampler.
    193 * @details Round and clamp at the byte boundary, not during interpolation. */
    194 template<typename SamplePixel>
    195 static void writeRow(unsigned char* destination, int width, SamplePixel&& sample) {
    196 for (int column = 0; column < width; ++column) {
    197 float values[4]; store(values, sample(column));
    198 for (int channel = 0; channel < 4; ++channel)
    199 destination[column * 4 + channel] = static_cast<unsigned char>(
    200 std::clamp(static_cast<int>(values[channel] + .5f), 0, 255));
    201 }
    202 }

    204 /** @brief Precompute bilinear indices and fractions along one output axis.
    205 * @param[out] result One entry per output sample.
    206 * @param count Output samples on the axis.
    207 * @param sourceSize Positive source extent in pixels.
    208 * @param offset Crop offset in source pixels.
    209 * @param scale Source pixels per output sample.
    210 * @details Map pixel centers and clamp before computing neighbors.
    211 * Reusing these entries avoids repeated coordinate arithmetic in each row. */
    212 static void samples(std::vector<Sample>& result, int count, int sourceSize,
    213 double offset, double scale) {
    214 result.resize(count);
    215 for (int i = 0; i < count; ++i) {
    216 const double position = std::clamp(offset + (i + 0.5) * scale - 0.5,
    217 0.0, static_cast<double>(sourceSize - 1));
    218 const int first = static_cast<int>(position);
    219 result[i] = {first, std::min(first + 1, sourceSize - 1),
    220 static_cast<float>(position - first)};
    221 }
    222 }

    224 /** @brief Source-space crop and uniform scale needed for an output frame. */
    225 struct Crop {
    226 double x, y, width, height, scale;
    227 };
    228 /**
    229 * @brief Fit an output aspect ratio inside source bounds and rotation margin.
    230 * @param[in] sourceWidth Source image width in pixels.
    231 * @param[in] sourceHeight Source image height in pixels.
    232 * @param[in] width Requested output width in pixels.
    233 * @param[in] height Requested output height in pixels.
    234 * @param[in] inset Minimum source-edge margin in pixels.
    235 * @param[in] rotationBoundDegrees Maximum absolute rotation to accommodate.
    236 * @return Centered crop rectangle and the common scale for both axes.
    237 * @throws std::runtime_error If dimensions, margin, or rotation are invalid.
    238 */
    239 static Crop crop(int sourceWidth, int sourceHeight, int width, int height, int inset = 0,
    240 double rotationBoundDegrees = 0) {
    241 validateJpegSize(width, height);
    242 if (sourceWidth < 1 || sourceHeight < 1 || sourceWidth > 16384 || sourceHeight > 16384)
    243 throw std::runtime_error("Invalid source dimensions for JPEG framing");
    244 if (inset < 0 || sourceWidth <= 2 * inset || sourceHeight <= 2 * inset)
    245 throw std::runtime_error("Capture inset leaves no usable image");
    246 if (!std::isfinite(rotationBoundDegrees) || std::abs(rotationBoundDegrees) > 45)
    247 throw std::runtime_error("Image framing rotation must be within +/-45 degrees");
    248 const double angle = std::abs(rotationBoundDegrees) * 0.01745329251994329577;
    249 const double c = std::cos(angle), s = std::sin(angle);
    250 const double scale = std::min((sourceWidth - 2.0 * inset) / (width * c + height * s),
    251 (sourceHeight - 2.0 * inset) / (height * c + width * s));
    252 const double cropWidth = width * scale, cropHeight = height * scale;
    253 return {(sourceWidth - cropWidth) / 2, (sourceHeight - cropHeight) / 2,
    254 cropWidth, cropHeight, scale};
    255 }

    262 /**
    263 * @brief Frame BGRA pixels into a crop, resampled image, or rotated image.
    264 * @param[in] bgra Borrowed source pixel buffer.
    265 * @param[in] sourceWidth Source width in pixels.
    266 * @param[in] sourceHeight Source height in pixels.
    267 * @param[in] stride Source row stride in bytes.
    268 * @param[in] width Requested output width in pixels.
    269 * @param[in] height Requested output height in pixels.
    270 * @param[in] inset Excluded border in source pixels.
    271 * @param[in] rotationDegrees Applied rotation in the renderer's image coordinates.
    272 * @param[in] rotationBoundDegrees Maximum rotation used when selecting a safe crop.
    273 * @return BGRA view and the source crop represented by its pixels.
    274 * @throws std::runtime_error If input bounds or the rotated footprint are invalid.
    275 * @note A returned pointer to internal storage is invalidated by the next call.
    276 */
    277 View apply(const unsigned char* bgra, int sourceWidth, int sourceHeight, int stride,
    278 int width, int height, int inset = 0, double rotationDegrees = 0,
    279 double rotationBoundDegrees = 0) {
    280 if (!std::isfinite(rotationDegrees))
    281 throw std::runtime_error("Nonfinite image framing rotation");
    282 const auto bounds = crop(sourceWidth, sourceHeight, width, height, inset,
    283 std::max(std::abs(rotationDegrees), std::abs(rotationBoundDegrees)));
    284 if (!bgra || sourceWidth < 1 || sourceHeight < 1 ||
    285 sourceWidth > 16384 || sourceHeight > 16384 || stride < sourceWidth * 4)
    286 throw std::runtime_error("Invalid source BGRA dimensions/stride for JPEG framing");
    287 // A single sampling scale on both axes prevents distortion, even for fractional crop edges.
    288 const double scale = bounds.scale, cropWidth = bounds.width, cropHeight = bounds.height;
    289 const double cropX = bounds.x, cropY = bounds.y;
    290 if (rotationDegrees != 0) {
    291 buffer_.resize(static_cast<std::size_t>(width) * height * 4);
    292 const double angle = rotationDegrees * 0.01745329251994329577;
    293 const double c = std::cos(angle), s = std::sin(angle);
    294 for (int y = 0; y < height; ++y) {
    295 const double dy = (y + 0.5 - height / 2.0) * scale;
    296 writeRow(buffer_.data() + static_cast<std::size_t>(y) * width * 4, width, [&](int x) {
    297 const double dx = (x + 0.5 - width / 2.0) * scale;
    298 const double px = sourceWidth / 2.0 - 0.5 + c * dx - s * dy;
    299 const double py = sourceHeight / 2.0 - 0.5 + s * dx + c * dy;
    300 const int ix = static_cast<int>(std::floor(px)), iy = static_cast<int>(std::floor(py));
    301 if (ix < 0 || iy < 0 || ix + 1 >= sourceWidth || iy + 1 >= sourceHeight)
    302 throw std::runtime_error("Rotated image footprint lies outside capture pixels");
    303 const auto* top = bgra + static_cast<std::size_t>(iy) * stride + ix * 4;
    304 const auto* bottom = top + stride;
    305 const auto hx = splat(static_cast<float>(px - ix));
    306 const auto vy = splat(static_cast<float>(py - iy));
    307 const auto ta = channels(top), tb = channels(top + 4);
    308 const auto ba = channels(bottom), bb = channels(bottom + 4);
    309 const auto upper = sum(ta, product(difference(tb, ta), hx));
    310 const auto lower = sum(ba, product(difference(bb, ba), hx));
    311 return sum(upper, product(difference(lower, upper), vy));
    312 });
    313 }
    314 return {buffer_.data(), width, height, width * 4, cropX, cropY, cropWidth, cropHeight};
    315 }
    316 if (scale == 1 && std::floor(cropX) == cropX && std::floor(cropY) == cropY)
    317 return {bgra + static_cast<std::size_t>(cropY) * stride + static_cast<std::size_t>(cropX) * 4,
    318 width, height, stride, cropX, cropY, cropWidth, cropHeight};
    319 if (sourceWidth_ != sourceWidth || sourceHeight_ != sourceHeight ||
    320 width_ != width || height_ != height || inset_ != inset || rotationBound_ != rotationBoundDegrees) {
    321 samples(columns_, width, sourceWidth, cropX, scale);
    322 samples(rows_, height, sourceHeight, cropY, scale);
    323 buffer_.resize(static_cast<std::size_t>(width) * height * 4);
    324 sourceWidth_ = sourceWidth;
    325 sourceHeight_ = sourceHeight;
    326 width_ = width;
    327 height_ = height;
    328 inset_ = inset;
    329 rotationBound_ = rotationBoundDegrees;
    330 }
    331 if (scale < 1) {
    332 horizontal_[0].resize(static_cast<std::size_t>(width) * 4);
    333 horizontal_[1].resize(static_cast<std::size_t>(width) * 4);
    334 // Cache only within this image; neighboring output rows reuse source rows when upscaling.
    335 int cached[2] = {-1, -1};
    336 const auto horizontalRow = [&](int row, int protectedSlot) {
    337 for (int slot = 0; slot < 2; ++slot)
    338 if (cached[slot] == row) return slot;
    339 const int slot = protectedSlot == 0 ? 1 : 0;
    340 const auto* source = bgra + static_cast<std::size_t>(row) * stride;
    341 auto* destination = horizontal_[slot].data();
    342 for (int x = 0; x < width; ++x) {
    343 const auto& column = columns_[x];
    344 const auto a = channels(source + column.first * 4);
    345 const auto b = channels(source + column.second * 4);
    346 store(destination + x * 4,
    347 sum(a, product(difference(b, a), splat(column.weight))));
    348 }
    349 cached[slot] = row;
    350 return slot;
    351 };
    352 for (int y = 0; y < height; ++y) {
    353 const auto& row = rows_[y];
    354 const int first = horizontalRow(row.first, -1);
    355 const int second = horizontalRow(row.second, first);
    356 const auto* top = horizontal_[first].data();
    357 const auto* bottom = horizontal_[second].data();
    358 auto* destination = buffer_.data() + static_cast<std::size_t>(y) * width * 4;
    359 const auto vertical = splat(row.weight);
    360 writeRow(destination, width, [&](int x) {
    361 const auto upper = load(top + x * 4), lower = load(bottom + x * 4);
    362 return sum(upper, product(difference(lower, upper), vertical));
    363 });
    364 }
    365 return {buffer_.data(), width, height, width * 4, cropX, cropY, cropWidth, cropHeight};
    366 }
    367 for (int y = 0; y < height; ++y) {
    368 const auto& row = rows_[y];
    369 const auto* top = bgra + static_cast<std::size_t>(row.first) * stride;
    370 const auto* bottom = bgra + static_cast<std::size_t>(row.second) * stride;
    371 auto* destination = buffer_.data() + static_cast<std::size_t>(y) * width * 4;
    372 const auto vertical = splat(row.weight);
    373 writeRow(destination, width, [&](int x) {
    374 const auto& column = columns_[x];
    375 const auto horizontal = splat(column.weight);
    376 const auto a = channels(top + column.first * 4);
    377 const auto b = channels(top + column.second * 4);
    378 const auto d = channels(bottom + column.first * 4);
    379 const auto e = channels(bottom + column.second * 4);
    380 const auto upper = sum(a, product(difference(b, a), horizontal));
    381 const auto lower = sum(d, product(difference(e, d), horizontal));
    382 return sum(upper, product(difference(lower, upper), vertical));
    383 });
    384 }
    385 return {buffer_.data(), width, height, width * 4, cropX, cropY, cropWidth, cropHeight};
    386 }
    387};
