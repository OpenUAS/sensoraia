    89bool is_shot_value(const char* text) {
    90 const char* digit = text;
    91 if (*digit == '-' || *digit == '+') ++digit;
    92 if (*digit == '\0') return false;
    93 for (; *digit != '\0'; ++digit) {
    94 if (*digit < '0' || *digit > '9') return false;
    95 }
    96 errno = 0;
    97 char* end = nullptr;
    98 const long long value = std::strtoll(text, &end, 10);
    99 return errno != ERANGE && *end == '\0' &&
    100 value >= std::numeric_limits<int32_t>::min() &&
    101 value <= std::numeric_limits<int32_t>::max();
    102}

    104bool emulate_lens(const char* input, const char* profile, const char* cache,
    105 std::string& output) {
    106 char executable[4096];
    107 ssize_t length =
    108 readlink("/proc/self/exe", executable, sizeof(executable) - 1);
    109 if (length <= 0 || static_cast<size_t>(length) >= sizeof(executable) - 1)
    110 return false;
    111 executable[length] = '\0';
    112 std::string directory(executable);
    113 directory.resize(directory.find_last_of('/'));
    114 const std::string executable_path = directory + "/camera_lens";
    115 if (access(executable_path.c_str(), X_OK) != 0) {
    116 std::cerr << "SODA: native lens executable missing; run make camera-lens "
    117 "in the USHER directory\n";
    118 return false;
    119 }
    120 char derivative[4096];
    121 if (!image_lens_emulate(executable_path.c_str(), input, profile, cache,
    122 derivative, sizeof(derivative), nullptr))
    123 return false;
    124 output = derivative;
    125 return true;
    126}

    129/** @brief Select and run the postprocessor for one saved camera image.
    130 * @param argc Argument count including the program name.
    131 * @param argv Command-line argument vector.
    132 * @return Process status for argument handling and the selected processing path.
    133 * @details Resolve camera and optional lens settings before dispatching;
    134 * local-camera derivatives are explicitly selected, not inferred from a live shot. */
    135int main(int argc, char* argv[]) {
    136 const CameraHandler* selected = nullptr;
    137 const char* lens_profile = nullptr;
    138 const char* lens_cache = nullptr;
    139 std::vector<const char*> positional;
    140 bool options = true;
    141 bool local = false;
    142 bool help = false;
    143 bool version = false;
    144 for (int argument = 1; argument < argc; ++argument) {
    145 const char* value = argv[argument];
    146 if (options && (std::strcmp(value, "--lens-profile") == 0 ||
    147 std::strcmp(value, "--lens-cache") == 0)) {
    148 const char** destination = std::strcmp(value, "--lens-profile") == 0
    149 ? &lens_profile
    150 : &lens_cache;
    151 if (*destination != nullptr || argument + 1 >= argc ||
    152 argv[argument + 1][0] == '\0') {
    153 std::cerr << "SODA: missing or duplicate lens option\n";
    154 return 2;
    155 }
    156 *destination = argv[++argument];
    157 continue;
    158 }
    159 if (options && std::strcmp(value, "--") == 0) {
    160 options = false;
    161 continue;
    162 }
    163 if (options && std::strcmp(value, "--local") == 0) {
    164 if (local) {
    165 std::cerr << "SODA: duplicate --local option" << std::endl;
    166 return 2;
    167 }
    168 local = true;
    169 continue;
    170 }
    171 if (options && std::strcmp(value, "--help") == 0) {
    172 help = true;
    173 continue;
    174 }
    175 if (options && std::strcmp(value, "--version") == 0) {
    176 version = true;
    177 continue;
    178 }
    179 const CameraHandler* match = nullptr;
    180 if (options) {
    181 for (const CameraHandler& handler : handlers) {
    182 if (std::strcmp(value, handler.option) == 0) match = &handler;
    183 }
    184 }
    185 if (match != nullptr) {
    186 if (selected != nullptr) {
    187 std::cerr << "SODA: specify only one camera selector" << std::endl;
    188 return 2;
    189 }
    190 selected = match;
    191 } else if (options && value[0] == '-' && !is_shot_value(value)) {
    192 std::cerr << "SODA: unknown option " << value << std::endl;
    193 return 2;
    194 } else {
    195 positional.push_back(value);
    196 }
    197 }
    198 if (help) {
    199 print_usage(argv[0]);
    200 return 0;
    201 }
    202 if (version) {
    203 std::cout << SODA_BUILD_VERSION << std::endl;
    204 return 0;
    205 }
    206 if (positional.empty()) {
    207 std::cerr << "SODA: no image was provided" << std::endl;
    208 return 1;
    209 }
    210 if ((lens_cache != nullptr && lens_profile == nullptr) ||
    211 (lens_profile != nullptr && (!local || selected != &handlers[0]))) {
    212 std::cerr << "SODA: lens emulation requires --local --aicam --lens-profile "
    213 "FILE\n";
    214 return 2;
    215 }
    216 if (positional.size() != 1 && positional.size() != 11) {
    217 std::cerr << "SODA: expected IMAGE optionally followed by 10 shot integers"
    218 << std::endl;
    219 return 2;
    220 }
    221 for (size_t index = 1; index < positional.size(); ++index) {
    222 if (!is_shot_value(positional[index])) {
    223 std::cerr << "SODA: invalid shot integer " << positional[index]
    224 << std::endl;
    225 return 2;
    226 }
    227 }
    228
    229 std::ifstream image(positional[0], std::ios::binary | std::ios::ate);
    230 if (!image || image.tellg() <= 0) {
    231 std::cerr << "SODA: could not inspect image " << positional[0] << std::endl;
    232 return 1;
    233 }
    234
    235 if (local) std::cerr << "SODA: development-PC mode (--local)" << std::endl;
    236 std::string lens_output;
    237 const char* analysis_image = positional[0];
    238 if (lens_profile != nullptr) {
    239 if (!emulate_lens(positional[0], lens_profile, lens_cache, lens_output)) {
    240 std::cerr << "SODA: lens emulation failed; clean image retained, "
    241 "analysis skipped\n";
    242 return 1;
    243 }
    244 analysis_image = lens_output.c_str();
    245 std::cout << "SODA: lens-emulated analysis input " << analysis_image
    246 << '\n';
    247 }
    248 // Every backend's image is checked here, independent of `selected` -- usher's
    249 // own
    250 // "--aicam" selector looks identical whether or not --spottercam produced
    251 // this image, so the only reliable signal is whatever this image's own EXIF
    252 // says.
    253 if (!soda_process_object_detection(analysis_image)) {
    254 // A partial output failure must reach USHER's bounded retry policy.
    255 // Returning success here would silently abandon missing crops/CSV while
    256 // claiming completion.
    257 std::cerr << "SODA: object post-processing incomplete; original retained\n";
    258 return 1;
    259 }
    260 if (selected != nullptr) {
    261 selected->process();
    262 } else {
    263 std::cout << "SODA: I looked at the image. Cool, right?" << std::endl;
    264 }
    265 return 0;
    266}
