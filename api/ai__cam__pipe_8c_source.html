    100/** @brief Read the child supervision clock.
    101 * @return Monotonic milliseconds, or zero on clock failure.
    102 * @details Use elapsed time rather than calendar time for startup and shutdown. */
    103static int64_t monotonic_ms(void)
    104{
    105 struct timespec now;
    106 if (clock_gettime(CLOCK_MONOTONIC, &now) != 0) return 0;
    107 return (int64_t)now.tv_sec * 1000 + now.tv_nsec / 1000000;
    108}

    110/** @brief Wait until a short polling interval elapses.
    111 * @param milliseconds Requested wait in milliseconds.
    112 * @details After interruption, recompute the remaining budget instead of
    113 * restarting the full delay. */
    114static void wait_ms(int milliseconds)
    115{
    116 const int64_t deadline = monotonic_ms() + milliseconds;
    117 int64_t remaining;
    118 while ((remaining = deadline - monotonic_ms()) > 0) {
    119 if (poll(NULL, 0, remaining > INT_MAX ? INT_MAX : (int)remaining) >= 0 || errno != EINTR) break;
    120 }
    121}

    123/** @brief Poll the owned child and reap it if it exited.
    124 * @param[out] status Optional wait status; use only when waitpid reports exit.
    125 * @return True while the child remains owned.
    126 * @details Clear the PID only after reaping or ECHILD. Other wait errors
    127 * leave ownership intact rather than allowing a second camera process. */
    128static bool child_alive(int *status)
    129{
    130 if (child_pid <= 0) return false;
    131 pid_t result;
    132 do {
    133 result = waitpid(child_pid, status, WNOHANG);
    134 } while (result < 0 && errno == EINTR);
    135 if (result == 0 || (result < 0 && errno != ECHILD)) return true;
    136 child_pid = -1;
    137 return false;
    138}
    139
    140/** @brief Give the owned child a bounded interval to exit.
    141 * @param milliseconds Supervision budget in milliseconds.
    142 * @return True once no child is owned; false on timeout.
    143 * @details Poll so a driver that does not return cannot block a waitpid call. */
    144static bool wait_child(int milliseconds)
    145{
    146 const int64_t deadline = monotonic_ms() + milliseconds;
    147 while (child_alive(NULL)) {
    148 if (monotonic_ms() >= deadline) return false;
    149 wait_ms(10);
    150 }
    151 return true;
    152}

    154/** @brief Request shutdown and escalate if the camera child does not exit.
    155 * @return True when no child remains owned.
    156 * @details Use the persistent application's signal protocol when its
    157 * watch is active. Retain the PID if even forced termination cannot be reaped. */
    158static bool stop_child(void)
    159{
    160 if (!child_alive(NULL)) return true;
    161 kill(child_pid, watch_fd >= 0 ? SIGUSR2 : SIGTERM);
    162 if (wait_child(1000)) return true;
    163 kill(child_pid, SIGKILL);
    164 if (wait_child(1000)) return true;
    165 fprintf(stderr, "AI_CAM_PIPE:\tchild %ld has not exited; camera remains owned\n", (long)child_pid);
    166 return false;
    167}

    169/** @brief Retire the subprocess and its directory watch.
    170 * @return True if subprocess ownership was released.
    171 * @details Close the watch even after a timeout; subsequent starts must
    172 * still resolve the retained child PID before opening the camera. */
    173static bool persistent_stop(void)
    174{
    175 const bool stopped = stop_child();
    176 if (watch_fd >= 0) close(watch_fd);
    177 watch_fd = -1;
    178 return stopped;

    200/** @brief Reserve a private filename prefix for a camera child.
    201 * @param directory Existing output directory.
    202 * @return True if both path components fit and the counter did not wrap.
    203 * @details Include both process ID and session counter so a late event from
    204 * an earlier child is not accepted as a current capture. */
    205static bool new_session(const char *directory)
    206{
    207 if (session_number == UINT_MAX) return false;
    208 ++session_number;
    209 const int directory_length = snprintf(signal_directory, sizeof(signal_directory), "%s", directory);
    210 const int prefix_length = snprintf(signal_prefix, sizeof(signal_prefix), ".aicam_%ld_%u_",
    211 (long)getpid(), session_number);
    212 return directory_length > 0 && (size_t)directory_length < sizeof(signal_directory)
    213 && prefix_length > 0 && (size_t)prefix_length < sizeof(signal_prefix);
    214}

    216/** @brief Launch the configured camera application and record ownership.
    217 * @param arguments Null-terminated argument vector, including argv[0].
    218 * @return Zero on launch, or -1 on spawn failure.
    219 * @details Pass arguments directly rather than interpreting paths as shell input.
    220 * A successful spawn does not prove that the camera is ready. */
    221static int spawn_camera(char *const arguments[])
    222{
    223 const int result = posix_spawnp(&child_pid, USHER_AI_CAM_COMMAND, NULL, NULL, arguments, environ);
    224 if (result) {
    225 child_pid = -1;
    226 fprintf(stderr, "AI_CAM_PIPE:\tcannot start %s: %s\n", USHER_AI_CAM_COMMAND, strerror(result));
    227 return -1;
    228 }
    229 return 0;
    230}

    232/** @brief Start the signal-triggered camera application.
    233 * @param directory Existing output directory to watch and write.
    234 * @return Zero after launch, or -1 on ownership, watch, path, or spawn failure.
    235 * @details Attach the watch before launch to avoid missing completion events.
    236 * Record a warmup deadline; the first trigger waits for it outside startup. */
    237static int persistent_start(const char *directory)
    238{
    239 if (!persistent_stop() || !new_session(directory)) return -1;
    240 watch_fd = inotify_init1(IN_NONBLOCK | IN_CLOEXEC);
    241 if (watch_fd < 0 || inotify_add_watch(watch_fd, directory, IN_CLOSE_WRITE | IN_MOVED_TO) < 0) {
    242 persistent_stop();
    243 return -1;
    244 }
    245 char pattern[PATH_MAX];
    246 const int length = snprintf(pattern, sizeof(pattern), "%s/%s%%06d.jpg", directory, signal_prefix);
    247 if (length < 0 || (size_t)length >= sizeof(pattern)) {
    248 persistent_stop();
    249 return -1;
    250 }
    251 char *arguments[20] = {
    252 (char *)USHER_AI_CAM_COMMAND, (char *)"-n", (char *)"-t", (char *)"0",
    253 (char *)"--signal", (char *)"-o", pattern,
    254 (char *)"--width", (char *)USHER_AI_CAM_WIDTH,
    255 (char *)"--height", (char *)USHER_AI_CAM_HEIGHT
    256 };
    257 size_t count = 11;
    258 signal_zsl = zsl_enabled();
    259 if (signal_zsl) {
    260 arguments[count++] = (char *)"--zsl";
    261 arguments[count++] = (char *)"--buffer-count";
    262 arguments[count++] = (char *)"2";
    263 arguments[count++] = (char *)"--viewfinder-width";
    264 arguments[count++] = (char *)"640";
    265 arguments[count++] = (char *)"--viewfinder-height";
    266 arguments[count++] = (char *)"480";
    267 }
    268 arguments[count] = NULL;
    269 if (spawn_camera(arguments)) {
    270 persistent_stop();
    271 return -1;
    272 }
    273 ready_ms = monotonic_ms() + USHER_AI_CAM_STARTUP_MS;
    274 return 0;
    275}

    277/** @brief Publish a completed private JPEG at the caller's staging path.
    278 * @param source Private camera output, required to be nonempty and regular.
    279 * @param destination New sibling path; existing files are not replaced.
    280 * @return Zero if linked, even if private-name cleanup failed, or -1 if not linked.
    281 * @details A hard link makes the complete file visible without copying pixels
    282 * or reopening the JPEG for writing. */
    283static int publish_image(const char *source, const char *destination)
    284{
    285 struct stat image;
    286 /* Linking the completed private file publishes atomically and refuses overwrite. */
    287 if (lstat(source, &image) || !S_ISREG(image.st_mode) || image.st_size <= 0
    288 || link(source, destination)) return -1;
    289 if (unlink(source)) fprintf(stderr, "AI_CAM_PIPE:\tprivate image retained: %s\n", source);
    290 return 0;
    291}

    293/** @brief Request and collect one persistent-child capture.
    294 * @param destination New staging path for the completed JPEG.
    295 * @return Saved, not triggered, or failed after a trigger may have been delivered.
    296 * @details Drain old notifications before signaling. Require a completed event
    297 * with the current child's private prefix. Once signaled, an ambiguous outcome
    298 * is terminal; replay could duplicate an exposure. */
    299static enum capture_result persistent_shoot(const char *destination)
    300{
    301 if (watch_fd < 0 || !child_alive(NULL)) return CAPTURE_NOT_TRIGGERED;
    302 const int64_t remaining = ready_ms - monotonic_ms();
    303 if (remaining > 0) wait_ms((int)remaining);
    304 if (!child_alive(NULL)) return CAPTURE_NOT_TRIGGERED;
    305 _Alignas(struct inotify_event) char buffer[4096];
    306 while (read(watch_fd, buffer, sizeof(buffer)) > 0) {}
    307 if (kill(child_pid, SIGUSR1)) return CAPTURE_NOT_TRIGGERED;
    308 const int64_t deadline = monotonic_ms() + USHER_AI_CAM_SIGNAL_TIMEOUT_MS;
    309 while (monotonic_ms() < deadline) {
    310 struct pollfd watch = {.fd = watch_fd, .events = POLLIN};
    311 const int result = poll(&watch, 1, 20);
    312 if (result < 0 && errno != EINTR) break;
    313 if (result > 0 && (watch.revents & POLLIN)) {
    314 const ssize_t length = read(watch_fd, buffer, sizeof(buffer));
    315 size_t offset = 0;
    316 while (length > 0 && offset + sizeof(struct inotify_event) <= (size_t)length) {
    317 const struct inotify_event *event = (const struct inotify_event *)(buffer + offset);
    318 const size_t event_size = sizeof(*event) + event->len;
    319 if (event_size > (size_t)length - offset) break;
    320 offset += event_size;
    321 const size_t prefix_length = strlen(signal_prefix);
    322 if (!event->len || !memchr(event->name, '\0', event->len)
    323 || !(event->mask & (IN_CLOSE_WRITE | IN_MOVED_TO))
    324 || strncmp(event->name, signal_prefix, prefix_length)) continue;
    325 const char *sequence = event->name + prefix_length;
    326 const size_t digits = strspn(sequence, "0123456789");
    327 if (!digits || strcmp(sequence + digits, ".jpg")) continue;
    328 char source[PATH_MAX];
    329 const int source_length = snprintf(source, sizeof(source), "%s/%s", signal_directory, event->name);
    330 if (source_length < 0 || (size_t)source_length >= sizeof(source)) return CAPTURE_FAILED;
    331 return publish_image(source, destination) ? CAPTURE_FAILED : CAPTURE_SAVED;
    332 }
    333 }
    334 if (!child_alive(NULL)) break;
    335 }
    336 fprintf(stderr, "AI_CAM_PIPE:\tshot completion ambiguous; no replay, private prefix %s/%s\n",
    337 signal_directory, signal_prefix);
    338 return CAPTURE_FAILED;
    339}

    341/** @brief Capture using a fresh camera subprocess.
    342 * @param directory Existing directory for private output.
    343 * @param destination New staging path for the completed JPEG.
    344 * @return Zero after successful child exit and publication, or -1 on failure.
    345 * @details Do not launch until the previous child is reaped. Use an unused
    346 * private path and publish only after the child exits successfully. */
    347static int oneshot_shoot(const char *directory, const char *destination)
    348{
    349 if (!persistent_stop() || !new_session(directory)) return -1;
    350 char source[PATH_MAX];
    351 const int length = snprintf(source, sizeof(source), "%s/%s000000.jpg", directory, signal_prefix);
    352 if (length < 0 || (size_t)length >= sizeof(source)) return -1;
    353 struct stat existing;
    354 if (!lstat(source, &existing) || errno != ENOENT) return -1;
    355 char *arguments[] = {
    356 (char *)USHER_AI_CAM_COMMAND, (char *)"-o", source,
    357 (char *)"--width", (char *)USHER_AI_CAM_WIDTH,
    358 (char *)"--height", (char *)USHER_AI_CAM_HEIGHT,
    359 (char *)"-t", (char *)"200", (char *)"-n", NULL
    360 };
    361 if (spawn_camera(arguments)) return -1;
    362 const int64_t deadline = monotonic_ms() + (int64_t)USHER_AI_CAM_TIMEOUT_SECONDS * 1000;
    363 int status = -1;
    364 while (child_alive(&status)) {
    365 if (monotonic_ms() >= deadline) {
    366 stop_child();
    367 fprintf(stderr, "AI_CAM_PIPE:\tone-shot timed out; private image retained if present: %s\n", source);
    368 return -1;
    369 }
    370 wait_ms(20);
    371 }
    372 if (status < 0 || !WIFEXITED(status) || WEXITSTATUS(status)) return -1;
    373 return publish_image(source, destination);
    374}

    430int ai_cam_pipe_shoot(char *filename, size_t filename_size, int image_number)
    431{
    432 if (native_enabled) return imx500_pipe_shoot(&native_capture, filename, filename_size, image_number);
    433 if (!filename || !filename_size) return -1;
    434 filename[0] = '\0';
    435 if (image_number < 0) return -1;
    436 char resolved[PATH_MAX];
    437 const char *directory = usher_resolve_path(photo_directory, resolved, sizeof(resolved));
    438 const int length = snprintf(filename, filename_size, "%s/a%06d.jpg", directory, image_number);
    439 if (length < 0 || (size_t)length >= filename_size || usher_prepare_photo(filename, filename_size) != 0) {
    440 filename[0] = '\0';
    441 return -1;
    442 }
    443 const int64_t started = monotonic_ms();
    444 enum capture_result result = CAPTURE_NOT_TRIGGERED;
    445 const bool persistent = persistent_enabled();
    446 if (persistent && started >= retry_ms) {
    447 if (child_pid <= 0) persistent_start(directory);
    448 result = persistent_shoot(filename);
    449 if (result != CAPTURE_SAVED) {
    450 if (signal_zsl) zsl_failed = true;
    451 if (failures < 2) ++failures;
    452 if (failures >= 2) retry_ms = monotonic_ms() + USHER_AI_CAM_RETRY_MS;
    453 persistent_stop();
    454 } else {
    455 failures = 0;
    456 retry_ms = 0;
    457 }
    458 }
    459 /* Only a proven pre-trigger failure permits the one-shot fallback. */
    460 if (result == CAPTURE_NOT_TRIGGERED) {
    461 result = oneshot_shoot(directory, filename) ? CAPTURE_FAILED : CAPTURE_SAVED;
    462 if (persistent && monotonic_ms() >= retry_ms && child_pid <= 0) persistent_start(directory);
    463 }
    464 if (timing_enabled) fprintf(stderr, "AI_CAM_PIPE TIMING:\tshot=%d capture_total_ms=%lld result=%d\n",
    465 image_number, (long long)(monotonic_ms() - started), result);
    466 if (result != CAPTURE_SAVED) filename[0] = '\0';
    467 return result == CAPTURE_SAVED ? 0 : -1;
    468}
