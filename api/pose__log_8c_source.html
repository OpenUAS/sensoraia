    19 * @file pose_log.c
    20 * @brief Bounded asynchronous CSV recorder for flight-controller pose evidence.
    21 * @details The UART path uses trylock and drops rather than blocks when the writer is

    95/** Configure the writer wake condition against the same monotonic clock as its deadline. */
    96static void initialize_condition(void)
    97{
    98 pthread_condattr_t attributes;
    99 condition_error = pthread_condattr_init(&attributes);
    100 if (condition_error != 0) return;
    101 condition_error = pthread_condattr_setclock(&attributes, CLOCK_MONOTONIC);
    102 if (condition_error == 0) condition_error = pthread_cond_init(&available, &attributes);
    103 pthread_condattr_destroy(&attributes);
    104}

    106/** Serialize one validated wire sample and its optional clock mapping as a CSV row.
    107 * @param record Queue-owned immutable snapshot.
    108 * @return Zero when written, otherwise errno-style I/O or bounds error. */
    109static int write_record(const struct pose_record *record)
    110{
    111 const union usher_pose_sample_union *sample = &record->sample;
    112 const union dc_shot_union *shot = &sample->data.shot;
    113 char line[1024];
    114 int length = snprintf(line, sizeof(line),
    115 "%d,%s,%" PRIu64 ",%" PRIu64 ",%" PRIu64 ",%" PRIu32 ",%" PRIu32 ",%" PRIu32
    116 ",%" PRId32 ",%" PRId32 ",%" PRId32 ",%" PRId32 ",%" PRId32 ",%" PRId32
    117 ",%" PRId32 ",%" PRId32 ",%" PRId32 ",%" PRId32 ",%" PRId32 ",%" PRId32
    118 ",%" PRId32 ",%" PRIu32 ",%" PRIu32 ",%" PRIu32 ",%" PRIu32 ",%" PRIu32
    119 ",%" PRIu32 ",%" PRIu32 ",%" PRIu32 ",%" PRIu32,
    120 POSE_LOG_SCHEMA_VERSION, boot_id, record->received_us, record->ordinal, record->dropped,
    121 sample->data.sequence, sample->data.sample_begin_us, sample->data.sample_end_us,
    122 shot->data.nr, shot->data.lat, shot->data.lon, shot->data.alt,
    123 shot->data.phi, shot->data.theta, shot->data.psi, shot->data.vground,
    124 shot->data.course, shot->data.groundalt, sample->data.velocity_north_bfp,
    125 sample->data.velocity_east_bfp, sample->data.velocity_down_bfp,
    126 sample->data.gps_tow_ms, sample->data.gps_week, sample->data.gps_hacc_cm,
    127 sample->data.gps_vacc_cm, sample->data.gps_sacc_cm_s, sample->data.gps_fix,
    128 sample->data.gps_num_sv, sample->data.gps_valid_fields, sample->data.flags);
    129 if (length < 0 || (size_t)length >= sizeof(line)) return EOVERFLOW;
    130 const struct pose_clock_evidence *clock = &record->clock;
    131 int extra = snprintf(line + length, sizeof(line) - (size_t)length,
    132 ",%" PRIu32 ",%" PRIu32 ",%d,%" PRIu64 ",%" PRIu64 ",%" PRIu64
    133 ",%" PRIu64 ",%" PRIu32 ",%" PRIu32 ",%" PRIu64 "\n",
    134 clock->token.data.token_low, clock->token.data.token_high, clock->mapped,
    135 clock->sample_time.earliest_us, clock->sample_time.latest_us,
    136 clock->probe_sent_us, clock->probe_received_us,
    137 clock->probe_receive_fc_us, clock->probe_transmit_fc_us, CLOCK_ALIGNMENT_DRIFT_PPM);
    138 if (extra < 0 || (size_t)extra >= sizeof(line) - (size_t)length) return EOVERFLOW;
    139 length += extra;
    140 if (bytes_written > POSE_LOG_MAX_BYTES || (size_t)length > POSE_LOG_MAX_BYTES - bytes_written) return EFBIG;
    141 if (fwrite(line, 1, (size_t)length, output) != (size_t)length) return errno ? errno : EIO;
    142 bytes_written += (size_t)length;
    143 return 0;
    144}

    146/** Drain bounded batches off the UART thread and fdatasync before counting them synced.
    147 * @param unused pthread entry-point argument; unused. */
    148static void *write_queue(void *unused)
    149{
    150 (void)unused;
    151 for (;;) {
    152 struct pose_record batch[POSE_LOG_BATCH];
    153 struct timespec deadline;
    154 if (clock_gettime(CLOCK_MONOTONIC, &deadline) != 0) {
    155 int error = errno;
    156 pthread_mutex_lock(&mutex);
    157 log_error = error;
    158 pthread_mutex_unlock(&mutex);
    159 return NULL;
    160 }
    161 ++deadline.tv_sec;
    162 pthread_mutex_lock(&mutex);
    163 while (!stopping && count < POSE_LOG_BATCH) {
    164 int result = pthread_cond_timedwait(&available, &mutex, &deadline);
    165 if (result == ETIMEDOUT) break;
    166 if (result != 0) {
    167 log_error = result;
    168 pthread_mutex_unlock(&mutex);
    169 return NULL;
    170 }
    171 }
    172 if (stopping && count == 0) {
    173 pthread_mutex_unlock(&mutex);
    174 return NULL;
    175 }
    176 size_t batch_count = 0;
    177 while (count > 0 && batch_count < POSE_LOG_BATCH) {
    178 batch[batch_count++] = queue[head];
    179 head = (head + 1) % POSE_LOG_CAPACITY;
    180 --count;
    181 }
    182 pthread_mutex_unlock(&mutex);
    183 if (batch_count == 0) continue;
    184 int error = 0;
    185 for (size_t index = 0; index < batch_count && error == 0; ++index) error = write_record(&batch[index]);
    186 if (error == 0 && (fflush(output) != 0 || fdatasync(fileno(output)) != 0)) error = errno ? errno : EIO;
    187 pthread_mutex_lock(&mutex);
    188 if (error == 0) synced += batch_count;
    189 else log_error = error;
    190 pthread_mutex_unlock(&mutex);
    191 if (error != 0) return NULL;
    192 }
    193}

    195/** Start one durable CSV session; an empty path leaves optional logging disabled.
    196 * @return Zero on active/disabled success, otherwise -1 with errno set. */
    197int pose_log_start(const char *directory)
    198{
    199 pthread_mutex_lock(&mutex);
    200 if (active) {
    201 pthread_mutex_unlock(&mutex);
    202 errno = EBUSY;
    203 return -1;
    204 }
    205 head = count = bytes_written = 0;
    206 accepted = synced = rejected = 0;
    207 atomic_store(&dropped, 0);
    208 log_error = 0;
    209 stopping = false;
    210 if (directory == NULL || directory[0] == '\0') {
    211 pthread_mutex_unlock(&mutex);
    212 return 0;
    213 }
    214 int initialization = pthread_once(&condition_once, initialize_condition);
    215 if (initialization != 0 || condition_error != 0) {
    216 log_error = initialization != 0 ? initialization : condition_error;
    217 pthread_mutex_unlock(&mutex);
    218 errno = log_error;
    219 return -1;
    220 }
    221 char path[PATH_MAX] = {0};
    222 char resolved_dir[PATH_MAX];
    223 const char *target_dir = usher_resolve_path(directory, resolved_dir, sizeof(resolved_dir));
    224 usher_ensure_directory(target_dir);
    225
    226 usher_boot_id(boot_id);
    227 char stamp[32];
    228 struct tm utc;
    229 time_t now = time(NULL);
    230 if (gmtime_r(&now, &utc) == NULL || strftime(stamp, sizeof(stamp), "%Y%m%dT%H%M%SZ", &utc) == 0) {
    231 pthread_mutex_unlock(&mutex);
    232 errno = EINVAL;
    233 return -1;
    234 }
    235 char temporary_path[PATH_MAX] = {0};
    236 int length = snprintf(temporary_path, sizeof(temporary_path), "%s/pose-%s-XXXXXX", target_dir, stamp);
    237 if (length < 0 || (size_t)length >= sizeof(temporary_path)) {
    238 pthread_mutex_unlock(&mutex);
    239 errno = ENAMETOOLONG;
    240 return -1;
    241 }
    242 int directory_fd = open(target_dir, O_RDONLY | O_DIRECTORY | O_CLOEXEC);
    243 if (directory_fd < 0) {
    244 log_error = errno;
    245 pthread_mutex_unlock(&mutex);
    246 return -1;
    247 }
    248 int descriptor = mkstemp(temporary_path);
    249 bool renamed = false;
    250 int error = descriptor < 0 ? errno : 0;
    251 if (error == 0 && fcntl(descriptor, F_SETFD, FD_CLOEXEC) != 0) error = errno;
    252 if (error == 0) {
    253 length = snprintf(path, sizeof(path), "%s.csv", temporary_path);
    254 if (length < 0 || (size_t)length >= sizeof(path)) error = ENAMETOOLONG;
    255 else if (rename(temporary_path, path) != 0) error = errno;
    256 else renamed = true;
    257 }
    258 if (error == 0) {
    259 output = fdopen(descriptor, "w");
    260 if (output == NULL) {
    261 error = errno;
    262 }
    263 }
    264 if (error == 0) {
    265 bytes_written = sizeof(header) - 1;
    266 if (bytes_written > POSE_LOG_MAX_BYTES) error = EFBIG;
    267 else if (fwrite(header, 1, bytes_written, output) != bytes_written || fflush(output) != 0
    268 || fsync(fileno(output)) != 0 || fsync(directory_fd) != 0) error = errno ? errno : EIO;
    269 }
    270 close(directory_fd);
    271 if (error == 0) error = pthread_create(&writer, NULL, write_queue, NULL);
    272 if (error != 0) {
    273 if (output != NULL) fclose(output);
    274 else if (descriptor >= 0) close(descriptor);
    275 output = NULL;
    276 if (descriptor >= 0) {
    277 unlink(temporary_path);
    278 if (renamed) unlink(path);
    279 }
    280 log_error = error;
    281 pthread_mutex_unlock(&mutex);
    282 errno = error;
    283 return -1;
    284 }
    285 active = true;
    286 pthread_mutex_unlock(&mutex);
    287 fprintf(stderr, "USHER POSE: logging to %s\n", path);
    288 return 0;
    289}

    301/** Validate and enqueue a pose with optional token-bound FC/MORA clock evidence.
    302 * @param payload Wire-format USHER_POSE_SAMPLE bytes.
    303 * @param length Must equal the packed message size.
    304 * @param receive_monotonic_us UART receive timestamp from CLOCK_MONOTONIC.
    305 * @param evidence Optional probe and mapped interval copied into this row.
    306 * @return 1 queued, 0 best-effort drop, or -1 malformed timing/pose data. */
    307int pose_log_record_clocked(const uint8_t *payload, size_t length, uint64_t receive_monotonic_us,
    308 const struct pose_clock_evidence *evidence)
    309{
    310 if (pthread_mutex_trylock(&mutex) != 0) {
    311 atomic_fetch_add(&dropped, 1);
    312 return 0;
    313 }
    314 if (!active || stopping) {
    315 pthread_mutex_unlock(&mutex);
    316 return 0;
    317 }
    318 union usher_pose_sample_union sample = {0};
    319 bool valid = payload != NULL && length == sizeof(sample.bin) && receive_monotonic_us != 0;
    320 if (valid) {
    321 for (size_t index = 0; index < sizeof(sample.bin); ++index) sample.bin[index] = payload[index];
    322 valid = usher_pose_flags_valid(sample.data.flags)
    323 && (uint32_t)(sample.data.sample_end_us - sample.data.sample_begin_us) <= POSE_LOG_MAX_SAMPLE_WINDOW_US
    324 && sample.data.shot.data.lat >= -90 * POSE_LOG_COORDINATE_SCALE
    325 && sample.data.shot.data.lat <= 90 * POSE_LOG_COORDINATE_SCALE
    326 && sample.data.shot.data.lon >= -180 * POSE_LOG_COORDINATE_SCALE
    327 && sample.data.shot.data.lon <= 180 * POSE_LOG_COORDINATE_SCALE;
    328 }
    329 if (valid && evidence != NULL && evidence->mapped) {
    330 valid = (evidence->token.data.token_low != 0 || evidence->token.data.token_high != 0)
    331 && evidence->sample_time.earliest_us > 0
    332 && evidence->sample_time.earliest_us <= evidence->sample_time.latest_us
    333 && evidence->sample_time.latest_us <= receive_monotonic_us
    334 && evidence->probe_sent_us > 0 && evidence->probe_sent_us <= evidence->probe_received_us
    335 && evidence->probe_received_us <= receive_monotonic_us;
    336 }
    337 if (!valid) {
    338 ++rejected;
    339 pthread_mutex_unlock(&mutex);
    340 return -1;
    341 }
    342 if (log_error != 0 || count == POSE_LOG_CAPACITY) {
    343 atomic_fetch_add(&dropped, 1);
    344 pthread_mutex_unlock(&mutex);
    345 return 0;
    346 }
    347 struct pose_record *record = &queue[(head + count) % POSE_LOG_CAPACITY];
    348 record->sample = sample;
    349 record->received_us = receive_monotonic_us;
    350 record->ordinal = ++accepted;
    351 record->dropped = atomic_load(&dropped);
    352 record->clock = evidence != NULL ? *evidence : (struct pose_clock_evidence) {0};
    353 if (!record->clock.mapped) {
    354 const union usher_clock_request_union token = record->clock.token;
    355 record->clock = (struct pose_clock_evidence) {.token = token};
    356 }
    357 ++count;
    358 pthread_cond_signal(&available);
    359 pthread_mutex_unlock(&mutex);
    360 return 1;
    361}

    373/** Stop accepting records, drain the queue, join the writer, and close the CSV.
    374 * @return Zero after orderly durable shutdown, otherwise -1 with errno set. */
    375int pose_log_stop(void)
    376{
    377 pthread_mutex_lock(&mutex);
    378 if (!active) {
    379 pthread_mutex_unlock(&mutex);
    380 return 0;
    381 }
    382 stopping = true;
    383 pthread_cond_signal(&available);
    384 pthread_mutex_unlock(&mutex);
    385 int error = pthread_join(writer, NULL);
    386 if (error != 0) {
    387 errno = error;
    388 return -1;
    389 }
    390 if (fclose(output) != 0) error = errno ? errno : EIO;
    391 pthread_mutex_lock(&mutex);
    392 output = NULL;
    393 active = false;
    394 if (log_error == 0) log_error = error;
    395 error = log_error;
    396 pthread_mutex_unlock(&mutex);
    397 if (error != 0) errno = error;
    398 return error == 0 ? 0 : -1;
    399}
