    93/**
    94 * @brief Read the local clock used for capture deadlines and pose alignment.
    95 * @return Monotonic microseconds, or zero when the clock read fails.
    96 * @details Use the same time base for age and ordering checks so wall-clock
    97 * corrections do not make old evidence appear fresh.
    98 */
    99static uint64_t monotonic_us(void)
    100{
    101 struct timespec now;
    102 if (clock_gettime(CLOCK_MONOTONIC, &now)) return 0;
    103 return (uint64_t)now.tv_sec * 1000000 + now.tv_nsec / 1000;
    104}

    106/**
    107 * @brief Obtain a nonzero token for a new survey session.
    108 * @param[out] token Destination; use only on success.
    109 * @return True for a complete, nonzero token and successful descriptor close.
    110 * @details Accumulate short reads and retry interruptions instead of accepting
    111 * a partial token. Zero is reserved for inactive session state.
    112 */
    113static bool random_session(uint32_t *token)
    114{
    115 const int descriptor = open("/dev/urandom", O_RDONLY | O_CLOEXEC);
    116 if (descriptor < 0) return false;
    117 size_t received = 0;
    118 while (received < sizeof(*token)) {
    119 const ssize_t count = read(descriptor, (unsigned char *)token + received, sizeof(*token) - received);
    120 if (count < 0 && errno == EINTR) continue;
    121 if (count <= 0) break;
    122 received += (size_t)count;
    123 }
    124 const bool valid = close(descriptor) == 0 && received == sizeof(*token) && *token != 0;
    125 return valid;
    126}

    146/**
    147 * @brief Retire an intention and record its terminal outcome.
    148 * @param[in,out] intent Owned entry to clear.
    149 * @param outcome Result code to report to each camera in the intent mask.
    150 * @details Report per camera before clearing the slot. The caller holds the
    151 * arbiter lock, and result queue overflow must not prevent retirement.
    152 */
    153static void expire(struct capture_intent *intent, uint32_t outcome)
    154{
    155 for (unsigned camera_id = 1; camera_id <= COVERAGE_CAMERAS; ++camera_id)
    156 if (intent->mask & (1U << (camera_id - 1))) report(intent, camera_id, outcome);
    157 *intent = (struct capture_intent) {0};
    158}

    172/** @brief Convert FC units and estimate geometric detail using this camera's calibration. */
    173static bool project(const union dc_shot_union *shot, unsigned camera_id, double uncertainty,
    174 struct coverage_candidate *candidate)
    175{
    176 if (!origin_set || camera_id < 1 || camera_id > COVERAGE_CAMERAS || !calibrated[camera_id - 1]) return false;
    177 const struct coverage_pose pose = {
    178 .latitude_deg = shot->data.lat / 1e7, .longitude_deg = shot->data.lon / 1e7,
    179 .agl_m = shot->data.groundalt / 256.0,
    180 .roll_deg = shot->data.phi * (57.29577951308232 / 4096.0),
    181 .pitch_deg = shot->data.theta * (57.29577951308232 / 4096.0),
    182 .yaw_deg = shot->data.psi * (57.29577951308232 / 4096.0), .uncertainty_m = uncertainty
    183 };
    184 *candidate = (struct coverage_candidate) {.camera_id = camera_id};
    185 const struct coverage_camera *camera = &cameras[camera_id - 1];
    186 if (!coverage_project(camera, &pose, latitude, longitude, &candidate->footprint)) return false;
    187 double twice_area = 0;
    188 for (size_t index = 0; index < 4; ++index) {
    189 const struct coverage_point *first = &candidate->footprint.corner[index];
    190 const struct coverage_point *second = &candidate->footprint.corner[(index + 1) % 4];
    191 twice_area += first->east * second->north - first->north * second->east;
    192 }
    193 const double metres_per_pixel = sqrt(fabs(twice_area) / (2.0 * camera->width * camera->height));
    194 if (!isfinite(metres_per_pixel) || metres_per_pixel <= 0) return false;
    195 candidate->quality = (uint8_t)fmax(1.0, fmin(255.0, floor(1 / metres_per_pixel)));
    196 return true;
    197}

    199/** @brief Take the short arc across heading wrap, using widened subtraction before conversion. */
    200static double angle_delta(int32_t first, int32_t second)
    201{
    202 return remainder(((int64_t)second - first) / 4096.0, 6.283185307179586);
    203}

    205/** @brief Forecast a short continuation only when mapped motion is suitably well behaved. */
    206static bool predict(uint64_t time_us, union dc_shot_union *shot, double *uncertainty)
    207{
    208 if (pose_count < 2) return false;
    209 const struct capture_pose *latest = &poses[(pose_head + pose_count - 1) % ARBITER_COUNT(poses)];
    210 const struct capture_pose *previous = &poses[(pose_head + pose_count - 2) % ARBITER_COUNT(poses)];
    211 /* Do not extend an old straight segment through a turn or missing telemetry. */
    212 if (time_us < latest->time_us || time_us - latest->time_us > 500000
    213 || latest->time_us <= previous->time_us || latest->time_us - previous->time_us > 200000) return false;
    214 const double elapsed = (time_us - latest->time_us) / 1e6;
    215 const double sample_time = (latest->time_us - previous->time_us) / 1e6;
    216 const double roll_rate = angle_delta(previous->shot.data.phi, latest->shot.data.phi) / sample_time;
    217 const double pitch_rate = angle_delta(previous->shot.data.theta, latest->shot.data.theta) / sample_time;
    218 const double yaw_rate = angle_delta(previous->shot.data.psi, latest->shot.data.psi) / sample_time;
    219 if (fabs(roll_rate) > 0.15 || fabs(pitch_rate) > 0.15 || fabs(yaw_rate) > 0.25
    220 || fabs(latest->shot.data.phi / 4096.0) > 0.6 || fabs(latest->shot.data.theta / 4096.0) > 0.6) return false;
    221 *shot = latest->shot;
    222 const double radians = 0.017453292519943295;
    223 const double speed = shot->data.vground / 524288.0, course = shot->data.course / 4096.0;
    224 const double predicted_latitude = shot->data.lat / 1e7 + speed * elapsed * cos(course) / 6378137 / radians;
    225 const double predicted_longitude = shot->data.lon / 1e7
    226 + speed * elapsed * sin(course) / (6378137 * cos(shot->data.lat / 1e7 * radians)) / radians;
    227 if (fabs(predicted_latitude) > 85 || fabs(predicted_longitude) > 180) return false;
    228 shot->data.lat = (int32_t)llround(predicted_latitude * 1e7);
    229 shot->data.lon = (int32_t)llround(predicted_longitude * 1e7);
    230 shot->data.phi += (int32_t)llround(roll_rate * elapsed * 4096);
    231 shot->data.theta += (int32_t)llround(pitch_rate * elapsed * 4096);
    232 shot->data.psi = (int32_t)llround(remainder(shot->data.psi / 4096.0 + yaw_rate * elapsed, 6.283185307179586) * 4096);
    233 *uncertainty = latest->accuracy_m + speed * (latest->error_us / 1e6 + elapsed * 0.25)
    234 + 5 * elapsed * elapsed + shot->data.groundalt / 256.0 * (0.02 + elapsed * 0.1);
    235 return isfinite(*uncertainty) && *uncertainty <= 15;
    236}

    238bool capture_arbiter_load_camera(const char *filename)
    239{
    240 struct usher_camera_profile profile;
    241 if (!usher_camera_profile_read(filename, &profile)) {
    242 fprintf(stderr, "USHER SURVEY: camera profile rejected: invalid bounded flat profile\n");
    243 return false;
    244 }
    245 if (profile.camera_id < 1 || profile.camera_id > COVERAGE_CAMERAS) {
    246 fprintf(stderr, "USHER SURVEY: camera profile rejected: prefix must be c, a or l\n");
    247 return false;
    248 }
    249 const struct coverage_camera *camera = &profile.calibration;
    250 const struct coverage_pose pose = {.agl_m = 10};
    251 struct coverage_footprint footprint;
    252 if (!coverage_project(camera, &pose, 0, 0, &footprint)) {
    253 fprintf(stderr, "USHER SURVEY: camera profile rejected: invalid calibration\n");
    254 return false;
    255 }
    256
    257 bool accepted = false;
    258 const char *error = "coverage initialization";
    259 pthread_mutex_lock(&guard);
    260 if (!covered) {
    261 /* 1024 cells * 4 m = 4096 m per side. Centre the ledger on the first trusted
    262 * saved exposure (chosen later); -2048 is a metre offset, not image resolution. */
    263 if (!coverage_init(&coverage_ledger, -2048, -2048, 4)) goto unlock;
    264 uint32_t token = 0;
    265 error = "session token";
    266 if (!random_session(&token)) goto unlock;
    267 covered = &coverage_ledger;
    268 session = token;
    269 }
    270 cameras[profile.camera_id - 1] = *camera;
    271 calibrated[profile.camera_id - 1] = true;
    272 accepted = true;
    273unlock:
    274 pthread_mutex_unlock(&guard);
    275 if (!accepted) fprintf(stderr, "USHER SURVEY: camera profile rejected: %s\n", error);
    276 return accepted;
    277}

    312void capture_arbiter_set_scope(const union usher_capture_arbiter_state_union *scope, uint64_t now_us)
    313{
    314 if (!scope || scope->data.active > 1 || scope->data.camera_mask > USHER_CAMERA_MASK_ALL) return;
    315 pthread_mutex_lock(&guard);
    316 if (!session || scope->data.session != session) goto done;
    317 scope_active = scope->data.active != 0;
    318 scope_mask = scope->data.camera_mask & (USHER_CAMERA_MASK_CHDK | USHER_CAMERA_MASK_AICAM | USHER_CAMERA_MASK_LWIR);
    319 scope_us = now_us;
    320 for (size_t index = 0; index < ARBITER_COUNT(waiting); ++index) {
    321 struct capture_intent *intent = &waiting[index];
    322 if (!scope_active) expire(intent, USHER_CAPTURE_ARBITER_EXPIRED);
    323 else {
    324 struct capture_intent removed = *intent;
    325 removed.mask &= ~scope_mask;
    326 expire(&removed, USHER_CAPTURE_ARBITER_EXPIRED);
    327 intent->mask &= scope_mask;
    328 }
    329 }
    330done:
    331 pthread_mutex_unlock(&guard);
    332}

    334bool capture_arbiter_offer(const union usher_capture_arbiter_shoot_union *request, uint64_t now_us)
    335{
    336 if (!request || request->data.camera_mask > USHER_CAMERA_MASK_ALL) return false;
    337 bool accepted = false;
    338 pthread_mutex_lock(&guard);
    339 if (!session || request->data.session != session) goto done;
    340 accepted = true;
    341 struct capture_intent offered = {.shot = request->data.shot, .mask = request->data.camera_mask & 7U, .received_us = now_us};
    342 if (!offered.mask) goto done;
    343 if (!fresh_scope(now_us)) {
    344 expire(&offered, USHER_CAPTURE_ARBITER_EXPIRED);
    345 goto done;
    346 }
    347 offered.mask &= scope_mask;
    348 /* Do not create another exposure for a request still owned by that camera. */
    349 for (size_t index = 0; index < ARBITER_COUNT(active); ++index)
    350 if (active[index].mask && active[index].shot.data.nr == offered.shot.data.nr) offered.mask &= ~active[index].mask;
    351 for (unsigned camera_index = 0; camera_index < COVERAGE_CAMERAS; ++camera_index) {
    352 const uint32_t bit = 1U << camera_index;
    353 if (!(offered.mask & bit)) continue;
    354 const size_t begin = camera_index * COVERAGE_CANDIDATES;
    355 size_t free_entry = ARBITER_COUNT(waiting), oldest = begin;
    356 bool duplicate = false;
    357 for (size_t index = begin; index < begin + COVERAGE_CANDIDATES; ++index) {
    358 const struct capture_intent *intent = &waiting[index];
    359 if (intent->mask && intent->shot.data.nr == offered.shot.data.nr) duplicate = true;
    360 if (!intent->mask && free_entry == ARBITER_COUNT(waiting)) free_entry = index;
    361 if (intent->received_us < waiting[oldest].received_us) oldest = index;
    362 }
    363 if (duplicate) continue;
    364 const size_t destination = free_entry != ARBITER_COUNT(waiting) ? free_entry : oldest;
    365 if (waiting[destination].mask) expire(&waiting[destination], USHER_CAPTURE_ARBITER_SUPERSEDED);
    366 waiting[destination] = (struct capture_intent) {.shot = offered.shot, .mask = bit, .received_us = now_us};
    367 }
    368done:
    369 pthread_mutex_unlock(&guard);
    370 return accepted;
    371}

    373/** @brief Make one admission decision while guard is held; no callbacks or dynamic allocation. */
    374static bool take_locked(uint32_t ready_mask, uint64_t now_us, union dc_shot_union *shot, uint8_t *mask)
    375{
    376 for (size_t index = 0; index < ARBITER_COUNT(waiting); ++index) {
    377 struct capture_intent *intent = &waiting[index];
    378 if (intent->mask && (!fresh_scope(now_us) || now_us < intent->received_us
    379 || now_us - intent->received_us > intent_lifetime_us)) expire(intent, USHER_CAPTURE_ARBITER_EXPIRED);
    380 }
    381 unsigned free_slots = 0;
    382 for (size_t index = 0; index < ARBITER_COUNT(active); ++index) if (!active[index].mask) ++free_slots;
    383 if (!free_slots) return false;
    384 for (unsigned camera_id = 1; camera_id <= COVERAGE_CAMERAS; ++camera_id) {
    385 unsigned in_flight = 0;
    386 for (size_t index = 0; index < ARBITER_COUNT(active); ++index)
    387 if (active[index].mask & (1U << (camera_id - 1))) ++in_flight;
    388 if (in_flight >= capacity[camera_id - 1]) ready_mask &= ~(1U << (camera_id - 1));
    389 }
    390 /* An unavailable sensor must not head-of-line block another bit of the FC mask. */
    391 size_t oldest = ARBITER_COUNT(waiting);
    392 for (size_t index = 0; index < ARBITER_COUNT(waiting); ++index)
    393 if ((waiting[index].mask & ready_mask) && (oldest == ARBITER_COUNT(waiting)
    394 || waiting[index].received_us < waiting[oldest].received_us)) oldest = index;
    395 if (oldest == ARBITER_COUNT(waiting)) return false;
    396 struct capture_intent *intent = &waiting[oldest];
    397 uint32_t selected = intent->mask & ready_mask;
    398 for (unsigned camera_id = 1; camera_id <= COVERAGE_CAMERAS; ++camera_id) {
    399 const uint32_t bit = 1U << (camera_id - 1);
    400 if (!(selected & bit)) continue;
    401 struct coverage_candidate candidate;
    402 union dc_shot_union current;
    403 double uncertainty = 0;
    404 if (predict(now_us, &current, &uncertainty) && project(&current, camera_id, uncertainty, &candidate)) {
    405 const double gain = coverage_gain(covered, &candidate);
    406 struct coverage_candidate opportunities[COVERAGE_CANDIDATES] = {0};
    407 size_t count = 0;
    408 for (unsigned step = 0; step < ARBITER_COUNT(opportunities); ++step) {
    409 const uint64_t planned = now_us + step * 75000;
    410 union dc_shot_union prediction;
    411 double error = 0;
    412 struct coverage_candidate opportunity;
    413 if (!predict(planned, &prediction, &error) || !project(&prediction, camera_id, error, &opportunity)) break;
    414 opportunity.ready_us = planned;
    415 opportunity.deadline_us = planned + 75000;
    416 opportunities[count++] = opportunity;
    417 }
    418 const int choice = coverage_choose(covered, opportunities, count, now_us, service_us[camera_id - 1]);
    419 fprintf(stderr, "USHER SURVEY: shot=%d camera=%u coverage_score_m2=%.1f mode=%s\n",
    420 intent->shot.data.nr, camera_id, gain, enforce ? "enforce" : "observe");
    421 if (choice > 0) fprintf(stderr, "USHER SURVEY: lookahead=%d defer_us=%llu\n", choice,
    422 (unsigned long long)(opportunities[choice].ready_us - now_us));
    423 /* Zero gain is not proof of redundancy: uncertainty-expanded coverage is. */
    424 if (enforce && coverage_wholly_saved(covered, &candidate)) {

    447 return true;

    491 have_saved[camera_id - 1] = true;

    493 }
    494 }

    630/**
    631 * @brief Interpolate a scalar fixed-point pose field.
    632 * @param first Value at the earlier sample.
    633 * @param second Value at the later sample.
    634 * @param fraction Position between the endpoints, from zero to one.
    635 * @return Nearest representable value in the original fixed-point units.
    636 * @details Widen before subtracting to accommodate opposite signed endpoints.
    637 * Use wrapped angular deltas instead for headings and attitude.
    638 */

    640{
    641 return (int32_t)llround(first + fraction * ((int64_t)second - first));

    644/**
    645 * @brief Interpolate a qualified pose from bounded sample history.
    646 * @param time_us Query time in local monotonic microseconds.
    647 * @param time_error_us Timing uncertainty around the query, in microseconds.
    648 * @param[out] shot Interpolated pose; use only on success.
    649 * @param[out] uncertainty_m Estimated position uncertainty in metres; use only on success.
    650 * @param[in,out] observation Optional destination for navigation provenance on success.
    651 * @return True if the query is bracketed and passes motion and uncertainty checks.
    652 * @details Do not extrapolate beyond recorded pose samples. Reject discontinuous
    653 * or widely spaced pairs before interpolating scalars and wrapped angles.
    654 * Combine timing, navigation accuracy, and attitude change as a conservative
    655 * operational check, not a surveyed accuracy guarantee.
    656 */
    657static bool pose_at(uint64_t time_us, uint64_t time_error_us, union dc_shot_union *shot,
    658 double *uncertainty_m, struct capture_observation *observation)
    659{
    660 if (!shot || !uncertainty_m || time_error_us > 100000) return false;
    661 bool valid = false;
    662 pthread_mutex_lock(&guard);
    663 for (size_t index = 1; index < pose_count; ++index) {
    664 const struct capture_pose *before = &poses[(pose_head + index - 1) % ARBITER_COUNT(poses)];
    665 const struct capture_pose *after = &poses[(pose_head + index) % ARBITER_COUNT(poses)];
    666 if (time_us < before->time_us || time_us > after->time_us) continue;
    667 const uint64_t span = after->time_us - before->time_us;
    668 if (!span || span > 200000) break;
    669 const double fraction = (double)(time_us - before->time_us) / span;
    670 const double roll_change = angle_delta(before->shot.data.phi, after->shot.data.phi);
    671 const double pitch_change = angle_delta(before->shot.data.theta, after->shot.data.theta);
    672 if (fabs(roll_change) > 0.1 || fabs(pitch_change) > 0.1
    673 || fabs(before->shot.data.phi / 4096.0) > 0.6 || fabs(after->shot.data.phi / 4096.0) > 0.6
    674 || fabs(before->shot.data.theta / 4096.0) > 0.6 || fabs(after->shot.data.theta / 4096.0) > 0.6
    675 || llabs((long long)after->shot.data.lat - before->shot.data.lat) > 10000
    676 || llabs((long long)after->shot.data.lon - before->shot.data.lon) > 10000) break;
    677 *shot = before->shot;
    678 shot->data.lat = interpolate(before->shot.data.lat, after->shot.data.lat, fraction);
    679 shot->data.lon = interpolate(before->shot.data.lon, after->shot.data.lon, fraction);
    680 shot->data.alt = interpolate(before->shot.data.alt, after->shot.data.alt, fraction);
    681 shot->data.groundalt = interpolate(before->shot.data.groundalt, after->shot.data.groundalt, fraction);
    682 shot->data.phi = (int32_t)llround(before->shot.data.phi + fraction * roll_change * 4096);
    683 shot->data.theta = (int32_t)llround(before->shot.data.theta + fraction * pitch_change * 4096);
    684 shot->data.psi = (int32_t)llround(before->shot.data.psi
    685 + fraction * angle_delta(before->shot.data.psi, after->shot.data.psi) * 4096);
    686 shot->data.course = (int32_t)llround(before->shot.data.course
    687 + fraction * angle_delta(before->shot.data.course, after->shot.data.course) * 4096);
    688 shot->data.vground = interpolate(before->shot.data.vground, after->shot.data.vground, fraction);
    689 const double speed = fmax(before->shot.data.vground, after->shot.data.vground) / 524288.0;
    690 /* Clock width, sample spacing and attitude change are an engineering error
    691 * margin, not a statistical covariance or a guarantee of surveyed accuracy. */
    692 *uncertainty_m = fmax(before->accuracy_m, after->accuracy_m)
    693 + speed * (time_error_us + before->error_us + after->error_us + span / 2) / 1e6
    694 + shot->data.groundalt / 256.0 * tan(hypot(roll_change, pitch_change));
    695 valid = isfinite(*uncertainty_m) && *uncertainty_m <= 15;
    696 if (valid && observation) {
    697 observation->navigation_projected = before->navigation_projected || after->navigation_projected;
    698 observation->navigation_age_ms = before->navigation_age_ms > after->navigation_age_ms
    699 ? before->navigation_age_ms : after->navigation_age_ms;
    700 }
    701 break;
    702 }
    703 pthread_mutex_unlock(&guard);
    704 return valid;
    705}
